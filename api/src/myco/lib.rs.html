1//! myco — the engine as a library. Everything that used to live
2//! in the binary lives here: the verb registry, clap tree, ops,
3//! serve/batch, views, continuity. `run()` is the compat CLI's
4//! whole body; drivers (`myco` bin, `tamias-driver-myco run`)
5//! split it into [`parse`]/[`parse_from`] + [`dispatch`] — argv
6//! in, exit code out, verbs callable as `dispatch(&matches)`.
7//!
8//! myco — project management, grown not built. A mycorrhizal
9//! network for your agents: tasks, signals, handoffs.
10//!
11//! Agent-first PM: markdown in `project/` is canonical and
12//! human-editable; every deterministic write goes through this CLI
13//! (auto-commit per verb, `--json` everywhere). The skill teaches
14//! agents the verbs; `myco manifest`/`myco schema` are the
15//! self-describing surface.
16//!
17//! Phase 6 wires continuity — `handoff`/`resume` over append-only
18//! `handoffs/<ID>.md` files (one per item, any kind); sepia hooks
19//! call them at session boundaries. Phase 7 lands the Jira-parity
20//! slice — `epic:`/`ms:` item metadata + grouping views,
21//! `absorb` importers (comment sweep, Jira CSV, GitHub dumps).
22//! P-007 makes the verb surface data: one [`registry`] table
23//! drives the clap tree, `manifest`, and serve/batch dispatch;
24//! `[verbs]` in `myco.toml` declares repo/user verbs.

68fn manifest() -> serde_json::Value {
69    serde_json::json!({
70        "name": "myco",
71        "version": env!("CARGO_PKG_VERSION"),
72        "protocol": 1,
73        "description": env!("CARGO_PKG_DESCRIPTION"),
74        "repository": env!("CARGO_PKG_REPOSITORY"),
75        "capabilities": {
76            "store": "markdown",
77            "index": "redb",
78            "protocol": 1,
79            "serve": {
80                "protocol": 1,
81                "transport": "stdio",
82                "wire": "jsonrpc-2.0",
83            },
84        },
85        "verbs": registry::manifest_verbs(),
86        "implemented": registry::manifest_implemented(),
87        "notes": "markdown in project/ is canonical; every verb is one \
88                  atomic commit on the `project` branch carrying an \
89                  `Actor:` trailer (the verb's actor — `--actor`/`--owner` \
90                  where it takes one, else the resolved actor: `$MYCO_ACTOR` \
91                  → `human:<git user.name>` → `cli`); the redb index under \
92                  $MYCO_HOME|~/.local/share/myco is a disposable derived cache",
93    })
94}

111/// The `ui` verb (feature `tui`). The index universe for
112/// `source = "index"` views is injected as a loader — the TUI
113/// re-runs it on every reload, so `load_fresh`'s staleness
114/// logic governs `:w` too.
115#[cfg(feature = "tui")]
116fn h_ui(ctx: &registry::Ctx, a: &registry::Args) -> Result<i32, String> {
117    let view = a.opt("view").map(str::to_string).or_else(|| {
118        // config `default_view` is the landing view — cwd falls
119        // back for index-sourced views outside a repo (`~/myco.toml`)
120        let root = ctx
121            .root
122            .clone()
123            .or_else(|| ctx.dir.clone())
124            .or_else(|| std::env::current_dir().ok())
125            .unwrap_or_else(|| std::path::PathBuf::from("."));
126        myco_ops::config::default_view_for(&root, ops::CONFIG)
127    });
128    let org: myco_tui::OrgLoader = Box::new(|| {
129        let (views, _warnings) = view::index_universe()?;
130        Ok(views
131            .into_iter()
132            .map(|v| {
133                let (items, archived) = v.snap.map(|s| (s.items, s.archived)).unwrap_or_default();
134                myco_tui::OrgRepo {
135                    name: v.name,
136                    worktree: v.worktree.into_std_path_buf(),
137                    items,
138                    archived,
139                    error: v.error,
140                }
141            })
142            .collect())
143    });
144    myco_tui::run(
145        ctx.dir.as_deref(),
146        ctx.root.as_deref(),
147        ctx.no_commit,
148        view,
149        Some(org),
150    )
151    .map(|_| 0)
152}

154/// Process argv → parsed matches. Split from [`dispatch`] so
155/// drivers can parse *their* argv (a `run` verb, a re-exec)
156/// without owning process args. Clap errors/help still exit the
157/// process, same as argv parsing.
158pub fn parse() -> clap::ArgMatches {
159    registry::command().get_matches()
160}
161
162/// Same, from an argv iterator — `parse_from(["myco", "board"])`.
163pub fn parse_from<I, T>(itr: I) -> clap::ArgMatches
164where
165    I: IntoIterator<Item = T>,
166    T: Into<std::ffi::OsString> + Clone,
167{
168    registry::command().get_matches_from(itr)
169}
170
171/// The global flags → the handler context.
172fn ctx_of(m: &clap::ArgMatches) -> registry::Ctx {
173    registry::Ctx {
174        json: m.get_flag("json"),
175        no_commit: m.get_flag("no_commit"),
176        dir: m.get_one::<String>("dir").map(PathBuf::from),
177        root: m.get_one::<String>("root").map(PathBuf::from),
178    }
179}
180
181/// One parsed command line → exit code. The dispatch every
182/// driver shares; argv parsing is a caller concern.
183pub fn dispatch(m: &clap::ArgMatches) -> i32 {
184    let ctx = ctx_of(m);
185    match registry::dispatch(&ctx, m) {
186        Ok(code) => code,
187        Err(e) => {
188            if ctx.json {
189                ops::print_json(&serde_json::json!({"error": e}));
190            } else {
191                eprintln!("myco: error: {e}");
192            }
193            2
194        }
195    }
196}

198/// The compat CLI body — parse argv, dispatch, hand back the code.
199/// `sigpipe::reset` restores conventional-Unix SIGPIPE behavior
200/// (die silently on EPIPE) — `println!` would otherwise panic when
201/// piped to a reader that closes early (`myco board | head`).
202pub fn run() -> i32 {
203    sigpipe::reset();
204    dispatch(&parse())
205}
206
207/// Same, from an argv iterator — the `run` passthrough entry
208/// every driver re-execs (`tamias-driver-myco run <myco argv>`).
209pub fn run_from<I, T>(argv: I) -> i32
210where
211    I: IntoIterator<Item = T>,
212    T: Into<std::ffi::OsString> + Clone,
213{
214    sigpipe::reset();
215    dispatch(&parse_from(argv))
216}
