1//! Item-file body templates — embedded `format!` strings, not a
2//! templating engine. Four kinds, four static skeletons; pulling in
3//! `minijinja` + a `myco-template` crate would add a dependency for
4//! nothing these can't do. Revisit when templates grow variants
5//! (epics, milestones).
6
7use myco_core::{date, Item};
8
9fn stamp(item: &Item, label: &str) -> String {
10    let created = item.created.map(date::render).unwrap_or_default();
11    format!("Status: **{label}** · Created: {created}")
12}
13
14/// `issues/` body — mirrors the v0 skill template: Status /
15/// Description / Checklist / Resolution.
16pub fn issue(item: &Item) -> String {
17    format!(
18        "# {title}\n\n{status}\n\n\
19         ## Description\n\n\
20         What is wrong and steps to reproduce.\n\n\
21         ## Checklist\n\n\
22         - [ ] Action item\n\n\
23         ## Resolution\n",
24        title = item.title,
25        status = stamp(item, "open"),
26    )
27}
28
29/// `proposals/` body — goal / requirements / design / questions.
30pub fn proposal(item: &Item) -> String {
31    format!(
32        "# {title}\n\n{status}\n\n\
33         ## Goal & Overview\n\n\n\n\
34         ## Requirements\n\n\n\n\
35         ## Technical Design\n\n\n\n\
36         ## Open questions\n",
37        title = item.title,
38        status = stamp(item, "draft"),
39    )
40}
41
42/// `plans/` body — a phase checklist.
43pub fn plan(item: &Item) -> String {
44    format!(
45        "# {title}\n\n{status}\n\n\
46         ## Phases\n\n\
47         - [ ] Phase 1 — \n\
48         - [ ] Phase 2 — \n",
49        title = item.title,
50        status = stamp(item, "open"),
51    )
52}
53
54/// `adrs/` body — context / decision / consequences.
55pub fn adr(item: &Item) -> String {
56    format!(
57        "# {title}\n\n{status}\n\n\
58         ## Context\n\n\n\n\
59         ## Decision\n\n\n\n\
60         ## Consequences\n",
61        title = item.title,
62        status = stamp(item, "proposed"),
63    )
64}

66/// Append `text` at the end of the `## <heading>` section in
67/// `body`; the heading is created at EOF when missing. Used by
68/// `issue resolve` (Resolution) and `issue comment` (Comments).
69pub fn append_to_section(body: &mut String, heading: &str, text: &str) {
70    let lines: Vec<&str> = body.lines().collect();
71    if let Some(hi) = lines.iter().position(|l| l.trim() == heading) {
72        // the section runs until the next heading or EOF
73        let end = lines
74            .iter()
75            .enumerate()
76            .skip(hi + 1)
77            .find(|(_, l)| l.trim_start().starts_with('#'))
78            .map(|(i, _)| i)
79            .unwrap_or(lines.len());
80        let mut head: Vec<&str> = lines[..end].to_vec();
81        while head.last().is_some_and(|l| l.trim().is_empty()) {
82            head.pop();
83        }
84        let mut out = head.join("\n");
85        out.push_str("\n\n");
86        out.push_str(text.trim_end());
87        let tail = &lines[end..];
88        if !tail.is_empty() {
89            out.push_str("\n\n");
90            out.push_str(&tail.join("\n"));
91        }
92        out.push('\n');
93        *body = out;
94    } else {
95        let mut out = body.trim_end().to_string();
96        if !out.is_empty() {
97            out.push_str("\n\n");
98        }
99        out.push_str(heading);
100        out.push_str("\n\n");
101        out.push_str(text.trim_end());
102        out.push('\n');
103        *body = out;
104    }
105}

107#[cfg(test)]
108#[allow(clippy::unwrap_used)]
109mod tests {
110    use super::*;
111    use myco_core::{ItemId, ItemKind};
112
113    fn item() -> Item {
114        Item::new(ItemId::new(ItemKind::Issue, 1), "the bug")
115            .created(date::parse("2026-02-09").unwrap())
116    }
117
118    #[test]
119    fn templates_have_sections() {
120        let i = item();
121        assert!(issue(&i).contains("## Resolution"));
122        assert!(issue(&i).contains("Status: **open** · Created: 2026-02-09"));
123        for (t, secs) in [
124            (
125                proposal(&i),
126                vec![
127                    "Goal & Overview",
128                    "Requirements",
129                    "Technical Design",
130                    "Open questions",
131                ],
132            ),
133            (adr(&i), vec!["Context", "Decision", "Consequences"]),
134        ] {
135            for s in secs {
136                assert!(t.contains(&format!("## {s}")), "{s}");
137            }
138        }
139        let p = plan(&i);
140        assert!(p.contains("- [ ] Phase 1"));
141    }
142
143    #[test]
144    fn append_to_existing_section() {
145        let mut body = issue(&item());
146        append_to_section(&mut body, "## Resolution", "fixed in abc123");
147        assert!(body.contains("## Resolution\n\nfixed in abc123"), "{body}");
148        // second append lands after the first
149        append_to_section(&mut body, "## Resolution", "verified");
150        assert!(body.contains("fixed in abc123\n\nverified"), "{body}");
151    }
152
153    #[test]
154    fn append_creates_missing_section() {
155        let mut body = String::from("# x\n\nbody\n");
156        append_to_section(&mut body, "## Comments", "- 2026-01-01 hi");
157        assert_eq!(body, "# x\n\nbody\n\n## Comments\n\n- 2026-01-01 hi\n");
158        // empty body
159        let mut empty = String::new();
160        append_to_section(&mut empty, "## Comments", "- d t");
161        assert_eq!(empty, "## Comments\n\n- d t\n");
162    }
163
164    #[test]
165    fn append_before_next_heading() {
166        let mut body = String::from("## A\n\none\n\n## B\n\ntwo\n");
167        append_to_section(&mut body, "## A", "added");
168        assert_eq!(body, "## A\n\none\n\nadded\n\n## B\n\ntwo\n");
169    }
170}
