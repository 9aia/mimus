1//! `myco handoff` / `myco resume` / `myco interrupt` — the
2//! continuity trio. At session end `handoff` appends a
3//! `## <today> · <actor>` entry plus note to `handoffs/<ID>.md`
4//! (append-only, one file per item, uniform across kinds). At
5//! session start `resume` emits the minimal restart context — the
6//! item, its `blocks:` dep titles, the latest handoff entries,
7//! the open checklist — sized for a prompt, the single-item
8//! counterpart of `status --prompt`. `interrupt` is the abnormal
9//! end: a work-state item (claimed/in-progress/blocked) reverts
10//! to `open` and gets an `interrupted` handoff in one commit.
11//!
12//! sepia owns session lifecycle; it calls these verbs from hooks.
13//! `--session <id>` is the hook contract: `handoff`/`interrupt`
14//! tag their entry with it, and the claiming verbs
15//! (`next --claim`, `item set`) record it as a `claimed`
16//! handoff entry — the tag lives in the entry heading
17//! (`session:<id>`), no item field. An item is "claimed under"
18//! a session when its latest session-tagged handoff entry names
19//! it ([`claim_session`]); `interrupt --session <id>` (no item
20//! id) is the teardown — every live work-state item that session
21//! last tagged reverts in one commit, so a dying session
22//! releases exactly its own work.
23

32/// Handoff entries shown by default (`--last` overrides).
33const DEFAULT_LAST: usize = 10;

35/// The note prefix `interrupt` writes — also the derivation rule:
36/// an item "was interrupted" when its latest handoff note starts
37/// with this word. No dedicated status — the event lives in the
38/// append-only log; status/board/report surfaces derive the flag.
39pub const INTERRUPT_NOTE: &str = "interrupted";

41/// The note the claim-side verbs write when `--session` tags a
42/// claim (`next --claim`, `item set` on a work-state item) — the
43/// `session:<id>` heading is the tag `interrupt --session`
44/// filters on; the note is for humans reading the log.
45pub const CLAIM_NOTE: &str = "claimed";

47/// `--session` values must fit the `session:<id>` heading token.
48pub(crate) fn validate_session(s: &str) -> Result<(), String> {
49    if !is_valid_session(s) {
50        return Err(format!(
51            "invalid --session {s:?} — sessions match `[A-Za-z0-9._:-]+`"
52        ));
53    }
54    Ok(())
55}

57/// Which session holds the item's claim — the `session:` of its
58/// latest handoff entry that carries one. An untagged entry is
59/// anonymous: it neither claims nor transfers; another session's
60/// tagged write does. `None` = no session ever declared one.
61///
62/// Known edge: an item claimed under `s`, interrupted, then
63/// re-claimed *without* `--session` still reads as `s`'s — an
64/// untagged claim leaves no tag to supersede. Session-aware
65/// callers tag every claim (`--session`) and the rule stays
66/// exact.
67pub fn claim_session(store: &impl Store, id: ItemId) -> Result<Option<String>, String> {
68    let entries = store.read_handoffs(id).map_err(|e| e.to_string())?;
69    Ok(entries.iter().rev().find_map(|e| e.session.clone()))
70}

72/// An item's writable pre-image — the line ledger plus its item
73/// file when file-backed — snapshotted so a paired write (status
74/// revert + marker entry; claim mutation + tag entry) can roll
75/// the first half back when the second fails instead of leaving
76/// it on disk for the *next* verb's autocommit to sweep up under
77/// its own message.
78struct Preimage {
79    ledger: myco_core::Ledger,
80    file: Option<myco_core::ItemFile>,
81}
82
83fn preimage(store: &impl Store, id: ItemId) -> Result<Preimage, String> {
84    let ledger = items::load_ledger(store)?;
85    let file = match store.item(id) {
86        Ok(f) => Some(f),
87        Err(myco_store::Error::NotFound { .. }) => None,
88        Err(e) => return Err(e.to_string()),
89    };
90    Ok(Preimage { ledger, file })
91}
92
93/// Best-effort restore of a [`Preimage`] — returns the legs that
94/// wouldn't write so the caller can report what stuck.
95fn restore(store: &impl Store, pre: &Preimage) -> Vec<String> {
96    let mut stuck = Vec::new();
97    if let Err(r) = items::save_ledger(store, &pre.ledger) {
98        stuck.push(format!("ledger: {r}"));
99    }
100    if let Some(f) = &pre.file {
101        if let Err(r) = store.save_item(f) {
102            stuck.push(format!("item file: {r}"));
103        }
104    }
105    stuck
106}

108/// Did the item's latest handoff mark it interrupted? A read
109/// failure propagates — a malformed handoff file is a finding,
110/// not a silent "not interrupted".
111pub fn is_interrupted(store: &impl Store, id: ItemId) -> Result<bool, String> {
112    let entries = store.read_handoffs(id).map_err(|e| e.to_string())?;
113    Ok(entries
114        .last()
115        .is_some_and(|e| e.note.trim_start().starts_with(INTERRUPT_NOTE)))
116}

118/// The interrupted subset of `items` — for status/board markers.
119/// The first unreadable handoff file fails the set.
120pub fn interrupted_set<'a>(
121    store: &impl Store,
122    items: impl Iterator<Item = &'a myco_core::Item>,
123) -> Result<BTreeSet<ItemId>, String> {
124    items
125        .map(|i| is_interrupted(store, i.id).map(|hit| (i.id, hit)))
126        .filter_map(|r| match r {
127            Ok((id, true)) => Some(Ok(id)),
128            Ok(_) => None,
129            Err(e) => Some(Err(e)),
130        })
131        .collect()
132}

134/// `--note` inline, or `-` for stdin.
135fn read_note(note: &str) -> Result<String, String> {
136    let note = if note == "-" {
137        use std::io::Read;
138        let mut s = String::new();
139        std::io::stdin()
140            .read_to_string(&mut s)
141            .map_err(|e| format!("read --note from stdin: {e}"))?;
142        s
143    } else {
144        note.to_string()
145    };
146    let note = note.trim().to_string();
147    if note.is_empty() {
148        return Err("handoff: --note is empty".into());
149    }
150    Ok(note)
151}

153/// The handoff write — the item must exist somewhere in the
154/// ledger (any kind, live or archived). Validates the note and
155/// the session id, appends the entry, and returns it plus the
156/// worktree-relative path (the file lands wherever it already
157/// lived). `actor` is the resolved actor
158/// ([`ops::resolve_actor`] — callers fold `--actor` in first).
159/// Caller commits.
160pub fn record_handoff(
161    store: &impl Store,
162    worktree: &std::path::Path,
163    id: ItemId,
164    note: String,
165    actor: &Actor,
166    session: Option<String>,
167) -> Result<(HandoffEntry, String), String> {
168    if items::find_item(store, id)?.is_none() {
169        return Err(format!("{id}: no such item — handoffs attach to items"));
170    }
171    let note = note.trim().to_string();
172    if note.is_empty() {
173        return Err("handoff: note is empty".into());
174    }
175    if let Some(s) = &session {
176        validate_session(s)?;
177    }
178    let entry = HandoffEntry {
179        date: items::today(),
180        actor: actor.clone(),
181        session,
182        note,
183    };
184    store
185        .append_handoff(id, &entry)
186        .map_err(|e| e.to_string())?;
187    // the file lands wherever it already lived
188    let live = format!("handoffs/{id}.md");
189    let rel = if worktree.join(&live).exists() {
190        live
191    } else {
192        format!("handoffs/archive/{id}.md")
193    };
194    Ok((entry, rel))
195}

197/// The `handoff` verb — `-` reads the note from stdin.
198pub fn h_handoff(ctx: &crate::registry::Ctx, a: &crate::registry::Args) -> Result<i32, String> {
199    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
200    let id = a.req::<ItemId>("id")?;
201    let explicit: Option<Actor> = a.parsed("actor")?;
202    let me = ops::resolve_actor(Some(&proj.worktree), explicit.as_ref());
203    let (entry, rel) = record_handoff(
204        &store,
205        &proj.worktree,
206        id,
207        read_note(a.text("note")?)?,
208        &me,
209        a.opt("session").map(str::to_string),
210    )?;
211    let committed = items::commit(
212        &proj.worktree,
213        &format!("chore: handoff {id}"),
214        Some(&entry.actor),
215        ctx.no_commit,
216    )?;
217    items::report(
218        json!({"id": id.to_string(), "path": rel, "entry": entry}),
219        committed,
220        ctx.json,
221        &format!("handoff {id} → {rel}"),
222    )
223}

225/// The interrupt write — a session ended with work still
226/// claimed/in-progress/blocked. Reverts the item to `open`
227/// through the shared `item set` mutation path (relocation rules
228/// included) and appends the `interrupted` marker entry. Caller
229/// commits — one commit covers both writes.
230///
231/// `session` scopes as well as tags: when passed, the item's
232/// claim must be held by that session ([`claim_session`]) —
233/// `interrupt <id> --session <s>` can't clobber a claim it
234/// doesn't own. Without `session` the interrupt is unconditional
235/// (the escape hatch).
236pub fn interrupt_item(
237    store: &impl Store,
238    id: ItemId,
239    note: Option<&str>,
240    actor: &Actor,
241    session: Option<String>,
242) -> Result<HandoffEntry, String> {
243    let Some((item, _, _)) = items::find_item(store, id)? else {
244        return Err(format!("{id}: no such item"));
245    };
246    match item.status {
247        Status::Claimed | Status::InProgress | Status::Blocked => {}
248        s => {
249            return Err(format!(
250                "{id}: status is {s} — nothing to interrupt \
251                 (only claimed/in-progress/blocked revert to open)"
252            ));
253        }
254    }
255    if let Some(s) = &session {
256        validate_session(s)?;
257        match claim_session(store, id)? {
258            Some(held) if held == *s => {}
259            held => {
260                let who = held.map_or_else(|| "no session".into(), |h| format!("session {h}"));
261                return Err(format!(
262                    "{id}: claimed under {who}, not session {s} — \
263                     drop --session to interrupt anyway"
264                ));
265            }
266        }
267    }
268    let note = match note.map(str::trim).filter(|n| !n.is_empty()) {
269        Some(extra) => format!("{INTERRUPT_NOTE} — {extra}"),
270        None => INTERRUPT_NOTE.to_string(),
271    };
272    let entry = HandoffEntry {
273        date: items::today(),
274        actor: actor.clone(),
275        session,
276        note,
277    };
278    // two writes share the caller's one commit — snapshot the
279    // pre-image so a failed handoff append can roll the status
280    // revert back
281    let pre = preimage(store, id)?;
282    // revert through the shared mutation path (relocation rules
283    // included), then the marker handoff
284    crate::itemcmd::mutate_item(
285        store,
286        id,
287        &crate::itemcmd::SetFlags::status(Status::Open),
288        actor,
289    )?;
290    if let Err(e) = store.append_handoff(id, &entry) {
291        let stuck = restore(store, &pre);
292        return Err(if stuck.is_empty() {
293            format!("{id}: handoff append failed ({e}) — status revert rolled back")
294        } else {
295            format!(
296                "{id}: handoff append failed ({e}) — status revert left on disk \
297                 (rollback: {})",
298                stuck.join("; ")
299            )
300        });
301    }
302    Ok(entry)
303}

305/// The claim-side write behind `--session` on the claiming verbs
306/// (`next --claim`, `item set`): apply `flags` through the shared
307/// mutation path and record the claim as a `claimed` handoff
308/// entry tagged with the session — the tag
309/// [`claim_session`]/`interrupt --session` read. The item must
310/// land (or already sit) in a work state — `--session` tags a
311/// claim, and an open item has none. Caller commits — one commit
312/// covers both writes; a failed tag rolls the mutation back.
313/// `actor` names the claimer (falls back to the item's
314/// post-mutation owner, then `me` — the resolved actor).
315pub(crate) fn tag_claim(
316    store: &impl Store,
317    id: ItemId,
318    flags: &crate::itemcmd::SetFlags,
319    actor: Option<Actor>,
320    session: &str,
321    me: &Actor,
322) -> Result<(Option<myco_core::Item>, String, HandoffEntry), String> {
323    let Some((item, _, _)) = items::find_item(store, id)? else {
324        return Err(format!("{id}: no such item"));
325    };
326    let post = flags.status.unwrap_or(item.status);
327    if !matches!(post, Status::Claimed | Status::InProgress | Status::Blocked) {
328        return Err(format!(
329            "{id}: --session tags a claim — the item would be {post}, which has no claim"
330        ));
331    }
332    validate_session(session)?;
333    let pre = preimage(store, id)?;
334    let (item, location) = crate::itemcmd::mutate_item(store, id, flags, me)?;
335    let entry = HandoffEntry {
336        date: items::today(),
337        actor: actor
338            .or_else(|| item.as_ref().and_then(|i| i.owner.clone()))
339            .unwrap_or_else(|| me.clone()),
340        session: Some(session.to_string()),
341        note: CLAIM_NOTE.to_string(),
342    };
343    if let Err(e) = store.append_handoff(id, &entry) {
344        let stuck = restore(store, &pre);
345        return Err(if stuck.is_empty() {
346            format!("{id}: claim tag append failed ({e}) — mutation rolled back")
347        } else {
348            format!(
349                "{id}: claim tag append failed ({e}) — mutation left on disk \
350                 (rollback: {})",
351                stuck.join("; ")
352            )
353        });
354    }
355    Ok((item, location, entry))
356}

358/// A session teardown's outcome — every item that reverted plus
359/// the per-item failures the run walked past.
360pub struct SessionInterrupts {
361    /// `(id, marker entry)` per reverted item, in ledger order.
362    pub interrupted: Vec<(ItemId, HandoffEntry)>,
363    /// `(id, error)` per failure — collected, not fatal: the
364    /// items that did revert still commit.
365    pub errors: Vec<(ItemId, String)>,
366}
367
368/// `interrupt --session <s>` — the bulk teardown. Every live
369/// work-state item whose claim [`claim_session`] reads as
370/// `session`'s reverts + marks. Caller commits once for the whole
371/// sweep — a malformed handoff or failed revert reports per item
372/// and doesn't abort the rest.
373pub fn interrupt_session(
374    store: &impl Store,
375    session: &str,
376    note: Option<&str>,
377    actor: &Actor,
378) -> Result<SessionInterrupts, String> {
379    validate_session(session)?;
380    let scan = crate::digest::scan(store)?;
381    let candidates: Vec<ItemId> = scan
382        .live()
383        .filter(|i| {
384            matches!(
385                i.status,
386                Status::Claimed | Status::InProgress | Status::Blocked
387            )
388        })
389        .map(|i| i.id)
390        .collect();
391    let mut out = SessionInterrupts {
392        interrupted: Vec::new(),
393        errors: Vec::new(),
394    };
395    for id in candidates {
396        match claim_session(store, id) {
397            Ok(Some(held)) if held == session => {
398                match interrupt_item(store, id, note, actor, Some(session.to_string())) {
399                    Ok(entry) => out.interrupted.push((id, entry)),
400                    Err(e) => out.errors.push((id, e)),
401                }
402            }
403            Ok(_) => {}
404            Err(e) => out.errors.push((id, e)),
405        }
406    }
407    Ok(out)
408}

410/// The bulk report — `{session, interrupted[], errors[],
411/// committed}` for `--json`, one line plus per-item stderr
412/// errors for humans. Returns the exit code (1 when any item
413/// failed — the reverts still committed).
414fn session_interrupt_report(
415    session: &str,
416    out: &SessionInterrupts,
417    committed: bool,
418    json_out: bool,
419) -> i32 {
420    if json_out {
421        ops::print_json(&json!({
422            "session": session,
423            "interrupted": out.interrupted.iter().map(|(id, e)| {
424                json!({"id": id.to_string(), "status": "open", "entry": e})
425            }).collect::<Vec<_>>(),
426            "errors": out.errors.iter().map(|(id, e)| {
427                json!({"id": id.to_string(), "error": e})
428            }).collect::<Vec<_>>(),
429            "committed": committed,
430        }));
431    } else {
432        let ids: Vec<String> = out
433            .interrupted
434            .iter()
435            .map(|(id, _)| id.to_string())
436            .collect();
437        if ids.is_empty() {
438            println!("nothing claimed under session {session}");
439        } else {
440            println!("interrupted {} → open (session {session})", ids.join(", "));
441        }
442        for (id, e) in &out.errors {
443            eprintln!("myco: error: {id}: {e}");
444        }
445    }
446    if out.errors.is_empty() {
447        0
448    } else {
449        1
450    }
451}

453/// The `interrupt` verb — called by whoever owns session
454/// lifecycle (sepia hooks, CLI session-end hooks); myco never
455/// detects sessions itself. With an id it reverts that one item
456/// (`--session` asserts the claim is that session's); without an
457/// id, `--session <id>` is the teardown — every item claimed
458/// under the session reverts in one commit.
459pub fn h_interrupt(ctx: &crate::registry::Ctx, a: &crate::registry::Args) -> Result<i32, String> {
460    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
461    let id: Option<ItemId> = a.parsed("id")?;
462    let session = a.opt("session").map(str::to_string);
463    let explicit: Option<Actor> = a.parsed("actor")?;
464    let me = ops::resolve_actor(Some(&proj.worktree), explicit.as_ref());
465    match (id, session) {
466        (Some(id), session) => {
467            let entry = interrupt_item(&store, id, a.opt("note"), &me, session)?;
468            let committed = items::commit(
469                &proj.worktree,
470                &format!("chore: interrupt {id}"),
471                Some(&entry.actor),
472                ctx.no_commit,
473            )?;
474            items::report(
475                json!({"id": id.to_string(), "status": "open", "entry": entry}),
476                committed,
477                ctx.json,
478                &format!("interrupted {id} → open"),
479            )
480        }
481        (None, Some(session)) => {
482            let out = interrupt_session(&store, &session, a.opt("note"), &me)?;
483            let committed = items::commit(
484                &proj.worktree,
485                &format!("chore: interrupt session {session}"),
486                Some(&me),
487                ctx.no_commit,
488            )?;
489            Ok(session_interrupt_report(
490                &session, &out, committed, ctx.json,
491            ))
492        }
493        (None, None) => Err(
494            "interrupt: pass an item id, or --session <id> to interrupt \
495             everything a session claimed"
496                .into(),
497        ),
498    }
499}

501/// Unchecked `- [ ]` checkbox items in a markdown body — the open
502/// checklist `resume` surfaces (issue bodies, plan phases).
503fn open_checklist(body: &str) -> Vec<String> {
504    body.lines()
505        .filter_map(|l| {
506            let t = l.trim_start();
507            for m in ["- [ ]", "* [ ]", "+ [ ]"] {
508                if let Some(rest) = t.strip_prefix(m) {
509                    if rest.is_empty() || rest.starts_with(char::is_whitespace) {
510                        return Some(rest.trim().to_string());
511                    }
512                }
513            }
514            None
515        })
516        .collect()
517}

519/// The gathered `resume` context — the typed form behind
520/// [`resume_json`] and the human block.
521struct ResumeData {
522    item: myco_core::Item,
523    location: String,
524    blocks_titles: Vec<serde_json::Value>,
525    links_titles: Vec<serde_json::Value>,
526    shown: Vec<HandoffEntry>,
527    total: usize,
528    checklist: Vec<String>,
529}
530
531/// Gather the resume context — the item, `blocks:`/`links:` dep
532/// titles, the latest handoff entries, the open checklist.
533fn resume_data(store: &impl Store, id: ItemId, last: Option<usize>) -> Result<ResumeData, String> {
534    let Some((item, location, body)) = items::find_item(store, id)? else {
535        return Err(format!("{id}: no such item"));
536    };
537    let dep_titles = |ids: &std::collections::BTreeSet<myco_core::ItemId>| {
538        ids.iter()
539            .map(|b| {
540                let title = items::find_item(store, *b)?.map(|(i, _, _)| i.title);
541                Ok(json!({"id": b.to_string(), "title": title}))
542            })
543            .collect::<Result<Vec<_>, String>>()
544    };
545    let blocks_titles = dep_titles(&item.blocks)?;
546    let links_titles = dep_titles(&item.links)?;
547    let entries = store.read_handoffs(id).map_err(|e| e.to_string())?;
548    let total = entries.len();
549    let n = last.unwrap_or(DEFAULT_LAST);
550    let shown: Vec<HandoffEntry> = entries
551        .iter()
552        .rev()
553        .take(n)
554        .cloned()
555        .collect::<Vec<_>>()
556        .into_iter()
557        .rev()
558        .collect();
559    let checklist = body.as_deref().map(open_checklist).unwrap_or_default();
560    Ok(ResumeData {
561        item,
562        location,
563        blocks_titles,
564        links_titles,
565        shown,
566        total,
567        checklist,
568    })
569}

596/// One item's prompt-ready block — the compact item line first,
597/// sections only when they carry context. Identical for single-
598/// and multi-id output (blocks join on a blank line).
599fn resume_block(d: &ResumeData) -> String {
600    let mut out = format!("{}\nin: {}", items::compact(&d.item), d.location);
601    if !d.blocks_titles.is_empty() {
602        out.push_str("\nblocks:");
603        for b in &d.blocks_titles {
604            out.push_str(&format!(
605                "\n  {} {}",
606                b["id"].as_str().unwrap_or_default(),
607                b["title"].as_str().unwrap_or("(missing)")
608            ));
609        }
610    }
611    if !d.links_titles.is_empty() {
612        out.push_str("\nlinks:");
613        for l in &d.links_titles {
614            out.push_str(&format!(
615                "\n  {} {}",
616                l["id"].as_str().unwrap_or_default(),
617                l["title"].as_str().unwrap_or("(missing)")
618            ));
619        }
620    }
621    if !d.shown.is_empty() {
622        out.push_str("\nhandoffs:");
623        for e in &d.shown {
624            let mut head = format!("{} · {}", e.date, e.actor);
625            if let Some(s) = &e.session {
626                head.push_str(&format!(" · session:{s}"));
627            }
628            out.push_str(&format!("\n{head}"));
629            for line in e.note.lines() {
630                out.push_str(&format!("\n  {line}"));
631            }
632            out.push('\n');
633        }
634        if d.total > d.shown.len() {
635            out.push_str(&format!("\n… +{} earlier entries", d.total - d.shown.len()));
636        }
637    }
638    if !d.checklist.is_empty() {
639        out.push_str("\nopen checklist:");
640        for c in &d.checklist {
641            out.push_str(&format!("\n  - [ ] {c}"));
642        }
643    }
644    out
645}

647/// The `resume` verb — 1..n ids, one discover + store open.
648/// `--json` is the single-id payload for one id, else the
649/// `[{id, ok, payload|error}]` batch array. Human output is the
650/// prompt block per id in argument order; a bad id reports to
651/// stderr and exits 1 without aborting the rest.
652/// The `resume` verb — 1..n ids, one discover + store open.
653pub fn h_resume(ctx: &crate::registry::Ctx, a: &crate::registry::Args) -> Result<i32, String> {
654    let ids = a.many_parsed::<ItemId>("id")?;
655    let last: Option<usize> = a.parsed("last")?;
656    let json_out = ctx.json;
657    let (_proj, store) = items::open_read(ctx.dir.as_deref(), ctx.root.as_deref())?;
658    if ids.len() == 1 {
659        let d = resume_data(&store, ids[0], last)?;
660        if json_out {
661            ops::print_json(&resume_value(&d));
662        } else {
663            println!("{}", resume_block(&d));
664        }
665        return Ok(0);
666    }
667    if json_out {
668        let (entries, failed) = items::multi_json(&ids, |id| resume_json(&store, id, last));
669        ops::print_json(&entries);
670        return Ok(if failed { 1 } else { 0 });
671    }
672    let mut blocks = Vec::new();
673    let mut failed = false;
674    for id in ids {
675        match resume_data(&store, id, last) {
676            Ok(d) => blocks.push(resume_block(&d)),
677            Err(e) => {
678                failed = true;
679                eprintln!("myco: error: {e}");
680            }
681        }
682    }
683    if !blocks.is_empty() {
684        println!("{}", blocks.join("\n\n"));
685    }
686    Ok(if failed { 1 } else { 0 })
687}
