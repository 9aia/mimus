1//! `myco org` — cross-repo *scaffolding*; cross-repo *views* are
2//! index-sourced `myco view` specs (`view org-board` /
3//! `view org-status`, `--source index`).
4//!
5//! [`load_fresh`] is the shared index-universe read: every
6//! registered repo's snapshot is checked for freshness (stored
7//! `head_commit` vs the worktree's current HEAD); a stale or
8//! missing snapshot is rebuilt on the spot — index reads never
9//! serve old data silently. Repos whose worktree vanished keep
10//! their last snapshot and report an error inline.
11
12use camino::Utf8PathBuf;
13use myco_index::{Index, RepoSnapshot};

19/// One repo's snapshot load outcome for index-sourced views.
20pub struct RepoView {
21    /// Registry name — identity even when `snap` is gone.
22    pub name: String,
23    /// Canonical repo-root path — the registry key.
24    pub path: Utf8PathBuf,
25    /// The registered `project/` worktree — where a mutation on
26    /// this repo's items opens a store (index-sourced boards
27    /// write per repo).
28    pub worktree: Utf8PathBuf,
29    /// The snapshot (may be the last good one on rescan failure).
30    pub snap: Option<RepoSnapshot>,
31    /// Rescan/load error, when the fresh snapshot couldn't be had.
32    pub error: Option<String>,
33}

35/// The index universe as registry entries — lasius clones ∪
36/// `repos.toml` registrations (P-026), anchored at the process
37/// cwd. `ledger = "none"` entries are ledger-less by design:
38/// they are omitted outright — no snapshot, no error, no
39/// rescan, no row — so org views skip them instead of warning.
40fn universe_entries(index: &impl Index) -> Result<Vec<myco_index::RepoEntry>, String> {
41    let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
42    Ok(myco_ops::universe::repos(&cwd, index, crate::ops::CONFIG)?
43        .into_iter()
44        .map(|u| u.entry)
45        .collect())
46}

48/// Load every universe repo's snapshot, auto-rebuilding stale
49/// ones (head mismatch or missing). The per-repo `git`/fs work is
50/// parallel — redb opens lock the file, so index I/O stays serial.
51pub fn load_fresh(index: &impl Index) -> Result<Vec<RepoView>, String> {
52    let repos = universe_entries(index)?
53        .into_iter()
54        .filter(|r| r.has_ledger())
55        .collect::<Vec<_>>();
56    let heads = indexcmd::par_map(&repos, indexcmd::current_head);
57    let mut cached = Vec::with_capacity(repos.len());
58    for r in &repos {
59        cached.push(index.snapshot(&r.path).map_err(|e| e.to_string())?);
60    }
61    let fresh =
62        |i: usize| matches!((&cached[i], &heads[i]), (Some(s), Some(h)) if s.head_commit == *h);
63    let stale: Vec<usize> = (0..repos.len()).filter(|i| !fresh(*i)).collect();
64    let scans = indexcmd::par_map(&stale, |&i| indexcmd::scan_one(&repos[i]));
65    let mut rebuilt: std::collections::BTreeMap<usize, Result<RepoSnapshot, String>> =
66        stale.iter().copied().zip(scans).collect();
67    let mut out = Vec::with_capacity(repos.len());
68    for (i, c) in cached.iter().enumerate() {
69        let name = repos[i].name.clone();
70        let path = repos[i].path.clone();
71        let worktree = repos[i].worktree_path().unwrap_or_default();
72        if fresh(i) {
73            out.push(RepoView {
74                name,
75                path,
76                worktree,
77                snap: c.clone(),
78                error: None,
79            });
80            continue;
81        }
82        match rebuilt
83            .remove(&i)
84            .unwrap_or_else(|| Err("rescan lost".into()))
85        {
86            Ok(snap) => {
87                index.save_snapshot(&snap).map_err(|e| e.to_string())?;
88                out.push(RepoView {
89                    name,
90                    path,
91                    worktree,
92                    snap: Some(snap),
93                    error: None,
94                });
95            }
96            Err(e) => out.push(RepoView {
97                name,
98                path,
99                worktree,
100                snap: cached[i].clone(),
101                error: Some(e),
102            }),
103        }
104    }
105    Ok(out)
106}

108/// `init` over every universe repo — per-repo outcomes, errors
109/// collected (one bad repo doesn't stop the rollout).
110/// `ledger = "none"` entries are skipped outright: the rollout
111/// refreshes repos that *have* (or should have, per the legacy
112/// `local` reading) ledgers — it must never scaffold a
113/// `project/` branch into an intentionally ledger-less org
114/// member. Mark such repos `none` — `[index] ledger = "none"`
115/// in the repo's `myco.toml`, or `ledger = "none"` in
116/// `repos.toml` / a ledger-less `index add` (all honored).
117fn init_all(index: &impl Index, json_out: bool, no_commit: bool) -> Result<i32, String> {
118    let repos = universe_entries(index)?;
119    let mut out = Vec::new();
120    let mut failures = 0usize;
121    for r in &repos {
122        if !r.has_ledger() {
123            out.push(json!({"path": r.path, "status": "no-ledger"}));
124            continue;
125        }
126        let p = r.path.as_std_path().to_path_buf();
127        match crate::init::ensure(None, Some(&p), no_commit) {
128            Ok(o) => out.push(json!({
129                "path": r.path,
130                "files_created": o.files_created,
131                "gitignore_updated": o.gitignore_updated,
132                "agents_md_updated": o.agents_md_updated,
133                "committed": o.committed,
134            })),
135            Err(e) => {
136                failures += 1;
137                out.push(json!({"path": r.path, "error": e}));
138            }
139        }
140    }
141    if json_out {
142        ops::print_json(&json!({"repos": out, "failures": failures}));
143        return Ok(if failures == 0 { 0 } else { 1 });
144    }
145    for (r, o) in repos.iter().zip(&out) {
146        if o.get("status").and_then(|v| v.as_str()) == Some("no-ledger") {
147            println!("{:<24} no ledger — skipped", r.path.as_str());
148        } else if let Some(e) = o.get("error").and_then(|v| v.as_str()) {
149            println!("{:<24} ⚠ {e}", r.path.as_str());
150        } else {
151            let mut notes = Vec::new();
152            if o["gitignore_updated"].as_bool().unwrap_or(false) {
153                notes.push(".gitignore");
154            }
155            if o["agents_md_updated"].as_bool().unwrap_or(false) {
156                notes.push("AGENTS.md");
157            }
158            let n = o["files_created"].as_array().map_or(0, Vec::len);
159            if n > 0 {
160                notes.push("scaffold");
161            }
162            println!(
163                "{:<24} {}",
164                r.path.as_str(),
165                if notes.is_empty() {
166                    "already current".to_string()
167                } else {
168                    format!("updated {}", notes.join(" + "))
169                }
170            );
171        }
172    }
173    Ok(if failures == 0 { 0 } else { 1 })
174}

176/// The `org init` verb — rolls scaffold out (no snapshot reads).
177pub fn h_init(ctx: &Ctx, _a: &Args) -> Result<i32, String> {
178    let index = indexcmd::open_index()?;
179    init_all(&index, ctx.json, ctx.no_commit)
180}
