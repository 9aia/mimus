13/// The set-verb flag bundle — applied to an `Item` after the
14/// status transition check.
15#[derive(Default, Clone)]
16pub(crate) struct SetFlags {
17    pub(crate) status: Option<Status>,
18    /// `--steal` — take an item over from a different owner
19    /// (claim transitions and owner reassignment on work-state
20    /// items are guarded; see [`ownership_guard`]).
21    pub(crate) steal: bool,
22    pub(crate) pri: Option<Priority>,
23    pub(crate) owner: Option<Actor>,
24    pub(crate) label: Vec<String>,
25    pub(crate) unlabel: Vec<String>,
26    pub(crate) blocks: Vec<String>,
27    pub(crate) unblock: Vec<ItemId>,
28    pub(crate) links: Vec<String>,
29    pub(crate) unlink: Vec<ItemId>,
30    pub(crate) epic: Option<String>,
31    pub(crate) milestone: Option<String>,
32    /// Raw `--due` value — kept stringly so `""` clears and the
33    /// date validates at apply time (like the group names).
34    pub(crate) due: Option<String>,
35    /// Raw `--est` value — `""` clears, else a points count.
36    pub(crate) est: Option<String>,
37    pub(crate) cycle: Option<String>,
38}

40/// The decoded `item set` flags — shared by the CLI handler and
41/// the wire handler (`myco serve`'s `item.set`).
42pub(crate) fn set_flags(a: &Args) -> Result<SetFlags, String> {
43    Ok(SetFlags {
44        status: a.parsed("status")?,
45        steal: a.flag("steal"),
46        pri: a.parsed("pri")?,
47        owner: a.parsed("owner")?,
48        label: a.many("label"),
49        unlabel: a.many("unlabel"),
50        blocks: a.many("blocks"),
51        unblock: a.many_parsed("unblock")?,
52        links: a.many("links"),
53        unlink: a.many_parsed("unlink")?,
54        epic: a.opt("epic").map(str::to_string),
55        milestone: a.opt("milestone").map(str::to_string),
56        due: a.opt("due").map(str::to_string),
57        est: a.opt("est").map(str::to_string),
58        cycle: a.opt("cycle").map(str::to_string),
59    })
60}

62impl SetFlags {
63    /// A status-only mutation — `myco interrupt` drives a legal
64    /// step through the same relocation path as `item set`.
65    pub(crate) fn status(to: Status) -> Self {
66        SetFlags {
67            status: Some(to),
68            ..Default::default()
69        }
70    }
71
72    pub(crate) fn is_noop(&self) -> bool {
73        self.status.is_none()
74            && self.pri.is_none()
75            && self.owner.is_none()
76            && self.label.is_empty()
77            && self.unlabel.is_empty()
78            && self.blocks.is_empty()
79            && self.unblock.is_empty()
80            && self.links.is_empty()
81            && self.unlink.is_empty()
82            && self.epic.is_none()
83            && self.milestone.is_none()
84            && self.due.is_none()
85            && self.est.is_none()
86            && self.cycle.is_none()
87    }
88}

90/// `+x` → add `x`, `-x` → remove `x`, `x` → add — the `item set`
91/// add/remove convention.
92fn signed(s: &str) -> (bool, &str) {
93    match s.strip_prefix('-') {
94        Some(rest) => (false, rest),
95        None => (true, s.strip_prefix('+').unwrap_or(s)),
96    }
97}

99/// The ownership guard — assigning an item whose owner is someone
100/// else is refused unless `--steal` is passed. It fires on claim
101/// transitions (`→ claimed`) and on owner reassignment while the
102/// item sits in a work state (`claimed`/`in-progress`); open-item
103/// triage stays free. `me` is the resolved actor
104/// ([`ops::resolve_actor`] — `--owner` already folded in).
105fn ownership_guard(item: &Item, flags: &SetFlags, me: &Actor) -> Result<(), String> {
106    if flags.steal || flags.owner.is_none() {
107        return Ok(());
108    }
109    let Some(existing) = &item.owner else {
110        return Ok(());
111    };
112    if *existing == *me {
113        return Ok(());
114    }
115    let claim_transition = flags.status == Some(Status::Claimed);
116    let work_state = matches!(item.status, Status::Claimed | Status::InProgress);
117    if claim_transition || work_state {
118        return Err(format!(
119            "{}: owned by {existing} — `--steal` takes it over",
120            item.id
121        ));
122    }
123    Ok(())
124}

126/// Apply every set flag to `item` — status first (machine-checked),
127/// then the plain metadata.
128fn apply_set(item: &mut Item, flags: &SetFlags) -> Result<(), String> {
129    if let Some(to) = flags.status {
130        let from = item.status;
131        item.status = from.transition_to(to).map_err(|_| {
132            format!(
133                "{id}: invalid status transition {from} → {to} (valid from {from}: {t})",
134                id = item.id,
135                t = items::targets(from)
136            )
137        })?;
138    }
139    if let Some(p) = flags.pri {
140        item.priority = p;
141    }
142    if let Some(o) = &flags.owner {
143        item.owner = Some(o.clone());
144    }
145    for raw in &flags.label {
146        let (add, name) = signed(raw);
147        if !is_valid_label(name) {
148            return Err(format!("invalid label {name:?}"));
149        }
150        if add {
151            item.labels.insert(name.to_string());
152        } else {
153            item.labels.remove(name);
154        }
155    }
156    for name in &flags.unlabel {
157        item.labels.remove(name.as_str());
158    }
159    for raw in &flags.blocks {
160        let (add, s) = signed(raw);
161        let id: ItemId = s
162            .parse()
163            .map_err(|e: myco_core::ParseItemIdError| e.to_string())?;
164        if add {
165            item.blocks.insert(id);
166        } else {
167            item.blocks.remove(&id);
168        }
169    }
170    for id in &flags.unblock {
171        item.blocks.remove(id);
172    }
173    for raw in &flags.links {
174        let (add, s) = signed(raw);
175        let id: ItemId = s
176            .parse()
177            .map_err(|e: myco_core::ParseItemIdError| e.to_string())?;
178        if add {
179            item.links.insert(id);
180        } else {
181            item.links.remove(&id);
182        }
183    }
184    for id in &flags.unlink {
185        item.links.remove(id);
186    }
187    for (what, flag, slot) in [
188        ("epic", &flags.epic, &mut item.epic),
189        ("milestone", &flags.milestone, &mut item.milestone),
190        ("cycle", &flags.cycle, &mut item.cycle),
191    ] {
192        if let Some(name) = flag {
193            if name.is_empty() {
194                *slot = None; // `--epic ""` clears
195            } else if !myco_core::is_valid_group_name(name) {
196                return Err(format!("invalid {what} name {name:?}"));
197            } else {
198                *slot = Some(name.clone());
199            }
200        }
201    }
202    if let Some(d) = &flags.due {
203        if d.is_empty() {
204            item.due = None; // `--due ""` clears
205        } else {
206            item.due = Some(
207                myco_core::date::parse(d).map_err(|e| format!("invalid due date {d:?} — {e}"))?,
208            );
209        }
210    }
211    if let Some(e) = &flags.est {
212        if e.is_empty() {
213            item.estimate = None; // `--est ""` clears
214        } else {
215            let n: u32 = e
216                .parse()
217                .map_err(|_| format!("invalid est value {e:?} — expected a point count"))?;
218            if n > myco_core::MAX_ESTIMATE {
219                return Err(format!(
220                    "estimate {n} exceeds max {}",
221                    myco_core::MAX_ESTIMATE
222                ));
223            }
224            item.estimate = Some(n);
225        }
226    }
227    Ok(())
228}

236/// The link graph behind the related section — derived live, not
237/// read from the persisted index. One gather per `item show` call,
238/// shared across a multi-id batch (see [`myco_ops::links::live_graph`]
239/// for why the index isn't consulted here). The gather is
240/// per-source lenient — a corrupt file costs its own edges — and
241/// its warnings go to stderr: they're drift `check` reports, not
242/// show's output.
243fn graph(store: &impl Store) -> myco_ops::links::LiveGraph {
244    let (graph, warnings) = myco_ops::links::live_graph(store);
245    for w in warnings {
246        eprintln!("myco: warning: {w}");
247    }
248    graph
249}

251/// What [`set_item`] produced — the post-mutation item, the
252/// location label, the session claim entry when one was written,
253/// and the previous owner when `--steal` took the item over.
254pub(crate) struct SetOutcome {
255    /// The item after mutation (`None` when `done` removed a task
256    /// line).
257    pub item: Option<Item>,
258    /// Ledger location label for reporting.
259    pub location: String,
260    /// The `claimed` handoff entry `--session` wrote, if any.
261    pub claim: Option<myco_core::HandoffEntry>,
262    /// The owner `--steal` took the item from, if a takeover
263    /// happened — reported so callers can surface it.
264    pub taken_from: Option<Actor>,
265}
266
267/// `item set` + an optional `--session` claim tag — the shared
268/// mutation path when absent, [`crate::continuity::tag_claim`]
269/// (mutation + `claimed` entry, atomically) when present.
270/// `me` is the resolved actor ([`ops::resolve_actor`] with the
271/// verb's explicit actor already folded in). A `→ claimed`
272/// transition with no `--owner` assigns `me`; a `--steal`
273/// takeover appends a `taken over from <actor>` handoff entry —
274/// both ride the caller's one commit.
275pub(crate) fn set_item(
276    store: &impl Store,
277    id: ItemId,
278    flags: &SetFlags,
279    session: Option<String>,
280    me: &Actor,
281) -> Result<SetOutcome, String> {
282    let mut flags = flags.clone();
283    // →claimed assigns the resolved actor when no owner named —
284    // the claim always names its claimer
285    if flags.status == Some(Status::Claimed) && flags.owner.is_none() {
286        flags.owner = Some(me.clone());
287    }
288    // snapshot for takeover detection — decidable up front: the
289    // post-mutation owner is `flags.owner` whenever one is set
290    let prior_owner = items::find_item(store, id)?.and_then(|(i, _, _)| i.owner);
291    let taken_from = if flags.steal && flags.owner.is_some() {
292        prior_owner.filter(|p| flags.owner.as_ref() != Some(p))
293    } else {
294        None
295    };
296    let outcome = match session {
297        Some(s) => {
298            let actor = flags.owner.clone();
299            let (item, location, entry) =
300                crate::continuity::tag_claim(store, id, &flags, actor, &s, me)?;
301            SetOutcome {
302                item,
303                location,
304                claim: Some(entry),
305                taken_from: None, // filled below
306            }
307        }
308        None => {
309            let (item, location) = mutate_item(store, id, &flags, me)?;
310            SetOutcome {
311                item,
312                location,
313                claim: None,
314                taken_from: None, // filled below
315            }
316        }
317    };
318    // the takeover audit entry — same commit as the mutation
319    // (the caller's autocommit sweeps both). Skipped when the
320    // mutation removed the item entirely (`done` on a task) —
321    // there's nothing live left to hang the note on.
322    if let (Some(from), Some(_)) = (&taken_from, &outcome.item) {
323        let entry = myco_core::HandoffEntry {
324            date: items::today(),
325            actor: me.clone(),
326            session: None,
327            note: format!("taken over from {from}"),
328        };
329        store
330            .append_handoff(id, &entry)
331            .map_err(|e| e.to_string())?;
332    }
333    Ok(SetOutcome {
334        taken_from,
335        ..outcome
336    })
337}

339/// The `item set` verb.
340pub fn h_set(ctx: &Ctx, a: &Args) -> Result<i32, String> {
341    let id = a.req::<ItemId>("id")?;
342    let flags = set_flags(a)?;
343    let session = a.opt("session").map(str::to_string);
344    if flags.is_noop() && session.is_none() {
345        return Err("item set: nothing to change — pass at least one flag".into());
346    }
347    // the terminals are one-way — the `item set` worth confirming
348    if let Some(to @ (Status::Done | Status::Cancelled)) = flags.status {
349        if crate::confirm::needed(a.flag("yes"))
350            && !crate::confirm::ask(&format!("set {id} → {to}? {to} is terminal."))
351        {
352            eprintln!("myco: cancelled");
353            return Ok(1);
354        }
355    }
356    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
357    // `--owner` doubles as the actor here (the resolution chain's
358    // first rung) — the guard and the commit trailer agree
359    let me = ops::resolve_actor(Some(&proj.worktree), flags.owner.as_ref());
360    let outcome = set_item(&store, id, &flags, session, &me)?;
361    let committed = items::commit(
362        &proj.worktree,
363        &format!("chore: set {id}"),
364        Some(&me),
365        ctx.no_commit,
366    )?;
367    let human = outcome
368        .item
369        .as_ref()
370        .map(|i| format!("set {}", items::compact(i)))
371        .unwrap_or_else(|| {
372            format!(
373                "set {id} {} — line removed",
374                flags
375                    .status
376                    .map_or("done".to_string(), |s| s.token().to_string())
377            )
378        });
379    let mut payload = json!({"item": outcome.item, "location": outcome.location});
380    if let Some(entry) = outcome.claim {
381        payload["claim"] = json!(entry);
382    }
383    if let Some(from) = outcome.taken_from {
384        payload["taken_from"] = json!(from);
385    }
386    items::report(payload, committed, ctx.json, &human)
387}

389/// The gathered `item show` pieces — one id's load shared by

396    interrupted: bool,

403/// Load one id anywhere in the ledger — TODO/BACKLOG line or
404/// item file, live or archived — plus its related edges out of
405/// `graph`.
406fn show_data(
407    store: &impl Store,
408    id: ItemId,
409    graph: &myco_ops::links::LiveGraph,
410) -> Result<ShowData, String> {
411    match items::find_item(store, id)? {
412        Some((item, location, body)) => {
413            let handoffs = store.read_handoffs(id).map_err(|e| e.to_string())?.len();
414            let interrupted = crate::continuity::is_interrupted(store, id)?;
415            Ok(ShowData {
416                item,
417                location,
418                handoffs,
419                interrupted,
420                body,
421                related: graph.related(id),
422            })
423        }
424        None => Err(format!("{id}: no such item")),
425    }
426}

428/// The related section's human lines — the TUI's preview shape:
429/// one `kind:` line per edge kind in declaration order, neighbors
430/// joined by `·`, `ID title` when the ledger knows the other end
431/// (a `handoff` self-edge prints the file path instead of the
432/// item's own id). Empty graph → empty string.
433fn related_lines(related: &[myco_ops::links::RelatedItem]) -> String {
434    let mut out = String::from("\n  related:");
435    for kind in EdgeKind::ALL {
436        let dsts: Vec<String> = related
437            .iter()
438            .filter(|r| r.kind == kind)
439            .map(|r| {
440                if kind == EdgeKind::Handoff {
441                    return format!("handoffs/{}.md", r.other);
442                }
443                match &r.title {
444                    Some(t) if !t.is_empty() => format!("{} {}", r.other, t),
445                    _ => r.other.to_string(),
446                }
447            })
448            .collect();
449        if !dsts.is_empty() {
450            out.push_str(&format!("\n    {:<13} {}", kind.name(), dsts.join(" · ")));
451        }
452    }
453    out
454}

456/// The related section's JSON shape — `{kind: [{id, title?}]}`,
457/// kinds in declaration order, empty groups omitted, `title`
458/// only when the ledger resolves the neighbor.
459fn related_json(related: &[myco_ops::links::RelatedItem]) -> serde_json::Value {
460    let mut groups = serde_json::Map::new();
461    for kind in EdgeKind::ALL {
462        let items: Vec<serde_json::Value> = related
463            .iter()
464            .filter(|r| r.kind == kind)
465            .map(|r| {
466                let mut v = json!({"id": r.other.to_string()});
467                if let Some(t) = &r.title {
468                    v["title"] = json!(t);
469                }
470                v
471            })
472            .collect();
473        if !items.is_empty() {
474            groups.insert(kind.name().to_string(), json!(items));
475        }
476    }
477    serde_json::Value::Object(groups)
478}

480/// One item's human block — identical for single- and multi-id
481/// output (blocks join on a blank line).
482fn show_block(d: &ShowData) -> String {
483    let mut out = format!(
484        "{}\n  {}\n  handoffs: {}",
485        items::compact(&d.item),
486        d.location,
487        d.handoffs
488    );
489    if d.interrupted {
490        out.push_str(&format!(
491            "\n  ⚡ interrupted — latest handoff marks it; `myco resume {}`",
492            d.item.id
493        ));
494    }
495    if !d.related.is_empty() {
496        out.push_str(&related_lines(&d.related));
497    }
498    if let Some(body) = &d.body {
499        if !body.is_empty() {
500            out.push_str(&format!("\n\n{body}"));
501        }
502    }
503    out
504}

520/// The `item show` body — 1..n ids in one open. Single id keeps
521/// the classic block + bare payload; multi-id repeats the block
522/// in argument order (a bad id reports to stderr, exit 1) and
523/// `--json` becomes the `[{id, ok, payload|error}]` array.
524fn show(store: &impl Store, ids: &[ItemId], json_out: bool) -> Result<i32, String> {
525    let graph = graph(store);
526    if ids.len() == 1 {
527        let d = show_data(store, ids[0], &graph)?;
528        if json_out {
529            crate::ops::print_json(&show_value(&d));
530        } else {
531            println!("{}", show_block(&d));
532        }
533        return Ok(0);
534    }
535    if json_out {
536        let (entries, failed) = items::multi_json(ids, |id| {
537            show_data(store, id, &graph).map(|d| show_value(&d))
538        });
539        crate::ops::print_json(&entries);
540        return Ok(if failed { 1 } else { 0 });
541    }
542    let mut blocks = Vec::new();
543    let mut failed = false;
544    for id in ids {
545        match show_data(store, *id, &graph) {
546            Ok(d) => blocks.push(show_block(&d)),
547            Err(e) => {
548                failed = true;
549                eprintln!("myco: error: {e}");
550            }
551        }
552    }
553    if !blocks.is_empty() {
554        println!("{}", blocks.join("\n\n"));
555    }
556    Ok(if failed { 1 } else { 0 })
557}

559/// Re-land a `take`n top-level task in its origin section: the
560/// same-named section when it still exists (empty sections are
561/// dropped by `take`, so a `None` name re-matches the surviving
562/// unsectioned run at index 0), else re-created at its old index
563/// so the item can't drift into `sections[0]` under a heading it
564/// never lived under.
565fn put_back(
566    sections: &mut Vec<myco_core::Section>,
567    section: usize,
568    name: Option<String>,
569    item: Item,
570) {
571    match sections.iter_mut().find(|s| s.name == name) {
572        Some(s) => s.items.push(item),
573        None => sections.insert(
574            section.min(sections.len()),
575            myco_core::Section {
576                name,
577                items: vec![item],
578            },
579        ),
580    }
581}

583/// Apply `flags` to one item anywhere in the ledger and save —
584/// the shared mutation behind `item set` and `interrupt`. Writes
585/// but does NOT commit; the caller owns the commit message.
586/// `me` is the resolved actor — the ownership guard compares the
587/// item's owner against it. Returns the post-mutation item
588/// (`None` when done removed a task line) and a location label
589/// for reporting.
590pub(crate) fn mutate_item(
591    store: &impl Store,
592    id: ItemId,
593    flags: &SetFlags,
594    me: &Actor,
595) -> Result<(Option<Item>, String), String> {
596    let mut ledger = items::load_ledger(store)?;
597    if let Some(loc) = ledger.locate(id) {
598        // task item: mutate in place; a status change may
599        // relocate the line — done deletes the subtree,
600        // deferred detaches the node into BACKLOG.md
601        // (children park with it), an active status moves
602        // backlog nodes into TODO.md (children unpark).
603        // Nested children that keep their doc stay nested.
604        let status_change = flags.status.is_some();
605        let mut done_doc = None;
606        let Some(target) = ledger.item_mut(&loc) else {
607            return Err(format!("{id}: no such task"));
608        };
609        ownership_guard(target, flags, me)?;
610        apply_set(target, flags)?;
611        if status_change {
612            let Some(current) = ledger.item(&loc) else {
613                return Err(format!("{id}: no such task"));
614            };
615            match current.status {
616                Status::Done | Status::Cancelled => {
617                    let terminal = current.status;
618                    let mut item = ledger
619                        .take(&loc)
620                        .ok_or_else(|| format!("{id}: no such task"))?;
621                    // DONE.md holds the terminal pair — archive
622                    // the whole subtree under the status it
623                    // reached (done drops its token, cancelled
624                    // keeps `status:cancelled`)
625                    items::walk_descendants_to(&mut item, terminal)?;
626                    // stage before any write — a dup/parse
627                    // failure leaves the ledger untouched
628                    let mut doc = items::load_done(store)?;
629                    items::done_push(&mut doc, item)?;
630                    done_doc = Some(doc);
631                }
632                Status::Deferred => {
633                    let mut item = ledger
634                        .take(&loc)
635                        .ok_or_else(|| format!("{id}: no such task"))?;
636                    items::walk_descendants_to(&mut item, Status::Deferred)?;
637                    ledger.put_backlog(item, None);
638                }
639                _ if loc.backlog => {
640                    let mut item = ledger
641                        .take(&loc)
642                        .ok_or_else(|| format!("{id}: no such task"))?;
643                    items::walk_descendants_to(&mut item, Status::Open)?;
644                    ledger.put_todo(item, None);
645                }
646                _ if !loc.is_child() => {
647                    // top-level items re-land at the tail of
648                    // their *origin* section — carry its index +
649                    // name through the take (an emptied section
650                    // is dropped, so re-find/re-create by name)
651                    let section = loc.section;
652                    let name = ledger
653                        .todo
654                        .sections
655                        .get(section)
656                        .and_then(|s| s.name.clone());
657                    let Some(item) = ledger.take(&loc) else {
658                        return Err(format!("{id}: no such task"));
659                    };
660                    put_back(&mut ledger.todo.sections, section, name, item);
661                }
662                _ => {} // a TODO child stays nested
663            }
664        }
665        let item = ledger.locate(id).and_then(|l| ledger.item(&l).cloned());
666        items::save_ledger(store, &ledger)?;
667        if let Some(doc) = done_doc {
668            items::save_done(store, &doc)?;
669        }
670        Ok((item, "tasks".to_string()))
671    } else {
672        let mut file = items::item_for_edit(store, id)?;
673        ownership_guard(&file.meta, flags, me)?;
674        apply_set(&mut file.meta, flags)?;
675        let item = file.meta.clone();
676        store.save_item(&file).map_err(|e| e.to_string())?;
677        let location = items::item_rel_path(store, &item)?;
678        Ok((Some(item), location))
679    }
680}
