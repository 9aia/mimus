12/// The `adr new` verb.
13pub fn h_new(ctx: &Ctx, a: &Args) -> Result<i32, String> {
14    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
15    items::new_file(
16        &proj,
17        &store,
18        &items::NewItem {
19            kind: ItemKind::Adr,
20            title: a.text("title")?.to_string(),
21            body: items::body_args(a)?.as_deref(),
22            meta: &items::meta_args(a)?,
23            template: templates::adr,
24            word: "adr",
25            json_out: ctx.json,
26            no_commit: ctx.no_commit,
27        },
28    )
29}

31/// The `adr list` verb.
32pub fn h_list(ctx: &Ctx, a: &Args) -> Result<i32, String> {
33    let (_proj, store) = items::open_read(ctx.dir.as_deref(), ctx.root.as_deref())?;
34    items::list_files(
35        &store,
36        ItemKind::Adr,
37        false,
38        &items::list_args(a)?,
39        ctx.json,
40    )
41}

43/// The `adr supersede` verb.
44pub fn h_supersede(ctx: &Ctx, a: &Args) -> Result<i32, String> {
45    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
46    let id = a.req::<ItemId>("id")?;
47    let by = a.req::<ItemId>("by")?;
48    items::require_kind(id, ItemKind::Adr, "adr supersede")?;
49    items::require_kind(by, ItemKind::Adr, "adr supersede --by")?;
50    if id == by {
51        return Err(format!("{id} can't supersede itself"));
52    }
53    let mut old = items::item_for_edit(&store, id)?;
54    let mut new = items::item_for_edit(&store, by)?;
55    items::walk_to(&mut old.meta, Status::Done)?;
56    old.meta.superseded_by = Some(by);
57    new.meta.supersedes = Some(id);
58    store.save_item(&old).map_err(|e| e.to_string())?;
59    store.save_item(&new).map_err(|e| e.to_string())?;
60    let committed = items::commit(
61        &proj.worktree,
62        &format!("chore: supersede {id} by {by}"),
63        None,
64        ctx.no_commit,
65    )?;
66    items::report(
67        json!({"superseded": old.meta, "by": new.meta}),
68        committed,
69        ctx.json,
70        &format!("{id} superseded by {by}"),
71    )
72}
