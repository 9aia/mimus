1//! `myco next` — the single highest-priority open item:
2//! "what should I work on" (I-020, falls out of P-006). It's a
3//! view eval with `limit = 1` — the filter flags are `view`'s
4//! [`FilterFlags`] merged over an open tasks+issues default.
5//! `--claim` is the only new surface: it mutates the emitted
6//! item to `claimed` + an owner in one per-verb commit.
7

9use myco_store::Store;

15/// The `next` flag bundle — `view`'s filter flags plus `--sort`
16/// (the engine's `pri,id` when unset) and `--claim`. Decoded out
17/// of the shared `Args` bag.
18#[derive(Debug, Default)]
19pub(crate) struct NextFlags {
20    pub(crate) filter: view::FilterPatch,
21    pub(crate) sort: Vec<myco_core::SortKey>,
22    pub(crate) claim: bool,
23    /// `--steal` — take the item over from a different owner
24    /// (requires `--claim`).
25    pub(crate) steal: bool,
26    /// `--session` — tags the claim (requires `--claim`).
27    pub(crate) session: Option<String>,
28}
29
30/// The default `next` spec — the builtin `next` view (open
31/// tasks+issues, top item only), before config shadows/flags.
32#[cfg(test)]
33pub(crate) fn base_spec() -> ViewSpec {
34    myco_ops::view::builtins()
35        .into_iter()
36        .find(|v| v.name == "next")
37        .map(|v| v.spec)
38        .unwrap_or_default()
39}

41/// `next`'s args → the flag bundle. `--session` only makes sense
42/// under `--claim` — it tags the claim being taken; on a
43/// read-only `next` there's no claim to tag.
44pub(crate) fn next_flags(a: &Args) -> Result<NextFlags, String> {
45    let session = a.opt("session").map(str::to_string);
46    let steal = a.flag("steal");
47    if session.is_some() && !a.flag("claim") {
48        return Err("next: --session needs --claim (it tags the claim being taken)".into());
49    }
50    if steal && !a.flag("claim") {
51        return Err("next: --steal needs --claim (it overrides the ownership guard)".into());
52    }
53    Ok(NextFlags {
54        filter: view::filter_args(a)?,
55        sort: a.many_parsed("sort")?,
56        claim: a.flag("claim"),
57        steal,
58        session,
59    })
60}

62/// Resolve the `next` named view — a repo's `[views.next]`
63/// shadows the builtin like any other name — then merge flags,
64/// validated. Under `--claim` the `--owner` flag names the
65/// claimer, so it must not also restrict the pick to items that
66/// owner already holds.
67pub(crate) fn spec(root: &std::path::Path, flags: &NextFlags) -> Result<ViewSpec, String> {
68    let (view, warnings) = myco_ops::view::resolve_lenient(root, "next", ops::CONFIG)?;
69    view::warn(&warnings);
70    let mut spec = view.spec;
71    flags.filter.apply(&mut spec.filter);
72    if flags.claim {
73        spec.filter.owner = Vec::new();
74    }
75    if !flags.sort.is_empty() {
76        spec.sort = flags.sort.clone();
77    }
78    spec.validate()?;
79    Ok(spec)
80}

82/// The highest-priority live item under `spec` — the pick
83/// behind the verb and `serve`'s `next` method. The universe is
84/// top-level items only — children ride with their parent (the
85/// `myco view` rule).
86pub(crate) fn pick<'a>(
87    scan: &'a digest::Scan,
88    spec: &ViewSpec,
89    kinds: &myco_core::KindTable,
90) -> Option<&'a Item> {
91    let universe = scan.items.iter().filter(|i| !scan.archived.contains(&i.id));
92    eval(universe, spec, kinds)
93        .iter()
94        .find_map(|g| g.items.first().map(|s| s.item))
95}

97/// Who a `--claim` assigns when `--owner` names one — under
98/// `--claim` it names the claimer, not a filter. `None` means
99/// "use the resolved actor" ([`ops::resolve_actor`]). Several
100/// `--owner`s can't name one claimer, so that's an error.
101fn claim_owner(flags: &NextFlags) -> Result<Option<Actor>, String> {
102    match flags.filter.owner.as_slice() {
103        [] => Ok(None),
104        [owner] => Ok(Some(owner.clone())),
105        _ => Err("next --claim: pass one --owner (the claimer), not several".into()),
106    }
107}

109/// The one-line human output — `items::compact` + the `⚡`
110/// interrupted marker, same as a `list`-format view row.
111fn line(store: &impl Store, item: &Item) -> Result<String, String> {
112    let mut out = items::compact(item);
113    if continuity::is_interrupted(store, item.id)? {
114        out.push_str(" ⚡");
115    }
116    Ok(out)
117}

119/// The `next` verb — read-only unless `--claim`, which takes
120/// the mutating open path + per-verb commit (`chore: claim
121/// <ID>`; `--no-commit`/`$MYCO_NO_COMMIT` honored).
122pub fn h(ctx: &Ctx, a: &Args) -> Result<i32, String> {
123    let flags = next_flags(a)?;
124    let json_out = ctx.json;
125    let no_commit = ctx.no_commit;
126    let explicit_owner = flags
127        .claim
128        .then(|| claim_owner(&flags))
129        .transpose()?
130        .flatten();
131    let (proj, store) = if flags.claim {
132        items::open(ctx.dir.as_deref(), ctx.root.as_deref())?
133    } else {
134        items::open_read(ctx.dir.as_deref(), ctx.root.as_deref())?
135    };
136    // the claimer — `--owner` is the chain's first rung, then
137    // `$MYCO_ACTOR` → `human:<git user.name>` → `cli`
138    let me = flags
139        .claim
140        .then(|| ops::resolve_actor(Some(&proj.worktree), explicit_owner.as_ref()));
141    let spec = spec(&proj.root, &flags)?;
142    let scan = digest::scan(&store)?;
143    let Some(item) = pick(&scan, &spec, store.kinds()).cloned() else {
144        if json_out {
145            ops::print_json(&json!({"item": null}));
146        } else {
147            println!("{}: nothing next — backlog is clear", scan.name);
148        }
149        return Ok(0);
150    };
151    let Some(me) = me else {
152        if json_out {
153            ops::print_json(&json!({"item": item}));
154        } else {
155            println!("{}", line(&store, &item)?);
156        }
157        return Ok(0);
158    };
159    let id = item.id;
160    let set = itemcmd::SetFlags {
161        // identity isn't a transition — an already-claimed item
162        // just re-stamps the owner
163        status: (item.status != Status::Claimed).then_some(Status::Claimed),
164        owner: Some(me.clone()),
165        steal: flags.steal,
166        ..Default::default()
167    };
168    // `--session` tags the claim — the mutation and the `claimed`
169    // handoff entry pair through the atomic claim path
170    let outcome = itemcmd::set_item(&store, id, &set, flags.session, &me)?;
171    let committed = items::commit(
172        &proj.worktree,
173        &format!("chore: claim {id}"),
174        Some(&me),
175        no_commit,
176    )?;
177    let human = match &outcome.item {
178        Some(i) => line(&store, i)?,
179        None => format!("claimed {id}"),
180    };
181    let mut payload = json!({"item": outcome.item});
182    if let Some(from) = outcome.taken_from {
183        payload["taken_from"] = json!(from);
184    }
185    items::report(payload, committed, json_out, &human)
186}

194    /// A hermetic root — no myco.toml, no user config leaks.
195    fn spec_at(dir: &tempfile::TempDir, flags: &NextFlags) -> ViewSpec {
196        temp_env::with_var("MYCO_CONFIG", None::<&str>, || {
197            spec(dir.path(), flags).unwrap()
198        })
199    }
200
201    #[test]
202    fn default_spec_is_open_tasks_and_issues_limit_1() {
203        let dir = tempfile::tempdir().unwrap();
204        let spec = spec_at(&dir, &NextFlags::default());
205        assert_eq!(spec.limit, Some(1));
206        assert_eq!(spec.filter.kind, ["task".to_string(), "issue".to_string()]);
207        assert_eq!(spec.filter.status, [Status::Open]);
208        assert!(spec.sort.is_empty()); // engine default: pri,id
209    }
210
211    #[test]
212    fn flags_merge_over_the_default() {
213        let flags = NextFlags {
214            filter: view::FilterPatch {
215                kind: vec!["issue".to_string()],
216                label: vec!["bug".into()],
217                ..Default::default()
218            },
219            sort: vec!["-id".parse().unwrap()],
220            claim: false,
221            steal: false,
222            session: None,
223        };
224        let dir = tempfile::tempdir().unwrap();
225        let spec = spec_at(&dir, &flags);
226        assert_eq!(spec.filter.kind, ["issue".to_string()]);
227        // unset fields keep the base: status stays [open]
228        assert_eq!(spec.filter.status, [Status::Open]);
229        assert_eq!(spec.filter.label, ["bug"]);
230        assert_eq!(spec.sort.len(), 1);
231    }
232
233    #[test]
234    fn claim_owner_resolution() {
235        let mut flags = NextFlags::default();
236        // no --owner → the resolved actor chain decides
237        assert_eq!(claim_owner(&flags).unwrap(), None);
238        flags.filter.owner = vec!["devin".parse().unwrap()];
239        assert_eq!(claim_owner(&flags).unwrap().unwrap().as_str(), "devin");
240        flags.filter.owner = vec!["devin".parse().unwrap(), "human:x".parse().unwrap()];
241        assert!(claim_owner(&flags).is_err());
242    }
243
244    #[test]
245    fn pick_orders_pri_then_id() {
246        let items = vec![
247            Item::new(ItemId::new(ItemKind::Task, 1), "low").priority(myco_core::Priority::Low),
248            Item::new(ItemId::new(ItemKind::Task, 2), "crit")
249                .priority(myco_core::Priority::Critical),
250            Item::new(ItemId::new(ItemKind::Issue, 1), "hi").priority(myco_core::Priority::High),
251        ];
252        let scan = digest::Scan {
253            name: "p".into(),
254            items,
255            archived: Default::default(),
256        };
257        let spec = base_spec();
258        assert_eq!(
259            pick(&scan, &spec, myco_core::kind::prelude_table())
260                .unwrap()
261                .id,
262            ItemId::new(ItemKind::Task, 2)
263        );
264        // a done crit drops out of the open default
265        let mut scan2 = digest::Scan {
266            name: "p".into(),
267            items: scan.items.clone(),
268            archived: Default::default(),
269        };
270        scan2.items[1].status = Status::Done;
271        assert_eq!(
272            pick(&scan2, &spec, myco_core::kind::prelude_table())
273                .unwrap()
274                .id,
275            ItemId::new(ItemKind::Issue, 1)
276        );
277    }
