1//! Re-export — lives in `myco_ops::discover` so other front-ends
2//! (`myco-driver`) share it. `discover` itself is cli-local: ops
3//! takes the config loader as a parameter and the composition
4//! root (this crate) names the impl, so call sites stay two-arg.
5//!
6//! The cli `discover` adds the org-ledger fallback on top of ops'
7//! strict discovery: inside a *registered* repo with no ledger of
8//! its own, verbs resolve to the `ledger = "org"` entry's
9//! worktree. The index opens lazily — only when plain discovery
10//! already failed — so the happy path never touches it, and
11//! unregistered repos fail exactly as before.
12
13use std::path::Path;
14
15pub use myco_ops::discover::{qualifies, repo_root_for, worktree_path, Project, BRANCH, DIR};
16
17/// [`myco_ops::discover::discover`] wired to [`crate::ops::CONFIG`]
18/// plus the org-ledger fallback (see module docs).
19pub fn discover(dir: Option<&Path>, root: Option<&Path>) -> Result<Project, String> {
20    match myco_ops::discover::discover(dir, root, crate::ops::CONFIG) {
21        Ok(p) => Ok(p),
22        Err(e) => {
23            let Ok(index) = crate::indexcmd::open_index() else {
24                return Err(e);
25            };
26            myco_ops::discover::discover_with(dir, root, crate::ops::CONFIG, Some(&index))
27                .or(Err(e))
28        }
29    }
30}
