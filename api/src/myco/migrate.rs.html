1//! `myco migrate` — bring legacy ledgers to the v2 schema.
2//!
3//! Two modes, auto-detected:
4//!
5//! - **`pm/` → `project/`** — a `pm/` dir at the repo root is the
6//!   pre-worktree layout. `TODO.md`/`BACKLOG.md` bullets become
7//!   T-items, `ROADMAP.md` becomes a `plans/` file, any other
8//!   `*.md` becomes an issue; the `pm/` dir is removed and its
9//!   deletion committed on the current branch.
10//! - **legacy `project/` → v2** — files already on the `project`
11//!   branch are normalized in place: id-less items get ids,
12//!   `[x]`/done items drop, heading-like lines become sections,
13//!   frontmatter-less files in item dirs are adopted (the
14//!   filename's number is honored when free — `0001-foo.md` →
15//!   `ADR-0001` — collisions take the next unclaimed number with a
16//!   warning; synthesized frontmatter + canonical name),
17//!   everything re-emitted canonically.
18//!
19//! Either way, bullet shapes the target doc can't express demote
20//! to plain-task text with a warning instead of failing the run —
21//! a `status:deferred` in TODO.md folds into the title, a
22//! self-block drops, a textless `##` merges into the section
23//! before it ([`demote`]).
24

48/// What a `pm/` file converts into.
49enum PmDest {
50    Todo,
51    Backlog,
52    Plan,
53    Issue,
54}
55
56fn pm_dest(name: &str) -> Option<PmDest> {
57    if name.eq_ignore_ascii_case("todo.md") {
58        Some(PmDest::Todo)
59    } else if name.eq_ignore_ascii_case("backlog.md") {
60        Some(PmDest::Backlog)
61    } else if name.eq_ignore_ascii_case("roadmap.md") {
62        Some(PmDest::Plan)
63    } else if name.ends_with(".md") {
64        Some(PmDest::Issue)
65    } else {
66        None
67    }
68}

70/// First `#`-heading text in a markdown body, else a fallback.
71fn first_heading(src: &str, fallback: &str) -> String {
72    for line in src.lines() {
73        let t = line.trim();
74        if let Some(rest) = t.strip_prefix('#') {
75            let title = rest.trim_start_matches('#').trim();
76            if !title.is_empty() {
77                return title.to_string();
78            }
79        }
80    }
81    fallback.to_string()
82}

84/// A title for an adopted/converted item file — `·` is the line
85/// grammar's metadata separator and can't live in a title, so a
86/// heading or filename that uses it degrades to `-` rather than
87/// failing the emit.
88fn adopt_title(src: &str, fallback: &str) -> String {
89    first_heading(src, fallback).replace('·', "-")
90}

92/// Per-run counters for the JSON report.
93#[derive(Default)]
94struct Stats {
95    converted_items: usize,
96    ids_allocated: usize,
97    dropped_done: usize,
98    files_moved: BTreeMap<String, String>,
99    unparsed: Vec<String>,
100    warnings: Vec<String>,
101    /// Ids issued to ledger items this run — a kept source id is
102    /// honored only while free; a source naming it again collides
103    /// to a fresh alloc with a warning, never a duplicate.
104    issued: BTreeSet<ItemId>,
105}

107/// Filename-number claim table for frontmatter-less adoption —
108/// one per `migrate` run, spanning all item dirs. A frontmatter-less
109/// `NNNN-slug.md` is *borrowing* its number: adoption honors it when
110/// no real item owns it (a `taken` id), and a collision probes
111/// upward past numbers other unprocessed adoptees still want
112/// (`pending`) so `0001-a` + `0001-b` + `0002-c` land 1, 3, 2 —
113/// never a fresh counter id.
114#[derive(Default)]
115struct AdoptClaims {
116    /// Ids owned by real items — frontmatter ids, non-adoptee
117    /// filenames, TODO/BACKLOG items, archived files.
118    taken: BTreeSet<ItemId>,
119    /// Filename numbers adoptees not yet processed still want.
120    pending: BTreeMap<ItemId, usize>,
121}
122
123impl AdoptClaims {
124    /// Claim `desired` for the adoptee now being processed: the
125    /// number itself when free, else the next number neither taken
126    /// nor wanted by a remaining adoptee. Returns `(id, collided)`.
127    fn claim(&mut self, desired: ItemId) -> (ItemId, bool) {
128        // desired was registered into pending in the claim pass —
129        // or_insert keeps the unreachable unclaimed case honest
130        let c = self.pending.entry(desired).or_insert(1);
131        *c -= 1;
132        if *c == 0 {
133            self.pending.remove(&desired);
134        }
135        if !self.taken.contains(&desired) {
136            self.taken.insert(desired);
137            return (desired, false);
138        }
139        let mut n = desired.next();
140        while self.taken.contains(&n) || self.pending.contains_key(&n) {
141            n = n.next();
142        }
143        self.taken.insert(n);
144        (n, true)
145    }
146
147    /// Record an id allocated outside the claim path (e.g.
148    /// `alloc_id` for a filename without a number).
149    fn reserve(&mut self, id: ItemId) {
150        self.taken.insert(id);
151    }
152}

154/// Turn a legacy item into a real one — keep a same-kind id or
155/// allocate from the run's [`Registry`]. The store's `alloc_id`
156/// can't serve here: it scans `all_ids`, which strict-parses
157/// `TODO.md`/`BACKLOG.md` — the very files a project-mode
158/// migration is mid-way through normalizing.
159fn materialize(
160    reg: &mut Registry,
161    li: &myco_store_md::LegacyItem,
162    kind: ItemKind,
163    stats: &mut Stats,
164) -> Result<Item, String> {
165    let mut item = li.item.clone();
166    item.id = match li.id {
167        Some(id) if id.kind() == kind => {
168            if stats.issued.insert(id) {
169                // a kept id must still raise the counter floor —
170                // `all_ids` never sees pm/-source ids, so without
171                // this a later alloc could reissue it mid-run
172                reg.observe(id);
173                id
174            } else {
175                // the source named an id already issued this run —
176                // keeping it would write a duplicate (id'd lines
177                // must still validate), so the item converts on a
178                // fresh number with a warning
179                let fresh = reg.alloc(kind);
180                stats.issued.insert(fresh);
181                stats.ids_allocated += 1;
182                stats
183                    .warnings
184                    .push(format!("{id} already issued — item converted as {fresh}"));
185                fresh
186            }
187        }
188        _ => {
189            let fresh = reg.alloc(kind);
190            stats.issued.insert(fresh);
191            stats.ids_allocated += 1;
192            fresh
193        }
194    };
195    stats.converted_items += 1;
196    Ok(item)
197}

199/// Legacy sections → v2 `Section`s for `file` (`TODO.md` /
200/// `BACKLOG.md`), allocating ids as needed. Shapes the target
201/// doc's grammar can't express demote to plain-task text rather
202/// than failing the migration — see [`demote`].
203fn convert_sections(
204    reg: &mut Registry,
205    doc: &myco_store_md::LegacyDoc,
206    kind: ItemKind,
207    file: &'static str,
208    stats: &mut Stats,
209) -> Result<Vec<Section>, String> {
210    let mut out: Vec<Section> = Vec::new();
211    for sec in &doc.sections {
212        // a `##` heading that flattens to no text (`##`, `## ###`)
213        // marks no real boundary — an empty name can't emit, so the
214        // section's items merge into the one before it (or the
215        // leading unsectioned run when nothing precedes)
216        let name = sec.name.as_deref().filter(|n| !n.trim().is_empty());
217        if name.is_some() || out.is_empty() {
218            out.push(Section {
219                name: name.map(String::from),
220                items: Vec::new(),
221            });
222        }
223        // a nameless section that isn't first lands its items on
224        // the previous one — `out` is non-empty either way
225        let target = out.len() - 1;
226        for li in &sec.items {
227            let mut item = materialize(reg, li, kind, stats)?;
228            demote(&mut item, li.id, file, stats);
229            out[target].items.push(item);
230        }
231    }
232    Ok(out)
233}

235/// Demote the parts of a converted item `file`'s grammar can't
236/// express to plain text, instead of failing the whole migration:
237/// a `status:` the doc forbids folds back into the title (the
238/// ` - ` separator unknown `·` tokens already use), and a
239/// self-block drops — each with a warning, nothing silent. The
240/// id stands: an id'd line keeps its number, only its unemittable
241/// *state* demotes. `src_id` is the id the line carried — the
242/// item's own number may have moved on a collision.
243fn demote(item: &mut Item, src_id: Option<ItemId>, file: &str, stats: &mut Stats) {
244    // `status:deferred` has no home in TODO.md — the marker becomes
245    // title text and the task lands open like any freeform bullet
246    if file == "TODO.md" && item.status == Status::Deferred {
247        let tok = "status:deferred";
248        item.title = match item.title.is_empty() {
249            true => tok.to_string(),
250            false => format!("{} - {tok}", item.title),
251        };
252        item.status = Status::Open;
253        stats.warnings.push(format!(
254            "{file}: {} — deferred tasks live in BACKLOG.md, folded `{tok}` into the title",
255            item.id
256        ));
257    }
258    // a `blocks:` at the line's own source id is meaningless at
259    // any address — it drops whether the item kept its number or
260    // collided onto a fresh one
261    if src_id.is_some_and(|id| item.blocks.remove(&id)) {
262        stats
263            .warnings
264            .push(format!("{file}: {} — self-block dropped", item.id));
265    }
266}

268/// Mode A — `pm/` at the repo root.
269fn migrate_pm(
270    dir: Option<&Path>,
271    root_hint: Option<&Path>,
272    no_commit: bool,
273) -> Result<(serde_json::Value, i32), String> {
274    let root = discover::repo_root_for(dir, root_hint)?;
275    let pm = root.join("pm");
276    // run() only routes here when pm/ exists — a vanish between
277    // the two probes just fails at read_dir below
278    debug_assert!(pm.is_dir(), "migrate_pm without pm/ at {}", root.display());
279    // ensure the project worktree exists (init when needed) —
280    // its lock releases with it; the migration's own RMW takes
281    // a fresh one held through the commit
282    let ensured = init::ensure(dir, Some(&root), no_commit)?;
283    let wt = ensured.worktree.clone();
284    let _lock = ops::lock_worktree(&wt)?;
285    let store = ops::store_at(&wt, myco_core::kind::prelude_table())?;
286    let mut stats = Stats::default();
287
288    let mut todo = store.todo().map_err(|e| e.to_string())?;
289    let mut backlog = store.backlog().map_err(|e| e.to_string())?;
290    // run-scoped allocator — one persisted reconcile at the end
291    // instead of alloc_id's per-call meta rewrite
292    let mut meta = store.meta().map_err(|e| e.to_string())?;
293    let mut reg = meta.registry();
294    reg.observe_all(store.all_ids().map_err(|e| e.to_string())?);
295    let mut names: Vec<String> = std::fs::read_dir(&pm)
296        .map_err(|e| e.to_string())?
297        .flatten()
298        .filter(|e| e.path().is_file())
299        .filter_map(|e| e.file_name().to_str().map(String::from))
300        .collect();
301    names.sort();
302    let mut leftovers: Vec<String> = Vec::new();
303    for name in &names {
304        let src = std::fs::read_to_string(pm.join(name)).map_err(|e| e.to_string())?;
305        let rel_from = format!("pm/{name}");
306        match pm_dest(name) {
307            Some(PmDest::Todo) => {
308                let (_repo, doc) = myco_store_md::parse_legacy_todo(&src, "migrated");
309                stats.dropped_done += doc.dropped;
310                todo.sections.extend(convert_sections(
311                    &mut reg,
312                    &doc,
313                    ItemKind::Task,
314                    "TODO.md",
315                    &mut stats,
316                )?);
317                stats.files_moved.insert(rel_from, "TODO.md".into());
318            }
319            Some(PmDest::Backlog) => {
320                let doc = myco_store_md::parse_legacy_backlog(&src);
321                stats.dropped_done += doc.dropped;
322                let mut secs =
323                    convert_sections(&mut reg, &doc, ItemKind::Task, "BACKLOG.md", &mut stats)?;
324                for s in &mut secs {
325                    for i in &mut s.items {
326                        i.status = Status::Deferred;
327                    }
328                }
329                backlog.sections.extend(secs);
330                stats.files_moved.insert(rel_from, "BACKLOG.md".into());
331            }
332            Some(PmDest::Plan) => {
333                let id = reg.alloc(ItemKind::Plan);
334                stats.ids_allocated += 1;
335                let title = adopt_title(&src, "Roadmap");
336                let file = ItemFile::new(Item::new(id, &title)).body(src);
337                store.save_item(&file).map_err(|e| e.to_string())?;
338                let rel = format!(
339                    "plans/{}",
340                    pspec(ItemKind::Plan)
341                        .and_then(|s| myco_store_md::item_filename(s, &id, &title))
342                        .unwrap_or_default()
343                );
344                stats.converted_items += 1;
345                stats.files_moved.insert(rel_from, rel);
346            }
347            Some(PmDest::Issue) => {
348                let id = reg.alloc(ItemKind::Issue);
349                stats.ids_allocated += 1;
350                let stem = name.trim_end_matches(".md");
351                let title = adopt_title(&src, stem);
352                let file = ItemFile::new(Item::new(id, &title)).body(src);
353                store.save_item(&file).map_err(|e| e.to_string())?;
354                let rel = format!(
355                    "issues/{}",
356                    pspec(ItemKind::Issue)
357                        .and_then(|s| myco_store_md::item_filename(s, &id, &title))
358                        .unwrap_or_default()
359                );
360                stats.converted_items += 1;
361                stats.files_moved.insert(rel_from, rel);
362            }
363            None => leftovers.push(rel_from),
364        }
365    }
366    store.save_todo(&todo).map_err(|e| e.to_string())?;
367    store.save_backlog(&backlog).map_err(|e| e.to_string())?;
368    meta.next = reg.counters();
369    store.save_meta(&meta).map_err(|e| e.to_string())?;
370
371    // remove converted files; the dir only goes when it's empty
372    for name in &names {
373        if pm_dest(name).is_some() {
374            let _ = std::fs::remove_file(pm.join(name));
375        }
376    }
377    let empty = std::fs::read_dir(&pm)
378        .map(|mut it| it.next().is_none())
379        .unwrap_or(false);
380    let pm_removed = if empty {
381        std::fs::remove_dir(&pm).is_ok()
382    } else {
383        false
384    };
385
386    let enabled = ops::commits_enabled(no_commit);
387    let msg = format!(
388        "chore: migrate pm/ → project/ ({} items)",
389        stats.converted_items
390    );
391    // pm/ files are already deleted — a failed commit must not
392    // leave them gone-but-uncommitted; restore them on the way out
393    let commits = (|| -> Result<(bool, bool), String> {
394        Ok((
395            ops::autocommit(&wt, &msg, None, enabled)?,
396            commit_pm_removal(&root, enabled)?,
397        ))
398    })();
399    let (c1, c2) = match commits {
400        Ok(pair) => pair,
401        Err(e) => {
402            let _ = myco_ops::vcs::git_out(&root, &["restore", "--source=HEAD", "--", "pm"]);
403            return Err(e);
404        }
405    };
406    let committed = c1 || c2;
407
408    let out = json!({
409        "mode": "pm",
410        "converted_items": stats.converted_items,
411        "ids_allocated": stats.ids_allocated,
412        "dropped_done": stats.dropped_done,
413        "files_moved": stats.files_moved.iter()
414            .map(|(from, to)| json!({"from": from, "to": to}))
415            .collect::<Vec<_>>(),
416        "pm_removed": pm_removed,
417        "pm_leftover": leftovers,
418        "warnings": stats.warnings,
419        "worktree": wt.to_string_lossy(),
420        "committed": committed,
421    });
422    Ok((out, 0))
423}

425/// `git add -u pm/ && git commit -- pm/` — stage only tracked
426/// deletions so untracked leftovers never get swept in.
427fn commit_pm_removal(root: &Path, enabled: bool) -> Result<bool, String> {
428    if !enabled {
429        return Ok(false);
430    }
431    // `add -u` errors when pm/ holds no tracked files at all —
432    // untracked leftovers aren't ours to commit
433    if myco_ops::vcs::git_out(root, &["ls-files", "--", "pm"])
434        .map_err(|e| e.to_string())?
435        .is_empty()
436    {
437        return Ok(false);
438    }
439    myco_ops::vcs::git_out(root, &["add", "-u", "--", "pm"]).map_err(|e| e.to_string())?;
440    let dirty = myco_ops::vcs::git_out(root, &["status", "--porcelain", "--", "pm"])
441        .map_err(|e| e.to_string())?;
442    if dirty.is_empty() {
443        return Ok(false);
444    }
445    let msg = ops::commit_msg("chore: migrate pm/ → project/", None);
446    myco_ops::vcs::git_ident(root, &["commit", "-m", &msg, "--", "pm"])
447        .map_err(|e| e.to_string())
448        .map(|_| true)
449}

451/// Mode B — normalize files already on the `project` branch.
452fn migrate_project(
453    dir: Option<&Path>,
454    root_hint: Option<&Path>,
455    no_commit: bool,
456) -> Result<(serde_json::Value, i32), String> {
457    // strict discovery, no org-ledger fallback — migrate
458    // rewrites *this* repo's ledger; silently landing on the org
459    // ledger would rewrite the wrong tree
460    let proj = myco_ops::discover::discover(dir, root_hint, crate::ops::CONFIG)?;
461    let wt = &proj.worktree;
462    let _lock = ops::lock_worktree(wt)?;
463    let store = ops::store_at(wt, myco_core::kind::prelude_table())?;
464    let mut stats = Stats::default();
465
466    // seed counters from ids we're about to keep so allocation
467    // can't collide with them. A fresh default meta takes the
468    // worktree's dir name ("project") — the repo dir name is the
469    // better guess.
470    let mut meta = store.meta().map_err(|e| e.to_string())?;
471    if !wt.join("meta.toml").exists() {
472        if let Some(n) = proj.root.file_name() {
473            meta.name = n.to_string_lossy().into_owned();
474        }
475    }
476    // run-scoped allocator over the tolerant id scan — the store's
477    // `all_ids` strict-parses the ledger docs, which fails on
478    // exactly the legacy files migrate exists to fix
479    let mut reg = meta.registry();
480    reg.observe_all(migrate_ids(wt)?);
481
482    // --- TODO.md ---
483    if let Some(src) = read_str(wt, "TODO.md") {
484        let doc = match myco_store_md::parse_todo(&src) {
485            Ok(doc) => doc, // already v2 — re-emit normalizes drift
486            Err(_) => {
487                let (repo, legacy) = myco_store_md::parse_legacy_todo(&src, &meta.name);
488                stats.dropped_done += legacy.dropped;
489                let mut doc = TodoDoc::new(if repo.is_empty() { &meta.name } else { &repo });
490                doc.sections =
491                    convert_sections(&mut reg, &legacy, ItemKind::Task, "TODO.md", &mut stats)?;
492                doc
493            }
494        };
495        store.save_todo(&doc).map_err(|e| e.to_string())?;
496    }
497
498    // --- BACKLOG.md ---
499    if let Some(src) = read_str(wt, "BACKLOG.md") {
500        let doc = match myco_store_md::parse_backlog(&src) {
501            Ok(doc) => doc,
502            Err(_) => {
503                let legacy = myco_store_md::parse_legacy_backlog(&src);
504                stats.dropped_done += legacy.dropped;
505                let mut secs =
506                    convert_sections(&mut reg, &legacy, ItemKind::Task, "BACKLOG.md", &mut stats)?;
507                for s in &mut secs {
508                    for i in &mut s.items {
509                        i.status = Status::Deferred;
510                    }
511                }
512                BacklogDoc { sections: secs }
513            }
514        };
515        store.save_backlog(&doc).map_err(|e| e.to_string())?;
516    }
517
518    // --- item dirs: normalize filenames + adopt unfrontmattered ---
519    //
520    // Claim table for adoption: `taken` = ids a real item owns —
521    // everything `all_ids` sees plus frontmatter ids (which can
522    // outrun the filename view), minus numbers merely *borrowed* by
523    // adoptee filenames. `pending` = filename numbers adoptees
524    // still want, so a collision probes past them instead of
525    // jumping to the counter.
526    let mut claims = AdoptClaims {
527        taken: migrate_ids(wt)?,
528        pending: BTreeMap::new(),
529    };
530    for spec in file_specs() {
531        let dir_name = spec.dir_name();
532        let dir = wt.join(&dir_name);
533        let Ok(read) = std::fs::read_dir(&dir) else {
534            continue;
535        };
536        let mut owned: BTreeSet<ItemId> = BTreeSet::new();
537        let mut desired: Vec<ItemId> = Vec::new();
538        for entry in read.flatten().filter(|e| e.path().is_file()) {
539            let Some(name) = entry.file_name().to_str().map(String::from) else {
540                continue;
541            };
542            if !name.ends_with(".md") || name.starts_with('.') {
543                continue;
544            }
545            let fname_id = myco_store_md::parse_item_filename(spec, &name);
546            match read_str(wt, &format!("{dir_name}/{name}"))
547                .as_deref()
548                .and_then(frontmatter_id)
549            {
550                // adoptee — the filename number is a claim to honor
551                None => desired.extend(fname_id),
552                // real item — it lands on its frontmatter id, so
553                // that number is off-limits (the filename number is
554                // vacated by the canonical rename, not reserved)
555                Some(fm_id) => {
556                    owned.insert(fm_id);
557                }
558            }
559        }
560        // archived files keep their numbers reserved forever
561        if let Ok(read) = std::fs::read_dir(dir.join("archive")) {
562            for entry in read.flatten().filter(|e| e.path().is_file()) {
563                if let Some(name) = entry.file_name().to_str() {
564                    owned.extend(myco_store_md::parse_item_filename(spec, name));
565                }
566            }
567        }
568        for d in &desired {
569            if !owned.contains(d) {
570                claims.taken.remove(d);
571            }
572            *claims.pending.entry(*d).or_default() += 1;
573        }
574        claims.taken.extend(owned);
575    }
576
577    let mut renames: BTreeMap<String, String> = BTreeMap::new();
578    for spec in file_specs() {
579        let dir_name = spec.dir_name();
580        let dir = wt.join(&dir_name);
581        if !dir.is_dir() {
582            continue;
583        }
584        let mut names: Vec<String> = std::fs::read_dir(&dir)
585            .map_err(|e| e.to_string())?
586            .flatten()
587            .filter(|e| e.path().is_file())
588            .filter_map(|e| e.file_name().to_str().map(String::from))
589            .filter(|n| n.ends_with(".md") && !n.starts_with('.'))
590            .collect();
591        names.sort();
592        for name in names {
593            let rel = format!("{dir_name}/{name}");
594            let src = match read_str(wt, &rel) {
595                Some(s) => s,
596                None => continue,
597            };
598            match adopt_or_normalize(&store, &mut reg, &src, spec, &rel, &mut stats, &mut claims)? {
599                Some(new_rel) if new_rel != rel => {
600                    renames.insert(rel.clone(), new_rel.clone());
601                    stats.files_moved.insert(rel, new_rel);
602                }
603                _ => {}
604            }
605        }
606    }
607
608    // index.md: repoint links at renamed files (raw passthrough,
609    // so a plain path replace is the whole fix)
610    if !renames.is_empty() {
611        if let Some(mut index) = read_str(wt, "index.md") {
612            let orig = index.clone();
613            for (from, to) in &renames {
614                index = index.replace(from.as_str(), to.as_str());
615            }
616            if index != orig {
617                store.save_index(&index).map_err(|e| e.to_string())?;
618            }
619        }
620    }
621
622    // meta.toml: create if missing + reconcile counters
623    {
624        reg.observe_all(store.all_ids().map_err(|e| e.to_string())?);
625        meta.next = reg.counters();
626        store.save_meta(&meta).map_err(|e| e.to_string())?;
627    }
628
629    // fill any remaining scaffold gaps (dirs, index, docs)
630    let created = ops::scaffold(&store, wt, &meta.name).map_err(|e| e.to_string())?;
631
632    let enabled = ops::commits_enabled(no_commit);
633    let msg = format!(
634        "chore: migrate project/ → v2 schema ({} items)",
635        stats.converted_items
636    );
637    let committed = ops::autocommit(wt, &msg, None, enabled)?;
638
639    let out = json!({
640        "mode": "project",
641        "converted_items": stats.converted_items,
642        "ids_allocated": stats.ids_allocated,
643        "dropped_done": stats.dropped_done,
644        "files_moved": stats.files_moved.iter()
645            .map(|(from, to)| json!({"from": from, "to": to}))
646            .collect::<Vec<_>>(),
647        "unparsed": stats.unparsed,
648        "warnings": stats.warnings,
649        "scaffold_created": created,
650        "worktree": wt.to_string_lossy(),
651        "committed": committed,
652    });
653    Ok((out, 0))
654}

656/// The id scan migrate needs mid-run — `Store::all_ids`
657/// strict-parses `TODO.md`/`BACKLOG.md`, which fails on exactly
658/// the legacy files a migration is there to fix. Same coverage,
659/// tolerant reads: an unparseable ledger doc falls back to the
660/// legacy parser (its kept ids still reserve their numbers), and
661/// item files reserve by filename with the same filename-vs-
662/// frontmatter cross-check `MdStore` does.
663fn migrate_ids(wt: &Path) -> Result<BTreeSet<ItemId>, String> {
664    let mut ids = BTreeSet::new();
665    if let Some(src) = read_str(wt, "TODO.md") {
666        match myco_store_md::parse_todo(&src) {
667            Ok(doc) => ids.extend(doc.items().flat_map(|i| i.subtree()).map(|i| i.id)),
668            Err(_) => {
669                let (_, legacy) = myco_store_md::parse_legacy_todo(&src, "");
670                ids.extend(legacy.items().filter_map(|li| li.id));
671            }
672        }
673    }
674    if let Some(src) = read_str(wt, "BACKLOG.md") {
675        match myco_store_md::parse_backlog(&src) {
676            Ok(doc) => ids.extend(doc.items().flat_map(|i| i.subtree()).map(|i| i.id)),
677            Err(_) => {
678                let legacy = myco_store_md::parse_legacy_backlog(&src);
679                ids.extend(legacy.items().filter_map(|li| li.id));
680            }
681        }
682    }
683    for spec in file_specs() {
684        let dir_name = spec.dir_name();
685        for archived in [false, true] {
686            let rel_dir = match archived {
687                false => dir_name.to_string(),
688                true => format!("{dir_name}/archive"),
689            };
690            let Ok(read) = std::fs::read_dir(wt.join(&rel_dir)) else {
691                continue;
692            };
693            for entry in read.flatten().filter(|e| e.path().is_file()) {
694                let Some(name) = entry.file_name().to_str().map(String::from) else {
695                    continue;
696                };
697                let Some(fname_id) = myco_store_md::parse_item_filename(spec, &name) else {
698                    continue;
699                };
700                let rel = format!("{rel_dir}/{name}");
701                match read_str(wt, &rel).as_deref().and_then(frontmatter_id) {
702                    Some(fm_id) if fm_id != fname_id => {
703                        return Err(format!(
704                            "{rel}: filename says {fname_id} but frontmatter id is {fm_id}"
705                        ));
706                    }
707                    // a frontmatter-less file borrows its filename
708                    // number — reserved either way
709                    Some(id) => {
710                        ids.insert(id);
711                    }
712                    None => {
713                        ids.insert(fname_id);
714                    }
715                };
716            }
717        }
718    }
719    Ok(ids)
720}

722/// Bring one file in an item dir to spec. Returns its new rel path
723/// when it moved (or was adopted), `None` when left as-is.
724fn adopt_or_normalize(
725    store: &myco_store_md::MdStore,
726    reg: &mut Registry,
727    src: &str,
728    spec: &myco_core::KindSpec,
729    rel: &str,
730    stats: &mut Stats,
731    claims: &mut AdoptClaims,
732) -> Result<Option<String>, String> {
733    let wt = store.root();
734    let kind = spec.id_kind();
735    // 1) parses under its own kind → save_item re-emits canonically
736    //    and renames the file when the slug drifted
737    if let Ok(file) = myco_store_md::parse_item_file(src, spec, rel) {
738        let id = file.meta.id;
739        store.save_item(&file).map_err(|e| e.to_string())?;
740        let canonical = myco_store_md::item_filename(spec, &id, &file.meta.title).map(|f| {
741            format!(
742                "{}/{}",
743                myco_store_md::item_dir(spec).unwrap_or_default(),
744                f
745            )
746        });
747        // save_item removes a drifted name only when it can tie it
748        // to the id — a name without a number is ours to clean up
749        if canonical.as_deref() != Some(rel) {
750            let _ = std::fs::remove_file(wt.join(rel));
751        }
752        return Ok(canonical.filter(|c| *c != rel));
753    }
754    // 2) frontmatter id of a different kind → move + re-emit
755    if let Some(id) = frontmatter_id(src) {
756        if id.kind() != kind {
757            let moved = myco_core::kind::prelude_table()
758                .spec_of(id.kind())
759                .and_then(|s2| myco_store_md::parse_item_file(src, s2, rel).ok());
760            if let Some(file) = moved {
761                let s2 = pspec(id.kind());
762                store.save_item(&file).map_err(|e| e.to_string())?;
763                let _ = std::fs::remove_file(wt.join(rel));
764                let canonical = s2.and_then(|s2| {
765                    let d = myco_store_md::item_dir(s2)?;
766                    myco_store_md::item_filename(s2, &id, &file.meta.title)
767                        .map(|f| format!("{d}/{f}"))
768                });
769                return Ok(canonical);
770            }
771            // has frontmatter but can't be parsed under any kind —
772            // leave it for `myco check` to report
773            stats.unparsed.push(rel.to_string());
774            return Ok(None);
775        }
776        // same kind but unparseable — leave to check
777        stats.unparsed.push(rel.to_string());
778        return Ok(None);
779    }
780    // 2b) `---` YAML frontmatter — hand-authored files reach for
781    //     the ubiquitous convention; rewrite as `+++` TOML, then
782    //     re-emit canonically like step 1. Unparseable repairs
783    //     are left for `myco check` to report.
784    if let Some(rs) = myco_store_md::repair_frontmatter(src) {
785        return match myco_store_md::parse_item_file(&rs, spec, rel) {
786            Ok(file) => {
787                let id = file.meta.id;
788                store.save_item(&file).map_err(|e| e.to_string())?;
789                let canonical =
790                    myco_store_md::item_filename(spec, &id, &file.meta.title).map(|f| {
791                        format!(
792                            "{}/{}",
793                            myco_store_md::item_dir(spec).unwrap_or_default(),
794                            f
795                        )
796                    });
797                if canonical.as_deref() != Some(rel) {
798                    let _ = std::fs::remove_file(wt.join(rel));
799                }
800                Ok(canonical.filter(|c| *c != rel))
801            }
802            Err(_) => {
803                stats.unparsed.push(rel.to_string());
804                Ok(None)
805            }
806        };
807    }
808    // 3) no frontmatter at all → adopt: honor the filename number
809    //    when free (legacy ids like `0001-gha-compat.md` encode
810    //    real numbering), else probe the next number no other
811    //    adoptee wants; number-less names take a fresh alloc
812    let name = rel.rsplit('/').next().unwrap_or(rel);
813    let id = match myco_store_md::parse_item_filename(spec, name) {
814        Some(desired) => {
815            let (id, collided) = claims.claim(desired);
816            if collided {
817                stats
818                    .warnings
819                    .push(format!("{rel}: {desired} already taken — adopted as {id}"));
820            }
821            reg.observe(id);
822            id
823        }
824        None => {
825            let id = reg.alloc(kind);
826            claims.reserve(id);
827            id
828        }
829    };
830    stats.ids_allocated += 1;
831    stats.converted_items += 1;
832    let stem = name.trim_end_matches(".md");
833    let title = adopt_title(src, stem);
834    let file = ItemFile::new(Item::new(id, &title)).body(src.to_string());
835    store.save_item(&file).map_err(|e| e.to_string())?;
836    // an already-canonical adoptee lands on its own name — deleting
837    // `rel` then would remove the file we just wrote
838    let canonical = myco_store_md::item_filename(spec, &id, &title)
839        .map(|f| format!("{}/{f}", myco_store_md::item_dir(spec).unwrap_or_default()));
840    if canonical.as_deref() != Some(rel) {
841        let _ = std::fs::remove_file(wt.join(rel));
842    }
843    Ok(canonical)
844}

846fn read_str(wt: &Path, rel: &str) -> Option<String> {
847    std::fs::read_to_string(wt.join(rel)).ok()
848}

850/// The `migrate` verb.
851pub fn h(ctx: &crate::registry::Ctx, _a: &crate::registry::Args) -> Result<i32, String> {
852    let (dir, root, json_out, no_commit) =
853        (ctx.dir.clone(), ctx.root.clone(), ctx.json, ctx.no_commit);
854    let repo = discover::repo_root_for(dir.as_deref(), root.as_deref());
855    let pm_mode = repo
856        .as_ref()
857        .map(|r| r.join("pm").is_dir())
858        .unwrap_or(false);
859    let (out, code) = if pm_mode {
860        migrate_pm(dir.as_deref(), root.as_deref(), no_commit)?
861    } else {
862        migrate_project(dir.as_deref(), root.as_deref(), no_commit)?
863    };
864    if json_out {
865        ops::print_json(&out);
866    } else {
867        println!(
868            "myco migrate ({}): {} items, {} ids allocated, {} done dropped",
869            out["mode"], out["converted_items"], out["ids_allocated"], out["dropped_done"]
870        );
871        for m in out["files_moved"].as_array().into_iter().flatten() {
872            println!("  {} → {}", m["from"], m["to"]);
873        }
874        for f in out["unparsed"].as_array().into_iter().flatten() {
875            eprintln!("  unparsed: {f}");
876        }
877        for w in out["warnings"].as_array().into_iter().flatten() {
878            eprintln!("  warning: {}", w.as_str().unwrap_or("?"));
879        }
880    }
881    Ok(code)
882}
