10/// The `backlog add` mutation — allocs a T-id straight into
11/// BACKLOG.md (backlog items are born deferred). Caller commits.
12pub fn add_item(
13    store: &impl Store,
14    title: &str,
15    section: Option<&str>,
16    meta: &myco_core::NewItemMeta,
17) -> Result<Item, String> {
18    let id = store.alloc_id(ItemKind::Task).map_err(|e| e.to_string())?;
19    // created deferred — backlog items are born parked
20    let mut item = Item::new(id, title)
21        .status(Status::Deferred)
22        .created(items::today());
23    meta.apply(&mut item)?;
24    let mut ledger = items::load_ledger(store)?;
25    ledger.put_backlog(item.clone(), section);
26    items::save_ledger(store, &ledger)?;
27    Ok(item)
28}

30/// The `backlog promote` mutation — `deferred → open`, moves the
31/// line (subtree included) into TODO.md. Caller commits.
32pub fn promote_item(store: &impl Store, id: ItemId) -> Result<Item, String> {
33    items::require_kind(id, ItemKind::Task, "backlog promote")?;
34    let mut ledger = items::load_ledger(store)?;
35    let loc = ledger
36        .locate(id)
37        .ok_or_else(|| format!("{id}: no such task"))?;
38    if !loc.backlog {
39        return Err(format!("{id} isn't in BACKLOG.md — nothing to promote"));
40    }
41    let mut item = ledger
42        .take(&loc)
43        .ok_or_else(|| format!("{id}: no such task"))?;
44    items::walk_to(&mut item, Status::Open)?;
45    // the subtree unparks with its parent — deferred
46    // children come back to open (TODO.md holds no
47    // deferred items)
48    items::walk_descendants_to(&mut item, Status::Open)?;
49    ledger.put_todo(item.clone(), None);
50    items::save_ledger(store, &ledger)?;
51    Ok(item)
52}

54/// The `backlog add` verb.
55pub fn h_add(ctx: &Ctx, a: &Args) -> Result<i32, String> {
56    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
57    let meta = items::meta_args(a)?;
58    let item = add_item(&store, a.text("title")?, a.opt("section"), &meta)?;
59    let committed = items::commit(
60        &proj.worktree,
61        &format!("chore: add {}", item.id),
62        meta.owner.as_ref(),
63        ctx.no_commit,
64    )?;
65    items::report(
66        json!({"item": item, "file": "BACKLOG.md"}),
67        committed,
68        ctx.json,
69        &format!("added {}", items::compact(&item)),
70    )
71}

73/// The `backlog promote` verb.
74pub fn h_promote(ctx: &Ctx, a: &Args) -> Result<i32, String> {
75    let (proj, store) = items::open(ctx.dir.as_deref(), ctx.root.as_deref())?;
76    let id = a.req::<ItemId>("id")?;
77    let item = promote_item(&store, id)?;
78    let committed = items::commit(
79        &proj.worktree,
80        &format!("chore: promote {id}"),
81        None,
82        ctx.no_commit,
83    )?;
84    items::report(
85        json!({"item": item, "file": "TODO.md"}),
86        committed,
87        ctx.json,
88        &format!("promoted {id} → TODO.md"),
89    )
90}

92/// The `backlog list` verb.
93pub fn h_list(ctx: &Ctx, a: &Args) -> Result<i32, String> {
94    let (_proj, store) = items::open_read(ctx.dir.as_deref(), ctx.root.as_deref())?;
95    let list = items::list_args(a)?;
96    let all = crate::todo::doc_items(&store, true)?;
97    let found: Vec<(usize, &Item)> = items::tree_view(&all)
98        .into_iter()
99        .filter(|(_, i)| list.matches(i))
100        .take(list.take())
101        .collect();
102    items::print_tree(&found, ctx.json)
103}
