1//! The single verb registry (P-007) — one [`VERBS`] table drives
2//! all three front-ends: the clap command tree ([`command`],
3//! builder mode, no derive), `manifest().verbs`/`implemented`
4//! ([`manifest_verbs`]), and serve/batch dispatch
5//! ([`crate::serve::Server::dispatch`]). A verb can't exist on one
6//! surface and not the others: `cli`/`wire` handlers share one
7//! `params` schema — the wire schema and the CLI surface are one
8//! declaration decoded into the same [`Args`] bag.
9//!
10//! Config-declared verbs (`[verbs]` in `myco.toml`) aren't in the
11//! table — they're data, resolved per call through
12//! `myco_ops::verb` and executed by `cfgverb`. Builtins always
13//! win a name collision.
14

29/// Which front-ends a param applies to — the CLI filter flags
30/// are `Cli`-only, `view`/`next`'s `spec` patch is `Wire`-only,
31/// everything else is `Both`.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum On {
34    /// CLI flag + wire key.
35    Both,
36    /// CLI only.
37    Cli,
38    /// Wire only.
39    Wire,
40}

42/// How a param is taken on the CLI — wire params decode by name
43/// regardless (positionals are required keys).
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub enum PKind {
46    /// A positional argument.
47    Pos,
48    /// A `--flag` boolean.
49    Flag,
50    /// A `--opt <value>`.
51    Opt,
52}

54/// The value type — drives the clap validator and the wire-side
55/// parse. `Spec` is wire-only: a `SpecPatch` object merged into a
56/// `ViewSpec` (`view`/`next`).
57#[derive(Debug, Clone, Copy, PartialEq, Eq)]
58pub enum Val {
59    /// Free text.
60    Str,
61    /// A filesystem path.
62    Path,
63    /// An `ItemId` (`T-001`, `ADR-0004`).
64    Id,
65    /// A status token (`in-progress`).
66    Status,
67    /// A priority (`low|med|high|crit`, names too).
68    Pri,
69    /// An actor id.
70    Actor,
71    /// A non-negative integer.
72    Count,
73    /// A sort key (`pri`, `-created`).
74    Sort,
75    /// An item kind (`task|issue|proposal|adr|plan`).
76    Kind,
77    /// A view source (`repo|index`).
78    Source,
79    /// A group axis (`status|priority|kind|owner|epic|milestone|repo`).
80    Axis,
81    /// A view format (`list|board|repos`).
82    Format,
83    /// A wire-only `ViewSpec` patch object.
84    Spec,
85}

87/// One declared param — CLI flag/positional *and* wire key.
88/// Fields are `pub` so the table reads as data; the `const`
89/// builders keep entries terse.
90#[derive(Debug, Clone)]
91pub struct Param {
92    /// The canonical name — clap gets `--name` (`_` → `-`), the
93    /// wire gets the `"name"` key.
94    pub name: &'static str,
95    /// Help text.
96    pub help: &'static str,
97    /// Pos / Flag / Opt.
98    pub kind: PKind,
99    /// The value type.
100    pub val: Val,
101    /// Which front-ends see it.
102    pub on: On,
103    /// Repeatable / multi-value (CLI `num_args`, wire array).
104    pub multi: bool,
105    /// Comma-separated values also accepted (CLI).
106    pub delimited: bool,
107    /// Values may start with `-` (CLI).
108    pub hyphen: bool,
109    /// Positionals are required by default; opts/flags aren't.
110    pub required: bool,
111    /// A baked default (`--to backlog`).
112    pub default: Option<&'static str>,
113    /// A short flag (`-m`).
114    pub short: Option<char>,
115    /// A visible alias (`--priority` for `--pri`).
116    pub alias: Option<&'static str>,
117    /// Mutually exclusive param name.
118    pub conflicts: Option<&'static str>,
119    /// `value_name` in help.
120    pub vname: Option<&'static str>,
121}

123const fn p(name: &'static str, kind: PKind, val: Val, help: &'static str) -> Param {
124    Param {
125        name,
126        help,
127        kind,
128        val,
129        on: On::Both,
130        multi: false,
131        delimited: false,
132        hyphen: false,
133        required: matches!(kind, PKind::Pos),
134        default: None,
135        short: None,
136        alias: None,
137        conflicts: None,
138        vname: None,
139    }
140}

222/// One decoded param value — strings keep the raw token (typed
223/// parses happen in accessors; `Spec` keeps the JSON object).
224#[derive(Debug, Clone)]
225pub enum ArgVal {
226    /// A `--flag` — `true` when present (wire: `"k": true`).
227    Flag(bool),
228    /// A single value.
229    One(String),
230    /// A multi value.
231    Many(Vec<String>),
232    /// A `SpecPatch` JSON object (wire-only).
233    Spec(Value),
234}

236/// The decoded argument bag one handler sees — identical shape
237/// whether it came from clap matches, wire JSON params, or an
238/// alias's baked argv.
239pub struct Args {
240    vals: BTreeMap<&'static str, ArgVal>,
241}
242
243/// `s` parses as `v`'s type — the clap validator and the wire
244/// check share it.
245fn validate(v: Val, s: &str) -> Result<(), String> {
246    macro_rules! p {
247        ($t:ty) => {
248            s.parse::<$t>().map(|_| ()).map_err(|e| e.to_string())
249        };
250    }
251    match v {
252        Val::Str | Val::Path | Val::Spec => Ok(()),
253        Val::Id => p!(myco_core::ItemId),
254        Val::Status => p!(myco_core::Status),
255        Val::Pri => p!(myco_core::Priority),
256        Val::Actor => p!(myco_core::Actor),
257        Val::Count => p!(u64),
258        Val::Sort => p!(myco_core::SortKey),
259        Val::Kind => p!(myco_core::ItemKind),
260        Val::Source => p!(myco_core::ViewSource),
261        Val::Axis => p!(myco_core::GroupAxis),
262        Val::Format => p!(myco_core::ViewFormat),
263    }
264}

266impl Args {
267    /// clap `ArgMatches` → the bag (CLI-facing params only).
268    pub fn from_matches(v: &Verb, m: &ArgMatches) -> Args {
269        let mut vals = BTreeMap::new();
270        for p in v.params() {
271            if p.on == On::Wire {
272                continue;
273            }
274            match (p.kind, p.multi) {
275                (PKind::Flag, _) => vals.insert(p.name, ArgVal::Flag(m.get_flag(p.name))),
276                (_, true) => {
277                    let many: Vec<String> = m
278                        .get_many::<String>(p.name)
279                        .map(|vs| vs.cloned().collect())
280                        .unwrap_or_default();
281                    vals.insert(p.name, ArgVal::Many(many))
282                }
283                _ => {
284                    if let Some(s) = m.get_one::<String>(p.name) {
285                        vals.insert(p.name, ArgVal::One(s.clone()))
286                    } else {
287                        continue;
288                    }
289                }
290            };
291        }
292        Args { vals }
293    }

295    /// Wire `params` → the bag (wire-facing params only; unknown
296    /// keys and bad values are decode errors — `-32602` at the
297    /// caller). `null`/`{}` are both "no params".
298    pub fn from_json(v: &Verb, params: &Value) -> Result<Args, String> {
299        let mut vals = BTreeMap::new();
300        let obj = if params.is_null() {
301            serde_json::Map::new()
302        } else {
303            params
304                .as_object()
305                .ok_or_else(|| "params must be an object".to_string())?
306                .clone()
307        };
308        for key in obj.keys() {
309            if v.params().all(|p| p.on == On::Cli || p.name != key) {
310                return Err(format!("unknown param \"{key}\""));
311            }
312        }
313        for p in v.params() {
314            if p.on == On::Cli {
315                continue;
316            }
317            let Some(raw) = obj.get(p.name) else {
318                if p.required {
319                    return Err(format!("missing required param \"{}\"", p.name));
320                }
321                continue;
322            };
323            let entry = match (p.kind, p.multi) {
324                (PKind::Flag, _) => ArgVal::Flag(
325                    raw.as_bool()
326                        .ok_or_else(|| format!("\"{}\" must be a boolean", p.name))?,
327                ),
328                (_, true) => {
329                    let items: Vec<String> = match raw {
330                        Value::Array(xs) => xs.iter().map(json_scalar).collect::<Result<_, _>>()?,
331                        other => vec![json_scalar(other)?],
332                    };
333                    for s in &items {
334                        validate(p.val, s).map_err(|e| format!("\"{}\": {e}", p.name))?;
335                    }
336                    ArgVal::Many(items)
337                }
338                _ => {
339                    if p.val == Val::Spec {
340                        ArgVal::Spec(raw.clone())
341                    } else {
342                        let s = json_scalar(raw)?;
343                        validate(p.val, &s).map_err(|e| format!("\"{}\": {e}", p.name))?;
344                        ArgVal::One(s)
345                    }
346                }
347            };
348            vals.insert(p.name, entry);
349        }
350        Ok(Args { vals })
351    }

353    /// Baked/interactive argv tokens → the bag — the alias path.
354    /// Global flags (`--json`, `--no-commit`, `--dir`, `--root`)
355    /// fold into `ctx` rather than the bag. `wire` restricts to
356    /// wire-facing params: a CLI-only flag in an alias's args is
357    /// a dispatch error on the wire, not a silent drop.
358    pub fn from_argv(
359        v: &Verb,
360        tokens: &[String],
361        ctx: &mut Ctx,
362        wire: bool,
363    ) -> Result<Args, String> {
364        let mut vals = BTreeMap::new();
365        let mut positional = 0usize;
366        let mut i = 0;
367        let mut no_flags = false;
368        let set =
369            |vals: &mut BTreeMap<&'static str, ArgVal>, p: &Param, s: &str| -> Result<(), String> {
370                validate(p.val, s).map_err(|e| format!("--{}: {e}", p.name))?;
371                match (p.kind, p.multi) {
372                    (_, true) => match vals.entry(p.name).or_insert_with(|| ArgVal::Many(vec![])) {
373                        ArgVal::Many(xs) => xs.extend(s.split(',').map(str::to_string)),
374                        _ => unreachable!("multi param bag is Many"),
375                    },
376                    _ => {
377                        vals.insert(p.name, ArgVal::One(s.to_string()));
378                    }
379                }
380                Ok(())
381            };
382        while i < tokens.len() {
383            let t = &tokens[i];
384            i += 1;
385            if !no_flags && t == "--" {
386                no_flags = true;
387                continue;
388            }
389            if !no_flags && t.starts_with("--") {
390                let body = &t[2..];
391                let (key, inline) = match body.split_once('=') {
392                    Some((k, x)) => (k, Some(x.to_string())),
393                    None => (body, None),
394                };
395                // global flags fold into ctx
396                match key {
397                    "json" => {
398                        ctx.json = true;
399                        continue;
400                    }
401                    "no-commit" => {
402                        ctx.no_commit = true;
403                        continue;
404                    }
405                    "dir" | "root" => {
406                        let val = take_value(key, inline, tokens, &mut i)?;
407                        let p = PathBuf::from(val);
408                        if key == "dir" {
409                            ctx.dir = Some(p);
410                        } else {
411                            ctx.root = Some(p);
412                        }
413                        continue;
414                    }
415                    _ => {}
416                }
417                let name = key.replace('-', "_");
418                let p = v
419                    .params()
420                    .find(|p| {
421                        p.name.replace('_', "-") == key || p.name == name || p.alias == Some(key)
422                    })
423                    .ok_or_else(|| format!("unknown argument \"--{key}\" for \"{}\"", v.name))?;
424                if wire && p.on == On::Cli {
425                    return Err(format!(
426                        "argument \"--{key}\" isn't serveable — \"{}\" is CLI-only",
427                        p.name
428                    ));
429                }
430                if p.kind == PKind::Flag {
431                    if let Some(x) = inline {
432                        ctx_bool(&x)
433                            .map(|b| vals.insert(p.name, ArgVal::Flag(b)))
434                            .map_err(|e| format!("--{key}: {e}"))?;
435                    } else {
436                        vals.insert(p.name, ArgVal::Flag(true));
437                    }
438                } else {
439                    let val = take_value(key, inline, tokens, &mut i)?;
440                    set(&mut vals, p, &val)?;
441                }
442            } else if !no_flags && t.starts_with('-') && t.len() > 1 && !looks_negative(t) {
443                // short flag — `-m msg`
444                let c = t.chars().nth(1).unwrap_or('-');
445                let p = v
446                    .params()
447                    .find(|p| p.short == Some(c))
448                    .ok_or_else(|| format!("unknown argument \"{t}\" for \"{}\"", v.name))?;
449                let inline = t.get(2..).map(str::to_string);
450                let val = take_value(&c.to_string(), inline, tokens, &mut i)?;
451                set(&mut vals, p, &val)?;
452            } else {
453                // positional — next unfilled pos param
454                let pos_params: Vec<&Param> = v.params().filter(|p| p.kind == PKind::Pos).collect();
455                let p = pos_params
456                    .iter()
457                    .enumerate()
458                    .find(|(n, _)| *n == positional)
459                    .map(|(_, p)| *p)
460                    .or_else(|| pos_params.last().filter(|p| p.multi).copied());
461                match p {
462                    Some(p) => {
463                        if !p.multi {
464                            positional += 1;
465                        } else if positional < pos_params.len() - 1 {
466                            positional = pos_params.len() - 1;
467                        }
468                        set(&mut vals, p, t)?;
469                    }
470                    None => {
471                        return Err(format!(
472                            "unexpected argument \"{t}\" — \"{}\" takes no positionals",
473                            v.name
474                        ))
475                    }
476                }
477            }
478        }
479        for p in v.params() {
480            if p.required && !vals.contains_key(p.name) {
481                return Err(format!(
482                    "missing required \"{}\" for \"{}\"",
483                    p.name, v.name
484                ));
485            }
486            if let Some(d) = p.default {
487                vals.entry(p.name)
488                    .or_insert_with(|| ArgVal::One(d.to_string()));
489            }
490        }
491        Ok(Args { vals })
492    }

545    /// Every value of a multi param parsed as `T`.

547    where
548        T::Err: std::fmt::Display,
549    {
550        self.many(name)
551            .iter()
552            .map(|s| s.parse().map_err(|e: T::Err| format!("\"{name}\": {e}")))
553            .collect()
554    }

556    /// A single `T` out of a param that may be multi (wire sends
557    /// one id even where the CLI takes several).
558    pub fn one<T: FromStr>(&self, name: &str) -> Result<T, String>
559    where
560        T::Err: std::fmt::Display,
561    {
562        let xs = self.many_parsed::<T>(name)?;
563        match xs.len() {
564            1 => Ok(xs.into_iter().next().unwrap_or_else(|| unreachable!())),
565            0 => Err(format!("missing required \"{name}\"")),
566            _ => Err(format!("\"{name}\" takes a single value")),
567        }
568    }

570    /// The `Spec` wire object, when present.
571    pub fn spec(&self, name: &str) -> Result<Option<crate::serve::SpecPatch>, String> {
572        match self.get(name) {
573            Some(ArgVal::Spec(v)) => serde_json::from_value(v.clone())
574                .map(Some)
575                .map_err(|e| format!("\"{name}\": {e}")),
576            _ => Ok(None),
577        }
578    }

581/// `--flag=<val>` inline values parse as booleans.
582fn ctx_bool(s: &str) -> Result<bool, String> {
583    match s {
584        "true" | "1" | "yes" => Ok(true),
585        "false" | "0" | "no" => Ok(false),
586        _ => Err(format!("expected a boolean, got {s:?}")),
587    }
588}

590/// `-5`-looking tokens are values, not flags.
591fn looks_negative(t: &str) -> bool {
592    t[1..].chars().next().is_some_and(|c| c.is_ascii_digit())
593}
594
595/// `--opt value` / `--opt=value` — the value half.
596fn take_value(
597    key: &str,
598    inline: Option<String>,
599    tokens: &[String],
600    i: &mut usize,
601) -> Result<String, String> {
602    if let Some(v) = inline {
603        return Ok(v);
604    }
605    let v = tokens
606        .get(*i)
607        .ok_or_else(|| format!("\"--{key}\" needs a value"))?;
608    *i += 1;
609    Ok(v.clone())
610}
611
612/// A wire scalar → its string form (numbers stringify so
613/// `"limit": 5` and `"5"` decode alike).
614fn json_scalar(v: &Value) -> Result<String, String> {
615    match v {
616        Value::String(s) => Ok(s.clone()),
617        Value::Number(n) => Ok(n.to_string()),
618        Value::Bool(b) => Ok(b.to_string()),
619        _ => Err(format!("expected a string or number, got {v}")),
620    }
621}

638/// The CLI handler — decoded args in, exit code out.

643/// One verb — name, help, param schema, kind-lock, mutability,
644/// and its front-end handlers.
645pub struct Verb {
646    /// The canonical name — the serve method name; `.` splits the
647    /// CLI path (`todo.add` → `myco todo add`).
648    pub name: &'static str,
649    /// Help text (clap `about` + generated docs).
650    pub about: &'static str,
651    /// The param schema — shared by the clap tree, the wire
652    /// decoder, and (for aliases) argv baking. Slices-of-slices so
653    /// shared bundles (`META`, `LIST`, `FILTER`) declare once.
654    pub params: &'static [&'static [Param]],
655    /// The item kind the verb family locks to, when one exists —
656    /// part of the declared schema (P-007); the kind-aware bits
657    /// that read it (alias lint, completions) land with them.
658    #[allow(dead_code)]
659    pub kind_lock: Option<myco_core::ItemKind>,
660    /// Can write the ledger/index — alias targets must be
661    /// read-only, so `mutates` disqualifies a builtin.

663    /// Hidden from help (docs pipeline verbs still work).

1846/// `Param` → a clap `Arg` — validator per `Val`, multi/delimited/
1847/// hyphen/conflicts honored, `_` → `-` on the long name.
1848fn param_arg(p: &Param) -> Arg {
1849    // `--pri-min` for `pri_min` — the set is tiny and built once,
1850    // so leaking the few `String`s beats a `Str`-typed field
1851    let long: &'static str = if p.name.contains('_') {
1852        Box::leak(p.name.replace('_', "-").into_boxed_str())
1853    } else {
1854        p.name
1855    };
1856    let val = p.val;
1857    let mut a = Arg::new(p.name).help(p.help);
1858    if p.kind != PKind::Flag {
1859        a = a.value_parser(move |s: &str| validate(val, s).map(|()| s.to_string()));
1860    }
1861    match p.kind {
1862        PKind::Pos => {
1863            a = a.required(p.required);
1864            if p.multi {
1865                a = a.num_args(1..);
1866            }
1867        }
1868        PKind::Flag => {
1869            a = a.long(long).action(ArgAction::SetTrue);
1870        }
1871        PKind::Opt => {
1872            a = a.long(long).action(if p.multi {
1873                ArgAction::Append
1874            } else {
1875                ArgAction::Set
1876            });
1877        }
1878    }
1879    if p.delimited {
1880        a = a.value_delimiter(',');
1881    }
1882    if p.hyphen {
1883        a = a.allow_hyphen_values(true);
1884    }
1885    if p.required && p.kind == PKind::Opt {
1886        a = a.required(true);
1887    }
1888    if let Some(d) = p.default {
1889        a = a.default_value(d);
1890    }
1891    if let Some(c) = p.short {
1892        a = a.short(c);
1893    }
1894    if let Some(al) = p.alias {
1895        a = a.visible_alias(al);
1896    }
1897    if let Some(c) = p.conflicts {
1898        a = a.conflicts_with(c);
1899    }
1900    if let Some(v) = p.vname {
1901        a = a.value_name(v);
1902    }
1903    a
1904}

1906/// One verb → its clap `Command` (leaf name after `.`).
1907fn verb_command(v: &Verb) -> Command {
1908    let leaf = v.name.rsplit('.').next().unwrap_or(v.name);
1909    let mut c = Command::new(leaf).about(v.about).hide(v.hidden);
1910    for p in v.params() {
1911        if p.on == On::Wire {
1912            continue;
1913        }
1914        c = c.arg(param_arg(p));
1915    }
1916    c
1917}

1946/// The full `myco` command — registry verbs under their groups,
1947/// global flags on top, and `allow_external_subcommands` so a
1948/// `[verbs.<name>]` config verb parses as `myco <name> [args…]`.
1949pub fn command() -> Command {
1950    let mut cmd = Command::new("myco")
1951        .version(env!("CARGO_PKG_VERSION"))
1952        .about("myco — agent-first project management for a repo's `project/` worktree.")
1953        .allow_external_subcommands(true)
1954        .subcommand_required(true)
1955        .args(global_args());
1956    for g in GROUPS {
1957        let mut gc = Command::new(g.name)
1958            .about(g.about)
1959            .subcommand_required(true);
1960        for v in VERBS.iter().filter(|v| {
1961            v.cli.is_some()
1962                && v.name.starts_with(g.name)
1963                && v.name.as_bytes().get(g.name.len()) == Some(&b'.')
1964        }) {
1965            gc = gc.subcommand(verb_command(v));
1966        }
1967        cmd = cmd.subcommand(gc);
1968    }
1969    for v in VERBS
1970        .iter()
1971        .filter(|v| v.cli.is_some() && !v.name.contains('.'))
1972    {
1973        cmd = cmd.subcommand(verb_command(v));
1974    }
1975    cmd
1976}

1980/// The top-level dispatch — builtin verb (leaf or group),
1981/// else a `[verbs.<name>]` config-declared verb.
1982pub fn dispatch(ctx: &Ctx, m: &ArgMatches) -> Result<i32, String> {
1983    let Some((name, sub)) = m.subcommand() else {
1984        return Err("a subcommand is required — `myco help` lists verbs".into());
1985    };
1986    if let Some(v) = find(name) {
1987        return run_builtin(ctx, v, sub);
1988    }
1989    if GROUPS.iter().any(|g| g.name == name) {
1990        let Some((leaf, lm)) = sub.subcommand() else {
1991            return Err(format!("myco {name}: a subcommand is required"));
1992        };
1993        let dotted = format!("{name}.{leaf}");
1994        let v =
1995            find(&dotted).ok_or_else(|| format!("myco {name}: unknown subcommand \"{leaf}\""))?;
1996        return run_builtin(ctx, v, lm);
1997    }
1998    // external subcommand — a config-declared verb (or a typo)
1999    let tokens: Vec<String> = sub
2000        .get_many::<OsString>("")
2001        .map(|vs| vs.map(|s| s.to_string_lossy().into_owned()).collect())
2002        .unwrap_or_default();
2003    crate::cfgverb::run(ctx, name, &tokens)
2004}
2005
2006/// Builtin leaf — warn when a config verb shadowed it (the
2007/// builtin wins; the warning mirrors view-shadowing).
2008fn run_builtin(ctx: &Ctx, v: &Verb, sub: &ArgMatches) -> Result<i32, String> {
2009    if !v.name.contains('.') {
2010        crate::cfgverb::warn_if_shadowed(ctx, v.name);
2011    }
2012    let cli = v
2013        .cli
2014        .ok_or_else(|| format!("verb \"{}\" has no CLI handler", v.name))?;
2015    cli(ctx, &Args::from_matches(v, sub))
2016}
