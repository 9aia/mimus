1//! `myco sync` — reconcile the ledger branch with its remote.
2//!
3//! The `project` branch is the audit trail, and it's shared: two
4//! clones committing independently fork it. Sync is the gate
5//! mutating verbs enforce (`ensure_synced` refuses while behind)
6//! and the healer: fetch, fast-forward when possible, rebase
7//! local commits onto `origin/<branch>` when diverged, then repair
8//! what a merge can break — `meta.toml` counters reconcile via
9//! `check --fix`, and duplicate ids (two writers minting the same
10//! `T-NNN`) are renumbered remote-wins by
11//! [`crate::check::renumber_duplicates`].
12//!
13//! A rebase conflict stops mid-flight with the conflicted files
14//! reported — resolve them (`git add` + `git rebase --continue`)
15//! and run `myco sync` again, or `myco sync --abort` to bail.
16
17use serde_json::json;

19use crate::discover;
20use crate::ops;

22/// The `sync` verb.
23pub fn h(ctx: &crate::registry::Ctx, a: &crate::registry::Args) -> Result<i32, String> {
24    let json_out = ctx.json;
25    let no_commit = ctx.no_commit;
26    let proj = discover::discover(ctx.dir.as_deref(), ctx.root.as_deref())?;
27    let wt = proj.worktree.clone();
28
29    // --abort bails a stopped rebase/merge without touching refs
30    if a.flag("abort") {
31        myco_ops::vcs::rebase_abort(&wt).map_err(|e| format!("rebase --abort: {e}"))?;
32        if json_out {
33            ops::print_json(&json!({"action": "abort"}));
34        } else {
35            println!("myco sync: rebase aborted — ledger back at its pre-sync tip");
36        }
37        return Ok(0);
38    }
39
40    let _lock = ops::lock_worktree(&wt)?;
41
42    // a rebase already stopped mid-flight: report state, don't restart
43    if myco_ops::vcs::rebase_in_progress(&wt) {
44        let conflicts = myco_ops::vcs::unmerged_files(&wt);
45        return finish(
46            json_out,
47            "conflict",
48            &conflicts,
49            &[],
50            1,
51            "myco sync: rebase is stopped on conflicts — resolve the files, \
52             `git add` them, `git rebase --continue`, then `myco sync` again \
53             (or `myco sync --abort` to bail)",
54        );
55    }
56
57    let Some(branch) = myco_ops::vcs::worktree_branch(&wt) else {
58        return Err(format!(
59            "{} is not on a branch — run `myco check --fix` to reattach it",
60            wt.display()
61        ));
62    };
63    let remote = format!("origin/{branch}");
64    let remote_ref = format!("refs/remotes/{remote}");
65
66    // fetch is best-effort — offline sync still reconciles against
67    // the last-known remote refs; a failed fetch is reported, not fatal
68    let fetched = myco_ops::vcs::sync(&proj.root);
69    let offline = if fetched {
70        ""
71    } else {
72        " (fetch failed — offline?)"
73    };
74    if !myco_ops::vcs::has_commit(&proj.root, &remote_ref) {
75        return finish(
76            json_out,
77            "no-remote",
78            &[],
79            &[],
80            0,
81            &format!(
82                "myco sync: no `{remote}` ref — the ledger was never published; push it when ready"
83            ),
84        );
85    }
86
87    let ahead = myco_ops::vcs::commits_ahead(&wt, &remote_ref).unwrap_or(0);
88    let behind = myco_ops::vcs::commits_behind(&wt, &remote_ref).unwrap_or(0);
89
90    // fast-forward when local has nothing to replay; rebase the
91    // local commits onto the remote tip when diverged. Up-to-date
92    // skips the git op but still runs the heal tail — a conflicted
93    // rebase resumed by hand (`--continue`) lands here with the
94    // duplicate ids it produced still in the tree.
95    let action = if behind == 0 {
96        "up-to-date"
97    } else if ahead == 0 {
98        "ff"
99    } else {
100        "rebase"
101    };
102    if behind > 0 {
103        let result = if ahead == 0 {
104            myco_ops::vcs::merge_ff(&wt, &remote)
105        } else {
106            myco_ops::vcs::rebase(&wt, &remote)
107        };
108        if let Err(e) = result {
109            let conflicts = myco_ops::vcs::unmerged_files(&wt);
110            if conflicts.is_empty() {
111                return Err(format!("git {action} onto `{remote}` failed: {e}"));
112            }
113            return finish(
114                json_out,
115                "conflict",
116                &conflicts,
117                &[],
118                1,
119                "myco sync: rebase stopped on conflicts — resolve the files, \
120                 `git add` them, `git rebase --continue`, then `myco sync` again \
121                 (or `myco sync --abort` to bail)",
122            );
123        }
124    }
125
126    // heal what the merge broke: duplicate ids get renumbered
127    // (remote wins), counters/refs/drift get repaired — the check
128    // --fix tail owns its own lock and commit, so drop ours first
129    let kinds = crate::check::repo_kinds(&proj.root);
130    let (renumbered, renum_report) = crate::check::renumber_duplicates(&wt, &kinds, &remote)
131        .unwrap_or_else(|e| (0, vec![format!("renumber failed: {e}")]));
132    let committed = renumbered > 0
133        && ops::autocommit(
134            &wt,
135            "chore: renumber colliding ids (myco sync)",
136            None,
137            ops::commits_enabled(no_commit),
138        )
139        .unwrap_or(false);
140    drop(_lock);
141
142    // the canonical repair pass — counters, emit drift, filename
143    // drift, and the final verdict all ride `check --fix`. It runs
144    // in human mode even under `--json`: its findings go to stderr,
145    // keeping stdout a single sync payload.
146    let check_code = crate::check::run_on(&proj, true, false, no_commit)?;
147    finish(
148        json_out,
149        action,
150        &[],
151        &renum_report,
152        check_code,
153        &format!(
154            "myco sync: {action} onto `{remote}` ({behind} behind, {ahead} ahead){offline}{}",
155            if committed {
156                " — colliding ids renumbered"
157            } else {
158                ""
159            }
160        ),
161    )
162}

164/// Shared exit — JSON payload or one human line, plus the code.
165fn finish(
166    json_out: bool,
167    action: &str,
168    conflicts: &[String],
169    renumbered: &[String],
170    code: i32,
171    human: &str,
172) -> Result<i32, String> {
173    if json_out {
174        ops::print_json(&json!({
175            "action": action,
176            "conflicts": conflicts,
177            "renumbered": renumbered,
178        }));
179    } else {
180        println!("{human}");
181        for c in conflicts {
182            println!("  conflict: {c}");
183        }
184        for r in renumbered {
185            println!("  renumbered: {r}");
186        }
187    }
188    Ok(code)
189}
