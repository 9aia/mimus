1//! `myco item move` — the cross-repo transfer. The item is
2//! re-created in the destination ledger's own id sequence (kind,
3//! title, body, `pri`/`owner`/`#label`/`epic`/`ms` carried;
4//! `blocks:`/`links:` dropped — they name source-ledger ids), both ledgers
5//! get a provenance handoff (`moved from`/`moved to`), then the
6//! source walks to `done`: a file-backed item keeps its file with
7//! status `done`; a task's line leaves the ledger and its handoff
8//! file drops to `handoffs/archive/` — the `check`-prescribed
9//! home for items that are gone. Two commits, one per worktree,
10//! both under the usual `--no-commit`/`$MYCO_NO_COMMIT` rules.
11//!
12//! There's no distributed transaction across the two worktrees —
13//! the destination half lands first (a copy left behind beats an
14//! item deleted on both ends), and a source-side failure reports
15//! the new id so the close can be finished by hand.
16

26/// What a move produced — the CLI report and the wire payload
27/// share it.
28pub(crate) struct Moved {
29    /// The source id (closed `done` / line removed).
30    pub from: ItemId,
31    /// The destination copy — fresh id in the dest sequence.
32    pub item: Item,
33    /// Destination repo label (`meta.toml` name, else the root
34    /// basename).
35    pub repo: String,
36    /// The destination repo root.
37    pub root: PathBuf,
38    /// The source-side handoff entry — the audit pointer.
39    pub handoff: HandoffEntry,
40    /// Source-worktree autocommit result.
41    pub committed_source: bool,
42    /// Destination-worktree autocommit result.
43    pub committed_dest: bool,
44    /// `--finish` found the source already closed — only commits
45    /// (if any) were needed.
46    pub already_closed: bool,
47}

49/// Canonical form for comparisons — never fails, like discovery's.
50fn canon(p: &Path) -> PathBuf {
51    p.canonicalize().unwrap_or_else(|_| p.to_path_buf())
52}
53
54/// Same ledger on both ends — the same worktree path, or two
55/// worktrees sharing one git common dir (linked worktrees of a
56/// single repo). Plain-dir ledgers (no git) compare by path only.
57fn same_repo(a: &Path, b: &Path) -> bool {
58    if canon(a) == canon(b) {
59        return true;
60    }
61    match (myco_ops::vcs::common_dir(a), myco_ops::vcs::common_dir(b)) {
62        (Ok(x), Ok(y)) => x == y,
63        _ => false,
64    }
65}

67/// `--to` → the destination project. A path resolves like
68/// `index add`'s arg (repo root *or* its `project/` worktree); a
69/// non-path falls back to the registry's repo *names* (`myco
70/// index list`).
71fn resolve_dest(to: &str) -> Result<discover::Project, String> {
72    if let Ok(p) = indexcmd::repo_at(Path::new(to)) {
73        return Ok(p);
74    }
75    if let Ok(index) = indexcmd::open_index() {
76        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
77        if let Ok(uni) = myco_ops::universe::repos(&cwd, &index, crate::ops::CONFIG) {
78            if let Some(u) = uni.iter().find(|u| u.entry.name == to) {
79                return indexcmd::repo_at(u.entry.path.as_std_path());
80            }
81        }
82    }
83    Err(format!(
84        "{to}: not a myco repo — pass a repo root, its project/ worktree, \
85         or a repo name from `myco index list`"
86    ))
87}

89/// The `repo` half of the `repo:ID` pointer — `meta.toml`'s name,
90/// else the root basename.
91fn repo_label(store: &impl Store, root: &Path) -> String {
92    store
93        .meta()
94        .map(|m| m.name)
95        .ok()
96        .filter(|n| !n.is_empty())
97        .or_else(|| root.file_name().map(|n| n.to_string_lossy().into_owned()))
98        .unwrap_or_else(|| "repo".into())
99}

101/// The metadata that survives the crossing. `blocks:`/`links:` are
102/// dropped on purpose — they name source-ledger ids and would
103/// dangle in the destination (`check` would flag them); the drop
104/// is recorded as plain text in the provenance handoffs
105/// ([`dropped_refs_line`]).
106fn carried_meta(src: &Item) -> NewItemMeta {
107    NewItemMeta {
108        priority: Some(src.priority),
109        owner: src.owner.clone(),
110        labels: src.labels.iter().cloned().collect(),
111        blocks: Vec::new(),
112        links: Vec::new(),
113        epic: src.epic.clone(),
114        milestone: src.milestone.clone(),
115        due: src.due,
116        estimate: src.estimate,
117        cycle: src.cycle.clone(),
118    }
119}

121/// Re-create `src` in `dest` — a task lands at TODO.md's tail via
122/// the `todo add` path; file kinds go through `save_item` with the
123/// source body verbatim. The id is the destination sequence's
124/// next; `created` carries over (a moved item keeps its filing
125/// date). Caller commits.
126fn create_in(dest: &impl Store, src: &Item, body: Option<&str>) -> Result<Item, String> {
127    let meta = carried_meta(src);
128    if src.id.kind() == ItemKind::Task {
129        let (item, _file) = todo::add_item(dest, &src.title, None, None, &meta)?;
130        return Ok(item);
131    }
132    let id = dest.alloc_id(src.id.kind()).map_err(|e| e.to_string())?;
133    let mut item = Item::new(id, &src.title).created(src.created.unwrap_or_else(items::today));
134    meta.apply(&mut item)?;
135    let file = ItemFile::new(item.clone()).body(body.unwrap_or_default().to_string());
136    dest.save_item(&file).map_err(|e| e.to_string())?;
137    Ok(item)
138}

140/// The `dropped refs:` note line — the source-ledger ids that
141/// didn't carry (`blocks:`/`links:` name ids in the *source*
142/// sequence). `None` when there was nothing to drop.
143fn dropped_refs_line(src: &Item) -> Option<String> {
144    let mut parts = Vec::new();
145    if !src.blocks.is_empty() {
146        parts.push(format!(
147            "blocks {}",
148            src.blocks
149                .iter()
150                .map(|i| i.to_string())
151                .collect::<Vec<_>>()
152                .join(", ")
153        ));
154    }
155    if !src.links.is_empty() {
156        parts.push(format!(
157            "links {}",
158            src.links
159                .iter()
160                .map(|i| i.to_string())
161                .collect::<Vec<_>>()
162                .join(", ")
163        ));
164    }
165    (!parts.is_empty()).then(|| format!("dropped refs: {}", parts.join("; ")))
166}

168/// `moved to <repo>:<ID>` pointer parse — the first line of a
169/// source-side handoff note, ` — <extra>` stripped, plus the
170/// `dest: <root>` resume line later moves carry (the canonical
171/// destination root — the label alone only resolves through the
172/// repo index). The repo label may itself contain `:`; the id is
173/// always the last `:`-separated token.
174fn parse_moved_to(note: &str) -> Option<(String, ItemId, Option<PathBuf>)> {
175    let line = note.lines().next()?;
176    let head = line.strip_prefix("moved to ")?;
177    let head = head.split(" — ").next().unwrap_or(head);
178    let (label, id) = head.rsplit_once(':')?;
179    let id = id.trim().parse().ok()?;
180    let root = note
181        .lines()
182        .find_map(|l| l.strip_prefix("dest: "))
183        .map(PathBuf::from);
184    Some((label.to_string(), id, root))
185}

187/// The source close after the audit handoff lands. A task's line
188/// is taken out of its ledger doc and its handoff file moves to
189/// `handoffs/archive/` (where `check` wants the files of gone
190/// items); a file-backed item walks the state machine to `done`
191/// (an already-done item just keeps its file).
192fn close_source(store: &impl Store, worktree: &Path, id: ItemId) -> Result<(), String> {
193    if id.kind() == ItemKind::Task {
194        let mut ledger = items::load_ledger(store)?;
195        let loc = ledger
196            .locate(id)
197            .ok_or_else(|| format!("{id}: no such task"))?;
198        ledger.take(&loc);
199        items::save_ledger(store, &ledger)?;
200        // the item is gone — its handoff goes cold, exactly the
201        // rename `check --fix` prescribes for orphan files. Both
202        // files existing is an already-inconsistent ledger: don't
203        // clobber the cold one, leave the live file for `check`
204        // to flag.
205        let name = myco_store_md::handoff_filename(&id);
206        let live = worktree.join("handoffs").join(&name);
207        let archive = worktree.join("handoffs").join("archive").join(&name);
208        if live.is_file() && !archive.exists() {
209            std::fs::create_dir_all(archive.parent().unwrap_or(worktree))
210                .map_err(|e| format!("handoffs/archive: {e}"))?;
211            std::fs::rename(&live, &archive).map_err(|e| format!("archive {name}: {e}"))?;
212        }
213        return Ok(());
214    }
215    let mut file = items::item_for_edit(store, id)?;
216    if file.meta.status != Status::Done {
217        items::walk_to(&mut file.meta, Status::Done)?;
218        store.save_item(&file).map_err(|e| e.to_string())?;
219    }
220    Ok(())
221}

223/// The move itself — shared by the CLI handler and `item.move` on
224/// the wire. `dir`/`root` are the global discovery flags (the
225/// source side); `to` resolves the destination.
226pub(crate) fn do_move(
227    dir: Option<&Path>,
228    root: Option<&Path>,
229    id: ItemId,
230    to: &str,
231    note: Option<&str>,
232    actor: Option<Actor>,
233    no_commit: bool,
234) -> Result<Moved, String> {
235    let (src_proj, src) = items::open(dir, root)?;
236    let (item, _location, body) =
237        items::find_item(&src, id)?.ok_or_else(|| format!("{id}: no such item"))?;
238    if src.is_archived(id).map_err(|e| e.to_string())? {
239        return Err(format!("{id} is archived — unarchive it before moving"));
240    }
241    if !item.children.is_empty() {
242        return Err(format!(
243            "{id} has {} child item(s) — move them individually first \
244             (`done`/`defer` acts on the whole subtree)",
245            item.children.len()
246        ));
247    }
248    let dest_proj = resolve_dest(to)?;
249    if same_repo(&src_proj.worktree, &dest_proj.worktree) {
250        return Err(format!(
251            "{to}: the destination is this repo — move is cross-repo only"
252        ));
253    }
254    let (dest_proj, dst) = items::open(Some(&dest_proj.worktree), Some(&dest_proj.root))?;
255    let src_name = repo_label(&src, &src_proj.root);
256    let dest_name = repo_label(&dst, &dest_proj.root);
257    // one resolved actor for both worktrees' handoffs + commits
258    let me = ops::resolve_actor(Some(&src_proj.worktree), actor.as_ref());
259
260    // destination first: create + provenance. `blocks:`/`links:`
261    // name source-ledger ids — they don't carry; the drop rides
262    // both provenance handoffs as plain text.
263    let new_item =
264        create_in(&dst, &item, body.as_deref()).map_err(|e| format!("{dest_name}: {e}"))?;
265    let dropped = dropped_refs_line(&item);
266    let mut from_note = format!("moved from {src_name}:{id}");
267    if let Some(d) = &dropped {
268        from_note.push_str(&format!("\n{d}"));
269    }
270    continuity::record_handoff(&dst, &dest_proj.worktree, new_item.id, from_note, &me, None)
271        .map_err(|e| format!("{dest_name}: {e}"))?;
272
273    // then the source audit pointer — written *before* the dest
274    // commit so a crash mid-move leaves the resume pointer on
275    // disk (`item move <id> --finish` picks it up)
276    let mut note_text = format!("moved to {dest_name}:{}", new_item.id);
277    if let Some(extra) = note.map(str::trim).filter(|n| !n.is_empty()) {
278        note_text.push_str(&format!(" — {extra}"));
279    }
280    // the resume handle for `--finish` — the canonical root,
281    // since the label only resolves through the repo index
282    note_text.push_str(&format!("\ndest: {}", dest_proj.root.display()));
283    if let Some(d) = &dropped {
284        note_text.push_str(&format!("\n{d}"));
285    }
286    let left = |e: String| {
287        format!(
288            "{e} — destination {dest_name}:{} was created; \
289             run `myco item move {id} --finish` to complete the source close",
290            new_item.id
291        )
292    };
293    let (handoff, _rel) =
294        continuity::record_handoff(&src, &src_proj.worktree, id, note_text, &me, None)
295            .map_err(left)?;
296    let committed_dest = items::commit(
297        &dest_proj.worktree,
298        &format!("chore: move {} in from {src_name}:{id}", new_item.id),
299        Some(&me),
300        no_commit,
301    )?;
302
303    // source close last — a failure here is resumable via
304    // `--finish`, which finds the pointer above
305    close_source(&src, &src_proj.worktree, id).map_err(left)?;
306    let committed_source = items::commit(
307        &src_proj.worktree,
308        &format!("chore: move {id} to {dest_name}:{}", new_item.id),
309        Some(&me),
310        no_commit,
311    )?;
312
313    Ok(Moved {
314        from: id,
315        item: new_item,
316        repo: dest_name,
317        root: dest_proj.root.clone(),
318        handoff,
319        committed_source,
320        committed_dest,
321        already_closed: false,
322    })
323}

325/// A read-only preview of [`do_move`] — the same guards, nothing
326/// written. The prospective destination id comes from the dest
327/// registry (in-memory only — no `meta.toml` touched).
328pub(crate) struct MovePlan {
329    /// Source id.
330    pub from: ItemId,
331    /// The source item.
332    pub item: Item,
333    /// Source repo label.
334    pub src_name: String,
335    /// Destination label + root (`to` resolved).
336    pub repo: String,
337    pub root: PathBuf,
338    /// The id the destination copy would get.
339    pub dest_id: ItemId,
340    /// `blocks:`/`links:` that won't carry — plain text for the
341    /// provenance notes.
342    pub dropped: Option<String>,
343    /// User-supplied extra note (as given, `--note`).
344    pub note: Option<String>,
345}

347/// `item move --dry-run` — validate the whole move, report the
348/// plan. No writes anywhere.
349pub(crate) fn move_plan(
350    dir: Option<&Path>,
351    root: Option<&Path>,
352    id: ItemId,
353    to: &str,
354    note: Option<&str>,
355) -> Result<MovePlan, String> {
356    let (src_proj, src) = items::open(dir, root)?;
357    let (item, _location, _body) =
358        items::find_item(&src, id)?.ok_or_else(|| format!("{id}: no such item"))?;
359    if src.is_archived(id).map_err(|e| e.to_string())? {
360        return Err(format!("{id} is archived — unarchive it before moving"));
361    }
362    if !item.children.is_empty() {
363        return Err(format!(
364            "{id} has {} child item(s) — move them individually first",
365            item.children.len()
366        ));
367    }
368    let dest_proj = resolve_dest(to)?;
369    if same_repo(&src_proj.worktree, &dest_proj.worktree) {
370        return Err(format!(
371            "{to}: the destination is this repo — move is cross-repo only"
372        ));
373    }
374    let (dest_proj, dst) = items::open(Some(&dest_proj.worktree), Some(&dest_proj.root))?;
375    let src_name = repo_label(&src, &src_proj.root);
376    let dest_name = repo_label(&dst, &dest_proj.root);
377    // prospective id — observe the live ids so `alloc` lands on
378    // the same number the real move would take
379    let mut reg = dst.meta().map_err(|e| e.to_string())?.registry();
380    reg.observe_all(dst.all_ids().map_err(|e| e.to_string())?);
381    let dest_id = reg.alloc(item.id.kind());
382    let dropped = dropped_refs_line(&item);
383    Ok(MovePlan {
384        from: id,
385        item,
386        src_name,
387        repo: dest_name,
388        root: dest_proj.root.clone(),
389        dest_id,
390        dropped,
391        note: note.map(str::to_string),
392    })
393}

395/// `item move --finish` — complete a move that died after the
396/// destination copy + pointer landed but before the source close
397/// committed. The `moved to <repo>:<ID>` handoff entry is the
398/// resume pointer; the dest copy must exist, then both sides
399/// commit (dest first — it may be uncommitted from the crash).
400/// Idempotent: an already-closed source only commits what's
401/// pending.
402pub(crate) fn finish_move(
403    dir: Option<&Path>,
404    root: Option<&Path>,
405    id: ItemId,
406    actor: Option<Actor>,
407    no_commit: bool,
408) -> Result<Moved, String> {
409    let (src_proj, src) = items::open(dir, root)?;
410    let me = ops::resolve_actor(Some(&src_proj.worktree), actor.as_ref());
411    // the newest `moved to` pointer — live or archived handoff
412    let entries = src.read_handoffs(id).map_err(|e| e.to_string())?;
413    let pointer = entries
414        .iter()
415        .rev()
416        .find_map(|e| parse_moved_to(&e.note).map(|p| (p, e.clone())));
417    let Some(((label, dest_id, dest_root), entry)) = pointer else {
418        return Err(format!(
419            "{id}: no `moved to` handoff — nothing to finish \
420             (a finished move leaves the pointer in the archive)"
421        ));
422    };
423    // the `dest:` root is canonical; the label is the fallback
424    // for pointers written before it existed
425    let dest_proj = match dest_root {
426        Some(root) => indexcmd::repo_at(&root).map_err(|e| format!("{label}: {e}"))?,
427        None => resolve_dest(&label)?,
428    };
429    let (dest_proj, dst) = items::open(Some(&dest_proj.worktree), Some(&dest_proj.root))?;
430    // the destination copy must exist — committed or still
431    // pending in the worktree (the store reads the files)
432    let Some((ditem, _loc, _body)) = items::find_item(&dst, dest_id).map_err(|e| e.to_string())?
433    else {
434        return Err(format!(
435            "{label}:{dest_id} not found in the destination — refusing \
436             to close {id} here; reconcile by hand"
437        ));
438    };
439    let src_name = repo_label(&src, &src_proj.root);
440    let dest_name = repo_label(&dst, &dest_proj.root);
441
442    // source side: close only what's still open — `close_source`
443    // is itself idempotent for done file items; a task whose line
444    // is already gone needs nothing
445    let already_closed = match items::find_item(&src, id).map_err(|e| e.to_string())? {
446        None => true,
447        Some((i, _, _)) => i.status == Status::Done,
448    };
449    if !already_closed {
450        close_source(&src, &src_proj.worktree, id).map_err(|e| {
451            format!(
452                "{e} — destination {dest_name}:{dest_id} exists; \
453                 retry `myco item move {id} --finish`"
454            )
455        })?;
456    }
457    let committed_dest = items::commit(
458        &dest_proj.worktree,
459        &format!("chore: move {dest_id} in from {src_name}:{id}"),
460        Some(&me),
461        no_commit,
462    )?;
463    let committed_source = items::commit(
464        &src_proj.worktree,
465        &format!("chore: move {id} to {dest_name}:{dest_id}"),
466        Some(&me),
467        no_commit,
468    )?;
469    Ok(Moved {
470        from: id,
471        item: ditem,
472        repo: dest_name,
473        root: dest_proj.root.clone(),
474        handoff: entry,
475        committed_source,
476        committed_dest,
477        already_closed,
478    })
479}

481/// The `--json` payload — the destination item plus both sides'
482/// pointers and commit flags (two worktrees, two autocommits).
483/// `committed` itself stays the scalar [`items::report`] merges.
484pub(crate) fn moved_json(m: &Moved) -> serde_json::Value {
485    json!({
486        "id": m.from.to_string(),
487        "item": m.item,
488        "to": {"repo": m.repo, "root": m.root, "id": m.item.id.to_string()},
489        "handoff": m.handoff,
490        "committed_source": m.committed_source,
491        "committed_destination": m.committed_dest,
492        "already_closed": m.already_closed,
493    })
494}

509/// The `item move` verb — `to` opens a fresh move, `--dry-run`
510/// previews it, `--finish` completes one that died mid-flight.
511pub fn h_move(ctx: &Ctx, a: &Args) -> Result<i32, String> {
512    let id = a.req::<ItemId>("id")?;
513    let finish = a.flag("finish");
514    let dry_run = a.flag("dry-run");
515    let to = a.opt("to");
516    if finish && (dry_run || to.is_some()) {
517        return Err("item move: --finish takes no --to/--dry-run".into());
518    }
519    if !finish && to.is_none() {
520        return Err("item move: --to <repo|path> is required (or --finish)".into());
521    }
522    if dry_run && !finish {
523        let p = move_plan(
524            ctx.dir.as_deref(),
525            ctx.root.as_deref(),
526            id,
527            to.unwrap_or_default(),
528            a.opt("note"),
529        )?;
530        return items::report(
531            plan_json(&p),
532            true,
533            ctx.json,
534            &format!(
535                "would move {} → {}:{} ({})",
536                p.from,
537                p.repo,
538                p.dest_id,
539                p.root.display()
540            ),
541        );
542    }
543    if crate::confirm::needed(a.flag("yes")) {
544        let what = match &to {
545            Some(t) => format!("move {id} → {t}? the source closes."),
546            None => format!("finish the move of {id}? the source closes."),
547        };
548        if !crate::confirm::ask(&what) {
549            eprintln!("myco: cancelled");
550            return Ok(1);
551        }
552    }
553    let m = if finish {
554        finish_move(
555            ctx.dir.as_deref(),
556            ctx.root.as_deref(),
557            id,
558            a.parsed("actor")?,
559            ctx.no_commit,
560        )?
561    } else {
562        do_move(
563            ctx.dir.as_deref(),
564            ctx.root.as_deref(),
565            id,
566            to.unwrap_or_default(),
567            a.opt("note"),
568            a.parsed("actor")?,
569            ctx.no_commit,
570        )?
571    };
572    items::report(
573        moved_json(&m),
574        m.committed_source && m.committed_dest,
575        ctx.json,
576        &format!("moved {} → {}:{}", m.from, m.repo, m.item.id),
577    )
578}

586    fn git(dir: &Path, args: &[&str]) {
587        let st = Command::new("git")
588            .arg("-C")
589            .arg(dir)
590            .args(args)
591            .stdout(std::process::Stdio::null())
592            .status()
593            .unwrap();
594        assert!(st.success(), "git {args:?}");
595    }

597    /// A repo with a `project/` worktree on the `project` branch.
598    fn repo_with_project() -> tempfile::TempDir {
599        let d = tempfile::tempdir().unwrap();
600        let r = d.path();
601        git(r, &["init", "-q"]);
602        git(r, &["config", "user.email", "t@t"]);
603        git(r, &["config", "user.name", "t"]);
604        myco_ops::vcs::create_orphan_branch(r, "project", "init").unwrap();
605        git(r, &["worktree", "add", "project", "project"]);
606        std::fs::write(r.join("project/TODO.md"), "# TODO — x\n").unwrap();
607        d
608    }

610    /// Drive a verb like dispatch does — argv through the
611    /// registry decoder into the handler, source `--dir` pinned.
612    fn run_verb(dir: &Path, name: &str, argv: &[String]) -> Result<i32, String> {
613        let mut ctx = crate::registry::Ctx {
614            json: true,
615            no_commit: true,
616            dir: Some(dir.to_path_buf()),
617            ..Default::default()
618        };
619        let v = crate::registry::find(name).unwrap();
620        let a = crate::registry::Args::from_argv(v, argv, &mut ctx, false)?;
621        (v.cli.unwrap())(&ctx, &a)
622    }

628    /// A move against two scratch repos — `$MYCO_HOME` pinned so
629    /// the registry-name fallback never touches the real index.
630    fn with_repos(f: impl Fn(&Path, &Path)) {
631        let home = tempfile::tempdir().unwrap();
632        temp_env::with_var("MYCO_HOME", Some(home.path().to_str().unwrap()), || {
633            let src = repo_with_project();
634            let dst = repo_with_project();
635            f(&src.path().join("project"), &dst.path().join("project"));
636        });
637    }

639    #[test]
640    fn move_task_end_to_end() {
641        with_repos(|src_wt, dst_wt| {
642            run_verb(
643                src_wt,
644                "todo.add",
645                &[
646                    s("port the thing"),
647                    s("--pri"),
648                    s("high"),
649                    s("--label"),
650                    s("core"),
651                ],
652            )
653            .unwrap();
654            run_verb(
655                src_wt,
656                "item.move",
657                &[
658                    s("T-001"),
659                    s("--to"),
660                    dst_wt.parent().unwrap().to_string_lossy().into_owned(),
661                ],
662            )
663            .unwrap();
664            // destination got a fresh id in its own sequence
665            let dst = crate::ops::store_at(dst_wt, myco_core::kind::prelude_table()).unwrap();
666            let (item, _, _) = items::find_item(&dst, "T-001".parse().unwrap())
667                .unwrap()
668                .unwrap();
669            assert_eq!(item.title, "port the thing");
670            assert_eq!(item.priority.to_string(), "high");
671            assert!(item.labels.contains("core"));
672            assert_eq!(item.status, Status::Open);
673            // provenance on the destination copy
674            let entries = dst.read_handoffs(item.id).unwrap();
675            assert!(entries[0].note.contains("moved from"), "{entries:?}");
676            // source line is gone; the handoff went cold with the
677            // `moved to` pointer intact
678            let src = crate::ops::store_at(src_wt, myco_core::kind::prelude_table()).unwrap();
679            assert!(items::find_item(&src, "T-001".parse().unwrap())
680                .unwrap()
681                .is_none());
682            assert!(!src_wt.join("handoffs/T-001.md").exists());
683            let archived = src_wt.join("handoffs/archive/T-001.md");
684            assert!(archived.is_file());
685            let text = std::fs::read_to_string(archived).unwrap();
686            assert!(text.contains("moved to"), "{text}");
687            assert!(text.contains("T-001"), "{text}");
688        });
689    }

691    #[test]
692    fn move_issue_closes_source_file() {
693        with_repos(|src_wt, dst_wt| {
694            run_verb(src_wt, "issue.new", &[s("leaky vault")]).unwrap();
695            run_verb(
696                src_wt,
697                "item.move",
698                &[s("I-001"), s("--to"), dst_wt.to_string_lossy().into_owned()],
699            )
700            .unwrap();
701            let src = crate::ops::store_at(src_wt, myco_core::kind::prelude_table()).unwrap();
702            let (old, _, _) = items::find_item(&src, "I-001".parse().unwrap())
703                .unwrap()
704                .unwrap();
705            assert_eq!(old.status, Status::Done);
706            // the audit entry stays live — the item file still
707            // exists, only its status moved
708            let entries = src.read_handoffs(old.id).unwrap();
709            assert!(
710                entries.last().unwrap().note.contains("moved to"),
711                "{entries:?}"
712            );
713            let dst = crate::ops::store_at(dst_wt, myco_core::kind::prelude_table()).unwrap();
714            let (new, _, body) = items::find_item(&dst, "I-001".parse().unwrap())
715                .unwrap()
716                .unwrap();
717            assert_eq!(new.title, "leaky vault");
718            assert!(body.is_some());
719        });
720    }

724        with_repos(|src_wt, dst_wt| {
725            run_verb(src_wt, "todo.add", &[s("x")]).unwrap();
726            // unknown id
727            assert!(run_verb(
728                src_wt,
729                "item.move",
730                &[s("T-099"), s("--to"), dst_wt.to_string_lossy().into_owned()],
731            )
732            .unwrap_err()
733            .contains("no such item"));
734            // same repo — the source worktree is not a destination
735            let e = run_verb(
736                src_wt,
737                "item.move",
738                &[s("T-001"), s("--to"), src_wt.to_string_lossy().into_owned()],
739            )
740            .unwrap_err();
741            assert!(e.contains("this repo"), "{e}");
742            // a dest that isn't a myco repo
743            let plain = tempfile::tempdir().unwrap();
744            let e = run_verb(
745                src_wt,
746                "item.move",
747                &[
748                    s("T-001"),
749                    s("--to"),
750                    plain.path().to_string_lossy().into_owned(),
751                ],
752            )
753            .unwrap_err();
754            assert!(e.contains("not a myco repo"), "{e}");
755            // nothing burned: T-001 is still live, the dest ledger
756            // is still empty
757            let src = crate::ops::store_at(src_wt, myco_core::kind::prelude_table()).unwrap();
758            assert!(items::find_item(&src, "T-001".parse().unwrap())
759                .unwrap()
760                .is_some());
761            let dst = crate::ops::store_at(dst_wt, myco_core::kind::prelude_table()).unwrap();
762            assert!(items::find_item(&dst, "T-001".parse().unwrap())
763                .unwrap()
764                .is_none());
765        });
