1//! Re-export — lives in `myco_ops::ops` so other front-ends
2//! (`myco-driver`) share it.
3//! `store_at` is cli-local: the concrete `MdStore` factory —
4//! `myco-ops` is generic over [`myco_store::Store`], so the
5//! composition root (this crate) names the impl.
6pub use myco_ops::ops::*;
7
8/// `MdStore` rooted at a worktree path, resolving item layout
9/// through the repo's kind table (`[[kinds]]`, P-008).
10pub fn store_at(
11    worktree: &std::path::Path,
12    kinds: &myco_core::KindTable,
13) -> Result<myco_store_md::MdStore, String> {
14    let root = camino::Utf8PathBuf::from_path_buf(worktree.to_path_buf())
15        .map_err(|p| format!("{}: non-utf8 path", p.display()))?;
16    Ok(myco_store_md::MdStore::with_kinds(root, kinds.clone()))
17}

19/// The config impl ops verbs share — the scarab-backed TOML
