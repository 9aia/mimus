1//! myco-index-redb — the redb impl of [`myco_index::Index`].
2//!
3//! Layout under the index dir (`$MYCO_HOME`, else
4//! `$XDG_DATA_HOME/myco`, else `~/.local/share/myco`):
5//!
6//! - `index.redb` — the `snapshots` table (canonical repo path →
7//!   JSON [`RepoSnapshot`]) plus the link-index tables (P-005):
8//!   `files` (`repo\0rel` → JSON [`FileStamp`]), `file_edges`
9//!   (`repo\0rel` → JSON edge set — the provenance that makes a
10//!   file's outgoing edges swappable), `edges` (multimap
11//!   `repo\0src\0kind` → dst), and `rev_edges` (multimap
12//!   `repo\0dst` → `kind\0src`) for O(1) `related`.
13//! - `repos.toml` — the registry (`[[repo]] path name`), kept as a
14//!   plain file so it stays human-inspectable.
15//!
16//! Everything here is a derived cache — `rm -rf` the dir and
17//! `myco index rebuild` derives it back from the worktrees.
18

28/// `snapshots` table — repo path → JSON [`RepoSnapshot`].
29const SNAPSHOTS: TableDefinition<&str, &str> = TableDefinition::new("snapshots");
30
31/// `files` table — `repo\0rel` → JSON [`FileStamp`]. Presence is
32/// also the sweep manifest: a row outlives its file → swept.
33const FILES: TableDefinition<&str, &str> = TableDefinition::new("files");
34
35/// `file_edges` table — `repo\0rel` → JSON `Vec<Edge>`. The
36/// provenance table: `record` must delete exactly the edges the
37/// file last produced, so it keeps them verbatim.
38const FILE_EDGES: TableDefinition<&str, &str> = TableDefinition::new("file_edges");
39
40/// `edges` multimap — `repo\0src\0kind` → `dst` (canonical id
41/// string). One value per edge; the multimap keeps per-file swaps
42/// to add/remove-delta.
43const EDGES: MultimapTableDefinition<&str, &str> = MultimapTableDefinition::new("edges");
44
45/// `rev_edges` multimap — `repo\0dst` → `kind\0src`. The incoming
46/// half of `related`: value carries the kind so one lookup gets
47/// every direction at once, mirrored at query time.
48const REV_EDGES: MultimapTableDefinition<&str, &str> = MultimapTableDefinition::new("rev_edges");
49
50/// The registry file name inside the index dir.
51const REGISTRY: &str = "repos.toml";
52
53/// The registry lock file — serializes the `repos.toml`
54/// read-modify-write across concurrent `myco` processes.
55const REGISTRY_LOCK: &str = "repos.toml.lock";
56
57/// The redb file name inside the index dir.
58const DB: &str = "index.redb";
59
60/// Total time a contended index lock is retried before giving up —
61/// lets concurrent `myco` processes queue instead of crash.
62const LOCK_RETRY_BUDGET: Duration = Duration::from_secs(2);
63
64/// First backoff after a busy lock; doubles each retry.
65const LOCK_RETRY_MIN: Duration = Duration::from_millis(10);
66
67/// Longest backoff between retries.
68const LOCK_RETRY_MAX: Duration = Duration::from_millis(320);
69
70/// The index dir: `$MYCO_HOME` → `$XDG_DATA_HOME/myco` →
71/// `~/.local/share/myco`. The `$MYCO_HOME` override is how tests
72/// (and unusual setups) relocate the whole cache. `Err` when no
73/// base resolves (HOME unset) — better than writing the cache
74/// into whatever directory `myco` happened to run from.
75pub fn data_dir() -> Result<PathBuf, Error> {
76    if let Some(d) = std::env::var_os("MYCO_HOME") {
77        return Ok(PathBuf::from(d));
78    }
79    if let Some(d) = std::env::var_os("XDG_DATA_HOME") {
80        return Ok(PathBuf::from(d).join("myco"));
81    }
82    std::env::var("HOME")
83        .map(|h| PathBuf::from(h).join(".local/share/myco"))
84        .map_err(|_| {
85            Error::Backend("no index dir — HOME unset (set MYCO_HOME or XDG_DATA_HOME)".into())
86        })
87}

96/// Open `index.redb` at `path`, retrying on lock contention
97/// ([`redb::DatabaseError::DatabaseAlreadyOpen`]) with exponential
98/// backoff until `budget` is spent. Anything else — corrupt file,
99/// IO, upgrade required — fails immediately.
100fn open_db(path: &Utf8Path, budget: Duration) -> Result<Database, Error> {
101    let start = Instant::now();
102    let mut backoff = LOCK_RETRY_MIN;
103    let mut attempt = 0u64;
104    loop {
105        match Database::create(path.as_std_path()) {
106            Ok(db) => return Ok(db),
107            Err(redb::DatabaseError::DatabaseAlreadyOpen) => {
108                let left = budget.saturating_sub(start.elapsed());
109                if left.is_zero() {
110                    return Err(Error::Backend(format!(
111                        "index busy — another myco process holds {DB} (waited ~{:.1}s)",
112                        budget.as_secs_f64()
113                    )));
114                }
115                // counter-based stagger — spreads contenders, no rand dep
116                let jitter = Duration::from_millis((attempt % 4) * 5);
117                std::thread::sleep((backoff + jitter).min(left));
118                backoff = (backoff * 2).min(LOCK_RETRY_MAX);
119                attempt += 1;
120            }
121            Err(e) => return Err(Error::Backend(format!("open {path}: {e}"))),
122        }
123    }
124}

131impl RedbIndex {
132    /// Open the default index ([`data_dir`]), creating it.
133    pub fn open() -> Result<Self, Error> {
134        let dir = Utf8PathBuf::from_path_buf(data_dir()?)
135            .map_err(|p| Error::Backend(format!("{}: non-utf8 index dir", p.display())))?;
136        Self::open_at(dir)
137    }

139    /// Open an index rooted at `dir`, creating it when missing.
140    pub fn open_at(dir: impl Into<Utf8PathBuf>) -> Result<Self, Error> {
141        let dir = dir.into();
142        std::fs::create_dir_all(&dir)?;
143        Ok(RedbIndex { dir })
144    }
145
146    /// The index dir.
147    pub fn dir(&self) -> &Utf8Path {
148        &self.dir
149    }
150
151    fn registry_path(&self) -> Utf8PathBuf {
152        self.dir.join(REGISTRY)
153    }
154
155    fn db_path(&self) -> Utf8PathBuf {
156        self.dir.join(DB)
157    }
158
159    fn load_registry(&self) -> Result<Registry, Error> {
160        match std::fs::read_to_string(self.registry_path()) {
161            Ok(src) => toml::from_str(&src).map_err(|e| Error::Codec(format!("{REGISTRY}: {e}"))),
162            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(Registry::default()),
163            Err(e) => Err(Error::Io(e)),
164        }
165    }
166
167    /// Take the registry lock — `try_lock` with the same retry
168    /// budget/backoff as [`open_db`]. Held across the
169    /// load→mutate→save of `register`/`remove`; a contended
170    /// registry queues instead of losing updates.
171    fn lock_registry(&self) -> Result<std::fs::File, Error> {
172        let path = self.dir.join(REGISTRY_LOCK);
173        let file = std::fs::OpenOptions::new()
174            .create(true)
175            .truncate(false)
176            .write(true)
177            .open(&path)?;
178        let start = Instant::now();
179        let mut backoff = LOCK_RETRY_MIN;
180        let mut attempt = 0u64;
181        loop {
182            match file.try_lock() {
183                Ok(()) => return Ok(file),
184                Err(std::fs::TryLockError::WouldBlock) => {
185                    let left = LOCK_RETRY_BUDGET.saturating_sub(start.elapsed());
186                    if left.is_zero() {
187                        return Err(Error::Backend(format!(
188                            "index busy — another myco process holds {REGISTRY_LOCK} \
189                             (waited ~{:.1}s)",
190                            LOCK_RETRY_BUDGET.as_secs_f64()
191                        )));
192                    }
193                    // same stagger as open_db — spreads contenders
194                    let jitter = Duration::from_millis((attempt % 4) * 5);
195                    std::thread::sleep((backoff + jitter).min(left));
196                    backoff = (backoff * 2).min(LOCK_RETRY_MAX);
197                    attempt += 1;
198                }
199                Err(std::fs::TryLockError::Error(e)) => return Err(Error::Io(e)),
200            }
201        }
202    }
203
204    fn save_registry(&self, reg: &Registry) -> Result<(), Error> {
205        let src = toml::to_string(reg).map_err(|e| Error::Codec(format!("{REGISTRY}: {e}")))?;
206        let path = self.registry_path();
207        // pid-tagged tmp — writers are serialized by the lock, but
208        // a stale tmp from a killed process must not be clobbered
209        // by a live one mid-rename. No fsync: the registry is a
210        // disposable cache — the rename is the atomicity, not
211        // power-loss durability — and the sync dominates the
212        // critical section contenders queue behind.
213        let tmp = path.with_extension(format!("toml.{}.tmp", std::process::id()));
214        let write = (|| -> std::io::Result<()> {
215            std::fs::write(&tmp, src)?;
216            std::fs::rename(&tmp, &path)
217        })();
218        if write.is_err() {
219            let _ = std::fs::remove_file(&tmp);
220        }
221        write.map_err(Error::Io)
222    }
223
224    fn db(&self) -> Result<Database, Error> {
225        open_db(&self.db_path(), LOCK_RETRY_BUDGET)
226    }
227
228    fn write_snapshot(&self, path: &Utf8Path, value: Option<&str>) -> Result<(), Error> {
229        let db = self.db()?;
230        let tx = db
231            .begin_write()
232            .map_err(|e| Error::Backend(format!("begin_write: {e}")))?;
233        {
234            let mut t = tx
235                .open_table(SNAPSHOTS)
236                .map_err(|e| Error::Backend(format!("open snapshots: {e}")))?;
237            match value {
238                Some(v) => {
239                    t.insert(path.as_str(), v)
240                        .map_err(|e| Error::Backend(format!("insert: {e}")))?;
241                }
242                None => {
243                    t.remove(path.as_str())
244                        .map_err(|e| Error::Backend(format!("remove: {e}")))?;
245                }
246            }
247        }
248        tx.commit()
249            .map_err(|e| Error::Backend(format!("commit: {e}")))
250    }
251}

253impl Index for RedbIndex {
254    fn register(&self, repo: &RepoEntry) -> Result<(), Error> {
255        let _lock = self.lock_registry()?;
256        let mut reg = self.load_registry()?;
257        match reg.repo.iter_mut().find(|r| r.path == repo.path) {
258            Some(r) => {
259                r.name = repo.name.clone();
260                r.worktree = repo.worktree.clone();
261                r.ledger = repo.ledger;
262            }
263            None => reg.repo.push(repo.clone()),
264        }
265        self.save_registry(&reg)
266    }
267
268    fn remove(&self, path: &Utf8Path) -> Result<bool, Error> {
269        let _lock = self.lock_registry()?;
270        let mut reg = self.load_registry()?;
271        let before = reg.repo.len();
272        reg.repo.retain(|r| r.path != path);
273        if reg.repo.len() == before {
274            return Ok(false);
275        }
276        // drop the snapshot before the registry write — a failure
277        // here leaves a registered repo with no snapshot (stale →
278        // rebuild), not an orphan snapshot keyed to nothing. The
279        // link records go with it — `purge` is just a sweep with
280        // an empty live set.
281        self.write_snapshot(path, None)?;
282        LinkIndex::purge(self, path.as_str())?;
283        self.save_registry(&reg)?;
284        Ok(true)
285    }
286
287    fn repos(&self) -> Result<Vec<RepoEntry>, Error> {
288        let mut repos = self.load_registry()?.repo;
289        repos.sort_by(|a, b| a.path.cmp(&b.path));
290        Ok(repos)
291    }
292
293    fn snapshot(&self, path: &Utf8Path) -> Result<Option<RepoSnapshot>, Error> {
294        let db = self.db()?;
295        let tx = db
296            .begin_read()
297            .map_err(|e| Error::Backend(format!("begin_read: {e}")))?;
298        let t = match tx.open_table(SNAPSHOTS) {
299            Ok(t) => t,
300            Err(_) => return Ok(None), // no table → no snapshots yet
301        };
302        let Some(v) = t
303            .get(path.as_str())
304            .map_err(|e| Error::Backend(format!("get: {e}")))?
305        else {
306            return Ok(None);
307        };
308        // a corrupt snapshot reads as a missing one — the index
309        // is a disposable cache, so callers see "never scanned",
310        // mark it stale, and rebuild instead of erroring every
311        // read (one bad row used to brick `org`/`index list`)
312        Ok(serde_json::from_str(v.value()).ok())
313    }
314
315    fn save_snapshot(&self, snapshot: &RepoSnapshot) -> Result<(), Error> {
316        let json = serde_json::to_string(snapshot)
317            .map_err(|e| Error::Codec(format!("snapshot {}: {e}", snapshot.path)))?;
318        self.write_snapshot(&snapshot.path, Some(&json))
319    }
320}

324/// `files`/`file_edges` row key — `repo\0rel`.
325fn file_key(repo: &str, rel: &Utf8Path) -> String {
326    format!("{repo}\0{rel}")
327}
328
329/// `edges` multimap key — `repo\0src\0kind`.
330fn edge_key(repo: &str, src: ItemId, kind: EdgeKind) -> String {
331    format!("{repo}\0{src}\0{}", kind.name())
332}
333
334/// `rev_edges` multimap key — `repo\0dst`.
335fn rev_key(repo: &str, dst: ItemId) -> String {
336    format!("{repo}\0{dst}")
337}
338
339/// `rev_edges` multimap value — `kind\0src`.
340fn rev_val(kind: EdgeKind, src: ItemId) -> String {
341    format!("{}\0{src}", kind.name())
342}
343
344/// Remove every `edges`/`rev_edges` pair an edge set owns —
345/// the delete half of `record`'s swap and `sweep`'s cleanup.
346fn drop_edges(
347    repo: &str,
348    edges: &[Edge],
349    ed: &mut redb::MultimapTable<&str, &str>,
350    rv: &mut redb::MultimapTable<&str, &str>,
351) -> Result<(), Error> {
352    for e in edges {
353        ed.remove(
354            edge_key(repo, e.src, e.kind).as_str(),
355            e.dst.to_string().as_str(),
356        )
357        .map_err(|e| Error::Backend(format!("edges remove: {e}")))?;
358        rv.remove(
359            rev_key(repo, e.dst).as_str(),
360            rev_val(e.kind, e.src).as_str(),
361        )
362        .map_err(|e| Error::Backend(format!("rev_edges remove: {e}")))?;
363    }
364    Ok(())
365}

367/// The stored edge set one file produced — corrupt JSON reads as
368/// empty, same "disposable cache" stance as `snapshot`.
369fn read_file_edges(
370    t: &impl redb::ReadableTable<&'static str, &'static str>,
371    key: &str,
372) -> Result<Vec<Edge>, Error> {
373    let Some(v) = t
374        .get(key)
375        .map_err(|e| Error::Backend(format!("file_edges get: {e}")))?
376    else {
377        return Ok(Vec::new());
378    };
379    Ok(serde_json::from_str(v.value()).unwrap_or_default())
380}

382impl LinkIndex for RedbIndex {
383    fn stamps(&self, repo: &str) -> Result<BTreeMap<Utf8PathBuf, FileStamp>, Error> {
384        let db = self.db()?;
385        let tx = db
386            .begin_read()
387            .map_err(|e| Error::Backend(format!("begin_read: {e}")))?;
388        let t = match tx.open_table(FILES) {
389            Ok(t) => t,
390            Err(_) => return Ok(BTreeMap::new()),
391        };
392        let prefix = format!("{repo}\0");
393        let mut out = BTreeMap::new();
394        let iter = t
395            .iter()
396            .map_err(|e| Error::Backend(format!("files iter: {e}")))?;
397        for row in iter {
398            let (k, v) = row.map_err(|e| Error::Backend(format!("files iter: {e}")))?;
399            let Some(rel) = k.value().strip_prefix(&prefix) else {
400                continue;
401            };
402            if let Ok(stamp) = serde_json::from_str::<FileStamp>(v.value()) {
403                out.insert(Utf8PathBuf::from(rel), stamp);
404            }
405        }
406        Ok(out)
407    }
408
409    fn record(
410        &self,
411        repo: &str,
412        rel: &Utf8Path,
413        stamp: &FileStamp,
414        edges: &[Edge],
415    ) -> Result<(), Error> {
416        let json =
417            serde_json::to_string(edges).map_err(|e| Error::Codec(format!("edges {rel}: {e}")))?;
418        let stamp_json =
419            serde_json::to_string(stamp).map_err(|e| Error::Codec(format!("stamp {rel}: {e}")))?;
420        let db = self.db()?;
421        let tx = db
422            .begin_write()
423            .map_err(|e| Error::Backend(format!("begin_write: {e}")))?;
424        {
425            let key = file_key(repo, rel);
426            let mut fe = tx
427                .open_table(FILE_EDGES)
428                .map_err(|e| Error::Backend(format!("open file_edges: {e}")))?;
429            let mut ed = tx
430                .open_multimap_table(EDGES)
431                .map_err(|e| Error::Backend(format!("open edges: {e}")))?;
432            let mut rv = tx
433                .open_multimap_table(REV_EDGES)
434                .map_err(|e| Error::Backend(format!("open rev_edges: {e}")))?;
435            // swap: out with the file's old set, in with the new —
436            // one commit, so a crash can't leave either half
437            let old = read_file_edges(&fe, &key)?;
438            drop_edges(repo, &old, &mut ed, &mut rv)?;
439            fe.insert(key.as_str(), json.as_str())
440                .map_err(|e| Error::Backend(format!("file_edges insert: {e}")))?;
441            for e in edges {
442                ed.insert(
443                    edge_key(repo, e.src, e.kind).as_str(),
444                    e.dst.to_string().as_str(),
445                )
446                .map_err(|e| Error::Backend(format!("edges insert: {e}")))?;
447                rv.insert(
448                    rev_key(repo, e.dst).as_str(),
449                    rev_val(e.kind, e.src).as_str(),
450                )
451                .map_err(|e| Error::Backend(format!("rev_edges insert: {e}")))?;
452            }
453            {
454                let mut fs = tx
455                    .open_table(FILES)
456                    .map_err(|e| Error::Backend(format!("open files: {e}")))?;
457                fs.insert(key.as_str(), stamp_json.as_str())
458                    .map_err(|e| Error::Backend(format!("files insert: {e}")))?;
459            }
460        }
461        tx.commit()
462            .map_err(|e| Error::Backend(format!("commit: {e}")))
463    }

465    fn restamp(&self, repo: &str, rel: &Utf8Path, stamp: &FileStamp) -> Result<(), Error> {
466        let json =
467            serde_json::to_string(stamp).map_err(|e| Error::Codec(format!("stamp {rel}: {e}")))?;
468        let db = self.db()?;
469        let tx = db
470            .begin_write()
471            .map_err(|e| Error::Backend(format!("begin_write: {e}")))?;
472        {
473            let mut fs = tx
474                .open_table(FILES)
475                .map_err(|e| Error::Backend(format!("open files: {e}")))?;
476            fs.insert(file_key(repo, rel).as_str(), json.as_str())
477                .map_err(|e| Error::Backend(format!("files insert: {e}")))?;
478        }
479        tx.commit()
480            .map_err(|e| Error::Backend(format!("commit: {e}")))
481    }

483    fn related(
484        &self,
485        repo: &str,
486        id: ItemId,
487        alive: &BTreeSet<ItemId>,
488    ) -> Result<Vec<Related>, Error> {
489        let db = self.db()?;
490        let tx = db
491            .begin_read()
492            .map_err(|e| Error::Backend(format!("begin_read: {e}")))?;
493        let mut out = BTreeSet::new();
494        // outgoing — only extractable kinds can be stored forward
495        if let Ok(ed) = tx.open_multimap_table(EDGES) {
496            for kind in EdgeKind::ALL.into_iter().filter(|k| k.extracted()) {
497                let vals = match ed.get(edge_key(repo, id, kind).as_str()) {
498                    Ok(v) => v,
499                    Err(_) => continue,
500                };
501                for v in vals {
502                    let v = v.map_err(|e| Error::Backend(format!("edges get: {e}")))?;
503                    if let Ok(dst) = v.value().parse::<ItemId>() {
504                        if alive.contains(&dst) {
505                            out.insert(Related { other: dst, kind });
506                        }
507                    }
508                }
509            }
510        }
511        // incoming — the reverse table mirrors the kind for us
512        if let Ok(rv) = tx.open_multimap_table(REV_EDGES) {
513            if let Ok(vals) = rv.get(rev_key(repo, id).as_str()) {
514                for v in vals {
515                    let v = v.map_err(|e| Error::Backend(format!("rev_edges get: {e}")))?;
516                    let Some((kind, src)) = v.value().split_once('\0') else {
517                        continue;
518                    };
519                    let (Some(kind), Ok(src)) = (EdgeKind::from_name(kind), src.parse::<ItemId>())
520                    else {
521                        continue;
522                    };
523                    // a self-edge (handoff) is already reported by
524                    // the outgoing pass — the set dedups it anyway
525                    if alive.contains(&src) {
526                        out.insert(Related {
527                            other: src,
528                            kind: kind.mirror(),
529                        });
530                    }
531                }
532            }
533        }
534        Ok(out.into_iter().collect())
535    }

537    fn reach(
538        &self,
539        repo: &str,
540        id: ItemId,
541        kind: EdgeKind,
542        max_depth: usize,
543    ) -> Result<Vec<ItemId>, Error> {
544        let db = self.db()?;
545        let tx = db
546            .begin_read()
547            .map_err(|e| Error::Backend(format!("begin_read: {e}")))?;
548        let ed = match tx.open_multimap_table(EDGES) {
549            Ok(t) => t,
550            Err(_) => return Ok(Vec::new()),
551        };
552        // BFS with a visited set — cyclic input terminates at the
553        // depth cap, never on a loop
554        let mut seen = BTreeSet::from([id]);
555        let mut frontier = vec![id];
556        for _ in 0..max_depth {
557            let mut next = Vec::new();
558            for cur in &frontier {
559                let vals = match ed.get(edge_key(repo, *cur, kind).as_str()) {
560                    Ok(v) => v,
561                    Err(_) => continue,
562                };
563                for v in vals {
564                    let v = v.map_err(|e| Error::Backend(format!("edges get: {e}")))?;
565                    if let Ok(dst) = v.value().parse::<ItemId>() {
566                        if seen.insert(dst) {
567                            next.push(dst);
568                        }
569                    }
570                }
571            }
572            if next.is_empty() {
573                break;
574            }
575            frontier = next;
576        }
577        seen.remove(&id);
578        Ok(seen.into_iter().collect())
579    }
580
581    fn sweep(&self, repo: &str, live: &BTreeSet<Utf8PathBuf>) -> Result<usize, Error> {
582        let db = self.db()?;
583        let tx = db
584            .begin_write()
585            .map_err(|e| Error::Backend(format!("begin_write: {e}")))?;
586        let mut removed = 0;
587        {
588            let mut fs = tx
589                .open_table(FILES)
590                .map_err(|e| Error::Backend(format!("open files: {e}")))?;
591            let mut fe = tx
592                .open_table(FILE_EDGES)
593                .map_err(|e| Error::Backend(format!("open file_edges: {e}")))?;
594            let mut ed = tx
595                .open_multimap_table(EDGES)
596                .map_err(|e| Error::Backend(format!("open edges: {e}")))?;
597            let mut rv = tx
598                .open_multimap_table(REV_EDGES)
599                .map_err(|e| Error::Backend(format!("open rev_edges: {e}")))?;
600            // collect first — can't mutate a table while an
601            // iterator borrows it
602            let prefix = format!("{repo}\0");
603            let dead: Vec<String> = {
604                let iter = fs
605                    .iter()
606                    .map_err(|e| Error::Backend(format!("files iter: {e}")))?;
607                let mut dead = Vec::new();
608                for row in iter {
609                    let (k, _) = row.map_err(|e| Error::Backend(format!("files iter: {e}")))?;
610                    let Some(rel) = k.value().strip_prefix(&prefix) else {
611                        continue;
612                    };
613                    if !live.contains(Utf8Path::new(rel)) {
614                        dead.push(k.value().to_string());
615                    }
616                }
617                dead
618            };
619            for key in dead {
620                let old = read_file_edges(&fe, &key)?;
621                drop_edges(repo, &old, &mut ed, &mut rv)?;
622                fe.remove(key.as_str())
623                    .map_err(|e| Error::Backend(format!("file_edges remove: {e}")))?;
624                fs.remove(key.as_str())
625                    .map_err(|e| Error::Backend(format!("files remove: {e}")))?;
626                removed += 1;
627            }
628        }
629        tx.commit()
630            .map_err(|e| Error::Backend(format!("commit: {e}")))?;
631        Ok(removed)
632    }
633}
