1//! Graph adapter — set-level checking (ADR-0002): index-edge
2//! classes against the kuda index (phase 1) plus index-free prose
3//! contradiction buckets (phase 2). Consumes the index *file
4//! format*, not the kuda crates — the schema is the contract,
5//! same as `mimus-check-symbol`.
6//!
7//! Classes:
8//! - `stale-reference` (error): an `imports`/`package-dep`/
9//!   `workspace-member` edge whose target resolves to no indexed
10//!   module — or whose *source* module is gone entirely. Internal
11//!   vs external is decided conservatively: an unresolved target
12//!   flags only when it shares the repo's own naming family (a
13//!   '/'-scope or a prefix segment carried by ≥2 indexed modules);
14//!   bare external-looking names (`serde`, `react`,
15//!   `github.com/ext/x`) are silence.
16//! - `unsupported-claim` (warn): prose of the shape
17//!   `X imports|depends on|requires|uses Y` where both sides
18//!   resolve to indexed modules and the index records no such
19//!   edge. Either side unresolvable → the triple is unproveable →
20//!   silence (evidence-or-silence, ADR-0001).
21//! - `contradiction` / `self-contradiction` (warn, phase 2):
22//!   prose claims reduce to `(subject, relation, object)` triples
23//!   (mimus-claims::triples) and conflicts fall out of the
24//!   `(subject, relation)` bucket — cross-source for doc↔doc,
25//!   reflexive when one file asserts two objects for the same
26//!   slot. No index needed: this pass runs on any claim set.
27//! - phase 3: a per-file triples cache (`cache` module — sqlite
28//!   at `.mimus/triples.db`, keyed on content hash + extractor
29//!   version) skips re-extracting unchanged files, and the prose
30//!   pass's `in_scope` gate lets `--diff` keep the buckets whole
31//!   while only reporting conflicts that touch changed files.
32//! - phase 5: the semantic tier is rescoped to *fuzzy-candidate
33//!   adjudication*. `semantic_candidates` turns the prose triples
34//!   the buckets leave unsettled (canon-differing subjects that
35//!   share a token, same relation, different objects) into the
36//!   agent's intake list — pre-filtered, so the agent adjudicates
37//!   pairs instead of scanning every claim.
38//! - `undocumented-fact` (warn, phase 4): reverse coverage — a
39//!   *public* module (one the index records as a dependency target)
40//!   that no claim names is code the docs never assert. Module
41//!   granularity, not edge: every indexed import is a fact nobody
42//!   documents, while a module is the smallest unit a doc page can
43//!   reasonably cover. `in_scope` gates it the way it gates the
44//!   buckets — under `--diff` only modules whose own sources changed
45//!   report.
46

61/// Load the kuda index into a `Graph` snapshot — read-only, never
62/// creates the file. Missing tables yield empty sets: a
63/// symbols-only index still produces a usable graph.
64pub fn load_graph(index: &Path) -> Result<Graph, String> {
65    let conn = Connection::open_with_flags(index, OpenFlags::SQLITE_OPEN_READ_ONLY)
66        .map_err(|e| e.to_string())?;
67    let mut g = Graph {
68        index: index.to_path_buf(),
69        ..Graph::default()
70    };
71    if table_exists(&conn, "edges")? {
72        g.edges = edges(&conn)?;
73    }
74    if table_exists(&conn, "symbols")? {
75        let (modules, symbols) = symbols(&conn)?;
76        g.modules = modules;
77        g.symbols = symbols;
78        g.module_files = module_files(&conn)?;
79    }
80    Ok(g)
81}

83/// Age of the index — `meta.generated_at` (unix secs, written by
84/// `kuda sync`) when present, else file mtime. `None` when the
85/// file can't be read at all; `mimus doctor` reports it.
86pub fn index_age(index: &Path) -> Option<Duration> {
87    meta_generated_at(index)
88        .or_else(|| mtime(index))
89        .and_then(|ts| {
90            std::time::UNIX_EPOCH
91                .checked_add(Duration::from_secs(ts))
92                .and_then(|t| t.elapsed().ok())
93        })
94}

96fn meta_generated_at(index: &Path) -> Option<u64> {
97    let conn = Connection::open_with_flags(index, OpenFlags::SQLITE_OPEN_READ_ONLY).ok()?;
98    conn.query_row("SELECT v FROM meta WHERE k = 'generated_at'", [], |r| {
99        r.get::<_, String>(0)
100    })
101    .ok()
102    .and_then(|v| v.parse().ok())
103}

105fn mtime(index: &Path) -> Option<u64> {
106    std::fs::metadata(index)
107        .ok()?
108        .modified()
109        .ok()?
110        .duration_since(std::time::UNIX_EPOCH)
111        .ok()
112        .map(|d| d.as_secs())
113}

115fn table_exists(conn: &Connection, name: &str) -> Result<bool, String> {
116    conn.query_row(
117        "SELECT COUNT(*) FROM sqlite_master WHERE type = 'table' AND name = ?1",
118        params![name],
119        |r| r.get::<_, i64>(0),
120    )
121    .map(|n| n > 0)
122    .map_err(|e| e.to_string())
123}

125fn ek_from(s: &str) -> EdgeKind {
126    match s {
127        "imports" => EdgeKind::Imports,
128        "package-dep" => EdgeKind::PackageDep,
129        "workspace-member" => EdgeKind::WorkspaceMember,

145#[allow(unexpected_cfgs)] // `cfg(coverage)` like mimus-check-symbol
146fn edges(conn: &Connection) -> Result<Vec<Edge>, String> {
147    let mut st = conn
148        .prepare("SELECT from_mod,to_mod,kind FROM edges")
149        .map_err(|e| e.to_string())?;
150    let rows = st.query_map([], |r| {
151        Ok(Edge {
152            from: r.get(0)?,
153            to: r.get(1)?,
154            kind: ek_from(&r.get::<_, String>(2)?),
155        })
156    });
157    // `?` on query_map is unreachable — [] can't fail to bind
158    // after a successful prepare; under coverage the Err arm
159    // would be a permanently red line
160    #[cfg(coverage)]
161    let rows = rows.unwrap();
162    #[cfg(not(coverage))]
163    let rows = rows.map_err(|e| e.to_string())?;
164    // row-level decode errors (NULL cols on hand-built indexes)
165    // are skipped, not fatal — same as SymbolChecker's flatten()
166    Ok(rows.flatten().collect())
167}

169/// `(modules, symbols)` — one pass over the symbols table; the
170/// module set matches kuda's `DISTINCT module FROM symbols`.
171#[allow(unexpected_cfgs)] // `cfg(coverage)` like mimus-check-symbol
172fn symbols(conn: &Connection) -> Result<(HashSet<String>, HashSet<String>), String> {
173    let mut st = conn
174        .prepare("SELECT name,module FROM symbols")
175        .map_err(|e| e.to_string())?;
176    let rows = st.query_map([], |r| Ok((r.get::<_, String>(0)?, r.get::<_, String>(1)?)));
177    // same unreachable `?` as edges() — see above
178    #[cfg(coverage)]
179    let rows = rows.unwrap();
180    #[cfg(not(coverage))]
181    let rows = rows.map_err(|e| e.to_string())?;
182    let (mut modules, mut syms) = (HashSet::new(), HashSet::new());
183    for (name, module) in rows.flatten() {
184        syms.insert(name);
185        modules.insert(module);
186    }
187    Ok((modules, syms))
188}

190/// `module → files` — the phase-4 provenance. Optional by design:
191/// a minimal symbols table (`name`,`module` only) leaves it empty,
192/// and the graph pass then treats every module as in scope.
193#[allow(unexpected_cfgs)] // `cfg(coverage)` like mimus-check-symbol
194fn module_files(conn: &Connection) -> Result<HashMap<String, HashSet<PathBuf>>, String> {
195    if !column_exists(conn, "symbols", "file")? {
196        return Ok(HashMap::new());
197    }
198    let mut st = conn
199        .prepare("SELECT module,file FROM symbols")
200        .map_err(|e| e.to_string())?;
201    let rows = st.query_map([], |r| Ok((r.get::<_, String>(0)?, r.get::<_, String>(1)?)));
202    // same unreachable `?` as edges() — see above
203    #[cfg(coverage)]
204    let rows = rows.unwrap();
205    #[cfg(not(coverage))]
206    let rows = rows.map_err(|e| e.to_string())?;
207    let mut files: HashMap<String, HashSet<PathBuf>> = HashMap::new();
208    for (module, file) in rows.flatten() {
209        files.entry(module).or_default().insert(PathBuf::from(file));
210    }
211    Ok(files)
212}

214/// Does `table` carry `col`? PRAGMA-driven so a minimal index
215/// schema degrades instead of erroring.
216fn column_exists(conn: &Connection, table: &str, col: &str) -> Result<bool, String> {
217    let mut st = conn
218        .prepare(&format!("PRAGMA table_info({table})"))
219        .map_err(|e| e.to_string())?;
220    let rows = st
221        .query_map([], |r| r.get::<_, String>(1))
222        .map_err(|e| e.to_string())?;
223    for name in rows.flatten() {
224        if name == col {
225            return Ok(true);
226        }
227    }
228    Ok(false)
229}

244/// Prose contradiction pass — ADR-0002 phase 2. The bucket
245/// classes `contradiction` (cross-source, doc↔doc) and
246/// `self-contradiction` (reflexive: one file asserting ≥2 objects
247/// for the same slot). Index-free — prose triples don't need the
248/// kuda index, so this runs on every claim set, not just when
249/// `[check].kuda_index` is configured. Warn tier: extraction is
250/// heuristic, so these report but never gate.
251pub struct ProseContradictionChecker {
252    /// Subject aliases from `[check].aliases` — `(phrase,
253    /// canonical)` pairs resolved at bucket-key time.
254    pub aliases: Vec<(String, String)>,
255    /// Per-file triples cache (phase 3) — `Some` serves unchanged
256    /// files' triples from `.mimus/triples.db`; `None` extracts
257    /// every run.
258    pub cache: Option<TriplesCache>,
259    /// Triples from the last `run_prose` — the semantic tier's
260    /// candidate intake reuses them instead of re-extracting.
261    pub triples: Vec<Triple>,
262}
263
264impl ProseContradictionChecker {
265    /// claims → triples → buckets → warn-tier violations.
266    /// `in_scope` gates *findings*, not extraction: the buckets
267    /// see every file's triples (a contradiction needs both
268    /// sides) but a conflict emits only when ≥1 object site is
269    /// in scope — under `--diff` that's the changed-file side,
270    /// so unchanged↔unchanged drift stays silent.
271    pub fn run_prose(
272        &mut self,
273        claims: &[Claim],
274        in_scope: &dyn Fn(&Path) -> bool,
275    ) -> Vec<Violation> {
276        let triples = match self.cache.as_mut() {
277            Some(c) => extract_triples_cached(claims, c),
278            None => extract_triples(claims),
279        };
280        // kept for the semantic tier's intake (phase 5) — the agent
281        // gets candidates off the same triples, not a second extract
282        self.triples = triples;
283        contradictions_with_aliases(&self.triples, &self.aliases)
284            .iter()
285            .filter(|c| c.objects.iter().any(|(_, src, _)| in_scope(src)))
286            .map(|c| contradiction_violation(c, claims, in_scope))
287            .collect()
288    }
289}

291/// One fuzzy pair the mechanical buckets can't settle — the
292/// semantic tier's intake list. The bucket pass keeps the
293/// mechanical-grade conflicts (canon-equal subject, different
294/// object) to itself and hands the agent only what needs
295/// understanding: near-duplicate subjects with a shared content
296/// token asserting different objects for the same relation.
297#[derive(Debug, Clone, PartialEq, Eq)]
298pub struct SemanticCandidate {
299    /// Relation both triples assert (`uses`, `is`, …) — the slot
300    /// the two sides disagree on.
301    pub relation: String,
302    /// Left triple's subject, as spelled at its own site — the
303    /// canonical form is the filter, never the output.
304    pub left_subject: String,
305    /// Left triple's object.
306    pub left_object: String,
307    /// File the left triple was extracted from.
308    pub left_source: PathBuf,
309    /// 1-based line in `left_source`.
310    pub left_line: usize,
311    /// Right triple's subject, as spelled at its own site.
312    pub right_subject: String,
313    /// Right triple's object.
314    pub right_object: String,
315    /// File the right triple was extracted from.
316    pub right_source: PathBuf,
317    /// 1-based line in `right_source`.
318    pub right_line: usize,
319}

321/// Triples → the semantic tier's candidate list (ADR-0002 ph5).
322///
323/// The buckets own canon-equal subjects (`contradiction`,
324/// `self-contradiction`); this is the residue they can't decide.
325/// A pair becomes a candidate when all four hold:
326///
327/// - the canonical subjects *differ* — canon-equal is the buckets'
328///   job, and re-emitting it would spend an agent call on a
329///   mechanical conflict;
330/// - both triples assert the same relation — different relations
331///   are different slots, not a disagreement;
332/// - the canonical subjects share ≥1 content token — `auth` pairs
333///   with `auth flow`, never with `cache`. This is the synonym /
334///   near-duplicate heuristic that keeps the intake small;
335/// - the objects differ — agreeing triples are not a question.
336///
337/// Each unordered pair emits once, and the output is sorted on
338/// `(relation, left_subject, right_subject, left_object,
339/// right_object)` so two runs over the same triples print the same
340/// prompt. Original spellings survive into the struct: the agent
341/// needs the real `file:line` site, not the bucket key.
342pub fn semantic_candidates(
343    triples: &[Triple],
344    aliases: &[(String, String)],
345) -> Vec<SemanticCandidate> {
346    let canon: Vec<String> = triples
347        .iter()
348        .map(|t| canon_subject(&t.subject, aliases))
349        .collect();
350    // token sets once per triple — pairing is the hot loop, and a
351    // 10k-claim sweep has no room for re-splitting subjects inside it
352    let toks: Vec<HashSet<&str>> = canon
353        .iter()
354        .map(|c| c.split_whitespace().collect())
355        .collect();
356    // only same-relation triples can disagree about a slot
357    let mut by_rel: HashMap<&str, Vec<usize>> = HashMap::new();
358    for (i, t) in triples.iter().enumerate() {
359        by_rel.entry(t.relation.as_str()).or_default().push(i);
360    }
361    // unordered (subject canon, object) pair sides — one entry per
362    // structural disagreement, however many files spell it
363    let mut seen: HashSet<(String, String, String, String, String)> = HashSet::new();
364    let mut out = Vec::new();
365    for group in by_rel.values() {
366        for x in 0..group.len() {
367            for y in (x + 1)..group.len() {
368                let (i, j) = (group[x], group[y]);
369                let (a, b) = (&triples[i], &triples[j]);
370                if canon[i] == canon[j] || a.object == b.object || toks[i].is_disjoint(&toks[j]) {
371                    continue;
372                }
373                // deterministic side order — the smaller (canon,
374                // object) is "left", so duplicate spellings collapse
375                // onto one candidate instead of two mirrored ones
376                let (l, r) = if (canon[i].as_str(), a.object.as_str())
377                    <= (canon[j].as_str(), b.object.as_str())
378                {
379                    (i, j)
380                } else {
381                    (j, i)
382                };
383                let (lt, rt) = (&triples[l], &triples[r]);
384                let key = (
385                    lt.relation.clone(),
386                    canon[l].clone(),
387                    lt.object.clone(),
388                    canon[r].clone(),
389                    rt.object.clone(),
390                );
391                if !seen.insert(key) {
392                    continue;
393                }
394                out.push(SemanticCandidate {
395                    relation: lt.relation.clone(),
396                    left_subject: lt.subject.clone(),
397                    left_object: lt.object.clone(),
398                    left_source: lt.source.clone(),
399                    left_line: lt.line,
400                    right_subject: rt.subject.clone(),
401                    right_object: rt.object.clone(),
402                    right_source: rt.source.clone(),
403                    right_line: rt.line,
404                });
405            }
406        }
407    }
408    out.sort_by(|a, b| {
409        (
410            &a.relation,
411            &a.left_subject,
412            &a.right_subject,
413            &a.left_object,
414            &a.right_object,
415            &a.left_source,
416            a.left_line,
417            &a.right_source,
418            a.right_line,
419        )
420            .cmp(&(
421                &b.relation,
422                &b.left_subject,
423                &b.right_subject,
424                &b.left_object,
425                &b.right_object,
426                &b.left_source,
427                b.left_line,
428                &b.right_source,
429                b.right_line,
430            ))
431    });
432    out
433}

435/// `undocumented-fact` — ADR-0002 phase 4: reverse coverage
436/// (code→doc), warn tier. The *public surface* is the set of
437/// modules the index records as a dependency target — internal
438/// details nobody imports aren't contract. A public module emits
439/// when no document in the mined set names it, matched the way the
440/// buckets match subjects (`canon_subject`: case, `-`/`_`, plurals
441/// and `[check].aliases` all fold), so `mimus-core`, `mimus_core`
442/// and an aliased spelling each count as documenting it. The
443/// haystack is document text, not extracted claims: a doc that
444/// merely lists a crate (`[mimus-check-acp](…)`) documents it
445/// without asserting a checkable claim — claim-level matching would
446/// flag every crate index page.
447pub struct UndocumentedFactChecker<'a> {
448    /// Subject aliases from `[check].aliases` — `(phrase,
449    /// canonical)` pairs, the same map the contradiction buckets
450    /// resolve.
451    pub aliases: &'a [(String, String)],
452}
453
454impl UndocumentedFactChecker<'_> {
455    /// `in_scope` gates *findings*, not the scan: under `--diff`
456    /// only a module whose own indexed sources changed reports;
457    /// unscoped runs pass `|_| true`. The module set itself always
458    /// comes from the whole index — a snapshot has no per-run
459    /// scope to narrow. `docs` is the concatenated text of every
460    /// document the claim set was mined from (cards included).
461    pub fn run_facts(
462        &self,
463        docs: &str,
464        graph: &Graph,
465        in_scope: &dyn Fn(&Path) -> bool,
466    ) -> Vec<Violation> {
467        // padded canonical doc text — the haystack, so `mimus-core`
468        // can't match inside `mimus-core-extra`
469        let hay = format!(" {} ", canon_subject(docs, self.aliases));
470        public_modules(graph)
471            .into_iter()
472            .filter(|m| !documented(m, &hay, self.aliases))
473            .filter(|m| module_in_scope(graph, m, in_scope))
474            .map(|m| fact_violation(graph, m))
475            .collect()
476    }
477}

479/// Modules the index records as a dependency *target* — the public
480/// surface other code actually consumes. `calls` edges prove nothing
481/// about the API, an unresolved target is external, and a self-edge
482/// (`a → a`) marks nothing.
483fn public_modules(graph: &Graph) -> BTreeSet<&str> {
484    graph
485        .edges
486        .iter()
487        .filter(|e| {
488            matches!(
489                e.kind,
490                EdgeKind::Imports | EdgeKind::PackageDep | EdgeKind::WorkspaceMember
491            )
492        })
493        .filter_map(|e| {
494            let target = resolve(&e.to, &graph.modules)?;
495            (module_key(&e.from) != module_key(target)).then_some(target)
496        })
497        .collect()
498}

500/// Does any document name `module`? Canon-matched against the padded
501/// haystack, plus every `[check].aliases` spelling whose canonical
502/// target folds onto the module.
503fn documented(module: &str, hay: &str, aliases: &[(String, String)]) -> bool {
504    let canon = canon_subject(module, aliases);
505    if canon.is_empty() {
506        // a name made only of stopwords has no needle to look for —
507        // silence beats a substring match on padding
508        return true;
509    }
510    let spellings = std::iter::once(canon.clone()).chain(
511        aliases
512            .iter()
513            .filter(|(_, target)| canon_subject(target, aliases) == canon)
514            .map(|(alias, _)| canon_subject(alias, &[])),
515    );
516    spellings
517        .filter(|n| !n.is_empty())
518        .any(|n| hay.contains(&format!(" {n} ")))
519}

521/// `--diff` gate: does any of the module's own indexed sources sit
522/// in scope? A snapshot without file provenance (a hand-built
523/// `Graph`) can't answer — it reports rather than silently drop.
524fn module_in_scope(graph: &Graph, module: &str, in_scope: &dyn Fn(&Path) -> bool) -> bool {
525    match graph.module_files.get(module) {
526        Some(files) => files.iter().any(|f| in_scope(f)),
527        None => true,
528    }
529}

531/// One undocumented public module → one warn. `line: 0` — the
532/// finding is module-level, anchored on a real source file so the
533/// report points at code rather than the index.
534fn fact_violation(graph: &Graph, module: &str) -> Violation {
535    let files = graph.module_files.get(module);
536    let source = files
537        .and_then(|f| f.iter().min())
538        .cloned()
539        .unwrap_or_else(|| graph.index.clone());
540    let n = files.map_or(0, HashSet::len);
541    Violation {
542        claim: Claim {
543            kind: ClaimKind::Symbol,
544            text: module.to_owned(),
545            source,
546            line: 0,
547        },
548        evidence: format!(
549            "public module `{module}` ({n} indexed file(s)) is named by no claim — code asserts it, the docs never do"
550        ),
551        fix: Some("document the module's role — or drop it from the public surface".into()),
552        severity: Severity::Warn,
553    }
554}

556/// One contradiction → one warn violation: the class name, the
557/// bucket key it matched on, and every object with its
558/// `file:line` site is the evidence. The anchor claim prefers an
559/// in-scope site — under `--diff` that's the *new* side of the
560/// disagreement, so output points at the drifted file.
561fn contradiction_violation(
562    c: &Contradiction,
563    claims: &[Claim],
564    in_scope: &dyn Fn(&Path) -> bool,
565) -> Violation {
566    let class = match c.kind {
567        ContradictionKind::CrossSource => "contradiction",
568        ContradictionKind::Reflexive => "self-contradiction",
569    };
570    let sites = c
571        .objects
572        .iter()
573        .map(|(o, src, line)| format!("`{o}` ({}:{line})", src.display()))
574        .collect::<Vec<_>>()
575        .join(" vs ");
576    let anchor = c
577        .objects
578        .iter()
579        .position(|(_, src, _)| in_scope(src))
580        .unwrap_or(0);
581    let (obj, src, line) = &c.objects[anchor];
582    // a triple's (source, line) always came from a claim in this
583    // set — the position() fallback is unreachable, just total
584    let i = claims
585        .iter()
586        .position(|k| k.source == *src && k.line as usize == *line && k.text.contains(obj.as_str()))
587        .unwrap_or_default();
588    Violation {
589        claim: claims[i].clone(),
590        evidence: format!(
591            "{class} — bucket `{}` asserts conflicting objects: {sites}",
592            c.bucket
593        ),
594        fix: Some("reconcile the claims — pick the true object for this slot".into()),
595        severity: Severity::Warn,
596    }
597}

599/// `stale-reference` — error tier: the edge exists but its target
600/// (or its source module) doesn't. Mechanical-grade evidence.
601fn stale_references(g: &Graph, out: &mut Vec<Violation>) {
602    let keys: HashSet<String> = g.modules.iter().map(|m| module_key(m)).collect();
603    let mut seen = HashSet::new();
604    for e in &g.edges {
605        // an edge recorded *from* a module the index no longer
606        // holds — orphaned rows a partial sync left behind
607        if !keys.contains(&module_key(&e.from)) {
608            emit_stale(
609                g,
610                &mut seen,
611                out,
612                e,
613                format!(
614                    "edge recorded from `{from}`, which is absent from the index — stale rows",
615                    from = e.from
616                ),
617            );
618            continue;
619        }
620        if !matches!(
621            e.kind,
622            EdgeKind::Imports | EdgeKind::PackageDep | EdgeKind::WorkspaceMember
623        ) {
624            continue; // calls/unknown kinds assert no module dep
625        }
626        if keys.contains(&module_key(&e.to)) {
627            continue; // resolved target — healthy edge
628        }
629        // `workspace-member` targets are internal by definition;
630        // the rest flag only when internal-shaped — else the
631        // missing target is just an external dep
632        if e.kind == EdgeKind::WorkspaceMember || internal_shaped(&e.to, &g.modules) {
633            emit_stale(
634                g,
635                &mut seen,
636                out,
637                e,
638                format!(
639                    "`{from}` {kind} `{to}`, but `{to}` is absent from the index — removed, renamed, or a stale index",
640                    from = e.from,
641                    kind = kind_str(e.kind),
642                    to = e.to
643                ),
644            );
645        }
646    }
647}
648
649fn emit_stale(
650    g: &Graph,
651    seen: &mut HashSet<(String, String)>,
652    out: &mut Vec<Violation>,
653    e: &Edge,
654    evidence: String,
655) {
656    if !seen.insert((e.from.clone(), e.to.clone())) {
657        return; // one finding per from→to pair
658    }
659    out.push(Violation {
660        claim: Claim {
661            kind: ClaimKind::Symbol,
662            text: format!(
663                "{from} -[{kind}]-> {to}",
664                from = e.from,
665                kind = kind_str(e.kind),
666                to = e.to
667            ),
668            source: g.index.clone(),
669            line: 0, // index-backed, not file-backed
670        },
671        evidence,
672        fix: Some("run `kuda sync` — or remove the dead reference".into()),
673        severity: Severity::Error,
674    });
675}

677/// `unsupported-claim` — warn tier: prose asserts an import/dep
678/// edge the index lacks. Both sides must resolve to indexed
679/// modules — anything fuzzier is unproveable, silence.
680fn unsupported_claims(claims: &[Claim], g: &Graph, out: &mut Vec<Violation>) {
681    for c in claims {
682        let mut seen = HashSet::new();
683        for (x, y) in edge_assertions(&c.text) {
684            if !seen.insert((x.to_lowercase(), y.to_lowercase())) {
685                continue;
686            }
687            if module_key(x) == module_key(y) {
688                continue; // self-edge claims answer nothing
689            }
690            let Some((a, b)) = resolve(x, &g.modules).zip(resolve(y, &g.modules)) else {
691                continue; // can't prove both sides internal — silence
692            };
693            if edge_exists(g, a, b) {
694                continue;
695            }
696            out.push(Violation {
697                claim: c.clone(),
698                evidence: format!(
699                    "asserts `{a}` depends on `{b}`, but the index records no imports/package-dep edge — doc drifted or the edge was removed"
700                ),
701                fix: Some("update the doc — or run `kuda sync` if the edge should exist".into()),
702                severity: Severity::Warn,
703            });
704        }
705    }
706}

708/// `X <verb> Y` pairs in prose — `imports`, `depends on`,
709/// `requires`, `uses`. Token-level, no regex: module names are
710/// words of ident/import-specifier characters.
711fn edge_assertions(text: &str) -> Vec<(&str, &str)> {
712    let words: Vec<&str> = text
713        .split(|c: char| !(c.is_alphanumeric() || matches!(c, '_' | '-' | '.' | '@' | '/')))
714        .filter(|w| !w.is_empty())
715        .collect();
716    let mut out = Vec::new();
717    for (i, w) in words.iter().enumerate() {
718        // the verb's object is one token ahead — two for the
719        // phrasal `depends on`
720        let gap = match w.to_lowercase().as_str() {
721            "imports" | "requires" | "uses" => 1,
722            "depends"
723                if words
724                    .get(i + 1)
725                    .is_some_and(|n| n.eq_ignore_ascii_case("on")) =>
726            {
727                2
728            }
729            _ => continue,
730        };
731        let (Some(x), Some(y)) = (i.checked_sub(1).map(|j| words[j]), words.get(i + gap)) else {
732            continue;
733        };
734        // sentence-ending periods ride along — '.' is a legit
735        // interior char (go paths) but never a boundary one
736        out.push((x.trim_matches('.'), y.trim_matches('.')));
737    }
738    out
739}

741/// Does the graph record a dep-ish edge `a → b`? Compared on
742/// kuda's folded `module_key`, not raw spelling.
743fn edge_exists(g: &Graph, a: &str, b: &str) -> bool {
744    let (ka, kb) = (module_key(a), module_key(b));
745    g.edges.iter().any(|e| {
746        matches!(
747            e.kind,
748            EdgeKind::Imports | EdgeKind::PackageDep | EdgeKind::WorkspaceMember
749        ) && module_key(&e.from) == ka
750            && module_key(&e.to) == kb
751    })
752}
753
754/// Claim ident → indexed module name: folded match first, then
755/// subpath (`pkg/sub` claims resolve to `pkg`) — mirrors kuda's
756/// `resolve_module`. Longest match wins, ties break
757/// lexicographically — deterministic regardless of set order.
758fn resolve<'a>(name: &str, modules: &'a HashSet<String>) -> Option<&'a str> {
759    let want = module_key(name);
760    modules
761        .iter()
762        .filter(|m| {
763            let k = module_key(m);
764            k == want || want.starts_with(&format!("{k}/"))
765        })
766        .max_by_key(|m| (m.len(), m.as_str()))
767        .map(String::as_str)
768}
769
770/// kuda's `module_key` — dash/underscore/case fold, the same
771/// normalization the indexer applies to edge targets.
772fn module_key(name: &str) -> String {
773    name.replace('-', "_").to_lowercase()
774}
775
776/// First '-'/'_'/'.'-separated segment — the naming-family prefix.
777fn seg0(name: &str) -> &str {
778    // split yields ≥1 item on any input — unwrap_or avoids unwrap
779    name.split(['-', '_', '.']).next().unwrap_or(name)
780}
781
782/// Is `to` shaped like a workspace member of this repo? A missing
783/// target flags only when it shares the monorepo's own naming
784/// family — a '/'-scope or a ≥3-char flat prefix carried by ≥2
785/// indexed modules (a convention, not a coincidence). Anything
786/// else is an external dep, not a stale reference.
787fn internal_shaped(to: &str, modules: &HashSet<String>) -> bool {
788    if to.contains('/') {
789        let Some(sc) = scope(to) else {
790            return false; // two-segment names carry no reliable scope
791        };
792        return modules
793            .iter()
794            .filter(|m| scope(m).is_some_and(|s| s.eq_ignore_ascii_case(sc)))
795            .count()
796            >= 2;
797    }
798    let lower = to.to_lowercase();
799    let first = seg0(&lower);
800    first.len() >= 3
801        && modules
802            .iter()
803            .filter(|m| seg0(&m.to_lowercase()) == first)
804            .count()
805            >= 2
806}
807
808/// The '/'-scope a name belongs to: `@org` for `@org/pkg`;
809/// `host/org` for go-style `github.com/org/repo` — so an external
810/// `github.com/ext/x` can't collide with the monorepo's own org.
811/// `None` for flat names and two-segment paths.
812fn scope(name: &str) -> Option<&str> {
813    let (s1, rest) = name.split_once('/')?;
814    if s1.starts_with('@') {
815        Some(s1)
816    } else {
817        rest.split_once('/')
818            .map(|(s2, _)| &name[..s1.len() + 1 + s2.len()])
819    }
820}

1167        // (b) shared token, different objects → one candidate with
1168        // the original spellings and both sites
1169        let pair = [
1170            t("auth flow", "uses", "jwt", "a.md", 3),
1171            t("auth service", "uses", "sessions", "b.md", 7),
1172        ];
1173        let c = semantic_candidates(&pair, &[]);
1174        assert_eq!(c.len(), 1, "{c:?}");
1175        assert_eq!(c[0].relation, "uses");
1176        assert_eq!(c[0].left_subject, "auth service");
1177        assert_eq!(c[0].left_object, "sessions");
1178        assert_eq!(c[0].left_source, PathBuf::from("b.md"));
1179        assert_eq!(c[0].left_line, 7);
1180        assert_eq!(c[0].right_subject, "auth flow");
1181        assert_eq!(c[0].right_object, "jwt");
1182        assert_eq!(c[0].right_line, 3);
1183        // (c) unrelated subjects → silence, even with different objects
1184        let unrelated = [
1185            t("auth", "uses", "jwt", "a.md", 1),
1186            t("cache", "uses", "redis", "b.md", 2),
1187        ];
1188        assert!(semantic_candidates(&unrelated, &[]).is_empty());
1189        // (d) same object → nothing to adjudicate
1190        let agree = [
1191            t("auth flow", "uses", "jwt", "a.md", 1),
1192            t("auth service", "uses", "jwt", "b.md", 2),
1193        ];
1194        assert!(semantic_candidates(&agree, &[]).is_empty());
1195        // (e) different relations → different slots, not a conflict
1196        let rels = [
1197            t("auth flow", "uses", "jwt", "a.md", 1),
1198            t("auth service", "is", "sessions", "b.md", 2),
1199        ];
1200        assert!(semantic_candidates(&rels, &[]).is_empty());

1203    /// ADR-0002 ph5 — duplicates collapse onto one candidate, the
1204    /// output order is fixed, and an alias that folds two spellings
1205    /// onto one canon subject removes the pair from the intake.
1206    #[test]
1207    fn semantic_candidates_dedupe_and_order() {
1208        let t = |subj: &str, rel: &str, obj: &str, src: &str, line: usize| Triple {
1209            subject: subj.into(),
1210            relation: rel.into(),
1211            object: obj.into(),
1212            source: PathBuf::from(src),
1213            line,
1214        };
1215        // the same disagreement spelled twice → one candidate
1216        let dupes = [
1217            t("auth service", "uses", "sessions", "b.md", 2),
1218            t("auth flow", "uses", "jwt", "a.md", 1),
1219            t("auth flow", "uses", "jwt", "c.md", 4),
1220        ];
1221        let c = semantic_candidates(&dupes, &[]);
1222        assert_eq!(c.len(), 1, "{c:?}");
1223        assert_eq!(c[0].left_subject, "auth service");
1224        assert_eq!(c[0].left_source, PathBuf::from("b.md"));
1225        // sorted on (relation, left_subject, right_subject,
1226        // left_object, right_object) — not input order
1227        let mixed = [
1228            t("auth service", "uses", "sessions", "b.md", 2),
1229            t("auth flow", "uses", "jwt", "a.md", 1),
1230            t("auth cookie", "uses", "tokens", "d.md", 3),
1231        ];
1232        let c = semantic_candidates(&mixed, &[]);
1233        let order: Vec<(&str, &str)> = c
1234            .iter()
1235            .map(|c| (c.left_subject.as_str(), c.right_subject.as_str()))
1236            .collect();
1237        assert_eq!(
1238            order,
1239            [
1240                ("auth cookie", "auth flow"),
1241                ("auth service", "auth cookie"),
1242                ("auth service", "auth flow"),
1243            ],
1244            "{c:?}"
1245        );
1246        // without an alias the two spellings share "flow" → one
1247        // candidate; the alias folds them onto one canon subject, so
1248        // the pair becomes the buckets' problem and the intake is
1249        // empty
1250        let spellings = [
1251            t("login flow", "uses", "jwt", "a.md", 1),
1252            t("auth flow", "uses", "sessions", "b.md", 2),
1253        ];
1254        assert_eq!(semantic_candidates(&spellings, &[]).len(), 1);
1255        let aliases = [("login flow".to_owned(), "auth flow".to_owned())];
1256        assert!(semantic_candidates(&spellings, &aliases).is_empty());
1257    }

1259    /// ADR-0002 ph3 — `in_scope` gates *findings*, not the
1260    /// buckets: a conflict touching a scoped file emits and
1261    /// anchors on the in-scope side; one entirely outside stays
1262    /// silent. The cache path returns identical findings — once
1263    /// extracting, once reading stored rows.
1264    #[test]
1265    fn prose_scope_gate_and_cache() {
1266        let dir = std::env::temp_dir().join(format!("mimus-prose-{}", std::process::id()));
1267        let _ = std::fs::remove_dir_all(&dir);
1268        std::fs::create_dir_all(&dir).unwrap();
1269        let (a, b, c, d) = (
1270            dir.join("a.md"),
1271            dir.join("b.md"),
1272            dir.join("c.md"),
1273            dir.join("d.md"),
1274        );
1275        std::fs::write(&a, "auth uses sessions\n").unwrap();
1276        std::fs::write(&b, "auth uses jwt\n").unwrap();
1277        std::fs::write(&c, "pool uses postgres\n").unwrap();
1278        std::fs::write(&d, "pool uses redis\n").unwrap();
1279        let mk = |text: &str, src: &Path, line: u32| Claim {
1280            kind: ClaimKind::Behavior,
1281            text: text.into(),
1282            source: src.to_path_buf(),
1283            line,
1284        };
1285        // b.md first-seen so the scope-preferred anchor is
1286        // observable: in-scope a.md is the *second* object site
1287        let claims = [
1288            mk("auth uses jwt", &b, 1),
1289            mk("auth uses sessions", &a, 1),
1290            mk("pool uses postgres", &c, 1),
1291            mk("pool uses redis", &d, 1),
1292        ];
1293        let only_a: &dyn Fn(&Path) -> bool = &|p| p == a;
1294        // uncached path: a↔b emits (anchored on in-scope a.md,
1295        // though b.md's site is first-seen), c↔d stays silent
1296        let mut chk = ProseContradictionChecker {
1297            aliases: Vec::new(),
1298            cache: None,
1299            triples: Vec::new(),
1300        };
1301        let v = chk.run_prose(&claims, only_a);
1302        assert_eq!(v.len(), 1, "{v:?}");
1303        assert!(v[0].evidence.contains("auth|uses"), "{}", v[0].evidence);
1304        assert!(v[0].evidence.contains("b.md:1"), "{}", v[0].evidence);
1305        assert_eq!(v[0].claim.source, a, "anchor prefers the in-scope site");
1306        assert_eq!(v[0].claim.text, "auth uses sessions");
1307        // cached path — same findings across an extract run and
1308        // a stored-rows run
1309        let mut chk = ProseContradictionChecker {
1310            aliases: Vec::new(),
1311            cache: Some(TriplesCache::open(&dir.join("triples.db")).unwrap()),
1312            triples: Vec::new(),
1313        };
1314        for _ in 0..2 {
1315            let v = chk.run_prose(&claims, only_a);
1316            assert_eq!(v.len(), 1, "{v:?}");
1317            assert_eq!(v[0].claim.source, a);
1318        }
1319        // nothing in scope → silence; everything → both conflicts
1320        assert!(chk.run_prose(&claims, &|_| false).is_empty());
1321        let v = chk.run_prose(&claims, &|_| true);
1322        assert_eq!(v.len(), 2, "{v:?}");
1323        let _ = std::fs::remove_dir_all(&dir);
1324    }

1357    /// ADR-0002 ph4 — a public module no document names warns; any
1358    /// canon or alias spelling documents it. Provenance gates the
1359    /// finding under `--diff`; a snapshot without it reports rather
1360    /// than drops.
1361    #[test]
1362    fn undocumented_fact_class() {
1363        let g = Graph {
1364            modules: ["app-cli", "app-core"]
1365                .iter()
1366                .map(|s| s.to_string())
1367                .collect(),
1368            edges: vec![Edge {
1369                from: "app-cli".into(),
1370                to: "app-core".into(),
1371                kind: EdgeKind::Imports,
1372            }],
1373            index: PathBuf::from("idx.sqlite"),
1374            ..Graph::default()
1375        };
1376        let facts = UndocumentedFactChecker { aliases: &[] };
1377        let all: &dyn Fn(&Path) -> bool = &|_| true;
1378        // any canon spelling (dash, underscore, case) documents it
1379        assert!(facts
1380            .run_facts("app-core owns the domain", &g, all)
1381            .is_empty());
1382        assert!(facts
1383            .run_facts("the APP_CORE crate owns it", &g, all)
1384            .is_empty());
1385        // a name of stopwords only has no needle — silence
1386        assert!(documented("the", " app core ", &[]));
1387        // no document names it → one warn, anchored on the index
1388        let v = facts.run_facts("app-cli parses flags", &g, all);
1389        assert_eq!(v.len(), 1, "{v:?}");
1390        let e = &v[0].evidence;
1391        assert!(e.contains("app-core"), "{e}");
1392        assert!(e.contains("0 indexed file(s)"), "{e}");
1393        assert_eq!(v[0].severity, Severity::Warn);
1394        assert_eq!(v[0].claim.text, "app-core");
1395        assert_eq!(v[0].claim.line, 0);
1396        assert_eq!(v[0].claim.source, PathBuf::from("idx.sqlite"));
1397        assert!(v[0].fix.is_some());
1398        // an alias spelling documents the module it canonicalizes to
1399        let aliased = UndocumentedFactChecker {
1400            aliases: &[("the core crate".to_owned(), "app-core".to_owned())],
1401        };
1402        assert!(aliased
1403            .run_facts("the core crate owns the domain", &g, all)
1404            .is_empty());
1405        // a stopword alias has no needle, and unrelated aliases
1406        // don't widen the match
1407        let odd = UndocumentedFactChecker {
1408            aliases: &[
1409                ("the".to_owned(), "app-core".to_owned()),
1410                ("login flow".to_owned(), "auth".to_owned()),
1411            ],
1412        };
1413        assert_eq!(odd.run_facts("unrelated docs", &g, all).len(), 1);
1414        // provenance: `--diff` scope asks the module's own files
1415        let g = Graph {
1416            module_files: [(
1417                "app-core".to_owned(),
1418                [PathBuf::from("crates/core/src/lib.rs")]
1419                    .into_iter()
1420                    .collect(),
1421            )]
1422            .into_iter()
1423            .collect(),
1424            ..g
1425        };
1426        let in_scope: &dyn Fn(&Path) -> bool = &|p| p.ends_with("lib.rs");
1427        let v = facts.run_facts("app-cli parses flags", &g, in_scope);
1428        assert_eq!(v.len(), 1, "{v:?}");
1429        assert_eq!(v[0].claim.source, PathBuf::from("crates/core/src/lib.rs"));
1430        assert!(v[0].evidence.contains("1 indexed file(s)"), "{e}");
1431        // module untouched by the diff → silence
1432        assert!(facts
1433            .run_facts("app-cli parses flags", &g, &|_| false)
1434            .is_empty());
1435    }
