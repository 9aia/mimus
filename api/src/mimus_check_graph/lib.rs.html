1//! Graph adapter — set-level checking (ADR-0002): index-edge
2//! classes against the kuda index (phase 1) plus index-free prose
3//! contradiction buckets (phase 2). The kuda index is read through
4//! `kuda-store-sqlite`'s blessed file-format contract
5//! ([`SqliteStore::open_readonly`]) — never raw SQL; the only
6//! remaining sqlite surface is the `.mimus/triples.db` cache.
7//!
8//! Classes:
9//! - `stale-reference` (error): an `imports`/`package-dep`/
10//!   `workspace-member` edge whose target resolves to no indexed
11//!   module — or whose *source* module is gone entirely. Internal
12//!   vs external is decided conservatively: an unresolved target
13//!   flags only when it shares the repo's own naming family (a
14//!   '/'-scope or a prefix segment carried by ≥2 indexed modules);
15//!   bare external-looking names (`serde`, `react`,
16//!   `github.com/ext/x`) are silence.
17//! - `unsupported-claim` (warn): prose of the shape
18//!   `X imports|depends on|requires|uses Y` where both sides
19//!   resolve to indexed modules and the index records no such
20//!   edge. Either side unresolvable → the triple is unproveable →
21//!   silence (evidence-or-silence, ADR-0001).
22//! - `contradiction` / `self-contradiction` (warn, phase 2):
23//!   prose claims reduce to `(subject, relation, object)` triples
24//!   (mimus-claims::triples) and conflicts fall out of the
25//!   `(subject, relation)` bucket — cross-source for doc↔doc,
26//!   reflexive when one file asserts two objects for the same
27//!   slot. No index needed: this pass runs on any claim set.
28//! - phase 3: a per-file triples cache (`cache` module — sqlite
29//!   at `.mimus/triples.db`, keyed on content hash + extractor
30//!   version) skips re-extracting unchanged files, and the prose
31//!   pass's `in_scope` gate lets `--diff` keep the buckets whole
32//!   while only reporting conflicts that touch changed files.
33//! - phase 5: the semantic tier is rescoped to *fuzzy-candidate
34//!   adjudication*. `semantic_candidates` turns the prose triples
35//!   the buckets leave unsettled (canon-differing subjects that
36//!   share a token, same relation, different objects) into the
37//!   agent's intake list — pre-filtered, so the agent adjudicates
38//!   pairs instead of scanning every claim.
39//! - `undocumented-fact` (warn, phase 4): reverse coverage — a
40//!   *public* module (one the index records as a dependency target)
41//!   that no claim names is code the docs never assert. Module
42//!   granularity, not edge: every indexed import is a fact nobody
43//!   documents, while a module is the smallest unit a doc page can
44//!   reasonably cover. `in_scope` gates it the way it gates the
45//!   buckets — under `--diff` only modules whose own sources changed
46//!   report.
47

64/// Load the kuda index into a `Graph` snapshot — read-only, never
65/// creates the file. Missing tables yield empty sets: a
66/// symbols-only index still produces a usable graph.
67pub fn load_graph(index: &Path) -> Result<Graph, String> {
68    let store = SqliteStore::open_readonly(index)?;
69    let mut g = Graph {
70        index: index.to_path_buf(),
71        ..Graph::default()
72    };
73    if store.has_table("edges")? {
74        g.edges = store.all_edges()?.into_iter().map(edge_from).collect();
75    }
76    if store.has_table("symbols")? {
77        let syms = store.all_symbols()?;
78        g.modules = syms.iter().map(|s| s.module.clone()).collect();
79        g.symbols = syms.iter().map(|s| s.name.clone()).collect();
80        for s in &syms {
81            g.module_files
82                .entry(s.module.clone())
83                .or_default()
84                .insert(s.file.clone());
85        }
86    }
87    Ok(g)
88}

90/// kuda `DepEdge` → mimus `Edge` — `references` edges (record
91/// corpora) aren't a module-dep relation mimus checks, so they
92/// land on `Other`.
93fn edge_from(e: kuda_core::DepEdge) -> Edge {
94    Edge {
95        from: e.from,
96        to: e.to,
97        kind: match e.kind {
98            KudaEdgeKind::Imports => EdgeKind::Imports,
99            KudaEdgeKind::Calls => EdgeKind::Calls,
100            KudaEdgeKind::PackageDep => EdgeKind::PackageDep,
101            KudaEdgeKind::WorkspaceMember => EdgeKind::WorkspaceMember,
102            KudaEdgeKind::References => EdgeKind::Other,
103        },
104    }
105}

107/// Age of the index — `meta.generated_at` (unix secs, written by
108/// `kuda sync`) when present, else file mtime. `None` when the
109/// file can't be read at all; `mimus doctor` reports it.
110pub fn index_age(index: &Path) -> Option<Duration> {
111    meta_generated_at(index)
112        .or_else(|| mtime(index))
113        .and_then(|ts| {
114            std::time::UNIX_EPOCH
115                .checked_add(Duration::from_secs(ts))
116                .and_then(|t| t.elapsed().ok())
117        })
118}
119
120fn meta_generated_at(index: &Path) -> Option<u64> {
121    let store = SqliteStore::open_readonly(index).ok()?;
122    store
123        .meta("generated_at")
124        .ok()
125        .flatten()
126        .and_then(|v| v.parse().ok())
127}

129fn mtime(index: &Path) -> Option<u64> {
130    std::fs::metadata(index)
131        .ok()?
132        .modified()
133        .ok()?
134        .duration_since(std::time::UNIX_EPOCH)
135        .ok()
136        .map(|d| d.as_secs())
137}

162/// Prose contradiction pass — ADR-0002 phase 2. The bucket
163/// classes `contradiction` (cross-source, doc↔doc) and
164/// `self-contradiction` (reflexive: one file asserting ≥2 objects
165/// for the same slot). Index-free — prose triples don't need the
166/// kuda index, so this runs on every claim set, not just when
167/// `[check].kuda_index` is configured. Warn tier: extraction is
168/// heuristic, so these report but never gate.
169pub struct ProseContradictionChecker {
170    /// Subject aliases from `[check].aliases` — `(phrase,
171    /// canonical)` pairs resolved at bucket-key time.
172    pub aliases: Vec<(String, String)>,
173    /// Per-file triples cache (phase 3) — `Some` serves unchanged
174    /// files' triples from `.mimus/triples.db`; `None` extracts
175    /// every run.
176    pub cache: Option<TriplesCache>,
177    /// Triples from the last `run_prose` — the semantic tier's
178    /// candidate intake reuses them instead of re-extracting.
179    pub triples: Vec<Triple>,
180}
181
182impl ProseContradictionChecker {
183    /// claims → triples → buckets → warn-tier violations.
184    /// `in_scope` gates *findings*, not extraction: the buckets
185    /// see every file's triples (a contradiction needs both
186    /// sides) but a conflict emits only when ≥1 object site is
187    /// in scope — under `--diff` that's the changed-file side,
188    /// so unchanged↔unchanged drift stays silent.
189    pub fn run_prose(
190        &mut self,
191        claims: &[Claim],
192        in_scope: &dyn Fn(&Path) -> bool,
193    ) -> Vec<Violation> {
194        let triples = match self.cache.as_mut() {
195            Some(c) => extract_triples_cached(claims, c),
196            None => extract_triples(claims),
197        };
198        // kept for the semantic tier's intake (phase 5) — the agent
199        // gets candidates off the same triples, not a second extract
200        self.triples = triples;
201        contradictions_with_aliases(&self.triples, &self.aliases)
202            .iter()
203            .filter(|c| c.objects.iter().any(|(_, src, _)| in_scope(src)))
204            .map(|c| contradiction_violation(c, claims, in_scope))
205            .collect()
206    }
207}

209/// One fuzzy pair the mechanical buckets can't settle — the
210/// semantic tier's intake list. The bucket pass keeps the
211/// mechanical-grade conflicts (canon-equal subject, different
212/// object) to itself and hands the agent only what needs
213/// understanding: near-duplicate subjects with a shared content
214/// token asserting different objects for the same relation.
215#[derive(Debug, Clone, PartialEq, Eq)]
216pub struct SemanticCandidate {
217    /// Relation both triples assert (`uses`, `is`, …) — the slot
218    /// the two sides disagree on.
219    pub relation: String,
220    /// Left triple's subject, as spelled at its own site — the
221    /// canonical form is the filter, never the output.
222    pub left_subject: String,
223    /// Left triple's object.
224    pub left_object: String,
225    /// File the left triple was extracted from.
226    pub left_source: PathBuf,
227    /// 1-based line in `left_source`.
228    pub left_line: usize,
229    /// Right triple's subject, as spelled at its own site.
230    pub right_subject: String,
231    /// Right triple's object.
232    pub right_object: String,
233    /// File the right triple was extracted from.
234    pub right_source: PathBuf,
235    /// 1-based line in `right_source`.
236    pub right_line: usize,
237}

239/// Triples → the semantic tier's candidate list (ADR-0002 ph5).
240///
241/// The buckets own canon-equal subjects (`contradiction`,
242/// `self-contradiction`); this is the residue they can't decide.
243/// A pair becomes a candidate when all four hold:
244///
245/// - the canonical subjects *differ* — canon-equal is the buckets'
246///   job, and re-emitting it would spend an agent call on a
247///   mechanical conflict;
248/// - both triples assert the same relation — different relations
249///   are different slots, not a disagreement;
250/// - the canonical subjects share ≥1 content token — `auth` pairs
251///   with `auth flow`, never with `cache`. This is the synonym /
252///   near-duplicate heuristic that keeps the intake small;
253/// - the objects differ — agreeing triples are not a question.
254///
255/// Each unordered pair emits once, and the output is sorted on
256/// `(relation, left_subject, right_subject, left_object,
257/// right_object)` so two runs over the same triples print the same
258/// prompt. Original spellings survive into the struct: the agent
259/// needs the real `file:line` site, not the bucket key.
260pub fn semantic_candidates(
261    triples: &[Triple],
262    aliases: &[(String, String)],
263) -> Vec<SemanticCandidate> {
264    let canon: Vec<String> = triples
265        .iter()
266        .map(|t| canon_subject(&t.subject, aliases))
267        .collect();
268    // token sets once per triple — pairing is the hot loop, and a
269    // 10k-claim sweep has no room for re-splitting subjects inside it
270    let toks: Vec<HashSet<&str>> = canon
271        .iter()
272        .map(|c| c.split_whitespace().collect())
273        .collect();
274    // only same-relation triples can disagree about a slot
275    let mut by_rel: HashMap<&str, Vec<usize>> = HashMap::new();
276    for (i, t) in triples.iter().enumerate() {
277        by_rel.entry(t.relation.as_str()).or_default().push(i);
278    }
279    // unordered (subject canon, object) pair sides — one entry per
280    // structural disagreement, however many files spell it
281    let mut seen: HashSet<(String, String, String, String, String)> = HashSet::new();
282    let mut out = Vec::new();
283    for group in by_rel.values() {
284        for x in 0..group.len() {
285            for y in (x + 1)..group.len() {
286                let (i, j) = (group[x], group[y]);
287                let (a, b) = (&triples[i], &triples[j]);
288                if canon[i] == canon[j] || a.object == b.object || toks[i].is_disjoint(&toks[j]) {
289                    continue;
290                }
291                // deterministic side order — the smaller (canon,
292                // object) is "left", so duplicate spellings collapse
293                // onto one candidate instead of two mirrored ones
294                let (l, r) = if (canon[i].as_str(), a.object.as_str())
295                    <= (canon[j].as_str(), b.object.as_str())
296                {
297                    (i, j)
298                } else {
299                    (j, i)
300                };
301                let (lt, rt) = (&triples[l], &triples[r]);
302                let key = (
303                    lt.relation.clone(),
304                    canon[l].clone(),
305                    lt.object.clone(),
306                    canon[r].clone(),
307                    rt.object.clone(),
308                );
309                if !seen.insert(key) {
310                    continue;
311                }
312                out.push(SemanticCandidate {
313                    relation: lt.relation.clone(),
314                    left_subject: lt.subject.clone(),
315                    left_object: lt.object.clone(),
316                    left_source: lt.source.clone(),
317                    left_line: lt.line,
318                    right_subject: rt.subject.clone(),
319                    right_object: rt.object.clone(),
320                    right_source: rt.source.clone(),
321                    right_line: rt.line,
322                });
323            }
324        }
325    }
326    out.sort_by(|a, b| {
327        (
328            &a.relation,
329            &a.left_subject,
330            &a.right_subject,
331            &a.left_object,
332            &a.right_object,
333            &a.left_source,
334            a.left_line,
335            &a.right_source,
336            a.right_line,
337        )
338            .cmp(&(
339                &b.relation,
340                &b.left_subject,
341                &b.right_subject,
342                &b.left_object,
343                &b.right_object,
344                &b.left_source,
345                b.left_line,
346                &b.right_source,
347                b.right_line,
348            ))
349    });
350    out
351}

353/// `undocumented-fact` — ADR-0002 phase 4: reverse coverage
354/// (code→doc), warn tier. The *public surface* is the set of
355/// modules the index records as a dependency target — internal
356/// details nobody imports aren't contract. A public module emits
357/// when no document in the mined set names it, matched the way the
358/// buckets match subjects (`canon_subject`: case, `-`/`_`, plurals
359/// and `[check].aliases` all fold), so `mimus-core`, `mimus_core`
360/// and an aliased spelling each count as documenting it. The
361/// haystack is document text, not extracted claims: a doc that
362/// merely lists a crate (`[mimus-check-acp](…)`) documents it
363/// without asserting a checkable claim — claim-level matching would
364/// flag every crate index page.
365pub struct UndocumentedFactChecker<'a> {
366    /// Subject aliases from `[check].aliases` — `(phrase,
367    /// canonical)` pairs, the same map the contradiction buckets
368    /// resolve.
369    pub aliases: &'a [(String, String)],
370}
371
372impl UndocumentedFactChecker<'_> {
373    /// `in_scope` gates *findings*, not the scan: under `--diff`
374    /// only a module whose own indexed sources changed reports;
375    /// unscoped runs pass `|_| true`. The module set itself always
376    /// comes from the whole index — a snapshot has no per-run
377    /// scope to narrow. `docs` is the concatenated text of every
378    /// document the claim set was mined from (cards included).
379    pub fn run_facts(
380        &self,
381        docs: &str,
382        graph: &Graph,
383        in_scope: &dyn Fn(&Path) -> bool,
384    ) -> Vec<Violation> {
385        // padded canonical doc text — the haystack, so `mimus-core`
386        // can't match inside `mimus-core-extra`
387        let hay = format!(" {} ", canon_subject(docs, self.aliases));
388        public_modules(graph)
389            .into_iter()
390            .filter(|m| !documented(m, &hay, self.aliases))
391            .filter(|m| module_in_scope(graph, m, in_scope))
392            .map(|m| fact_violation(graph, m))
393            .collect()
394    }
395}

397/// Modules the index records as a dependency *target* — the public
398/// surface other code actually consumes. `calls` edges prove nothing
399/// about the API, an unresolved target is external, and a self-edge
400/// (`a → a`) marks nothing.
401fn public_modules(graph: &Graph) -> BTreeSet<&str> {
402    graph
403        .edges
404        .iter()
405        .filter(|e| {
406            matches!(
407                e.kind,
408                EdgeKind::Imports | EdgeKind::PackageDep | EdgeKind::WorkspaceMember
409            )
410        })
411        .filter_map(|e| {
412            let target = resolve(&e.to, &graph.modules)?;
413            (module_key(&e.from) != module_key(target)).then_some(target)
414        })
415        .collect()
416}

418/// Does any document name `module`? Canon-matched against the padded
419/// haystack, plus every `[check].aliases` spelling whose canonical
420/// target folds onto the module.
421fn documented(module: &str, hay: &str, aliases: &[(String, String)]) -> bool {
422    let canon = canon_subject(module, aliases);
423    if canon.is_empty() {
424        // a name made only of stopwords has no needle to look for —
425        // silence beats a substring match on padding
426        return true;
427    }
428    let spellings = std::iter::once(canon.clone()).chain(
429        aliases
430            .iter()
431            .filter(|(_, target)| canon_subject(target, aliases) == canon)
432            .map(|(alias, _)| canon_subject(alias, &[])),
433    );
434    spellings
435        .filter(|n| !n.is_empty())
436        .any(|n| hay.contains(&format!(" {n} ")))
437}

439/// `--diff` gate: does any of the module's own indexed sources sit
440/// in scope? A snapshot without file provenance (a hand-built
441/// `Graph`) can't answer — it reports rather than silently drop.
442fn module_in_scope(graph: &Graph, module: &str, in_scope: &dyn Fn(&Path) -> bool) -> bool {
443    match graph.module_files.get(module) {
444        Some(files) => files.iter().any(|f| in_scope(f)),
445        None => true,
446    }
447}

449/// One undocumented public module → one warn. `line: 0` — the
450/// finding is module-level, anchored on a real source file so the
451/// report points at code rather than the index.
452fn fact_violation(graph: &Graph, module: &str) -> Violation {
453    let files = graph.module_files.get(module);
454    let source = files
455        .and_then(|f| f.iter().min())
456        .cloned()
457        .unwrap_or_else(|| graph.index.clone());
458    let n = files.map_or(0, HashSet::len);
459    Violation {
460        claim: Claim {
461            kind: ClaimKind::Symbol,
462            text: module.to_owned(),
463            source,
464            line: 0,
465        },
466        evidence: format!(
467            "public module `{module}` ({n} indexed file(s)) is named by no claim — code asserts it, the docs never do"
468        ),
469        fix: Some("document the module's role — or drop it from the public surface".into()),
470        severity: Severity::Warn,
471    }
472}

474/// One contradiction → one warn violation: the class name, the
475/// bucket key it matched on, and every object with its
476/// `file:line` site is the evidence. The anchor claim prefers an
477/// in-scope site — under `--diff` that's the *new* side of the
478/// disagreement, so output points at the drifted file.
479fn contradiction_violation(
480    c: &Contradiction,
481    claims: &[Claim],
482    in_scope: &dyn Fn(&Path) -> bool,
483) -> Violation {
484    let class = match c.kind {
485        ContradictionKind::CrossSource => "contradiction",
486        ContradictionKind::Reflexive => "self-contradiction",
487    };
488    let sites = c
489        .objects
490        .iter()
491        .map(|(o, src, line)| format!("`{o}` ({}:{line})", src.display()))
492        .collect::<Vec<_>>()
493        .join(" vs ");
494    let anchor = c
495        .objects
496        .iter()
497        .position(|(_, src, _)| in_scope(src))
498        .unwrap_or(0);
499    let (obj, src, line) = &c.objects[anchor];
500    // a triple's (source, line) always came from a claim in this
501    // set — the position() fallback is unreachable, just total
502    let i = claims
503        .iter()
504        .position(|k| k.source == *src && k.line as usize == *line && k.text.contains(obj.as_str()))
505        .unwrap_or_default();
506    Violation {
507        claim: claims[i].clone(),
508        evidence: format!(
509            "{class} — bucket `{}` asserts conflicting objects: {sites}",
510            c.bucket
511        ),
512        fix: Some("reconcile the claims — pick the true object for this slot".into()),
513        severity: Severity::Warn,
514    }
515}

517/// `stale-reference` — error tier: the edge exists but its target
518/// (or its source module) doesn't. Mechanical-grade evidence.
519fn stale_references(g: &Graph, out: &mut Vec<Violation>) {
520    let keys: HashSet<String> = g.modules.iter().map(|m| module_key(m)).collect();
521    let mut seen = HashSet::new();
522    for e in &g.edges {
523        // an edge recorded *from* a module the index no longer
524        // holds — orphaned rows a partial sync left behind
525        if !keys.contains(&module_key(&e.from)) {
526            emit_stale(
527                g,
528                &mut seen,
529                out,
530                e,
531                format!(
532                    "edge recorded from `{from}`, which is absent from the index — stale rows",
533                    from = e.from
534                ),
535            );
536            continue;
537        }
538        if !matches!(
539            e.kind,
540            EdgeKind::Imports | EdgeKind::PackageDep | EdgeKind::WorkspaceMember
541        ) {
542            continue; // calls/unknown kinds assert no module dep
543        }
544        if keys.contains(&module_key(&e.to)) {
545            continue; // resolved target — healthy edge
546        }
547        // `workspace-member` targets are internal by definition;
548        // the rest flag only when internal-shaped — else the
549        // missing target is just an external dep
550        if e.kind == EdgeKind::WorkspaceMember || internal_shaped(&e.to, &g.modules) {
551            emit_stale(
552                g,
553                &mut seen,
554                out,
555                e,
556                format!(
557                    "`{from}` {kind} `{to}`, but `{to}` is absent from the index — removed, renamed, or a stale index",
558                    from = e.from,
559                    kind = kind_str(e.kind),
560                    to = e.to
561                ),
562            );
563        }
564    }
565}

567fn emit_stale(
568    g: &Graph,
569    seen: &mut HashSet<(String, String)>,
570    out: &mut Vec<Violation>,
571    e: &Edge,
572    evidence: String,
573) {
574    if !seen.insert((e.from.clone(), e.to.clone())) {
575        return; // one finding per from→to pair
576    }
577    out.push(Violation {
578        claim: Claim {
579            kind: ClaimKind::Symbol,
580            text: format!(
581                "{from} -[{kind}]-> {to}",
582                from = e.from,
583                kind = kind_str(e.kind),
584                to = e.to
585            ),
586            source: g.index.clone(),
587            line: 0, // index-backed, not file-backed
588        },
589        evidence,
590        fix: Some("run `kuda sync` — or remove the dead reference".into()),
591        severity: Severity::Error,
592    });
593}

595/// `unsupported-claim` — warn tier: prose asserts an import/dep
596/// edge the index lacks. Both sides must resolve to indexed
597/// modules — anything fuzzier is unproveable, silence.
598fn unsupported_claims(claims: &[Claim], g: &Graph, out: &mut Vec<Violation>) {
599    for c in claims {
600        let mut seen = HashSet::new();
601        for (x, y) in edge_assertions(&c.text) {
602            if !seen.insert((x.to_lowercase(), y.to_lowercase())) {
603                continue;
604            }
605            if module_key(x) == module_key(y) {
606                continue; // self-edge claims answer nothing
607            }
608            let Some((a, b)) = resolve(x, &g.modules).zip(resolve(y, &g.modules)) else {
609                continue; // can't prove both sides internal — silence
610            };
611            if edge_exists(g, a, b) {
612                continue;
613            }
614            out.push(Violation {
615                claim: c.clone(),
616                evidence: format!(
617                    "asserts `{a}` depends on `{b}`, but the index records no imports/package-dep edge — doc drifted or the edge was removed"
618                ),
619                fix: Some("update the doc — or run `kuda sync` if the edge should exist".into()),
620                severity: Severity::Warn,
621            });
622        }
623    }
624}

626/// `X <verb> Y` pairs in prose — `imports`, `depends on`,
627/// `requires`, `uses`. Token-level, no regex: module names are
628/// words of ident/import-specifier characters.
629fn edge_assertions(text: &str) -> Vec<(&str, &str)> {
630    let words: Vec<&str> = text
631        .split(|c: char| !(c.is_alphanumeric() || matches!(c, '_' | '-' | '.' | '@' | '/')))
632        .filter(|w| !w.is_empty())
633        .collect();
634    let mut out = Vec::new();
635    for (i, w) in words.iter().enumerate() {
636        // the verb's object is one token ahead — two for the
637        // phrasal `depends on`
638        let gap = match w.to_lowercase().as_str() {
639            "imports" | "requires" | "uses" => 1,
640            "depends"
641                if words
642                    .get(i + 1)
643                    .is_some_and(|n| n.eq_ignore_ascii_case("on")) =>
644            {
645                2
646            }
647            _ => continue,
648        };
649        let (Some(x), Some(y)) = (i.checked_sub(1).map(|j| words[j]), words.get(i + gap)) else {
650            continue;
651        };
652        // sentence-ending periods ride along — '.' is a legit
653        // interior char (go paths) but never a boundary one
654        out.push((x.trim_matches('.'), y.trim_matches('.')));
655    }
656    out
657}

659/// Does the graph record a dep-ish edge `a → b`? Compared on
660/// kuda's folded `module_key`, not raw spelling.
661fn edge_exists(g: &Graph, a: &str, b: &str) -> bool {
662    let (ka, kb) = (module_key(a), module_key(b));
663    g.edges.iter().any(|e| {
664        matches!(
665            e.kind,
666            EdgeKind::Imports | EdgeKind::PackageDep | EdgeKind::WorkspaceMember
667        ) && module_key(&e.from) == ka
668            && module_key(&e.to) == kb
669    })
670}

672/// Claim ident → indexed module name: folded match first, then
673/// subpath (`pkg/sub` claims resolve to `pkg`) — mirrors kuda's
674/// `resolve_module`. Longest match wins, ties break
675/// lexicographically — deterministic regardless of set order.
676fn resolve<'a>(name: &str, modules: &'a HashSet<String>) -> Option<&'a str> {
677    let want = module_key(name);
678    modules
679        .iter()
680        .filter(|m| {
681            let k = module_key(m);
682            k == want || want.starts_with(&format!("{k}/"))
683        })
684        .max_by_key(|m| (m.len(), m.as_str()))
685        .map(String::as_str)
686}

688/// First '-'/'_'/'.'-separated segment — the naming-family prefix.
689fn seg0(name: &str) -> &str {
690    // split yields ≥1 item on any input — unwrap_or avoids unwrap
691    name.split(['-', '_', '.']).next().unwrap_or(name)
692}
693
694/// Is `to` shaped like a workspace member of this repo? A missing
695/// target flags only when it shares the monorepo's own naming
696/// family — a '/'-scope or a ≥3-char flat prefix carried by ≥2
697/// indexed modules (a convention, not a coincidence). Anything
698/// else is an external dep, not a stale reference.
699fn internal_shaped(to: &str, modules: &HashSet<String>) -> bool {
700    if to.contains('/') {
701        let Some(sc) = scope(to) else {
702            return false; // two-segment names carry no reliable scope
703        };
704        return modules
705            .iter()
706            .filter(|m| scope(m).is_some_and(|s| s.eq_ignore_ascii_case(sc)))
707            .count()
708            >= 2;
709    }
710    let lower = to.to_lowercase();
711    let first = seg0(&lower);
712    first.len() >= 3
713        && modules
714            .iter()
715            .filter(|m| seg0(&m.to_lowercase()) == first)
716            .count()
717            >= 2
718}
719
720/// The '/'-scope a name belongs to: `@org` for `@org/pkg`;
721/// `host/org` for go-style `github.com/org/repo` — so an external
722/// `github.com/ext/x` can't collide with the monorepo's own org.
723/// `None` for flat names and two-segment paths.
724fn scope(name: &str) -> Option<&str> {
725    let (s1, rest) = name.split_once('/')?;
726    if s1.starts_with('@') {
727        Some(s1)
728    } else {
729        rest.split_once('/')
730            .map(|(s2, _)| &name[..s1.len() + 1 + s2.len()])
731    }
732}

1170        // the pair becomes the buckets' problem and the intake is
1171        // empty
1172        let spellings = [
1173            t("login flow", "uses", "jwt", "a.md", 1),
1174            t("auth flow", "uses", "sessions", "b.md", 2),
1175        ];
1176        assert_eq!(semantic_candidates(&spellings, &[]).len(), 1);
1177        let aliases = [("login flow".to_owned(), "auth flow".to_owned())];
1178        assert!(semantic_candidates(&spellings, &aliases).is_empty());
1179    }

1181    /// ADR-0002 ph3 — `in_scope` gates *findings*, not the
1182    /// buckets: a conflict touching a scoped file emits and
1183    /// anchors on the in-scope side; one entirely outside stays
1184    /// silent. The cache path returns identical findings — once
1185    /// extracting, once reading stored rows.
1186    #[test]
1187    fn prose_scope_gate_and_cache() {
1188        let dir = std::env::temp_dir().join(format!("mimus-prose-{}", std::process::id()));
1189        let _ = std::fs::remove_dir_all(&dir);
1190        std::fs::create_dir_all(&dir).unwrap();
1191        let (a, b, c, d) = (
1192            dir.join("a.md"),
1193            dir.join("b.md"),
1194            dir.join("c.md"),
1195            dir.join("d.md"),
1196        );
1197        std::fs::write(&a, "auth uses sessions\n").unwrap();
1198        std::fs::write(&b, "auth uses jwt\n").unwrap();
1199        std::fs::write(&c, "pool uses postgres\n").unwrap();
1200        std::fs::write(&d, "pool uses redis\n").unwrap();
1201        let mk = |text: &str, src: &Path, line: u32| Claim {
1202            kind: ClaimKind::Behavior,
1203            text: text.into(),
1204            source: src.to_path_buf(),
1205            line,
1206        };
1207        // b.md first-seen so the scope-preferred anchor is
1208        // observable: in-scope a.md is the *second* object site
1209        let claims = [
1210            mk("auth uses jwt", &b, 1),
1211            mk("auth uses sessions", &a, 1),
1212            mk("pool uses postgres", &c, 1),
1213            mk("pool uses redis", &d, 1),
1214        ];
1215        let only_a: &dyn Fn(&Path) -> bool = &|p| p == a;
1216        // uncached path: a↔b emits (anchored on in-scope a.md,
1217        // though b.md's site is first-seen), c↔d stays silent
1218        let mut chk = ProseContradictionChecker {
1219            aliases: Vec::new(),
1220            cache: None,
1221            triples: Vec::new(),
1222        };
1223        let v = chk.run_prose(&claims, only_a);
1224        assert_eq!(v.len(), 1, "{v:?}");
1225        assert!(v[0].evidence.contains("auth|uses"), "{}", v[0].evidence);
1226        assert!(v[0].evidence.contains("b.md:1"), "{}", v[0].evidence);
1227        assert_eq!(v[0].claim.source, a, "anchor prefers the in-scope site");
1228        assert_eq!(v[0].claim.text, "auth uses sessions");
1229        // cached path — same findings across an extract run and
1230        // a stored-rows run
1231        let mut chk = ProseContradictionChecker {
1232            aliases: Vec::new(),
1233            cache: Some(TriplesCache::open(&dir.join("triples.db")).unwrap()),
1234            triples: Vec::new(),
1235        };
1236        for _ in 0..2 {
1237            let v = chk.run_prose(&claims, only_a);
1238            assert_eq!(v.len(), 1, "{v:?}");
1239            assert_eq!(v[0].claim.source, a);
1240        }
1241        // nothing in scope → silence; everything → both conflicts
1242        assert!(chk.run_prose(&claims, &|_| false).is_empty());
1243        let v = chk.run_prose(&claims, &|_| true);
1244        assert_eq!(v.len(), 2, "{v:?}");
1245        let _ = std::fs::remove_dir_all(&dir);
1246    }

1279    /// ADR-0002 ph4 — a public module no document names warns; any
1280    /// canon or alias spelling documents it. Provenance gates the
1281    /// finding under `--diff`; a snapshot without it reports rather
1282    /// than drops.
1283    #[test]
1284    fn undocumented_fact_class() {
1285        let g = Graph {
1286            modules: ["app-cli", "app-core"]
1287                .iter()
1288                .map(|s| s.to_string())
1289                .collect(),
1290            edges: vec![Edge {
1291                from: "app-cli".into(),
1292                to: "app-core".into(),
1293                kind: EdgeKind::Imports,
1294            }],
1295            index: PathBuf::from("idx.sqlite"),
1296            ..Graph::default()
1297        };
1298        let facts = UndocumentedFactChecker { aliases: &[] };
1299        let all: &dyn Fn(&Path) -> bool = &|_| true;
1300        // any canon spelling (dash, underscore, case) documents it
1301        assert!(facts
1302            .run_facts("app-core owns the domain", &g, all)
1303            .is_empty());
1304        assert!(facts
1305            .run_facts("the APP_CORE crate owns it", &g, all)
1306            .is_empty());
1307        // a name of stopwords only has no needle — silence
1308        assert!(documented("the", " app core ", &[]));
1309        // no document names it → one warn, anchored on the index
1310        let v = facts.run_facts("app-cli parses flags", &g, all);
1311        assert_eq!(v.len(), 1, "{v:?}");
1312        let e = &v[0].evidence;
1313        assert!(e.contains("app-core"), "{e}");
1314        assert!(e.contains("0 indexed file(s)"), "{e}");
1315        assert_eq!(v[0].severity, Severity::Warn);
1316        assert_eq!(v[0].claim.text, "app-core");
1317        assert_eq!(v[0].claim.line, 0);
1318        assert_eq!(v[0].claim.source, PathBuf::from("idx.sqlite"));
1319        assert!(v[0].fix.is_some());
1320        // an alias spelling documents the module it canonicalizes to
1321        let aliased = UndocumentedFactChecker {
1322            aliases: &[("the core crate".to_owned(), "app-core".to_owned())],
1323        };
1324        assert!(aliased
1325            .run_facts("the core crate owns the domain", &g, all)
1326            .is_empty());
1327        // a stopword alias has no needle, and unrelated aliases
1328        // don't widen the match
1329        let odd = UndocumentedFactChecker {
1330            aliases: &[
1331                ("the".to_owned(), "app-core".to_owned()),
1332                ("login flow".to_owned(), "auth".to_owned()),
1333            ],
1334        };
1335        assert_eq!(odd.run_facts("unrelated docs", &g, all).len(), 1);
1336        // provenance: `--diff` scope asks the module's own files
1337        let g = Graph {
1338            module_files: [(
1339                "app-core".to_owned(),
1340                [PathBuf::from("crates/core/src/lib.rs")]
1341                    .into_iter()
1342                    .collect(),
1343            )]
1344            .into_iter()
1345            .collect(),
1346            ..g
1347        };
1348        let in_scope: &dyn Fn(&Path) -> bool = &|p| p.ends_with("lib.rs");
1349        let v = facts.run_facts("app-cli parses flags", &g, in_scope);
1350        assert_eq!(v.len(), 1, "{v:?}");
1351        assert_eq!(v[0].claim.source, PathBuf::from("crates/core/src/lib.rs"));
1352        assert!(v[0].evidence.contains("1 indexed file(s)"), "{e}");
1353        // module untouched by the diff → silence
1354        assert!(facts
1355            .run_facts("app-cli parses flags", &g, &|_| false)
1356            .is_empty());
1357    }
