1//! Document models — the in-memory shapes `myco-store-md` parses
2//! `project/` files into and emits back from. `sections` preserve
3//! file order; a `name: None` section is the unsectioned run before
4//! the first `##` heading (only legal at index 0).
5
6use crate::Item;

8/// A named run of items under a `## <section>` heading.
9#[derive(Debug, Clone, PartialEq, Eq, Default)]
10#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
11#[derive(serde::Serialize, serde::Deserialize)]
12pub struct Section {
13    /// Heading text; `None` = items before the first `##`.
14    pub name: Option<String>,
15    /// Items in file order.
16    pub items: Vec<Item>,
17}
18
19impl Section {
20    /// A named section with no items yet.
21    pub fn named(name: impl Into<String>) -> Self {
22        Section {
23            name: Some(name.into()),
24            items: Vec::new(),
25        }
26    }
27
28    /// The unsectioned run (name `None`) with no items yet.
29    pub fn unsectioned() -> Self {
30        Section::default()
31    }
32}

34/// `TODO.md` — `# TODO — <repo>` plus `## <section>` runs of
35/// `- [ ] T-NNN …` task items.
36#[derive(Debug, Clone, PartialEq, Eq, Default)]
37#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
38#[derive(serde::Serialize, serde::Deserialize)]
39pub struct TodoDoc {
40    /// Repo name from the `# TODO — <repo>` title.
41    pub repo: String,
42    /// Sections in file order.
43    pub sections: Vec<Section>,
44}
45
46impl TodoDoc {
47    /// Fresh doc for `myco init` — title only.
48    pub fn new(repo: impl Into<String>) -> Self {
49        TodoDoc {
50            repo: repo.into(),
51            sections: Vec::new(),
52        }
53    }
54
55    /// Every top-level item across all sections, in file order —
56    /// nested children stay inside their parents (flatten with
57    /// [`Item::subtree`]).
58    pub fn items(&self) -> impl Iterator<Item = &Item> {
59        self.sections.iter().flat_map(|s| s.items.iter())
60    }
61}

103/// One `issues|proposals|adrs|plans/*.md` file — TOML frontmatter
104/// in `+++` fences + a free-form markdown body.
105#[derive(Debug, Clone, PartialEq, Eq)]
106#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
107#[derive(serde::Serialize, serde::Deserialize)]
108pub struct ItemFile {
109    /// Frontmatter (`id`, `title`, `status`, `created`, `owner`,
110    /// `priority`, `labels`, `blocks`, `note`).
111    pub meta: Item,
112    /// Markdown body, verbatim (normalized via [`canonical_body`]).
113    /// Kind-specific section templates are `myco-template`'s job,
114    /// not the store's.
115    pub body: String,
116}
117
118impl ItemFile {
119    /// New file with an empty body.
120    pub fn new(meta: Item) -> Self {
121        ItemFile {
122            meta,
123            body: String::new(),
124        }
125    }
126
127    /// Builder-style body setter — stored in canonical form (see
128    /// [`canonical_body`]).
129    pub fn body(mut self, body: impl Into<String>) -> Self {
130        let body = body.into();
131        self.body = canonical_body(&body).to_string();
132        self
133    }
134}
135
136/// Canonical form of a free-text block — an item-file body, a
137/// handoff note: no leading blank lines, no trailing whitespace.
138/// Leading whitespace on the *first content line* survives — a
139/// full `trim()` would mangle a body that opens with an indented
140/// code block into a paragraph.
141pub fn canonical_body(s: &str) -> &str {
142    let s = s.trim_end();
143    let mut start = 0;
144    for line in s.split_inclusive('\n') {
145        if line.trim().is_empty() {
146            start += line.len();
147        } else {
148            break;
149        }
150    }
151    &s[start..]
152}

154#[cfg(test)]
155#[allow(clippy::unwrap_used)]
156mod tests {
157    use super::*;
158    use crate::{ItemId, ItemKind};
159
160    fn t(n: u32) -> Item {
161        Item::new(ItemId::new(ItemKind::Task, n), format!("t{n}"))
162    }
163
164    #[test]
165    fn section_ctors() {
166        assert_eq!(Section::named("Now").name.as_deref(), Some("Now"));
167        assert_eq!(Section::unsectioned().name, None);
168    }
169
170    #[test]
171    fn doc_items_iterate_in_order() {
172        let mut todo = TodoDoc::new("myco");
173        todo.sections.push(Section {
174            name: Some("a".into()),
175            items: vec![t(1), t(2)],
176        });
177        todo.sections.push(Section {
178            name: None,
179            items: vec![t(3)],
180        });
181        let ids: Vec<u32> = todo.items().map(|i| i.id.number()).collect();
182        assert_eq!(ids, [1, 2, 3]);
183
184        let mut bl = BacklogDoc::default();
185        bl.sections.push(Section {
186            name: Some("someday".into()),
187            items: vec![t(9)],
188        });
189        assert_eq!(bl.items().count(), 1);
190    }
191
192    #[test]
193    fn item_file_body_canonicalizes() {
194        let f = ItemFile::new(t(1)).body("\n\nhello \n");
195        assert_eq!(f.body, "hello");
196        assert_eq!(ItemFile::new(t(1)).body, "");
197        // leading blank lines go, but the first content line's
198        // own indentation stays — an indented code block must not
199        // be mangled into a paragraph (I-021)
200        let f = ItemFile::new(t(1)).body("\n\n    code()\n\nrest\n\n");
201        assert_eq!(f.body, "    code()\n\nrest");
202        // whitespace-only collapses to empty
203        assert_eq!(ItemFile::new(t(1)).body("  \n \n").body, "");
204    }
205
206    #[test]
207    fn canonical_body_edges() {
208        assert_eq!(canonical_body(""), "");
209        assert_eq!(canonical_body("   \n\n  "), "");
210        assert_eq!(canonical_body("a\nb  \n\n"), "a\nb");
211        assert_eq!(canonical_body("\n \n\nx"), "x");
212        // inner whitespace is verbatim — only the edges move
213        assert_eq!(canonical_body("a\n\n   b\n"), "a\n\n   b");
214        assert_eq!(canonical_body("x"), "x");
215    }
