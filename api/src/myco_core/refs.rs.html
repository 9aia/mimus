1//! refs — the pure edge scanner behind the link index (P-005).
2//!
3//! Every `project/` file contributes a set of [`Edge`]s, always
4//! with `src` = the item the file speaks for: `blocks:`/
5//! `supersedes:`/`superseded_by:`/`links:` frontmatter declares
6//! edges verbatim, `X-NNN` tokens in prose become `references`
7//! edges, and a
8//! `handoffs/<ID>.md` filename is a `handoff` edge (a filename
9//! fact — callers record it, nothing to scan).
10//!
11//! The index stores these forward edges plus a reverse table, so
12//! `blocked_by`/`referenced_by`/`superseded_by` appear only as
13//! mirrored kinds in query results — a file never declares an
14//! incoming edge. Frontmatter edges are kept as declared
15//! (self-blocks are the ledger's problem); mention self-hits are
16//! dropped — quoting your own id isn't a link.
17//!
18//! Bounds (the proposal's "bounded GC"): bodies scan at most
19//! [`MAX_SCAN_BYTES`] and emit at most [`MAX_MENTIONS`] reference
20//! edges, so one pathological file can't blow up the index.
21
22use std::collections::BTreeSet;
23
24use crate::{Item, ItemId, KindTable};

26/// The most body text one extract scans — 64 KiB. The cap applies
27/// to the byte window, not to items: a file is still scanned
28/// per-item, so every item gets its own 64 KiB.
29pub const MAX_SCAN_BYTES: usize = 64 * 1024;
30
31/// The most `references` edges one extract emits.
32pub const MAX_MENTIONS: usize = 64;
33
34/// The relationship an [`Edge`] asserts between two items. The
35/// `*_by` forms are the reverse views — extract only ever produces
36/// [`Handoff`](EdgeKind::Handoff), [`Blocks`](EdgeKind::Blocks),
37/// [`Supersedes`](EdgeKind::Supersedes),
38/// [`SupersededBy`](EdgeKind::SupersededBy),
39/// [`Associated`](EdgeKind::Associated), and
40/// [`References`](EdgeKind::References); the index materializes
41/// the mirror at query time.
42#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
43#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
44#[derive(serde::Serialize, serde::Deserialize)]
45#[serde(rename_all = "snake_case")]
46pub enum EdgeKind {
47    /// `handoffs/<ID>.md` exists — `src == dst`, an attachment
48    /// marker rather than a link between items.
49    Handoff,
50    /// `src` declares `blocks: dst`.
51    Blocks,
52    /// Reverse of [`Blocks`](EdgeKind::Blocks) — query-side only.
53    BlockedBy,
54    /// `src` declares `supersedes: dst`.
55    Supersedes,
56    /// `src` declares `superseded_by: dst`.
57    SupersededBy,
58    /// `src` declares `links: dst` — a kind-agnostic association
59    /// (its plan, its proposal, …). Symmetric: the mirror is the
60    /// kind itself, so incoming and outgoing land in one
61    /// `associated` group.
62    Associated,
63    /// `src`'s text mentions `dst`.
64    References,
65    /// Reverse of [`References`](EdgeKind::References) —
66    /// query-side only.
67    ReferencedBy,
68}
69
70impl EdgeKind {
71    /// All kinds, in declaration order.
72    pub const ALL: [EdgeKind; 8] = [
73        EdgeKind::Handoff,
74        EdgeKind::Blocks,
75        EdgeKind::BlockedBy,
76        EdgeKind::Supersedes,
77        EdgeKind::SupersededBy,
78        EdgeKind::Associated,
79        EdgeKind::References,
80        EdgeKind::ReferencedBy,
81    ];
82
83    /// Stable token — `handoff`, `blocks`, `blocked_by`, … Used as
84    /// the storage key segment and the display label.
85    pub fn name(self) -> &'static str {
86        match self {
87            EdgeKind::Handoff => "handoff",
88            EdgeKind::Blocks => "blocks",
89            EdgeKind::BlockedBy => "blocked_by",
90            EdgeKind::Supersedes => "supersedes",
91            EdgeKind::SupersededBy => "superseded_by",
92            EdgeKind::Associated => "associated",
93            EdgeKind::References => "references",
94            EdgeKind::ReferencedBy => "referenced_by",
95        }
96    }
97
98    /// Kind from a [`name`](EdgeKind::name) token; `None` when
99    /// unknown (forward-compat: a newer edge kind in an old
100    /// index reads as "skipped", not corrupt).
101    pub fn from_name(name: &str) -> Option<EdgeKind> {
102        EdgeKind::ALL.into_iter().find(|k| k.name() == name)
103    }
104
105    /// The reverse view: `Blocks` ↔ `BlockedBy`,
106    /// `Supersedes` ↔ `SupersededBy`,
107    /// `References` ↔ `ReferencedBy`; `Handoff` mirrors to itself.
108    /// An incoming `(other → id, k)` edge reads as
109    /// `(id, other, k.mirror())` to the item being queried.
110    pub fn mirror(self) -> EdgeKind {
111        match self {
112            EdgeKind::Blocks => EdgeKind::BlockedBy,
113            EdgeKind::BlockedBy => EdgeKind::Blocks,
114            EdgeKind::Supersedes => EdgeKind::SupersededBy,
115            EdgeKind::SupersededBy => EdgeKind::Supersedes,
116            EdgeKind::Associated => EdgeKind::Associated,
117            EdgeKind::References => EdgeKind::ReferencedBy,
118            EdgeKind::ReferencedBy => EdgeKind::References,
119            EdgeKind::Handoff => EdgeKind::Handoff,
120        }
121    }
122
123    /// Can `extract` produce this kind? The `*_by` forms have no
124    /// frontmatter to declare them (`blocked_by` exists only as a
125    /// `blocks` mirror) — except [`SupersededBy`](EdgeKind::SupersededBy),
126    /// which *is* a real frontmatter field on ADR-shaped items.
127    pub fn extracted(self) -> bool {
128        !matches!(self, EdgeKind::BlockedBy | EdgeKind::ReferencedBy)
129    }
130}

132/// One directed edge `src → dst` — the unit `extract` emits and
133/// the index stores. `Ord` sorts `(src, kind, dst)` for
134/// deterministic output.
135#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
136#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
137#[derive(serde::Serialize, serde::Deserialize)]
138pub struct Edge {
139    /// The item the file speaks for.
140    pub src: ItemId,
141    /// The item `src` points at.
142    pub dst: ItemId,
143    /// What `src` asserts about `dst`.
144    pub kind: EdgeKind,
145}
146
147impl PartialOrd for Edge {
148    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
149        Some(self.cmp(other))
150    }
151}
152
153impl Ord for Edge {
154    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
155        (self.src, self.kind, self.dst).cmp(&(other.src, other.kind, other.dst))
156    }
157}
158
159impl Edge {
160    /// `src --kind--> dst`.
161    pub const fn new(src: ItemId, dst: ItemId, kind: EdgeKind) -> Self {
162        Edge { src, dst, kind }
163    }
164
165    /// The `handoffs/<ID>.md` association — a self-edge recording
166    /// that `id` has handoff notes on disk. A filename fact: the
167    /// caller knows it, nothing to parse.
168    pub const fn handoff(id: ItemId) -> Self {
169        Edge::new(id, id, EdgeKind::Handoff)
170    }
171}

173/// The edges one file (or file section) declares for `id` —
174/// frontmatter links plus `X-NNN` mentions in `body`. `meta` is
175/// `None` for files without item frontmatter (handoffs); `id` is
176/// always the item the file speaks for (the caller resolves it —
177/// from frontmatter, filename, or ledger position).
178///
179/// Pure and deterministic: same input, same edge set. Dedup'd —
180/// mentioning `T-001` ten times is one edge. Mentioning `id`
181/// itself yields nothing (a self-mention isn't a link).
182///
183/// `kinds` gates the *prose* scan: since P-008 any `[A-Z]`-prefixed
184/// token is a grammatical id, so a mention only counts when its
185/// prefix is a declared kind — `WIP-1` in prose stays prose.
186/// Frontmatter edges are declared data and never filtered.
187pub fn extract(body: &str, meta: Option<&Item>, id: ItemId, kinds: &KindTable) -> BTreeSet<Edge> {
188    let mut edges = BTreeSet::new();
189    if let Some(m) = meta {
190        for b in &m.blocks {
191            edges.insert(Edge::new(id, *b, EdgeKind::Blocks));
192        }
193        if let Some(s) = m.supersedes {
194            edges.insert(Edge::new(id, s, EdgeKind::Supersedes));
195        }
196        if let Some(s) = m.superseded_by {
197            edges.insert(Edge::new(id, s, EdgeKind::SupersededBy));
198        }
199        for l in &m.links {
200            edges.insert(Edge::new(id, *l, EdgeKind::Associated));
201        }
202    }
203    for dst in mentions(body) {
204        if dst != id && kinds.spec_of(dst.kind()).is_some() {
205            edges.insert(Edge::new(id, dst, EdgeKind::References));
206        }
207    }
208    edges
209}

211/// `X-NNN` id tokens in `text` — `T-001`, `I-003`, `P-002`,
212/// `ADR-0004`, `PL-001`, plus any ≤3-char uppercase prefix a
213/// user kind might declare (P-008 — the scan is grammar-level;
214/// [`extract`] drops undeclared prefixes via the kind table).
215/// Deliberately not markdown-aware: a token in a code fence
216/// counts the same as one in prose (v1 deliberately over-counts
217/// rather than teaching the scanner the grammar). Sorted,
218/// dedup'd, bounded by [`MAX_SCAN_BYTES`] / [`MAX_MENTIONS`];
219/// malformed candidates (`T-0`, overflow numbers) fall through
220/// [`ItemId`]'s own parser.
221pub fn mentions(text: &str) -> Vec<ItemId> {
222    let b = &text.as_bytes()[..text.len().min(MAX_SCAN_BYTES)];
223    let mut out = BTreeSet::new();
224    let mut i = 0;
225    while i < b.len() && out.len() < MAX_MENTIONS {
226        if !b[i].is_ascii_uppercase() {
227            i += 1;
228            continue;
229        }
230        // an uppercase run is the prefix candidate — it must start
231        // on a token boundary so `XT-001` doesn't yield `T-001`
232        // via the `T` inside it
233        let start = i;
234        while i < b.len() && b[i].is_ascii_uppercase() {
235            i += 1;
236        }
237        let bounded_before = start == 0 || !b[start - 1].is_ascii_alphanumeric();
238        if !bounded_before || !(1..=3).contains(&(i - start)) || b.get(i) != Some(&b'-') {
239            continue;
240        }
241        let dstart = i + 1;
242        let mut d = dstart;
243        while d < b.len() && b[d].is_ascii_digit() {
244            d += 1;
245        }
246        // the token must end cleanly — `T-001x` isn't `T-001`,
247        // and a trailing `-` means a longer token we don't own
248        let bounded_after = d == b.len() || !(b[d].is_ascii_alphanumeric() || b[d] == b'-');
249        if d > dstart && bounded_after {
250            // [start..d] is ASCII — a valid str slice boundary
251            if let Ok(id) = text[start..d].parse::<ItemId>() {
252                out.insert(id);
253            }
254        }
255        i = d; // consumed the digits either way — no re-scan inside
256    }
257    out.into_iter().collect()
258}

278    #[test]
279    fn kind_names_and_mirror() {
280        for k in EdgeKind::ALL {
281            assert_eq!(EdgeKind::from_name(k.name()), Some(k), "{}", k.name());
282            assert_eq!(k.mirror().mirror(), k);
283        }
284        assert_eq!(EdgeKind::from_name("relates"), None);
285        assert!(EdgeKind::BlockedBy.mirror() == EdgeKind::Blocks);
286        assert!(!EdgeKind::BlockedBy.extracted());
287        assert!(EdgeKind::SupersededBy.extracted());
288        // associated is symmetric — mirror is itself
289        assert_eq!(EdgeKind::Associated.mirror(), EdgeKind::Associated);
290        assert!(EdgeKind::Associated.extracted());
291    }

295        let plan = ItemId::new(ItemKind::Plan, 1);
296        let meta = Item::new(t(1), "x")
297            .blocks(t(2))
298            .blocks(i(3))
299            .links(plan)

309                Edge::new(t(1), ItemId::new(ItemKind::Adr, 1), EdgeKind::Supersedes),
310                Edge::new(t(1), ItemId::new(ItemKind::Adr, 9), EdgeKind::SupersededBy),
311            ])

339    #[test]
340    fn mentions_all_kinds_and_boundaries() {
341        let ids = mentions("T-001 I-3 P-002 ADR-0004 PL-001");
342        assert_eq!(
343            ids,
344            [
345                t(1),
346                i(3),
347                ItemId::new(ItemKind::Proposal, 2),
348                ItemId::new(ItemKind::Adr, 4),
349                ItemId::new(ItemKind::Plan, 1)
350            ]
351        );
352        // code-fence tokens count the same as prose (v1 rule)
353        assert_eq!(mentions("```\nT-007\n```"), [t(7)]);
354        // the scanner's own bound: prefix runs past 3 chars are
355        // never candidates — `FOOADR-0001` can't yield `ADR-0001`
356        assert!(mentions("FOOADR-0001").is_empty());
357        // short foreign prefixes parse (grammar-level leniency,
358        // P-008) — `extract` drops them via the kind table
359        let foreign = mentions("XT-001 NOT-002");
360        assert_eq!(foreign.len(), 2);
361        assert!(extract("XT-001 NOT-002", None, t(2), &kinds()).is_empty());
362        // a declared user kind's mentions DO become edges
363        let spike = toml::from_str::<crate::KindSpec>(
364            "name = \"xt\"\nprefix = \"XT\"\nstorage = \"doc\"\n",
365        )
366        .unwrap();
367        let kinds2 = KindTable::new(vec![spike]).unwrap();
368        let edges = extract("see XT-001", None, t(2), &kinds2);
369        assert_eq!(
370            edges,
371            BTreeSet::from([Edge::new(
372                t(2),
373                "XT-001".parse().unwrap(),
374                EdgeKind::References
375            )])
376        );
377        // longer tokens we don't own — `T-001x`, `T-001-2`
378        assert!(mentions("T-001x T-001-2").is_empty());
379        // but a real mention after a separator still lands
380        assert_eq!(mentions("WELL-T-005,"), [t(5)]);
381        // zero / overflow / malformed rejected by ItemId's parser —
382        // `X-001` itself parses now (grammar leniency)
383        assert!(
384            mentions("T-0 T-00 X-001 T-99999999999999999999").contains(&"X-001".parse().unwrap())
385        );
386        assert!(mentions("T-0 T-00 T-99999999999999999999").is_empty());
387        // punctuation boundaries are fine
388        assert_eq!(mentions("(T-004), [I-009]."), [t(4), i(9)]);
389    }

391    #[test]
392    fn mentions_bounds() {
393        // over the mention cap the tail is just not recorded
394        let many: String = (1..=MAX_MENTIONS + 10).map(|n| format!("I-{n} ")).collect();
395        assert_eq!(mentions(&many).len(), MAX_MENTIONS);
396        // past the byte cap nothing is scanned
397        let pad = "x".repeat(MAX_SCAN_BYTES);
398        assert!(mentions(&format!("{pad}T-001")).is_empty());
399        // right at the cap boundary the token still lands
400        let pad = format!("{} ", "x".repeat(MAX_SCAN_BYTES - "T-001".len() - 1));
401        assert_eq!(mentions(&format!("{pad}T-001")), [t(1)]);
402    }
