1//! `time::Date` helpers — myco serializes dates as `YYYY-MM-DD`
2//! everywhere (TODO lines, frontmatter, JSON).
3
4use time::macros::format_description;
5use time::Date;
6
7const FORMAT: &[time::format_description::FormatItem<'static>] =
8    format_description!("[year]-[month]-[day]");
9
10/// Parse a `YYYY-MM-DD` date.
11pub fn parse(s: &str) -> Result<Date, time::error::Parse> {
12    Date::parse(s, FORMAT)
13}
14
15/// Render as `YYYY-MM-DD` (`time::Date` Display is already ISO 8601).
16pub fn render(d: Date) -> String {
17    d.to_string()
18}
19
20/// Serde adapter for a required `Date` — `#[serde(with = "date::req")]`.
21/// Serializes as `"YYYY-MM-DD"`.
22pub mod req {
23    use serde::{Deserialize, Deserializer, Serializer};
24    use time::Date;
25
26    /// Serialize as `"YYYY-MM-DD"`.
27    pub fn serialize<S: Serializer>(d: &Date, s: S) -> Result<S::Ok, S::Error> {
28        s.collect_str(&d.to_string())
29    }
30
31    /// Deserialize a `"YYYY-MM-DD"` string.
32    pub fn deserialize<'de, D: Deserializer<'de>>(d: D) -> Result<Date, D::Error> {
33        let s = String::deserialize(d)?;
34        super::parse(&s).map_err(serde::de::Error::custom)
35    }
36}
37
38/// Serde adapter for `Option<Date>` — `#[serde(with = "date::opt")]`.
39pub mod opt {
40    use serde::{Deserialize, Deserializer, Serializer};
41    use time::Date;
42
43    /// Serialize `Some(d)` as `"YYYY-MM-DD"`, `None` as absent/null.
44    pub fn serialize<S: Serializer>(d: &Option<Date>, s: S) -> Result<S::Ok, S::Error> {
45        match d {
46            Some(d) => s.collect_str(&d.to_string()),
47            None => s.serialize_none(),
48        }
49    }
50
51    /// Deserialize a `"YYYY-MM-DD"` string or null.
52    pub fn deserialize<'de, D: Deserializer<'de>>(d: D) -> Result<Option<Date>, D::Error> {
53        let s = Option::<String>::deserialize(d)?;
54        s.map(|s| super::parse(&s).map_err(serde::de::Error::custom))
55            .transpose()
56    }
57}

59#[cfg(test)]
60#[allow(clippy::unwrap_used)]
61mod tests {
62    use super::*;
63
64    #[test]
65    fn roundtrip() {
66        let d = parse("2026-02-09").unwrap();
67        assert_eq!(render(d), "2026-02-09");
68        assert!(parse("9/2/26").is_err());
69        assert!(parse("2026-13-01").is_err());
70        assert!(parse("").is_err());
71    }
72
73    #[test]
74    fn serde_opt() {
75        #[derive(serde::Serialize, serde::Deserialize)]
76        struct W {
77            #[serde(with = "opt")]
78            d: Option<time::Date>,
79        }
80        let json = serde_json::to_string(&W {
81            d: parse("2026-02-09").ok(),
82        })
83        .unwrap();
84        assert_eq!(json, r#"{"d":"2026-02-09"}"#);
85        let back: W = serde_json::from_str(&json).unwrap();
86        assert_eq!(back.d, parse("2026-02-09").ok());
87        let none: W = serde_json::from_str(r#"{"d":null}"#).unwrap();
88        assert!(none.d.is_none());
89        assert_eq!(
90            serde_json::to_string(&W { d: None }).unwrap(),
91            r#"{"d":null}"#
92        );
93        assert!(serde_json::from_str::<W>(r#"{"d":"bogus"}"#).is_err());
94    }
95
96    #[test]
97    fn serde_req() {
98        #[derive(serde::Serialize, serde::Deserialize)]
99        struct W {
100            #[serde(with = "req")]
101            d: time::Date,
102        }
103        let json = serde_json::to_string(&W {
104            d: parse("2026-10-29").unwrap(),
105        })
106        .unwrap();
107        assert_eq!(json, r#"{"d":"2026-10-29"}"#);
108        let back: W = serde_json::from_str(&json).unwrap();
109        assert_eq!(back.d, parse("2026-10-29").unwrap());
110        assert!(serde_json::from_str::<W>(r#"{"d":"nope"}"#).is_err());
111        assert!(serde_json::from_str::<W>(r#"{"d":null}"#).is_err());
112    }
113}
