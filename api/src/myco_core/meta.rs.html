1//! `meta.toml` model + the pure id-allocation logic.
2//!
3//! Counters are **persistent** — done tasks leave `TODO.md` and ids
4//! must never be reused, so `TODO.md` alone can't tell you the next
5//! `T-NNN`. [`Registry`] computes next ids from counters plus a set
6//! of observed ids (reconcile = counter ≥ max-seen + 1), which is
7//! what the store uses to recover from hand-edits.
8
9use crate::{ItemId, ItemKind};
10
11const fn one() -> u32 {
12    1
13}
14
15/// A `[next]` counter — ids are 1-based, so `0` mints a `*-000`
16/// id no file can be named for. A hand-edited `t = 0` is a parse
17/// error, not a counter reset.
18fn nonzero<'de, D>(d: D) -> Result<u32, D::Error>
19where
20    D: serde::Deserializer<'de>,
21{
22    let n = <u32 as serde::Deserialize>::deserialize(d)?;
23    if n == 0 {
24        return Err(serde::de::Error::custom(
25            "counter must be ≥ 1 — 0 mints an unrepresentable `*-000` id",
26        ));
27    }
28    Ok(n)
29}
30
31/// A user-kind `[next]` counter — same `nonzero` rule as the
32/// builtin keys.
33fn nonzero_map<'de, D>(d: D) -> Result<std::collections::BTreeMap<String, u32>, D::Error>
34where
35    D: serde::Deserializer<'de>,
36{
37    let m = <std::collections::BTreeMap<String, u32> as serde::Deserialize>::deserialize(d)?;
38    for (k, n) in &m {
39        if *n == 0 {
40            return Err(serde::de::Error::custom(format!(
41                "counter {k} must be ≥ 1 — 0 mints an unrepresentable `*-000` id"
42            )));
43        }
44    }
45    Ok(m)
46}

48/// Per-kind next-counters — `meta.toml`'s `[next]` table.
49/// Missing keys default to 1 so a partial `[next]` survives a
50/// hand-edit. Builtin kinds keep their fixed keys (`t`, `i`, …);
51/// user kinds (P-008) land in `extra` keyed by lowercase prefix
52/// (`s = 4`). Unknown keys ride `extra` too — the kind table, not
53/// serde, decides whether a key names a real kind (a typo'd
54/// counter can't reissue ids anyway: `observe` floors every
55/// counter above the ids on disk).
56#[derive(Debug, Clone, PartialEq, Eq)]
57#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
58#[derive(serde::Serialize, serde::Deserialize)]
59pub struct NextCounters {
60    /// Next `T-NNN`.
61    #[serde(default = "one", deserialize_with = "nonzero")]
62    pub t: u32,
63    /// Next `I-NNN`.
64    #[serde(default = "one", deserialize_with = "nonzero")]
65    pub i: u32,
66    /// Next `P-NNN`.
67    #[serde(default = "one", deserialize_with = "nonzero")]
68    pub p: u32,
69    /// Next `ADR-NNNN`.
70    #[serde(default = "one", deserialize_with = "nonzero")]
71    pub adr: u32,
72    /// Next `PL-NNN`.
73    #[serde(default = "one", deserialize_with = "nonzero")]
74    pub pl: u32,
75    /// User-kind counters — `[next]` keys beyond the five fixed
76    /// ones, keyed by the kind's `counter_key` (lowercase prefix).
77    #[serde(
78        default,
79        flatten,
80        deserialize_with = "nonzero_map",
81        skip_serializing_if = "std::collections::BTreeMap::is_empty"
82    )]
83    pub extra: std::collections::BTreeMap<String, u32>,
84}
85
86impl NextCounters {
87    /// Counter for `kind` — builtin key or the `extra` map
88    /// (absent keys read as 1, matching the missing-key default).
89    pub fn get(&self, kind: ItemKind) -> u32 {
90        match kind.builtin_index() {
91            Some(0) => self.t,
92            Some(1) => self.i,
93            Some(2) => self.p,
94            Some(3) => self.adr,
95            Some(4) => self.pl,
96            _ => self.extra.get(&kind.counter_key()).copied().unwrap_or(1),
97        }
98    }
99
100    /// Set counter for `kind`.
101    pub fn set(&mut self, kind: ItemKind, n: u32) {
102        match kind.builtin_index() {
103            Some(0) => self.t = n,
104            Some(1) => self.i = n,
105            Some(2) => self.p = n,
106            Some(3) => self.adr = n,
107            Some(4) => self.pl = n,
108            _ => {
109                self.extra.insert(kind.counter_key(), n);
110            }
111        }
112    }
113}
114
115impl Default for NextCounters {
116    /// Counters start at 1 — `T-000` is not a valid id.
117    fn default() -> Self {
118        NextCounters {
119            t: 1,
120            i: 1,
121            p: 1,
122            adr: 1,
123            pl: 1,
124            extra: std::collections::BTreeMap::new(),
125        }
126    }
127}

129/// `meta.toml` — the one piece of project state that isn't
130/// derivable from the item files. Unknown keys are rejected:
131/// a typo'd table (`[netx]`) would otherwise silently drop the
132/// real counters and restart id allocation at 1 — done ids get
133/// reissued, colliding with their handoffs and history.
134#[derive(Debug, Clone, PartialEq, Eq)]
135#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
136#[derive(serde::Serialize, serde::Deserialize)]
137#[serde(deny_unknown_fields)]
138pub struct ProjectMeta {
139    /// Repo/project name — used in `# TODO — <name>` and views.
140    pub name: String,
141    /// Next-id counters per kind.
142    #[serde(default)]
143    pub next: NextCounters,
144}
145
146impl ProjectMeta {
147    /// Fresh meta for `myco init`.
148    pub fn new(name: impl Into<String>) -> Self {
149        ProjectMeta {
150            name: name.into(),
151            next: NextCounters::default(),
152        }
153    }
154
155    /// A [`Registry`] seeded from these counters.
156    pub fn registry(&self) -> Registry {
157        Registry {
158            counters: self.next.clone(),
159        }
160    }
161}
162
163/// Id allocator — pure. Seed from [`ProjectMeta::registry`],
164/// [`observe_all`][Registry::observe_all] every known id, then
165/// [`alloc`][Registry::alloc]; take the bumped counters back out via
166/// [`counters`][Registry::counters] to persist.
167#[derive(Debug, Clone, Default)]
168pub struct Registry {
169    counters: NextCounters,
170}
171
172impl Registry {
173    /// The id `alloc` would return for `kind` without bumping.
174    /// A zeroed counter floors at 1 — `T-000` is not a valid id.
175    pub fn peek(&self, kind: ItemKind) -> ItemId {
176        ItemId::new(kind, self.counters.get(kind).max(1))
177    }
178
179    /// Record an existing id — bumps its kind's counter past it.
180    /// Never lowers a counter.
181    pub fn observe(&mut self, id: ItemId) {
182        let floor = id.number().saturating_add(1);
183        let cur = self.counters.get(id.kind());
184        if floor > cur {
185            self.counters.set(id.kind(), floor);
186        }
187    }
188
189    /// Record many existing ids — the reconcile pass after a scan.
190    pub fn observe_all(&mut self, ids: impl IntoIterator<Item = ItemId>) {
191        for id in ids {
192            self.observe(id);
193        }
194    }
195
196    /// Take the next id for `kind` and bump the counter.
197    pub fn alloc(&mut self, kind: ItemKind) -> ItemId {
198        let id = self.peek(kind);
199        self.counters.set(kind, id.number().saturating_add(1));
200        id
201    }
202
203    /// The counters to write back to `meta.toml`.
204    pub fn counters(&self) -> NextCounters {
205        self.counters.clone()
206    }
207}

214    #[test]
215    fn counters_start_at_one() {
216        let n = NextCounters::default();
217        for k in ItemKind::ALL {
218            assert_eq!(n.get(k), 1);
219        }
220        let meta = ProjectMeta::new("myco");
221        assert_eq!(meta.name, "myco");
222        assert_eq!(meta.next, n);
223    }
224
225    #[test]
226    fn alloc_bumps_per_kind() {
227        let mut r = Registry::default();
228        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 1));
229        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 2));
230        assert_eq!(r.alloc(ItemKind::Issue), ItemId::new(ItemKind::Issue, 1));
231        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 3));
232        assert_eq!(r.peek(ItemKind::Adr), ItemId::new(ItemKind::Adr, 1));
233    }
234
235    #[test]
236    fn observe_raises_floor_never_lowers() {
237        let mut r = Registry::default();
238        r.observe(ItemId::new(ItemKind::Task, 40));
239        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 41));
240        // observing a lower id doesn't lower the counter
241        r.observe(ItemId::new(ItemKind::Task, 5));
242        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 41));
243        // other kinds untouched
244        assert_eq!(r.peek(ItemKind::Plan), ItemId::new(ItemKind::Plan, 1));
245        // counters written back
246        let c = r.counters();
247        assert_eq!(c.t, 41);
248        assert_eq!(c.pl, 1);
249    }

251    #[test]
252    fn observe_all_reconciles() {
253        let mut meta = ProjectMeta::new("x");
254        meta.next.t = 2; // stale: file actually contains T-012
255        let mut r = meta.registry();
256        r.observe_all([
257            ItemId::new(ItemKind::Task, 12),
258            ItemId::new(ItemKind::Task, 3),
259        ]);
260        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 13));
261    }

274    #[test]
275    fn unknown_keys_land_in_extra() {
276        // P-008: `[next]` is open — user-kind counters and typo'd
277        // keys alike land in `extra`; the *table* name typo stays
278        // fatal (it would drop every real counter, not just one)
279        let n = serde_json::from_value::<NextCounters>(serde_json::json!({"tx": 9})).unwrap();
280        assert_eq!(n.extra["tx"], 9);
281        assert_eq!(n.t, 1);
282        assert!(serde_json::from_value::<ProjectMeta>(serde_json::json!({
283            "name": "x",
284            "netx": {"t": 9}
285        }))
286        .is_err());
287        let meta = serde_json::from_value::<ProjectMeta>(serde_json::json!({
288            "name": "x",
289            "next": {"t": 9, "s": 4}
290        }))
291        .unwrap();
292        assert_eq!(meta.next.t, 9);
293        // a user kind reads/writes its counter through extra
294        let s = ItemKind::from_prefix("S").unwrap();
295        assert_eq!(meta.next.get(s), 4);
296    }

298    #[test]
299    fn user_kind_counters_alloc() {
300        let s = ItemKind::for_spec("S", 4).unwrap();
301        let mut r = Registry::default();
302        assert_eq!(r.alloc(s).to_string(), "S-0001");
303        assert_eq!(r.alloc(s).to_string(), "S-0002");
304        let c = r.counters();
305        assert_eq!(c.extra["s"], 3);
306        // serde round-trips the extra key
307        let back: NextCounters = serde_json::from_str(&serde_json::to_string(&c).unwrap()).unwrap();
308        assert_eq!(back, c);
309        // and observe floors it like any counter
310        let mut r2 = Registry { counters: back };
311        r2.observe_all([ItemId::new(s, 40)]);
312        assert_eq!(r2.alloc(s).to_string(), "S-0041");
313    }

315    #[test]
316    fn zero_counter_is_rejected_and_floored() {
317        // `0` mints `T-000` — an id whose file name can't
318        // round-trip; reject it at the boundary…
319        assert!(serde_json::from_value::<NextCounters>(serde_json::json!({"t": 0})).is_err());
320        // …and floor it if a zeroed counter arrives anyway
321        let mut meta = ProjectMeta::new("x");
322        meta.next.t = 0;
323        let mut r = meta.registry();
324        assert_eq!(r.alloc(ItemKind::Task), ItemId::new(ItemKind::Task, 1));
325        assert_eq!(r.peek(ItemKind::Task), ItemId::new(ItemKind::Task, 2));
326    }
