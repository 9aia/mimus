1//! Status-transition walks — the shortest legal path `from → to`
2//! through the state machine (BFS over [`Status::transitions`]).
3//!
4//! Convenience verbs — `done`, `defer`, `promote`, `resolve`,
5//! proposal `status`, ADR `supersede` — walk multi-step paths
6//! (e.g. `open → in-progress → done`), so every hop stays
7//! machine-validated.
8
9use std::collections::{BTreeMap, VecDeque};
10
11use crate::{Item, ItemId, Status};
12
13/// The shortest legal path `from → to`, or the transition error
14/// naming the valid targets.
15pub fn status_path(id: ItemId, from: Status, to: Status) -> Result<Vec<Status>, String> {
16    if from == to {
17        return Err(format!("{id} is already {to}"));
18    }
19    let mut prev: BTreeMap<Status, Status> = BTreeMap::new();
20    let mut q = VecDeque::from([from]);
21    while let Some(cur) = q.pop_front() {
22        if cur == to {
23            break;
24        }
25        for &next in cur.transitions() {
26            if next != from && !prev.contains_key(&next) {
27                prev.insert(next, cur);
28                q.push_back(next);
29            }
30        }
31    }
32    if !prev.contains_key(&to) {
33        return Err(format!(
34            "{id}: invalid status transition {from} → {to} (valid from {from}: {})",
35            targets(from)
36        ));
37    }
38    let mut path = vec![to];
39    while let Some(&cur) = path.last() {
40        if cur == from {
41            break;
42        }
43        path.push(prev[&cur]);
44    }
45    path.pop();
46    path.reverse();
47    Ok(path)
48}
49
50/// `"claimed, in-progress, deferred"` / `"none (terminal)"` — the
51/// valid-targets suffix on transition errors.
52pub fn targets(from: Status) -> String {
53    let t: Vec<&str> = from.transitions().iter().map(|s| s.token()).collect();
54    if t.is_empty() {
55        "none (terminal)".into()
56    } else {
57        t.join(", ")
58    }
59}

61/// Walk `item` along [`status_path`] to `to`, validating each hop.
62pub fn walk_to(item: &mut Item, to: Status) -> Result<(), String> {
63    for step in status_path(item.id, item.status, to)? {
64        item.status = item.status.transition_to(step).map_err(|e| e.to_string())?;
65    }
66    Ok(())
67}

69/// Walk every descendant to `to` — subtree moves keep the ledger
70/// consistent: a deferred parent's children can't stay open in
71/// BACKLOG.md, and children park/unpark with their parent. Nodes
72/// already at `to` are left alone (identity isn't a transition).
73/// All-or-nothing: the whole subtree is validated before any
74/// status moves, so an unreachable target can't leave earlier
75/// siblings half-transitioned.
76pub fn walk_descendants_to(item: &mut Item, to: Status) -> Result<(), String> {
77    check_descendants(item, to)?;
78    apply_descendants(item, to);
79    Ok(())
80}
81
82/// Pass 1 — every descendant must reach `to` before any move.
83/// Iterative — a deep tree must not overflow the call stack.
84fn check_descendants(item: &Item, to: Status) -> Result<(), String> {
85    let mut stack: Vec<&Item> = item.children.iter().collect();
86    while let Some(c) = stack.pop() {
87        if c.status != to {
88            status_path(c.id, c.status, to)?;
89        }
90        stack.extend(c.children.iter());
91    }
92    Ok(())
93}
94
95/// Pass 2 — apply; `check_descendants` already proved each walk.
96fn apply_descendants(item: &mut Item, to: Status) {
97    let mut stack: Vec<&mut Item> = item.children.iter_mut().collect();
98    while let Some(c) = stack.pop() {
99        if c.status != to {
100            // prevalidated — walk_to can only fail via status_path
101            let _ = walk_to(c, to);
102        }
103        stack.extend(c.children.iter_mut());
104    }
105}

107#[cfg(test)]
108#[allow(clippy::unwrap_used)]
109mod tests {
110    use super::*;
111    use crate::ItemKind;
112
113    fn t(n: u32, s: Status) -> Item {
114        Item::new(ItemId::new(ItemKind::Task, n), "x").status(s)
115    }
116
117    #[test]
118    fn status_path_walks() {
119        let id = ItemId::new(ItemKind::Task, 1);
120        assert_eq!(
121            status_path(id, Status::Open, Status::Done).unwrap(),
122            vec![Status::InProgress, Status::Done]
123        );
124        assert_eq!(
125            status_path(id, Status::Open, Status::Deferred).unwrap(),
126            vec![Status::Deferred]
127        );
128        assert_eq!(
129            status_path(id, Status::Blocked, Status::Done).unwrap(),
130            vec![Status::InProgress, Status::Done]
131        );
132        // deferred reaches done only via open → in-progress
133        assert_eq!(
134            status_path(id, Status::Deferred, Status::Done).unwrap(),
135            vec![Status::Open, Status::InProgress, Status::Done]
136        );
137        // no path out of terminal states
138        assert!(status_path(id, Status::Done, Status::Open).is_err());
139        // same → error
140        assert!(status_path(id, Status::Open, Status::Open).is_err());
141        // error message lists valid targets
142        let e = status_path(id, Status::Done, Status::Open).unwrap_err();
143        assert!(e.contains("none (terminal)"), "{e}");
144    }
145
146    #[test]
147    fn walk_descendants_skips_already_there() {
148        let mut parent = t(1, Status::Open).child(t(2, Status::InProgress));
149        walk_descendants_to(&mut parent, Status::Deferred).unwrap();
150        assert_eq!(parent.children[0].status, Status::Deferred);
151        // already-deferred nodes don't re-error on the identity walk
152        let mut parked = t(3, Status::Deferred).child(t(4, Status::Deferred));
153        walk_descendants_to(&mut parked, Status::Deferred).unwrap();
154        // unpark: deferred children come back to open
155        walk_descendants_to(&mut parked, Status::Open).unwrap();
156        assert_eq!(parked.children[0].status, Status::Open);
157    }
158
159    #[test]
160    fn walk_descendants_is_all_or_nothing() {
161        // an unreachable target mid-subtree must not leave the
162        // earlier siblings transitioned
163        let mut parent = t(1, Status::Open)
164            .child(t(2, Status::Open))
165            .child(t(3, Status::Done));
166        assert!(walk_descendants_to(&mut parent, Status::Deferred).is_err());
167        assert_eq!(parent.children[0].status, Status::Open);
168        assert_eq!(parent.children[1].status, Status::Done);
169        // same at depth: a done grandchild blocks the walk before
170        // the sibling subtree moves
171        let mut parent = t(1, Status::Open)
172            .child(t(2, Status::Open))
173            .child(t(3, Status::Open).child(t(4, Status::Done)));
174        assert!(walk_descendants_to(&mut parent, Status::Deferred).is_err());
175        assert_eq!(parent.children[0].status, Status::Open);
176        assert_eq!(parent.children[1].status, Status::Open);
177        assert_eq!(parent.children[1].children[0].status, Status::Done);
178    }
