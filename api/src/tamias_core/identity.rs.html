1//! Package identity — `[@scope/]name@version`.
2//!
3//! A **scope** namespaces a package (npm-style `@9aia/ci-gate`) so two
4//! orgs can publish the same `name` without clashing. The **version** is
5//! semver (the `semver` crate). The identity is what the store path, the
6//! registry key, and the lockfile all key on.
7
8use std::fmt;
9
10use semver::Version;
11use serde::{Deserialize, Serialize};
12
13/// A package identity: optional scope, name, and semver version.
14#[derive(Clone, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
15pub struct Identity {
16    /// Namespace, without the leading `@` (`9aia` → `@9aia/…`).
17    pub scope: Option<String>,
18    /// The bare package name (`ci-gate`).
19    pub name: String,
20    /// Semver version.
21    pub version: Version,
22}
23
24/// Identity parse/validation failures.
25#[derive(Debug, thiserror::Error)]
26pub enum IdentityError {
27    /// The input isn't `[@scope/]name@version`.
28    #[error("invalid identity '{0}': expected [@scope/]name@version")]
29    Invalid(String),
30    /// The name/scope is empty or has a bad shape.
31    #[error("invalid {what} '{value}' in identity")]
32    Segment {
33        /// `scope` or `name`.
34        what: &'static str,
35        /// The offending value.
36        value: String,
37    },
38    /// The version isn't valid semver.
39    #[error("invalid version '{0}'")]
40    Version(String),
41}
42
43fn valid_segment(s: &str) -> bool {
44    !s.is_empty()
45        && s.bytes()
46            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
47}
48
49impl Identity {
50    /// Build from parts, validating the name segments and semver.
51    ///
52    /// # Errors
53    /// Returns [`IdentityError`] on a bad name/scope/version.
54    pub fn new(scope: Option<String>, name: &str, version: &str) -> Result<Self, IdentityError> {
55        if !valid_segment(name) {
56            return Err(IdentityError::Segment {
57                what: "name",
58                value: name.to_string(),
59            });
60        }
61        if let Some(scope) = &scope {
62            if !valid_segment(scope) {
63                return Err(IdentityError::Segment {
64                    what: "scope",
65                    value: scope.clone(),
66                });
67            }
68        }
69        let version =
70            Version::parse(version).map_err(|_| IdentityError::Version(version.to_string()))?;
71        Ok(Self {
72            scope,
73            name: name.to_string(),
74            version,
75        })
76    }
77
78    /// Parse `[@scope/]name@version`.
79    ///
80    /// # Errors
81    /// Returns [`IdentityError`] when the shape or version is invalid.
82    pub fn parse(input: &str) -> Result<Self, IdentityError> {
83        let input = input.trim();
84        let (scope, rest) = if let Some(r) = input.strip_prefix('@') {
85            let (scope, rest) = r
86                .split_once('/')
87                .ok_or_else(|| IdentityError::Invalid(input.to_string()))?;
88            (Some(scope.to_string()), rest)
89        } else {
90            (None, input)
91        };
92        let (name, version) = rest
93            .rsplit_once('@')
94            .ok_or_else(|| IdentityError::Invalid(input.to_string()))?;
95        Self::new(scope, name, version)
96    }
97
98    /// `@scope/name` (no version) — the store dir / registry namespace.
99    #[must_use]
100    pub fn scoped(&self) -> String {
101        match &self.scope {
102            Some(scope) => format!("@{scope}/{}", self.name),
103            None => self.name.clone(),
104        }
105    }
106}
107
108impl fmt::Display for Identity {
109    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
110        write!(f, "{}@{}", self.scoped(), self.version)
111    }
112}

114#[cfg(test)]
115#[cfg_attr(coverage, coverage(off))]
116mod tests {
117    use super::*;
118
119    #[test]
120    fn parse_scoped() {
121        let id = Identity::parse("@9aia/ci-gate@0.2.0").unwrap();
122        assert_eq!(id.scope.as_deref(), Some("9aia"));
123        assert_eq!(id.name, "ci-gate");
124        assert_eq!(id.version.to_string(), "0.2.0");
125        assert_eq!(id.scoped(), "@9aia/ci-gate");
126        assert_eq!(id.to_string(), "@9aia/ci-gate@0.2.0");
127    }
128
129    #[test]
130    fn parse_unscoped() {
131        let id = Identity::parse("ci-gate@1.0.0").unwrap();
132        assert_eq!(id.scope, None);
133        assert_eq!(id.name, "ci-gate");
134        assert_eq!(id.to_string(), "ci-gate@1.0.0");
135    }
136
137    #[test]
138    fn rejects_bad() {
139        assert!(Identity::parse("ci-gate").is_err()); // no version
140        assert!(Identity::parse("@9aia@1.0.0").is_err()); // scope needs /
141        assert!(Identity::parse("name@nope").is_err()); // bad semver
142        assert!(Identity::parse("@/x@1.0.0").is_err()); // empty scope
143    }
144
145    #[test]
146    fn roundtrip() {
147        let id = Identity::new(Some("acme".into()), "pack", "2.3.4").unwrap();
148        assert_eq!(Identity::parse(&id.to_string()).unwrap(), id);
149    }
150
151    #[test]
152    fn segment_errors() {
153        assert!(Identity::new(None, "", "1.0.0").is_err());
154        assert!(Identity::new(Some("bad scope".into()), "x", "1.0.0").is_err());
155    }
156}
