1//! The project lockfile — `tamias.lock` (YAML) pins resolved identity,
2//! source ref, and digest for reproducible installs.
3
4use std::path::Path;
5
6use serde::{Deserialize, Serialize};
7
8use crate::{Error, StoreRecord};
9
10/// The lockfile — resolved packages, sorted by (scope, name).
11///
12/// Written by `tamias add` ([`Lock::upsert_record`]), kept honest by
13/// `update`/`pin` ([`Lock::sync_record`]), and consumed by `tamias
14/// sync` — the dep-set reconcile that installs what's missing or
15/// drifted and refreshes entries via [`Lock::upsert_record`].
16#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
17#[serde(default)]
18pub struct Lock {
19    /// One entry per pinned package.
20    pub packages: Vec<LockEntry>,
21}
22
23/// A pinned package — everything needed to reproduce the install.
24#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
25pub struct LockEntry {
26    /// Bare name.
27    pub name: String,
28    /// Scope, when namespaced.
29    #[serde(default, skip_serializing_if = "Option::is_none")]
30    pub scope: Option<String>,
31    /// Declared version, when any.
32    #[serde(default, skip_serializing_if = "Option::is_none")]
33    pub version: Option<String>,
34    /// The ref this was installed from.
35    pub source_ref: String,
36    /// What the ref resolved to — commit sha or `sha256:<hex>`.
37    pub resolved: String,
38    /// `sha256:<hex>` of the payload.
39    pub digest: String,
40}
41
42impl LockEntry {
43    /// A lock entry from a store record.
44    #[must_use]
45    pub fn from_record(record: &StoreRecord) -> Self {
46        Self {
47            name: record.name.clone(),
48            scope: record.scope.clone(),
49            version: record.version.clone(),
50            source_ref: record.source_ref.clone(),
51            resolved: record.resolved.clone(),
52            digest: record.digest.clone(),
53        }
54    }
55}
56
57impl Lock {
58    /// Load `path` — a missing file is an empty lock, not an error.
59    ///
60    /// # Errors
61    /// Fails on read/parse error.
62    pub fn load(path: &Path) -> Result<Self, Error> {
63        if !path.is_file() {
64            return Ok(Self::default());
65        }
66        let text =
67            std::fs::read_to_string(path).map_err(Error::io(format!("read {}", path.display())))?;
68        serde_yaml_ng::from_str(&text)
69            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))
70    }
71
72    /// Write the lockfile to `path` as YAML.
73    ///
74    /// # Errors
75    /// Propagates serialize + write failures.
76    pub fn save(&self, path: &Path) -> Result<(), Error> {
77        let ctx = |e| Error::Failed(format!("serialize lock: {e}"));
78        let text = serde_yaml_ng::to_string(self).map_err(ctx)?;
79        crate::write_atomic(path, text.as_bytes())
80    }
81
82    /// Insert or replace the entry for (scope, name), keeping the list
83    /// sorted.
84    pub fn upsert(&mut self, entry: LockEntry) {
85        if let Some(existing) = self
86            .packages
87            .iter_mut()
88            .find(|e| e.name == entry.name && e.scope == entry.scope)
89        {
90            *existing = entry;
91        } else {
92            self.packages.push(entry);
93        }
94        self.packages
95            .sort_by(|a, b| (&a.scope, &a.name).cmp(&(&b.scope, &b.name)));
96    }
97
98    /// Refresh the pinned entry for `record`'s (scope, name) in the
99    /// lockfile at `path` — the `update`/`pin` "keep `tamias.lock`
100    /// honest" rule: when the file exists *and* already pins the
101    /// package, update `source_ref`/`resolved`/`digest`/`version`;
102    /// otherwise leave it alone. Never creates the file — that's
103    /// [`Self::upsert_record`]'s (`add`'s) job.
104    ///
105    /// # Errors
106    /// Fails on lock read/parse/write error.
107    pub fn sync_record(path: &Path, record: &StoreRecord) -> Result<(), Error> {
108        if !path.is_file() {
109            return Ok(());
110        }
111        let mut lock = Self::load(path)?;
112        let Some(entry) = lock
113            .packages
114            .iter_mut()
115            .find(|e| e.name == record.name && e.scope == record.scope)
116        else {
117            return Ok(());
118        };
119        entry.source_ref = record.source_ref.clone();
120        entry.resolved = record.resolved.clone();
121        entry.digest = record.digest.clone();
122        entry.version = record.version.clone();
123        lock.save(path)
124    }
125
126    /// Upsert `LockEntry::from_record(record)` into the lockfile at
127    /// `path`, creating it when missing — the `add`/dep-record tail.
128    ///
129    /// # Errors
130    /// Fails on lock read/parse/write error.
131    pub fn upsert_record(path: &Path, record: &StoreRecord) -> Result<(), Error> {
132        let mut lock = Self::load(path)?;
133        lock.upsert(LockEntry::from_record(record));
134        lock.save(path)
135    }
136}

138#[cfg(test)]
139#[cfg_attr(coverage, coverage(off))]
140mod tests {
141    use super::*;
142
143    fn record() -> StoreRecord {
144        StoreRecord {
145            name: "x".into(),
146            scope: Some("acme".into()),
147            version: Some("1.0.0".into()),
148            kind: Some("files".into()),
149            source_ref: "./x".into(),
150            resolved: "sha256:abc".into(),
151            digest: "sha256:def".into(),
152            signature: None,
153            signer: None,
154            trust: None,
155            installed_at: 7,
156        }
157    }
158
159    #[test]
160    fn load_missing_is_empty_and_roundtrips() {
161        let dir = tempfile::tempdir().unwrap();
162        let path = dir.path().join("tamias.lock");
163        assert_eq!(Lock::load(&path).unwrap(), Lock::default());
164        let mut lock = Lock::default();
165        lock.upsert(LockEntry::from_record(&record()));
166        lock.upsert(LockEntry::from_record(&record()));
167        assert_eq!(lock.packages.len(), 1);
168        lock.save(&path).unwrap();
169        let back = Lock::load(&path).unwrap();
170        assert_eq!(back.packages[0].name, "x");
171        assert_eq!(back.packages[0].scope.as_deref(), Some("acme"));
172    }
173
174    #[cfg(unix)]
175    #[test]
176    fn load_error_arms() {
177        use std::os::unix::fs::PermissionsExt;
178        let dir = tempfile::tempdir().unwrap();
179        let p = dir.path().join("tamias.lock");
180        std::fs::write(&p, ": not : yaml :").unwrap();
181        assert!(Lock::load(&p).is_err());
182        std::fs::set_permissions(&p, std::fs::Permissions::from_mode(0o000)).unwrap();
183        assert!(Lock::load(&p).is_err());
184        std::fs::set_permissions(&p, std::fs::Permissions::from_mode(0o644)).unwrap();
185    }
