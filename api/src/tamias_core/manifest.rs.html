13/// `[A-Za-z0-9][A-Za-z0-9._-]*` — the package name/scope charset.
14/// Always a single safe path component.
15#[must_use]
16pub fn is_valid_name(s: &str) -> bool {
17    !s.is_empty()
18        && s.bytes()
19            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'))
20        && s.bytes().next().is_some_and(|b| b.is_ascii_alphanumeric())
21}
22
23/// Version-label charset — `[A-Za-z0-9][A-Za-z0-9._+-]*` with no `..`.
24/// Covers semver, prerelease/build suffixes, dist-tags (`latest`), and
25/// the sha short-form fallback; always a single safe path/URL component.
26#[must_use]
27pub fn is_valid_version(s: &str) -> bool {
28    !s.is_empty()
29        && s.bytes()
30            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-' | b'+'))
31        && s.bytes().next().is_some_and(|b| b.is_ascii_alphanumeric())
32        && !s.contains("..")
33}
34
35/// Content-type charset — `[a-z0-9][a-z0-9._-]*` with no `..`. The
36/// kind becomes `tamias-driver-<kind>` in PATH/home lookups, so it
37/// must stay a single safe path component (no `/`, no separators, no
38/// traversal).
39#[must_use]
40pub fn is_valid_kind(s: &str) -> bool {
41    !s.is_empty()
42        && s.bytes().all(|b| {
43            b.is_ascii_lowercase() || b.is_ascii_digit() || matches!(b, b'.' | b'_' | b'-')
44        })
45        && s.bytes()
46            .next()
47            .is_some_and(|b| b.is_ascii_lowercase() || b.is_ascii_digit())
48        && !s.contains("..")
49}

51/// The package manifest.
52#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, schemars::JsonSchema)]
53#[serde(default)]
54pub struct Manifest {
55    /// Bare package name — charset `[A-Za-z0-9][A-Za-z0-9._-]*`.
56    pub name: String,
57    /// Optional namespace (`@scope/name`).
58    pub scope: Option<String>,
59    /// Author-declared version (semver-ish label).
60    pub version: Option<String>,
61    /// Free-text description.
62    pub description: Option<String>,
63    /// Content type (`peregrin-bundle`, `myco-config`, …).
64    #[serde(rename = "type")]
65    pub kind: Option<String>,
66    /// Base packages to compose under this payload — installed store
67    /// names or installable source refs. Local wins; earlier includes
68    /// win over later ones.
69    ///
70    /// Applied on install (`tamias-install` resolves each entry —
71    /// store name → that entry's payload dir, ref → a recursive
72    /// install — then [`crate::overlay::overlay`]s bases under the
73    /// local payload; the store record's digest covers the *composed*
74    /// tree and the manifest rides along unchanged for provenance) and
75    /// flattened at publish/pack time (`tamias-publish`, read-only —
76    /// `--keep-includes` ships the raw payload instead).
77    pub include: Vec<String>,
78    /// Direct dependencies — refs this project wants installed.
79    /// Recorded by `tamias add` (and the TUI deps form) into the
80    /// project manifest; `tamias.lock` pins each resolved record.
81    /// `tamias sync` (CLI/serve/TUI) reconciles this set against the
82    /// store + lock — install-if-missing-or-drifted, never removes.
83    pub deps: Vec<String>,
84    /// `provider:` — the declarative runtime spec a deploy driver
85    /// materializes (service units, ingress/forge hooks, health probe).
86    /// Absent = a plain package with nothing to deploy.
87    pub provider: Option<Provider>,
88    /// `bins:` — executable surfaces the package declares (P-023).
89    /// `tamias bin deploy` materializes them into the bin dir; `tamias
90    /// install` deploys them by default. Pure data — tamias never
91    /// learns a product's CLI shape.
92    pub bins: Vec<BinDecl>,
93}

95/// One `bins[]` entry — a command name plus *how* it materializes.
96///
97/// Exactly one of `shim`/`payload` is set:
98///
99/// - `shim: "cli"` + `driver: <kind>` — a generated front-end: tamias
100///   writes a small script into the bin dir that execs
101///   `tamias bin exec <label> <name> "$@"`, which resolves the kind
102///   driver (`tamias-driver-<kind>`, falling back through capabilities
103///   — e.g. a generic `tamias-driver-cli`) and dispatches its `cli`
104///   verb with the package's `--dir`/`--name`. The shim is a script,
105///   not a compiled helper: `tamias` itself is the helper binary, so
106///   driver discovery stays dynamic (PATH + `<tamias-home>/drivers/`,
107///   including drivers the package itself just installed).
108/// - `payload: tools/bin/<triple>/<file>` — a real binary shipped in
109///   the payload. The `<triple>` segment selects the host platform
110///   (`tamias_driver::target_triple`); declaring another triple's
111///   path resolves to the host's dir at deploy time, and a payload
112///   whose `tools/bin/` ships platform dirs but none for this host
113///   fails the install (same rule as `drivers/bin/`). A path without
114///   a triple dir (`tools/bin/<file>`) is a portable payload bin.
115#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, schemars::JsonSchema)]
116#[serde(default)]
117pub struct BinDecl {
118    /// Command name on `PATH` — `[A-Za-z0-9][A-Za-z0-9._-]*`, never
119    /// `tamias` itself (reserved for the real binary).
120    pub name: String,
121    /// Shim kind — only `"cli"` (generated driver front-end) exists.
122    pub shim: Option<String>,
123    /// Driver kind the `cli` shim dispatches to — resolved like a
124    /// content type (`tamias-driver-<kind>` → capabilities fallback).
125    /// Required iff `shim` is set.
126    pub driver: Option<String>,
127    /// Payload-relative path to a real binary — `tools/bin/…`.
128    /// Required iff `shim` is unset.
129    pub payload: Option<String>,
130}

132/// `provider:` — declarative service/runtime spec. Deploy drivers turn
133/// this into host resources (systemd units, forge webhooks); tamias
134/// itself never interprets it beyond validation.
135#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, schemars::JsonSchema)]
136#[serde(default)]
137pub struct Provider {
138    /// Provider/service identity — defaults to the package name.
139    pub name: Option<String>,
140    /// Services the provider runs (one unit each).
141    pub services: Vec<ProviderService>,
142    /// Public ingress endpoints — forge hooks, tunnels.
143    pub ingress: Vec<ProviderIngress>,
144    /// Advisory verb hints for the CLI surface (`run`, `logs`,
145    /// `health`, …) — not enforced.
146    pub verbs: Vec<String>,
147    /// Optional health probe.
148    pub health: Option<ProviderHealth>,
149}

151/// One service under `provider.services` — rendered as a systemd unit
152/// (or the platform equivalent) by the deploy driver.
153#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize, schemars::JsonSchema)]
154#[serde(default)]
155pub struct ProviderService {
156    /// Unit/service name — `[A-Za-z0-9_.@-]`, no `..` (same charset the

168    pub workdir: Option<String>,
169    /// Restart policy (`Restart=`) — `on-failure` when unset.
170    pub restart: Option<String>,
171}

207/// `[A-Za-z0-9_.@-]`, no `..`, not `.`/`..` — the unit-name charset the
208/// local deploy adapter enforces; a provider service name becomes a
209/// unit name, so the manifest rejects anything else up front.
210fn is_valid_unit_name(s: &str) -> bool {
211    !s.is_empty()
212        && s != "."
213        && s != ".."
214        && !s.contains("..")
215        && s.chars()
216            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '@' | '-'))
217}

219/// `bins[].payload` path rules — a relative `tools/bin/…` path made of
220/// plain components only (no `.`/`..`, no root/prefix, no separators
221/// beyond `/`, no NUL). The triple-selection and on-disk checks live
222/// in `tamias-bin`; this is the manifest-level safety gate.
223fn check_bin_payload_path(bin: &str, payload: &str) -> Result<(), Error> {
224    let bad = |why: &str| {
225        Err(Error::Failed(format!(
226            "bin '{bin}': invalid payload path '{payload}' — {why}"
227        )))
228    };
229    if payload.is_empty() || payload.contains(['\0', '\\']) {
230        return bad("empty or contains NUL/backslash");
231    }
232    let comps: Vec<&str> = payload.split('/').collect();
233    if comps
234        .iter()
235        .any(|c| c.is_empty() || *c == "." || *c == "..")
236    {
237        return bad("must be a clean relative path (no empty, '.', or '..' segments)");
238    }
239    if comps.len() < 3 || comps[0] != "tools" || comps[1] != "bin" {
240        return bad("must live under tools/bin/");
241    }
242    Ok(())
243}

245/// Reject line breaks/NUL in a field that lands on one unit-file line —
246/// mirrors the deploy adapter's `check_unit_fields`.
247fn check_single_line(what: &str, value: &str) -> Result<(), Error> {
248    if value.contains(['\n', '\r', '\0']) {
249        Err(Error::Failed(format!(
250            "provider {what} contains a line break or NUL"
251        )))
252    } else {
253        Ok(())
254    }
255}

257impl Provider {
258    /// Semantic checks the schema can't express — required fields,
259    /// unit-name charset, no duplicate service names.
260    ///
261    /// # Errors
262    /// Fails on a malformed provider spec.
263    fn check(&self) -> Result<(), Error> {
264        if let Some(name) = &self.name {
265            if !is_valid_name(name) {
266                return Err(Error::Failed(format!("invalid provider name '{name}'")));
267            }
268        }
269        let mut seen = std::collections::BTreeSet::new();
270        for service in &self.services {
271            if !is_valid_unit_name(&service.name) {
272                return Err(Error::Failed(format!(
273                    "invalid provider service name '{}' (expected [A-Za-z0-9_.@-], no '..')",
274                    service.name
275                )));
276            }
277            if !seen.insert(service.name.as_str()) {
278                return Err(Error::Failed(format!(
279                    "duplicate provider service name '{}'",
280                    service.name
281                )));
282            }
283            if service.exec.trim().is_empty() {
284                return Err(Error::Failed(format!(
285                    "provider service '{}' is missing exec",
286                    service.name
287                )));
288            }
289            for (what, value) in [
290                ("service exec", service.exec.as_str()),
291                (
292                    "service env_file",
293                    service.env_file.as_deref().unwrap_or(""),
294                ),
295                ("service workdir", service.workdir.as_deref().unwrap_or("")),
296                ("service restart", service.restart.as_deref().unwrap_or("")),
297            ] {
298                check_single_line(what, value)?;
299            }
300            for (key, value) in &service.env {
301                if key.is_empty() {
302                    return Err(Error::Failed(format!(
303                        "provider service '{}' has an empty env name",
304                        service.name
305                    )));
306                }
307                check_single_line("service env name", key)?;
308                check_single_line("service env value", value)?;
309            }
310        }
311        for ingress in &self.ingress {
312            if ingress.public_url.trim().is_empty() {
313                return Err(Error::Failed(
314                    "provider ingress is missing public_url".to_string(),
315                ));
316            }
317            for (what, value) in [
318                ("ingress listen", ingress.listen.as_deref().unwrap_or("")),
319                ("ingress path", ingress.path.as_deref().unwrap_or("")),
320            ] {
321                check_single_line(what, value)?;
322            }
323            for hook in &ingress.hooks {
324                if hook.forge.trim().is_empty() || hook.repo.trim().is_empty() {
325                    return Err(Error::Failed(
326                        "provider hook is missing forge or repo".to_string(),
327                    ));
328                }
329                if hook.events.iter().any(|e| e.trim().is_empty()) {
330                    return Err(Error::Failed(
331                        "provider hook has an empty event name".to_string(),
332                    ));
333                }
334            }
335        }
336        if self.verbs.iter().any(|v| v.trim().is_empty()) {
337            return Err(Error::Failed(
338                "provider verbs must be non-empty".to_string(),
339            ));
340        }
341        if let Some(health) = &self.health {
342            if health.cmd.trim().is_empty() {
343                return Err(Error::Failed("provider health is missing cmd".to_string()));
344            }
345            check_single_line("health cmd", &health.cmd)?;
346        }
347        Ok(())
348    }
349}

351impl Manifest {
352    /// The manifest file inside `dir`, trying [`MANIFEST_NAMES`].
353    #[must_use]
354    pub fn path_in(dir: &Path) -> Option<PathBuf> {
355        MANIFEST_NAMES
356            .iter()
357            .map(|name| dir.join(name))
358            .find(|path| path.is_file())
359    }
360
361    /// Load + validate the manifest inside `dir`.
362    ///
363    /// # Errors
364    /// Fails when no manifest exists or it doesn't validate.
365    pub fn load(dir: &Path) -> Result<Self, Error> {
366        let Some(path) = Self::path_in(dir) else {
367            return Err(Error::Failed(format!(
368                "no tamias.yaml/.yml/.toml in {}",
369                dir.display()
370            )));
371        };
372        Self::load_at(&path)
373    }
374
375    /// Load + validate a manifest file; syntax dispatched on the
376    /// `.yaml`/`.yml` extension (everything else parses as TOML).
377    ///
378    /// # Errors
379    /// Fails on read, parse, or validation error.
380    pub fn load_at(path: &Path) -> Result<Self, Error> {
381        let manifest = Self::parse_at(path)?;
382        manifest.validate()?;
383        Ok(manifest)
384    }
385
386    /// Load a *project* manifest file — `tamias.yaml` holding `deps:` /
387    /// `include:` — same parse as [`Self::load_at`] but validated with
388    /// [`Self::validate_project`], so a missing `name` is allowed: a
389    /// project needn't be a package.
390    ///
391    /// # Errors
392    /// Fails on read, parse, or validation error.
393    pub fn load_project_at(path: &Path) -> Result<Self, Error> {
394        let manifest = Self::parse_at(path)?;
395        manifest.validate_project()?;
396        Ok(manifest)
397    }
398
399    /// Parse a manifest file without validating it.
400    fn parse_at(path: &Path) -> Result<Self, Error> {
401        let text =
402            std::fs::read_to_string(path).map_err(Error::io(format!("read {}", path.display())))?;
403        let is_yaml = matches!(
404            path.extension().and_then(|e| e.to_str()),
405            Some("yaml" | "yml")
406        );
407        if is_yaml {
408            serde_yaml_ng::from_str::<Self>(&text)
409                .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))
410        } else {
411            toml::from_str(&text)
412                .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))
413        }
414    }
415
416    /// Semantic checks the schema can't express — a package manifest must
417    /// name itself.
418    ///
419    /// # Errors
420    /// Fails on a missing/invalid name, scope, or version.
421    pub fn validate(&self) -> Result<(), Error> {
422        self.check(true)
423    }
424
425    /// Semantic checks for a *project* manifest — the same field rules as
426    /// [`Self::validate`] but `name` may be empty.
427    ///
428    /// # Errors
429    /// Fails on an invalid name/scope/version.
430    pub fn validate_project(&self) -> Result<(), Error> {
431        self.check(false)
432    }
433
434    fn check(&self, require_name: bool) -> Result<(), Error> {
435        if self.name.is_empty() {
436            if require_name {
437                return Err(Error::Failed(
438                    "manifest is missing a package name".to_string(),
439                ));
440            }
441        } else if !is_valid_name(&self.name) {
442            return Err(Error::Failed(format!(
443                "invalid package name '{}' — expected [A-Za-z0-9][A-Za-z0-9._-]*",
444                self.name
445            )));
446        }
447        if let Some(scope) = &self.scope {
448            if !is_valid_name(scope) {
449                return Err(Error::Failed(format!("invalid scope '{scope}'")));
450            }
451        }
452        if let Some(version) = &self.version {
453            if !is_valid_version(version) {
454                return Err(Error::Failed(format!(
455                    "invalid version '{version}' — expected [A-Za-z0-9][A-Za-z0-9._+-]* with no '..'"
456                )));
457            }
458        }
459        if let Some(kind) = &self.kind {
460            if !is_valid_kind(kind) {
461                return Err(Error::Failed(format!(
462                    "invalid content type '{kind}' — expected [a-z0-9][a-z0-9._-]* with no '..'"
463                )));
464            }
465        }
466        if let Some(provider) = &self.provider {
467            provider.check()?;
468        }
469        self.check_bins()?;
470        Ok(())
471    }
472
473    /// Semantic checks for `bins[]` — name charset + uniqueness,
474    /// exactly one of `shim`/`payload`, known shim kinds, a driver
475    /// kind for shims, and a safe payload-relative path.
476    fn check_bins(&self) -> Result<(), Error> {
477        let mut seen = std::collections::BTreeSet::new();
478        for bin in &self.bins {
479            if !is_valid_name(&bin.name) || bin.name == "tamias" {
480                return Err(Error::Failed(format!(
481                    "invalid bin name '{}' — expected [A-Za-z0-9][A-Za-z0-9._-]*, not 'tamias'",
482                    bin.name
483                )));
484            }
485            if !seen.insert(bin.name.as_str()) {
486                return Err(Error::Failed(format!("duplicate bin name '{}'", bin.name)));
487            }
488            match (&bin.shim, &bin.payload) {
489                (Some(_), Some(_)) | (None, None) => {
490                    return Err(Error::Failed(format!(
491                        "bin '{}' must set exactly one of shim/payload",
492                        bin.name
493                    )));
494                }
495                (Some(shim), None) => {
496                    if shim != "cli" {
497                        return Err(Error::Failed(format!(
498                            "bin '{}': unknown shim kind '{shim}' (only 'cli' exists)",
499                            bin.name
500                        )));
501                    }
502                    match &bin.driver {
503                        Some(driver) if is_valid_kind(driver) => {}
504                        _ => {
505                            return Err(Error::Failed(format!(
506                                "bin '{}': shim 'cli' requires a valid driver kind",
507                                bin.name
508                            )));
509                        }
510                    }
511                }
512                (None, Some(payload)) => {
513                    check_bin_payload_path(&bin.name, payload)?;
514                }
515            }
516        }
517        Ok(())
518    }
519
520    /// Write this manifest to `path` as YAML.
521    ///
522    /// # Errors
523    /// Propagates serialize + write failures.
524    pub fn save(&self, path: &Path) -> Result<(), Error> {
525        let ctx = |e| Error::Failed(format!("serialize manifest: {e}"));
526        let text = serde_yaml_ng::to_string(self).map_err(ctx)?;
527        crate::write_atomic(path, text.as_bytes())
528    }
529}

539    fn load_yaml_toml_and_validate() {
540        let dir = tempfile::tempdir().unwrap();
541        std::fs::write(
542            dir.path().join("tamias.toml"),
543            "name = \"x\"\nversion = \"1.0.0\"\n",
544        )
545        .unwrap();
546        let manifest = Manifest::load(dir.path()).unwrap();
547        assert_eq!(manifest.name, "x");
548        assert!(Manifest::path_in(dir.path()).is_some());
549        let saved = dir.path().join("saved.yaml");
550        manifest.save(&saved).unwrap();
551        assert_eq!(Manifest::load_at(&saved).unwrap().name, "x");
552        let empty = tempfile::tempdir().unwrap();
553        assert!(Manifest::load(empty.path()).is_err());
554        assert!(Manifest {
555            name: "bad name".into(),
556            ..Default::default()
557        }
558        .validate()
559        .is_err());
560        assert!(Manifest {
561            name: "x".into(),
562            scope: Some("bad scope".into()),
563            ..Default::default()
564        }
565        .validate()
566        .is_err());
567        // a package manifest must name itself; a project manifest needn't
568        assert!(Manifest::default().validate().is_err());
569        assert!(Manifest::default().validate_project().is_ok());
570        assert!(Manifest {
571            name: "bad name".into(),
572            ..Default::default()
573        }
574        .validate_project()
575        .is_err());
576        // nameless yaml still loads as a project manifest
577        let proj = dir.path().join("proj.yaml");
578        std::fs::write(&proj, "deps:\n  - other/pkg\n").unwrap();
579        let project = Manifest::load_project_at(&proj).unwrap();
580        assert!(project.name.is_empty());
581        assert_eq!(project.deps, ["other/pkg"]);
582        assert!(Manifest::load_at(&proj).is_err());
583    }

711      exec: "peregrin campaign serve /srv/ci"

720      hooks:

724  verbs: [run, logs, health]

739        assert_eq!(provider.ingress[0].hooks[0].forge, "github");

881    #[test]
882    fn bins_parse_and_validate() {
883        let dir = tempfile::tempdir().unwrap();
884        std::fs::write(
885            dir.path().join("tamias.yaml"),
886            "name: x\nbins:\n  \
887             - name: x-cli\n    shim: cli\n    driver: x\n  \
888             - name: x-tool\n    payload: tools/bin/x86_64-unknown-linux-gnu/x-tool\n",
889        )
890        .unwrap();
891        let m = Manifest::load(dir.path()).unwrap();
892        assert_eq!(m.bins.len(), 2);
893        assert_eq!(m.bins[0].shim.as_deref(), Some("cli"));
894        assert_eq!(m.bins[0].driver.as_deref(), Some("x"));
895        assert_eq!(
896            m.bins[1].payload.as_deref(),
897            Some("tools/bin/x86_64-unknown-linux-gnu/x-tool")
898        );
899
900        let with_bins = |bins: Vec<BinDecl>| Manifest {
901            name: "x".into(),
902            bins,
903            ..Default::default()
904        };
905        let shim = |name: &str| BinDecl {
906            name: name.into(),
907            shim: Some("cli".into()),
908            driver: Some("x".into()),
909            payload: None,
910        };
911        let payload = |name: &str, p: &str| BinDecl {
912            name: name.into(),
913            shim: None,
914            driver: None,
915            payload: Some(p.into()),
916        };
917        // happy paths
918        assert!(with_bins(vec![shim("a"), payload("b", "tools/bin/t/a")])
919            .validate()
920            .is_ok());
921        assert!(
922            with_bins(vec![payload("b", "tools/bin/aarch64-apple-darwin/x/b")])
923                .validate()
924                .is_ok()
925        );
926        // name rules
927        for bad in ["", "-x", "a b", "tamias", "../x"] {
928            assert!(
929                with_bins(vec![shim(bad)]).validate().is_err(),
930                "bin name {bad:?} must fail"
931            );
932        }
933        // exactly one of shim/payload
934        assert!(with_bins(vec![BinDecl {
935            name: "a".into(),
936            shim: Some("cli".into()),
937            driver: Some("x".into()),
938            payload: Some("tools/bin/t/a".into()),
939        }])
940        .validate()
941        .is_err());
942        assert!(with_bins(vec![BinDecl {
943            name: "a".into(),
944            shim: None,
945            driver: None,
946            payload: None,
947        }])
948        .validate()
949        .is_err());
950        // unknown shim kind / missing driver
951        assert!(with_bins(vec![BinDecl {
952            name: "a".into(),
953            shim: Some("mcp".into()),
954            driver: Some("x".into()),
955            payload: None,
956        }])
957        .validate()
958        .is_err());
959        assert!(with_bins(vec![BinDecl {
960            name: "a".into(),
961            shim: Some("cli".into()),
962            driver: None,
963            payload: None,
964        }])
965        .validate()
966        .is_err());
967        assert!(with_bins(vec![BinDecl {
968            name: "a".into(),
969            shim: Some("cli".into()),
970            driver: Some("Bad Kind".into()),
971            payload: None,
972        }])
973        .validate()
974        .is_err());
975        // payload path rules
976        for bad in [
977            "",
978            "bin/x",
979            "tools/x",
980            "tools/bin",
981            "tools/bin/",
982            "tools//bin/x",
983            "tools/bin/../x",
984            "tools/bin/./x",
985            "/tools/bin/x",
986            "tools/bin/x/../../y",
987            "tools\\bin\\x",
988            "tools/bin/a\0b",
989        ] {
990            assert!(
991                with_bins(vec![payload("a", bad)]).validate().is_err(),
992                "payload {bad:?} must fail"
993            );
994        }
995        // duplicate names
996        assert!(with_bins(vec![shim("a"), shim("a")]).validate().is_err());
997    }
