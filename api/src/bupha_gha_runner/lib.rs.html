1//! bupha-gha-runner — bupha as a **GitHub Actions self-hosted
2//! runner**: the runner registers with GitHub, polls the message
3//! queue, executes the jobs it is offered, and reports results —
4//! enough of the actions/runner broker protocol that a workflow can
5//! say `runs-on: [self-hosted, bupha]` and GitHub schedules onto it.
6//!
7//! This is the *opposite direction* from [`bupha-gha`](https://docs.rs/bupha-gha):
8//! that crate reads `.github/workflows/*.yml` and executes them as
9//! bupha CI; this crate makes bupha a runner GitHub itself drives.
10//! The name was reserved for this by T-054 — `gha` stays the
11//! workflow dialect, `gha-runner` is the broker protocol.
12//!
13//! ## Protocol coverage
14//!
15//! Implemented (the flow `actions/runner`'s `run.sh` performs):
16//!
17//! * registration — `POST {api}/actions/runner-registration` trades a
18//!   short-lived registration token (or a PAT-minted one via
19//!   `POST /repos/{repo}/actions/runners/registration-token`) for the
20//!   Actions service URL + bearer token; `Default` pool lookup;
21//!   `POST …/pools/{pool}/agents` with an RSA public key.
22//! * session — `POST/DELETE …/pools/{pool}/sessions`; RSA-OAEP
23//!   session-key unwrap and AES-256-CBC message decryption when the
24//!   session carries a key.
25//! * queue — long-poll `GET …/pools/{pool}/messages`, ack via
26//!   `DELETE …/messages/{id}`; broker migration
27//!   (`BrokerMigration` → `POST /session`, `GET /message` on the
28//!   broker) and job offers (`RunnerJobRequest` → `POST
29//!   {run_service}/acquirejob`).
30//! * execution — script steps via the job's shell, `actions/checkout`
31//!   via git, `$GITHUB_ENV`/`$GITHUB_OUTPUT`/`$GITHUB_PATH` env
32//!   files, `success()`/`failure()`/`cancelled()`/`always()`
33//!   conditions, job lease renewal, cancellation messages.
34//! * results — `WorkflowStepsUpdate`, signed-URL log upload
35//!   (append-blob + metadata), `completejob` with the outcome and
36//!   the evaluated job `outputs:` map (the `jobOutputs` templates
37//!   the service sends are resolved against collected
38//!   `$GITHUB_OUTPUT` values for `steps.*`/`needs.*`/`env.*`/
39//!   `github.*` refs — the channel `needs.*.outputs` reads back).
40//!
41//! Explicitly out of scope (fail loudly, never silently): node/docker
42//! actions other than `actions/checkout`, container jobs, service
43//! containers, `${{ }}` expression evaluation inside step fields
44//! (the service pre-compiles them — expressions that survive to the
45//! runner are passed through literally), the realtime log feed
46//! (websocket), and artifact/cache/upload endpoints.

66/// Everything the runner needs that isn't a credential.
67///
68/// Deserializable from the runner's config file (TOML/YAML — the
69/// impl-owned `extra` map convention); secrets are *names*, never
70/// values: `token_env`/`pat_env` name env vars, `token_file` names a
71/// file.
72#[derive(Debug, Clone, Deserialize)]
73#[serde(default)]
74pub struct Config {
75    /// The repository the runner joins — `owner/repo` or an https
76    /// URL. Required for registration; unused with stored
77    /// credentials.
78    pub repo: String,
79    /// Repo web URL for the tenant call — derived from `repo` when
80    /// empty (`https://github.com/owner/repo`).
81    pub repo_url: String,
82    /// Runner name shown in the repo's runner list. Default:
83    /// `bupha-<host>-<pid>`.
84    pub name: String,
85    /// Extra user labels on top of the system ones
86    /// (`self-hosted`, OS, arch). `bupha` is always added so
87    /// `runs-on: [self-hosted, bupha]` matches.
88    #[serde(default)]
89    pub labels: Vec<String>,
90    /// Forge instance — `github.com` (default) or a GHES host. The
91    /// API base is derived (`api.github.com` vs `{host}/api/v3`).
92    pub instance: String,
93    /// Explicit API base override (rare; GHES with a nonstandard
94    /// path).
95    pub api_base: String,
96    /// Env var NAME holding a runner registration token (the
97    /// settings-UI/`config.sh` token). Empty + `pat_env` set → mint
98    /// one via the REST API instead.
99    pub token_env: String,
100    /// Env var NAME holding a PAT with `repo` (or `admin:org`)
101    /// scope — used to mint a registration token when `token_env`
102    /// is unset.
103    pub pat_env: String,
104    /// File the resolved registration token may also live in
105    /// (e.g. `~/.config/bupha/gha-runner.token`, mode 600).
106    pub token_file: PathBuf,
107    /// Where the runner credentials (`.runner` equivalent) persist
108    /// once registered.
109    pub credentials: PathBuf,
110    /// Root for per-job workspaces.
111    pub workdir: PathBuf,
112    /// Max parallel jobs — informational today (one job at a time).
113    pub max_parallelism: u32,
114    /// Short-lived registration tokens expire in an hour; re-register
115    /// when credentials are missing AND this is set.
116    pub ephemeral: bool,
117}

139impl Config {
140    /// The repo web URL the tenant call registers against.
141    pub fn repo_url(&self) -> String {
142        if !self.repo_url.is_empty() {
143            return self.repo_url.trim_end_matches('/').to_string();
144        }
145        let repo = self.repo.trim_end_matches('/');
146        if repo.starts_with("http://") || repo.starts_with("https://") {
147            return repo.to_string();
148        }
149        let host = if self.instance == "github.com" {
150            "github.com".to_string()
151        } else {
152            self.instance.clone()
153        };
154        format!("https://{host}/{repo}")
155    }
156
157    /// `owner/repo` for REST paths — from `repo` or the URL.
158    pub fn repo_slug(&self) -> String {
159        let repo = self.repo.trim_end_matches('/');
160        if let Some((_, rest)) = repo.split_once("://") {
161            let path = rest.split('/').skip(1).collect::<Vec<_>>().join("/");
162            return path.trim_end_matches(".git").to_string();
163        }
164        repo.to_string()
165    }
166
167    /// The forge REST API base.
168    pub fn api_base(&self) -> String {
169        if !self.api_base.is_empty() {
170            return self.api_base.trim_end_matches('/').to_string();
171        }
172        if self.instance == "github.com" {
173            "https://api.github.com".to_string()
174        } else {
175            format!("https://{}/api/v3", self.instance)
176        }
177    }
178
179    /// Runner name — explicit, else `bupha-<host>-<pid>`.
180    pub fn runner_name(&self) -> String {
181        if !self.name.is_empty() {
182            return self.name.clone();
183        }
184        let host = hostname();
185        format!("bupha-{host}-{}", std::process::id())
186    }
187
188    /// Labels the agent registers with: `self-hosted`, OS, arch, then
189    /// `bupha` + the configured extras (deduped, system first).
190    pub fn labels(&self) -> Vec<types::Label> {
191        let system = ["self-hosted", host_os(), host_arch()];
192        let mut out: Vec<types::Label> = system
193            .iter()
194            .map(|n| types::Label {
195                name: (*n).to_string(),
196                kind: "system",
197            })
198            .collect();
199        let mut user = vec!["bupha".to_string()];
200        user.extend(self.labels.iter().cloned());
201        for name in user {
202            if system.contains(&name.as_str()) {
203                continue;
204            }
205            if out.iter().any(|l| l.name == name) {
206                continue;
207            }
208            out.push(types::Label { name, kind: "user" });
209        }
210        out
211    }
212}

214/// Resolve the registration token: env var first, then the token
215/// file. Returns None when neither yields one.
216pub fn resolve_token(cfg: &Config) -> Option<String> {
217    if !cfg.token_env.is_empty() {
218        if let Ok(v) = std::env::var(&cfg.token_env) {
219            let v = v.trim().to_string();
220            if !v.is_empty() {
221                return Some(v);
222            }
223        }
224    }
225    if !cfg.token_file.as_os_str().is_empty() {
226        if let Ok(v) = std::fs::read_to_string(&cfg.token_file) {
227            let v = v.trim().to_string();
228            if !v.is_empty() {
229                return Some(v);
230            }
231        }
232    }
233    None
234}

248/// Errors the runner surfaces — protocol, transport, crypto.
249#[derive(Debug)]
250pub enum Error {
251    Io(std::io::Error),
252    /// Transport-level failure (DNS, TLS, connect).
253    Http(String),
254    Ureq(ureq::Error),
255    /// A non-2xx with the service's own body (truncated).
256    Status {
257        status: u16,
258        body: String,
259    },
260    /// The service answered something this crate cannot read.
261    Protocol(String),
262    Crypto(String),
263}
264
265impl std::fmt::Display for Error {
266    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
267        match self {
268            Self::Io(e) => write!(f, "io: {e}"),
269            Self::Http(e) => write!(f, "http: {e}"),
270            Self::Ureq(e) => write!(f, "transport: {e}"),
271            Self::Status { status, body } => write!(f, "status {status}: {body}"),
272            Self::Protocol(e) => write!(f, "protocol: {e}"),
273            Self::Crypto(e) => write!(f, "crypto: {e}"),
274        }
275    }
276}
277
278impl std::error::Error for Error {}
279
280impl From<std::io::Error> for Error {
281    fn from(e: std::io::Error) -> Self {
282        Self::Io(e)
283    }
284}
285
286impl From<ureq::Error> for Error {
287    fn from(e: ureq::Error) -> Self {
288        Self::Ureq(e)
289    }
290}
291
292impl From<crypto::CryptoError> for Error {
293    fn from(e: crypto::CryptoError) -> Self {
294        Self::Crypto(e.0)
295    }
296}

304/// `linux`/`windows`/`macos` — the `runs-on` OS label and the
305/// `runnerOS` acquire field.
306pub fn host_os() -> &'static str {
307    if cfg!(windows) {
308        "Windows"
309    } else if cfg!(target_os = "macos") {
310        "macOS"
311    } else {
312        "Linux"
313    }
314}
315
316/// `x64`/`arm64`/… — the arch label and the `architecture` poll field.
317pub fn host_arch() -> &'static str {
318    match std::env::consts::ARCH {
319        "aarch64" => "ARM64",
320        "x86_64" => "X64",
321        other => other,
322    }
323}
324
325fn hostname() -> String {
326    std::env::var("HOSTNAME")
327        .or_else(|_| std::env::var("COMPUTERNAME"))
328        .unwrap_or_else(|_| "localhost".to_string())
329}

331/// Serialize credentials for persistence — used by `register`'s
332/// caller and tests.
333pub fn write_json<T: Serialize>(path: &Path, value: &T) -> std::io::Result<()> {
334    if let Some(parent) = path.parent() {
335        std::fs::create_dir_all(parent)?;
336    }
337    std::fs::write(
338        path,
339        serde_json::to_string_pretty(value).unwrap_or_default(),
340    )
341}

343/// Read a persisted JSON document.
344pub fn read_json<T: for<'de> Deserialize<'de>>(path: &Path) -> std::io::Result<T> {
345    let text = std::fs::read_to_string(path)?;
346    serde_json::from_str(&text).map_err(std::io::Error::other)
