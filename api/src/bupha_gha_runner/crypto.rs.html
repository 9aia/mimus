1//! The cryptography the protocol needs: an RSA keypair per runner
2//! (public half registered, private half signs the token assertion
3//! and unwraps session keys), JWT bearer assertions, RSA-OAEP session
4//! key unwrap, and AES-256-CBC message decryption.

17/// Errors here are protocol failures, not I/O.
18#[derive(Debug)]
19pub struct CryptoError(pub String);
20
21impl std::fmt::Display for CryptoError {
22    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
23        write!(f, "crypto: {}", self.0)
24    }
25}
26
27impl std::error::Error for CryptoError {}
28
29fn err(e: impl std::fmt::Display) -> CryptoError {
30    CryptoError(e.to_string())
31}

33/// A fresh 2048-bit runner key.
34pub fn generate_key() -> Result<RsaPrivateKey, CryptoError> {
35    RsaPrivateKey::new(&mut rand::thread_rng(), 2048).map_err(err)
36}
37
38/// The public half, base64 big-endian — the shape `AgentRequest`
39/// sends.
40pub fn public_key(key: &RsaPrivateKey) -> PublicKey {
41    let encode = |n: &rsa::BigUint| STANDARD.encode(n.to_bytes_be());
42    PublicKey {
43        exponent: encode(key.e()),
44        modulus: encode(key.n()),
45    }
46}

48/// PKCS#1 PEM — what the credentials file stores.
49pub fn private_key_pem(key: &RsaPrivateKey) -> Result<String, CryptoError> {
50    key.to_pkcs1_pem(rsa::pkcs1::LineEnding::LF)
51        .map(|p| p.to_string())
52        .map_err(err)
53}
54
55/// Either spelling: registration writes PKCS#1; a key from elsewhere
56/// may be PKCS#8.
57pub fn read_key(pem: &str) -> Result<RsaPrivateKey, CryptoError> {
58    RsaPrivateKey::from_pkcs1_pem(pem)
59        .or_else(|_| RsaPrivateKey::from_pkcs8_pem(pem))
60        .map_err(err)
61}
62
63fn b64url(bytes: &[u8]) -> String {
64    URL_SAFE_NO_PAD.encode(bytes)
65}
66
67fn seconds_since_epoch() -> u64 {
68    std::time::SystemTime::now()
69        .duration_since(std::time::UNIX_EPOCH)
70        .map(|d| d.as_secs())
71        .unwrap_or_default()
72}

74/// The RS256 JWT the token endpoint trades for an access token.
75///
76/// Claims are `iss`/`sub` = client id, `aud` = token url, unique
77/// `jti`, and a five-minute life counted from `nbf` — the service
78/// rejects longer. `nbf` is backdated 60s so a slightly fast clock
79/// still validates.
80pub fn sign_assertion(credentials: &Credentials) -> Result<String, CryptoError> {
81    let key = read_key(&credentials.private_key)?;
82    let now = seconds_since_epoch();
83    let claims = serde_json::json!({
84        "iss": credentials.client_id,
85        "sub": credentials.client_id,
86        "aud": credentials.token_url,
87        "jti": uuid::Uuid::new_v4().to_string(),
88        "nbf": now.saturating_sub(60),
89        "exp": now.saturating_sub(60) + 300,
90    });
91    let header = b64url(br#"{"alg":"RS256","typ":"JWT"}"#);
92    let payload = b64url(serde_json::to_string(&claims).map_err(err)?.as_bytes());
93    let signing_input = format!("{header}.{payload}");
94    let digest = Sha256::digest(signing_input.as_bytes());
95    let signature = key
96        .sign(rsa::Pkcs1v15Sign::new::<Sha256>(), &digest)
97        .map_err(err)?;
98    Ok(format!("{signing_input}.{}", b64url(&signature)))
99}

101/// Unwrap the session's AES key — RSA-OAEP, SHA-256 under FIPS,
102/// SHA-1 otherwise (what the service defaults to).
103pub fn unwrap_session_key(
104    key: &RsaPrivateKey,
105    wrapped: &[u8],
106    fips: bool,
107) -> Result<Vec<u8>, CryptoError> {
108    let padding = if fips {
109        Oaep::new::<Sha256>()
110    } else {
111        Oaep::new::<sha1::Sha1>()
112    };
113    key.decrypt(padding, wrapped)
114        .map_err(|e| CryptoError(format!("cannot unwrap the session key: {e}")))
115}

117/// AES-256-CBC with PKCS#7 — what the service's `Aes.Create()`
118/// defaults to on the other side.
119pub fn decrypt_message(key: &[u8], iv: &[u8], body: &[u8]) -> Result<String, CryptoError> {
120    use aes::cipher::{BlockDecryptMut, KeyIvInit};
121    type Decryptor = cbc::Decryptor<aes::Aes256>;
122    let decryptor = Decryptor::new_from_slices(key, iv)
123        .map_err(|e| CryptoError(format!("bad key or iv: {e}")))?;
124    let mut buf = body.to_vec();
125    let plain = decryptor
126        .decrypt_padded_mut::<aes::cipher::block_padding::Pkcs7>(&mut buf)
127        .map_err(|e| CryptoError(format!("cannot decrypt the message: {e}")))?;
128    String::from_utf8(plain.to_vec()).map_err(|e| CryptoError(format!("message is not utf8: {e}")))
129}

137    #[test]
138    fn key_pem_round_trips() {
139        let key = generate_key().unwrap();
140        let pem = private_key_pem(&key).unwrap();
141        let back = read_key(&pem).unwrap();
142        assert_eq!(back.n(), key.n());
143    }

145    #[test]
146    fn assertion_is_a_verifiable_rs256_jwt() {
147        let key = generate_key().unwrap();
148        let pem = private_key_pem(&key).unwrap();
149        let creds = Credentials {
150            service_url: "https://svc.example".into(),
151            pool_id: 1,
152            agent_id: 2,
153            agent_name: "bupha".into(),
154            client_id: "cid".into(),
155            token_url: "https://auth.example/token".into(),
156            private_key: pem,
157        };
158        let jwt = sign_assertion(&creds).unwrap();
159        let mut parts = jwt.split('.');
160        let (h, p, s) = (
161            parts.next().unwrap(),
162            parts.next().unwrap(),
163            parts.next().unwrap(),
164        );
165        assert_eq!(parts.next(), None);
166        let header: serde_json::Value =
167            serde_json::from_slice(&URL_SAFE_NO_PAD.decode(h).unwrap()).unwrap();
168        assert_eq!(header["alg"], "RS256");
169        let claims: serde_json::Value =
170            serde_json::from_slice(&URL_SAFE_NO_PAD.decode(p).unwrap()).unwrap();
171        assert_eq!(claims["iss"], "cid");
172        assert_eq!(claims["aud"], "https://auth.example/token");
173        assert!(claims["exp"].as_u64().unwrap() - claims["nbf"].as_u64().unwrap() <= 300);
174        // Verify the signature with the public key.
175        let sig = URL_SAFE_NO_PAD.decode(s).unwrap();
176        let n = key.n().to_bytes_be();
177        let e = key.e().to_bytes_be();
178        let pubkey = rsa::RsaPublicKey::new(
179            rsa::BigUint::from_bytes_be(&n),
180            rsa::BigUint::from_bytes_be(&e),
181        )
182        .unwrap();
183        pubkey
184            .verify(
185                rsa::Pkcs1v15Sign::new::<Sha256>(),
186                Sha256::digest(format!("{h}.{p}").as_bytes()).as_slice(),
187                &sig,
188            )
189            .unwrap();
190    }

207        let key = [7u8; 32];
208        let iv = [3u8; 16];

216            .to_vec();
