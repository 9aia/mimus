1//! Claim extraction — find factual assertions in prose, comments,
2//! and cards. Conservative by design: extracting a claim that
3//! isn't one produces noise downstream (see ADR-0001).
4
5use mimus_core::{Claim, ClaimKind};
6use std::path::Path;
7
8/// Extract claims from document text. v1 mines backtick-quoted
9/// spans — the highest-precision source: writers quote exactly
10/// the things they assert (paths, commands, names).
11pub fn extract(text: &str, source: &Path) -> Vec<Claim> {
12    // <!-- mimus-ignore-file --> — intentional dead refs throughout.
13    if text.contains("mimus-ignore-file") {
14        return Vec::new();
15    }
16    let mut out = Vec::new();
17    let mut in_fence = false;
18    let mut para = String::new();
19    let mut para_start = 0usize;
20    let mut para_end = 0usize;
21    let mut suppress_next = false;
22    for (line_no, line) in text.lines().enumerate() {
23        // <!-- mimus-ignore --> — alone on a line it suppresses
24        // the NEXT line's claims; inline it suppresses its own.
25        if line.contains("mimus-ignore") {
26            suppress_next = !line.contains('`');
27            continue;
28        }
29        if std::mem::take(&mut suppress_next) {
30            continue;
31        }
32        if line.trim_start().starts_with("```") {
33            in_fence = !in_fence;
34            continue;
35        }
36        // fenced blocks are examples/output, not claims about this
37        // repo — a tutorial showing `nonexistent/dead.rs` as sample
38        // output must not become a dead-path violation
39        if in_fence {
40            continue;
41        }
42        // table rows are enumerations of syntax, not assertions —
43        // "| `path` | `src/lib.rs` |" is an example, not a claim
44        if line.trim_start().starts_with('|') {
45            continue;
46        }
47        for span in backtick_spans(line) {
48            // `[text](url)` link syntax backticked is not a path
49            if span.contains("](") {
50                continue;
51            }
52            if let Some(kind) = classify(span) {
53                out.push(Claim {
54                    kind,
55                    text: span.to_owned(),
56                    source: source.to_path_buf(),
57                    line: (line_no + 1) as u32,
58                });
59            }
60        }
61        // Paragraph accumulation: hard-wrapped md splits sentences
62        // across lines — join, flush on blank/heading/fence.
63        let t = line.trim();
64        let breaks = t.is_empty() || t.starts_with('#') || t.starts_with('|') || in_fence;
65        if breaks {
66            flush_para(&mut para, para_start, para_end, source, &mut out);
67            continue;
68        }
69        if t.starts_with('-') || t.starts_with('*') || t.starts_with('>') {
70            continue; // list items / quotes — fragments without context
71        }
72        if para.is_empty() {
73            para_start = line_no + 1;
74        }
75        para_end = line_no + 1;
76        if !para.is_empty() {
77            para.push(' ');
78        }
79        para.push_str(t);
80    }
81    flush_para(&mut para, para_start, para_end, source, &mut out);
82    out
83}

85/// Sentence-split a paragraph into Behavior claims — whole
86/// sentences only, capped per file.
87fn flush_para(
88    para: &mut String,
89    start: usize,
90    end: usize,
91    source: &Path,
92    out: &mut Vec<Claim>,
93) {
94    if para.is_empty() {
95        return;
96    }
97    let text = std::mem::take(para);
98    let seen = out
99        .iter()
100        .filter(|c| c.source == source && c.kind == ClaimKind::Behavior)
101        .count();
102    let mut budget = 10usize.saturating_sub(seen);
103    for sent in sentences(&text) {
104        if budget == 0 {
105            break;
106        }
107        if !is_assertion(sent) {
108            continue;
109        }
110        budget -= 1;
111        out.push(Claim {
112            kind: ClaimKind::Behavior,
113            text: sent.to_owned(),
114            source: source.to_path_buf(),
115            // line granularity: use paragraph start; precise
116            // mapping needs offsets, not worth it v1
117            line: start.max(end) as u32,
118        });
119    }
120}

122/// Naive sentence split on ". " / ".\n" — good enough for claim
123/// extraction (false splits cost a claim, not correctness).
124fn sentences(text: &str) -> Vec<&str> {
125    text.split(". ")
126        .map(|s| s.trim())
127        .filter(|s| !s.is_empty())
128        .collect()
129}
130
131fn is_assertion(s: &str) -> bool {
132    !s.ends_with(':') // list intro, not a claim
133        && s.split_whitespace().count() >= 6
134        && s.split_whitespace().any(|w| {
135            matches!(
136                w.trim_matches(|c: char| !c.is_alphabetic()),
137                "is" | "are" | "stores" | "uses" | "reads" | "writes"
138                    | "returns" | "contains" | "supports" | "defaults"
139            )
140        })
141}

143/// Text inside `…` spans on one line.
144fn backtick_spans(line: &str) -> Vec<&str> {
145    let mut out = Vec::new();
146    let bytes = line.as_bytes();
147    let mut i = 0;
148    while i < bytes.len() {
149        if bytes[i] == b'`' {
150            // skip run of 2+ backticks (code fences)
151            if bytes.get(i + 1) == Some(&b'`') {
152                i += 1;
153                while i < bytes.len() && bytes[i] == b'`' {
154                    i += 1;
155                }
156                continue;
157            }
158            if let Some(end) = line[i + 1..].find('`') {
159                // end == 0 is unreachable — adjacent backticks were
160                // consumed by the run-skip above
161                out.push(&line[i + 1..i + 1 + end]);
162                i += end + 2;
163                continue;
164            }
165        }
166        i += 1;
167    }
168    out
169}

171/// Classify a backticked span — None means "not a checkable
172/// claim" (formatting, code refs we can't verify, etc.).
173fn classify(span: &str) -> Option<ClaimKind> {
174    let s = span.trim();
175    if s.is_empty() || s.len() > 200 {
176        return None;
177    }
178    if s.starts_with("http://") || s.starts_with("https://") {
179        return Some(ClaimKind::Link);
180    }
181    // Path-ish: contains / and no spaces, or ends in a known ext.
182    if !s.contains(char::is_whitespace)
183        && (s.contains('/')
184            || s.ends_with(".rs")
185            || s.ends_with(".md")
186            || s.ends_with(".toml")
187            || s.ends_with(".yaml")
188            || s.ends_with(".yml")
189            || s.ends_with(".sh"))
190    {
191        // But not flags. (An all-caps span can't reach here —
192        // path-ish requires `/` or an extension, neither of which
193        // is uppercase/`_` — so only the `-` guard is needed.)
194        if s.starts_with('-') {
195            return None;
196        }
197        // `@scope/pkg`, `~/alias/...` — import specifiers and host
198        // paths, not repo-relative paths the fs checker can prove.
199        if s.starts_with('@') || s.starts_with('~') {
200            return None;
201        }
202        // `VAR=value` assignments and template/glob-placeholder
203        // spans name a shape, not a concrete file — unverifiable.
204        if looks_like_env_assignment(s) || looks_like_template(s) {
205            return None;
206        }
207        // `nuqs/adapters/tanstack-router` — a bare package subpath:
208        // every segment lowercase-word, no filename anywhere. Same
209        // ambiguity FsChecker already resolves as silence for `a/b`.
210        if looks_like_import_specifier(s) {
211            return None;
212        }
213        return Some(ClaimKind::Path);
214    }
215    // Command-ish: only when the first token is a known binary —
216    // "starts with a lowercase word" eats code keywords (pub, data).
217    if s.contains(' ') {
218        // a space implies a first token — unwrap_or avoids a dead `?`
219        let first = s.split_whitespace().next().unwrap_or("");
220        const BINARIES: &[&str] = &[
221            "cargo", "rustc", "rustup", "git", "gh", "npm", "npx", "yarn", "pnpm", "bun",
222            "deno", "node", "pip", "pip3", "uv", "python", "python3", "pytest", "poetry",
223            "go", "make", "cmake", "ninja", "meson", "bazel", "mvn", "gradle", "docker",
224            "kubectl", "helm", "terraform", "ansible", "systemctl", "journalctl", "dagger",
225            "curl", "wget", "ssh", "scp", "rsync", "jq", "tar", "gzip", "zip", "unzip",
226            "rg", "grep", "find", "sed", "awk", "sort", "diff", "patch", "xargs", "tee",
227            "sh", "bash", "zsh", "sqlite3", "psql", "mysql", "openssl", "ffmpeg",
228            "nextest", "cargo-nextest", "kuda", "mimus", "bupha", "hnl", "clippy",
229            "rustfmt", "rustdoc", "wasm-pack", "trunk", "vite", "tsc", "eslint",
230        ];
231        if BINARIES.contains(&first) {
232            return Some(ClaimKind::Command);
233        }
234        return None;
235    }
236    // Bare identifier → symbol candidate (snake/camel/Upper idents).
237    // Whole-span `<name>` is a placeholder, not a generic type.
238    if s.starts_with('<') && s.ends_with('>') {
239        return None;
240    }
241    if s.chars()
242        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | ':' | '<' | '>'))
243        && s.chars().any(|c| c.is_ascii_alphabetic())
244        && s.len() >= 3
245    {
246        return Some(ClaimKind::Symbol);
247    }
248    None
249}

251/// `nuqs/adapters/tanstack-router`, `pkg/sub/thing` — a bare
252/// node_modules import specifier: ≥2 segments, all lowercase word
253/// characters, and no `.` anywhere (a real repo path names a file
254/// with an extension). Extensionless lowercase `docs/api/v1`-style
255/// claims are the same unprovable shape FsChecker already silences.
256fn looks_like_import_specifier(s: &str) -> bool {
257    s.contains('/')
258        && !s.contains('.')
259        // s is non-empty (checked in classify) — starts_with avoids
260        // a dead None arm on chars().next()
261        && s.starts_with(|c: char| c.is_ascii_lowercase())
262        && s.split('/').all(|seg| {
263            !seg.is_empty()
264                && seg
265                    .bytes()
266                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-' || b == b'_')
267        })
268}

270/// `FOO=bar`, `NEURALCHAT_BFF_PUBLIC_URL=http://x` — an env
271/// assignment, not a path the repo can satisfy.
272fn looks_like_env_assignment(s: &str) -> bool {
273    match s.split_once('=') {
274        Some((k, _)) => {
275            !k.is_empty()
276                && !k.chars().next().unwrap().is_ascii_digit()
277                && k.chars()
278                    .all(|c| c.is_ascii_alphanumeric() || c == '_')
279        }
280        None => false,
281    }
282}

284/// Spans naming a *pattern* — `..._x.md`, `{a,b}`, `<name>.rs`,
285/// `adr-mkt-0N.md`, `.md` — not a concrete path claim.
286fn looks_like_template(s: &str) -> bool {
287    if s.contains("...")
288        || s.contains('{')
289        || s.contains('}')
290        || s.contains('<')
291        || s.contains('>')
292    {
293        return true;
294    }
295    // bare extension (`.md`) — a suffix, not a file. (rest can't
296    // be empty: `.` alone never reaches the path gate — no `/`,
297    // no extension suffix)
298    if let Some(rest) = s.strip_prefix('.') {
299        if !rest.contains(['.', '/', '\\']) {
300            return true;
301        }
302    }
303    // digit+uppercase fragment inside a segment (`0N`, `0NN`) —
304    // serial placeholders, not real basenames. `v18`/`rfc1234`
305    // (letter-then-digit or lowercase) stay checkable.
306    s.split('/').any(|seg| {
307        seg.as_bytes()
308            .windows(2)
309            .any(|w| w[0].is_ascii_digit() && w[1].is_ascii_uppercase())
310    })
311}

319        let claims = extract(
320            "run `cargo test -p foo` then see `crates/x/src/lib.rs` and `README.md`",

353    #[test]
354    fn mimus_ignore_suppresses() {
355        // marker alone on a line -> next line's claims
356        let c = extract(
357            "<!-- mimus-ignore -->\nsee `dead/x.md` and `real/README.md`",
358            "d".as_ref(),
359        );
360        assert!(c.is_empty(), "{:?}", c);
361        // inline marker -> same line only
362        let c = extract(
363            "see `dead/x.md` <!-- mimus-ignore -->\nbut `README.md` lives",
364            "d".as_ref(),
365        );
366        assert_eq!(c.len(), 1, "{:?}", c);
367        assert_eq!(c[0].text, "README.md");
368        // file-level marker
369        let c = extract("<!-- mimus-ignore-file -->\n`README.md`", "d".as_ref());
370        assert!(c.is_empty(), "{:?}", c);
371    }

374/// Extract assertive comment lines as Behavior claims — the
375/// code↔code consistency surface: comments that assert things the
376/// code may or may not do. `//`, `///`, `//!`, `#` lines; capped.
377pub fn extract_comments(text: &str, source: &Path) -> Vec<Claim> {
378    let mut out = Vec::new();
379    for (i, line) in text.lines().enumerate() {
380        let t = line.trim();
381        let body = t
382            .strip_prefix("///")
383            .or_else(|| t.strip_prefix("//!"))
384            .or_else(|| t.strip_prefix("//"))
385            .or_else(|| t.strip_prefix('#'))
386            .map(str::trim);
387        let Some(b) = body else { continue };
388        if b.len() < 20 || b.starts_with('!') || b.starts_with('[') {
389            continue; // doc attrs / annotations
390        }
391        if !is_assertion(b) {
392            continue;
393        }
394        out.push(Claim {
395            kind: ClaimKind::Behavior,
396            text: b.to_owned(),
397            source: source.to_path_buf(),
398            line: (i + 1) as u32,
399        });
400        if out.len() >= 10 {
401            break;
402        }
403    }
404    out
