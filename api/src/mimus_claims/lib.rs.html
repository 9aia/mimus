1//! Claim extraction — find factual assertions in prose, comments,
2//! and cards. Conservative by design: extracting a claim that
3//! isn't one produces noise downstream (see ADR-0001).
4
5use mimus_core::{Claim, ClaimKind};
6use std::path::Path;
7
8/// Extract claims from document text. v1 mines backtick-quoted
9/// spans — the highest-precision source: writers quote exactly
10/// the things they assert (paths, commands, names).
11pub fn extract(text: &str, source: &Path) -> Vec<Claim> {
12    let mut out = Vec::new();
13    for (line_no, line) in text.lines().enumerate() {
14        for span in backtick_spans(line) {
15            if let Some(kind) = classify(span) {
16                out.push(Claim {
17                    kind,
18                    text: span.to_owned(),
19                    source: source.to_path_buf(),
20                    line: (line_no + 1) as u32,
21                });
22            }
23        }
24    }
25    out
26}
27
28/// Text inside `…` spans on one line.
29fn backtick_spans(line: &str) -> Vec<&str> {
30    let mut out = Vec::new();
31    let bytes = line.as_bytes();
32    let mut i = 0;
33    while i < bytes.len() {
34        if bytes[i] == b'`' {
35            // skip run of 2+ backticks (code fences)
36            if bytes.get(i + 1) == Some(&b'`') {
37                i += 1;
38                while i < bytes.len() && bytes[i] == b'`' {
39                    i += 1;
40                }
41                continue;
42            }
43            if let Some(end) = line[i + 1..].find('`') {
44                let span = &line[i + 1..i + 1 + end];
45                if !span.is_empty() {
46                    out.push(span);
47                }
48                i += end + 2;
49                continue;
50            }
51        }
52        i += 1;
53    }
54    out
55}

57/// Classify a backticked span — None means "not a checkable
58/// claim" (formatting, code refs we can't verify, etc.).
59fn classify(span: &str) -> Option<ClaimKind> {
60    let s = span.trim();
61    if s.is_empty() || s.len() > 200 {
62        return None;
63    }
64    if s.starts_with("http://") || s.starts_with("https://") {
65        return Some(ClaimKind::Link);
66    }
67    // Path-ish: contains / and no spaces, or ends in a known ext.
68    if !s.contains(char::is_whitespace)
69        && (s.contains('/')
70            || s.ends_with(".rs")
71            || s.ends_with(".md")
72            || s.ends_with(".toml")
73            || s.ends_with(".yaml")
74            || s.ends_with(".yml")
75            || s.ends_with(".sh"))
76    {
77        // But not env vars, flags, or dotted identifiers.
78        if s.starts_with('-') || s.chars().all(|c| c.is_ascii_uppercase() || c == '_') {
79            return None;
80        }
81        return Some(ClaimKind::Path);
82    }
83    // Command-ish: only when the first token is a known binary —
84    // "starts with a lowercase word" eats code keywords (pub, data).
85    if s.contains(' ') {
86        let first = s.split_whitespace().next()?;
87        const BINARIES: &[&str] = &[
88            "cargo", "rustc", "rustup", "git", "gh", "npm", "npx", "yarn", "pnpm", "bun",
89            "deno", "node", "pip", "pip3", "uv", "python", "python3", "pytest", "poetry",
90            "go", "make", "cmake", "ninja", "meson", "bazel", "mvn", "gradle", "docker",
91            "kubectl", "helm", "terraform", "ansible", "systemctl", "journalctl", "dagger",
92            "curl", "wget", "ssh", "scp", "rsync", "jq", "tar", "gzip", "zip", "unzip",
93            "rg", "grep", "find", "sed", "awk", "sort", "diff", "patch", "xargs", "tee",
94            "sh", "bash", "zsh", "sqlite3", "psql", "mysql", "openssl", "ffmpeg",
95            "nextest", "cargo-nextest", "kuda", "mimus", "bupha", "hnl", "clippy",
96            "rustfmt", "rustdoc", "wasm-pack", "trunk", "vite", "tsc", "eslint",
97        ];
98        if BINARIES.contains(&first) {
99            return Some(ClaimKind::Command);
100        }
101        return None;
102    }
103    // Bare identifier → symbol candidate (snake/camel/Upper idents).
104    if s.chars()
105        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | ':' | '<' | '>'))
106        && s.chars().any(|c| c.is_ascii_alphabetic())
107        && s.len() >= 3
108    {
109        return Some(ClaimKind::Symbol);
110    }
111    None
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    #[test]
119    fn extracts_paths_and_commands() {
120        let claims = extract(
121            "run `cargo test -p foo` then see `crates/x/src/lib.rs` and `README.md`",
122            "doc.md".as_ref(),
123        );
124        let kinds: Vec<_> = claims.iter().map(|c| (c.kind, c.text.as_str())).collect();
125        assert!(kinds.contains(&(ClaimKind::Command, "cargo test -p foo")));
126        assert!(kinds.contains(&(ClaimKind::Path, "crates/x/src/lib.rs")));
127        assert!(kinds.contains(&(ClaimKind::Path, "README.md")));
128    }
129
130    #[test]
131    fn skips_formatting_and_env() {
132        let claims = extract("`KUDA_TOKEN` and `--flag` and `someVar_123`", "d".as_ref());
133        assert!(claims.iter().all(|c| c.kind == ClaimKind::Symbol));
134    }
