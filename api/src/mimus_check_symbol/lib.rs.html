1//! Symbol adapter — `symbol` claims resolve against a `kuda`
2//! index. Consumes the index *file format* (`.kuda/index.sqlite`),
3//! not the kuda crates — the schema is the contract.
4//!
5//! Evidence rules: a claim found in the index is verified;
6//! a *namespaced* claim (`mod::Sym`) absent from it is refuted;
7//! a bare identifier absent from it is silence — it may simply
8//! be external (std/dep) and unproveable.
9
10use mimus_check::Checker;
11use mimus_core::{Claim, ClaimKind, Severity, Violation};
12use rusqlite::{params, Connection};
13use std::path::{Path, PathBuf};
14use std::sync::Mutex;
15
16/// Checks `ClaimKind::Symbol` — `new(kuda_index_path)`.
17/// `conn` is `None` when the index is missing/unreadable: every
18/// claim then emits silence, not failures.
19pub struct SymbolChecker {
20    conn: Option<Mutex<Connection>>,
21}
22
23impl SymbolChecker {
24    pub fn new(index: impl Into<PathBuf>) -> Self {
25        let conn = Connection::open(index.into()).ok().map(Mutex::new);
26        Self { conn }
27    }
28
29    /// Whether a usable index exists at the configured path.
30    pub fn index_present(&self) -> bool {
31        self.conn.is_some()
32    }
33}
34
35/// Every indexed symbol — `(name, kind, file, module)` — for
36/// coverage analysis: which symbols no doc ever names.
37#[allow(unexpected_cfgs)]
38pub fn all_symbols(index: &Path) -> Result<Vec<(String, String, String, String)>, String> {
39    let conn = Connection::open(index).map_err(|e| e.to_string())?;
40    let mut st = conn
41        .prepare("SELECT name,kind,file,module FROM symbols")
42        .map_err(|e| e.to_string())?;
43    let rows = st.query_map([], |r| {
44        Ok((r.get(0)?, r.get(1)?, r.get(2)?, r.get(3)?))
45    });
46    // `?` on query_map is unreachable — params![] can't fail to
47    // bind after a successful prepare; under coverage the Err arm
48    // would be a permanently red line
49    #[cfg(coverage)]
50    let rows = rows.unwrap();
51    #[cfg(not(coverage))]
52    let rows = rows.map_err(|e| e.to_string())?;
53    rows.collect::<Result<Vec<_>, _>>().map_err(|e| e.to_string())
54}
55
56struct Hit {
57    file: String,
58    line: i64,
59    module: String,
60}
61
62impl Checker for SymbolChecker {
63    fn handles(&self) -> &[ClaimKind] {
64        &[ClaimKind::Symbol]
65    }
66
67    #[allow(unexpected_cfgs)]
68    fn check(&self, claim: &Claim) -> Result<Option<Violation>, String> {
69        let Some(conn) = &self.conn else {
70            return Ok(None); // no index — nothing to verify against
71        };
72        let text = claim.text.trim();
73        let (leaf, ns) = match text.rsplit_once("::") {
74            Some((ns, leaf)) => (leaf, Some(ns)),
75            None => (text, None),
76        };
77        let conn = conn.lock().map_err(|e| e.to_string())?;
78        let mut st = conn
79            .prepare("SELECT file,line,module FROM symbols WHERE name = ?1")
80            .map_err(|e| e.to_string())?;
81        let rows = st.query_map(params![leaf], |r| {
82            Ok(Hit {
83                file: r.get(0)?,
84                line: r.get(1)?,
85                module: r.get(2)?,
86            })
87        });
88        // same unreachable `?` as all_symbols — params![leaf] binds
89        // a String after a successful prepare; see above
90        #[cfg(coverage)]
91        let rows = rows.unwrap();
92        #[cfg(not(coverage))]
93        let rows = rows.map_err(|e| e.to_string())?;
94        let hits: Vec<Hit> = rows.flatten().collect();
95
96        match (ns, hits.is_empty()) {
97            // bare ident found — verified
98            (None, false) => Ok(None),
99            // namespaced and found — the claimed path must agree
100            (Some(ns), false) => {
101                let ok = hits.iter().any(|h| {
102                    h.file.replace('/', "::").contains(ns)
103                        || h.module.replace('-', "_").contains(ns)
104                });
105                if ok {
106                    return Ok(None);
107                }
108                let h = &hits[0];
109                Ok(Some(Violation {
110                    claim: claim.clone(),
111                    evidence: format!(
112                        "`{leaf}` lives at {}:{} (module {}), not `{ns}`",
113                        h.file, h.line, h.module
114                    ),
115                    fix: Some(format!("`{}::{leaf}`", h.file.trim_end_matches(".rs").replace('/', "::"))),
116                    severity: Severity::Error,
117                }))
118            }
119            // namespaced and absent — the index covers the repo, so
120            // the claimed path is refuted
121            (Some(_), true) => Ok(Some(Violation {
122                claim: claim.clone(),
123                evidence: format!("`{text}` not found in .kuda index — renamed, moved, or dead?"),
124                fix: Some("update the reference or run `kuda sync`".into()),
125                severity: Severity::Error,
126            })),
127            // bare ident absent — could be std/dep; unproveable
128            (None, true) => Ok(None),
129        }
130    }
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136
137    /// A poisoned `conn` mutex — lock() errors surface as Err, not
138    /// a panic or a hang (a checker must stay advisory).
139    #[test]
140    fn poisoned_conn_lock_errors() {
141        let dir = std::env::temp_dir()
142            .join(format!("mimus-sym-poison-{}", std::process::id()));
143        let _ = std::fs::remove_dir_all(&dir);
144        std::fs::create_dir_all(&dir).unwrap();
145        let idx = dir.join("i.sqlite");
146        // empty file opens as an empty db — conn is Some
147        std::fs::write(&idx, "").unwrap();
148        let c = SymbolChecker::new(&idx);
149        assert!(c.index_present());
150        let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
151            let _g = c.conn.as_ref().unwrap().lock().unwrap();
152            panic!("poison the lock");
153        }));
154        let cl = Claim {
155            kind: ClaimKind::Symbol,
156            text: "SomeSym".into(),
157            source: "d.md".into(),
158            line: 1,
159        };
160        assert!(c.check(&cl).is_err());
161        let _ = std::fs::remove_dir_all(&dir);
162    }
