2//! tamias-serve — the ndjson JSON-RPC surface for embedding: read one JSON
3//! request per line on stdin, write one JSON response per line on stdout.
4//! Same wire as `sepia serve` / `cuora serve` / `myco serve`.
5//!
6//! Ops: `install`, `plan`, `sync`, `list`, `info`, `verify`, `remove`,
7//! `search`, `publish`, `registry.list`, `registry.get`, `registry.add`,
8//! `registry.remove`. Requests are `{"op":"…", …}`; responses are
9//! `{"ok":true,"result":…}` or `{"ok":false,"error":"…"}`.
10//!
11//! `plan` is `tamias install --plan` — `{"op":"plan","ref":…,
12//! "allow_drivers":…}` fetches/plans and reports what a real install
13//! would do (includes, drivers, provider services, trust tier,
14//! would-be problems) without committing or dispatching.
15//!
16//! `sync` reconciles the *server's* cwd project dep set against the
17//! store + `tamias.lock` — the same `tamias sync` semantics
18//! (`{"op":"sync","locked":…,"from_lock":…,"allow_drivers":…,
19//! "dir":…}`; `dir` defaults to `.`). `from_lock: true` treats
20//! `tamias.lock` itself as the spec (no manifest consult, no lock
21//! rewrite). Per-dep outcomes come back in `result.deps`; a
22//! bundled-driver consent refusal is an `ok:false` error like
23//! `install`'s.
24//!
25//! `search`/`publish`/`registry.*` resolve named registry sources and
26//! config writes through `tamias-registry-open` — the same code the
27//! `tamias` CLI verbs use, so the wire surface can't diverge.
28//!
29//! **Trust boundary.** Stdin is fully trusted: `install` accepts
30//! `allow_drivers:true` (bundled-driver consent), and
31//! `publish`/`registry.add`/`registry.remove` mutate config and the
32//! registry. Embedders must treat the pipe as a privileged channel —
33//! never expose it to untrusted input.

43/// Store + cache + registry roots, plus the config file the
44/// registry-aware ops read/write.
45#[derive(Clone, Debug)]
46pub struct Env {
47    /// The local store root.
48    pub store_root: PathBuf,
49    /// The fetch-cache root.
50    pub cache_root: PathBuf,
51    /// The default local-registry root (`TAMIAS_REGISTRY`/XDG) — the
52    /// target when a `search`/`publish` `registry` value is absent.
53    pub registry_root: PathBuf,
54    /// Explicit config file for the registry-aware ops — `None` honors
55    /// `TAMIAS_CONFIG` then `./tamias.{yml,yaml,toml}` discovery, and
56    /// the file is re-read per op so a `registry.add` is visible to a
57    /// later `search`/`publish` in the same session.
58    pub config_file: Option<PathBuf>,
59}
60
61/// The effective config for read-side resolution (`search`/`publish`
62/// routing, `install`'s download cap) — mirrors the CLI's
63/// warn-and-ignore posture: a broken config warns and yields defaults
64/// rather than failing every op. Management ops (`registry.*`) read
65/// through `config_target` directly and propagate parse errors.
66fn env_config(env: &Env) -> tamias_config::Config {
67    match tamias_registry_open::load_config(env.config_file.as_deref()) {
68        Ok(config) => config,
69        Err(e) => {
70            eprintln!("tamias-serve: warning — ignoring config file: {e}");
71            tamias_config::Config::default()
72        }
73    }
74}

76/// Run the loop until stdin closes.
77///
78/// # Errors
79/// Propagates stdin/stdout failures.
80pub fn serve(env: Env) -> Result<(), Error> {
81    serve_loop(env)
82}
83
84/// The real blocking stdin loop — compiled out under `cargo-llvm-cov`
85/// (`cfg(coverage)`) so the coverage build doesn't hang on stdin.
86#[cfg(not(coverage))]
87fn serve_loop(env: Env) -> Result<(), Error> {
88    let stdin = std::io::stdin();
89    let mut stdout = std::io::stdout();
90    for line in stdin.lock().lines() {
91        let line = line.map_err(Error::io("read stdin"))?;
92        if line.trim().is_empty() {
93            continue;
94        }
95        let response = handle(&env, &line);
96        writeln!(stdout, "{response}").map_err(Error::io("write stdout"))?;
97        stdout.flush().map_err(Error::io("flush stdout"))?;
98    }
99    Ok(())
100}

102/// The coverage probe — a fast exit so the loop's lines aren't measured.
103#[cfg(coverage)]
104fn serve_loop(_env: Env) -> Result<(), Error> {
105    Err(Error::Failed(
106        "serve loop is excluded under coverage".to_string(),
107    ))
108}

110fn handle(env: &Env, line: &str) -> String {
111    match dispatch(env, line) {
112        Ok(result) => json!({"ok": true, "result": result}).to_string(),
113        Err(e) => json!({"ok": false, "error": e.to_string()}).to_string(),
114    }
115}

117fn str_field<'a>(req: &'a Value, key: &str) -> Result<&'a str, Error> {
118    req.get(key)
119        .and_then(Value::as_str)
120        .ok_or_else(|| Error::Failed(format!("missing '{key}'")))
121}

123fn dispatch(env: &Env, line: &str) -> Result<Value, Error> {
124    let req: Value =
125        serde_json::from_str(line).map_err(|e| Error::Failed(format!("bad request: {e}")))?;
126    let op = str_field(&req, "op")?;
127    match op {
128        "install" => {
129            let reference = str_field(&req, "ref")?;
130            // Optional consent for payload-shipped `drivers/` — same rule as
131            // `tamias install --allow-drivers` / `TAMIAS_ALLOW_DRIVERS=1`.
132            let allow_drivers = req
133                .get("allow_drivers")
134                .and_then(Value::as_bool)
135                .unwrap_or(false);
136            // The shared install pipeline — silent progress, `"allow_drivers"`
137            // consent hints (see `tamias_install::Surface::Rpc`).
138            let config = env_config(env);
139            let entry = tamias_install::install(
140                &tamias_install::Env {
141                    store_root: env.store_root.clone(),
142                    cache_root: env.cache_root.clone(),
143                    // Global `limits.max_download` caps http(s)/`cf:`
144                    // fetches — same resolution as the CLI's install.
145                    max_download: Some(config.effective_limits(None, None).max_download),
146                    // `signing.*` — package-signature verification policy.
147                    signing: tamias_sign::SigningPolicy::resolve(Some(&config)),
148                    bin_dir: None,
149                },
150                reference,
151                allow_drivers,
152            )?;
153            Ok(json!({
154                "label": entry.label(),
155                "resolved": entry.record.resolved,
156                "digest": entry.record.digest,
157                "signed": entry.record.signature.is_some(),
158            }))
159        }
160        "plan" => {
161            // `tamias install --plan` — the dry-run over the same
162            // store/cache roots + config resolution. Reports what a
163            // real install would do; commits and dispatches nothing.
164            let reference = str_field(&req, "ref")?;
165            let allow_drivers = req
166                .get("allow_drivers")
167                .and_then(Value::as_bool)
168                .unwrap_or(false);
169            let plan = tamias_install::plan_install(
170                &tamias_install::Env::for_config(
171                    env.store_root.clone(),
172                    env.cache_root.clone(),
173                    Some(&env_config(env)),
174                ),
175                reference,
176                allow_drivers,
177            )?;
178            Ok(serde_json::to_value(&plan).map_err(|e| Error::Failed(e.to_string()))?)
179        }
180        "sync" => {
181            // `tamias sync` — reconcile the project dep set against
182            // the store + `tamias.lock`. `dir` (default `.`) is the
183            // project dir; fs refs inside `deps:` still resolve
184            // against the server's cwd, same as `install`.
185            // `from_lock: true` swaps the spec to `tamias.lock`
186            // itself (the manifest's `deps:` is not consulted).
187            let allow_drivers = req
188                .get("allow_drivers")
189                .and_then(Value::as_bool)
190                .unwrap_or(false);
191            let locked = req.get("locked").and_then(Value::as_bool).unwrap_or(false);
192            let from_lock = req
193                .get("from_lock")
194                .and_then(Value::as_bool)
195                .unwrap_or(false);
196            let dir = req
197                .get("dir")
198                .and_then(Value::as_str)
199                .map_or_else(|| Path::new("."), Path::new);
200            let config = env_config(env);
201            let report = tamias_install::sync_project(
202                &tamias_install::Env {
203                    store_root: env.store_root.clone(),
204                    cache_root: env.cache_root.clone(),
205                    max_download: Some(config.effective_limits(None, None).max_download),
206                    signing: tamias_sign::SigningPolicy::resolve(Some(&config)),
207                    bin_dir: None,
208                },
209                dir,
210                tamias_install::Surface::Rpc,
211                tamias_install::SyncOptions {
212                    allow_drivers,
213                    locked,
214                    source: if from_lock {
215                        tamias_install::SyncSource::Lock
216                    } else {
217                        tamias_install::SyncSource::Deps
218                    },
219                },
220                &mut |_| {},
221            )?;
222            let deps: Vec<Value> = report
223                .outcomes
224                .iter()
225                .map(|o| {
226                    json!({
227                        "dep": o.dep,
228                        "status": o.status.as_str(),
229                        "detail": o.detail,
230                    })
231                })
232                .collect();
233            Ok(json!({"deps": deps, "failed": report.failed()}))
234        }
235        "list" => {
236            let entries = Store::new(&env.store_root).entries()?;
237            let items: Vec<Value> = entries
238                .iter()
239                .map(|e| {
240                    json!({
241                        "label": e.label(),
242                        "source_ref": e.record.source_ref,
243                        "digest": e.record.digest,
244                    })
245                })
246                .collect();
247            Ok(json!(items))
248        }
249        "info" => {
250            let name = str_field(&req, "name")?;
251            let entry = Store::new(&env.store_root).find(name)?;
252            Ok(json!({
253                "label": entry.label(),
254                "resolved": entry.record.resolved,
255                "digest": entry.record.digest,
256                "size": entry.size()?,
257            }))
258        }
259        "verify" => {
260            let name = str_field(&req, "name")?;
261            let store = Store::new(&env.store_root);
262            let report = store.verify(name)?;
263            // Signature status alongside the digest check — the record's
264            // `signature` re-verified against `signing.allowed_signers`.
265            let entry = store.find(name)?;
266            let config = env_config(env);
267            let policy = tamias_sign::SigningPolicy::resolve(Some(&config));
268            let signature = tamias_sign::record_signature_status(
269                &entry.record,
270                policy.allowed_signers.as_deref(),
271                &policy.namespace,
272            )?
273            .as_str();
274            // Trust tier + signer re-derived live (same rule as the
275            // CLI's `verify`) — recorded `signer` wins when present.
276            let (trust, live_signer) = tamias_sign::trust_and_signer(
277                &entry.record.source_ref,
278                entry.record.signature.as_deref(),
279                &entry.record.name,
280                entry.record.scope.as_deref(),
281                entry.record.version.as_deref(),
282                &entry.record.digest,
283                &policy,
284            )?;
285            let signer = entry.record.signer.clone().or(live_signer);
286            Ok(json!({
287                "name": report.name,
288                "ok": report.ok,
289                "signature": signature,
290                "trust": trust.as_str(),
291                "signer": signer,
292            }))
293        }
294        "remove" => {
295            let name = str_field(&req, "name")?;
296            // The shared lightweight uninstall (`tamias_install::remove`)
297            // — same teardown rule as `tamias ui`'s Remove job.
298            let record = tamias_install::remove(&env.store_root, name, &mut |m| {
299                eprintln!("tamias-serve: warning — {m}");
300            })?;
301            Ok(json!({"name": record.name}))
302        }
303        // --- registry sources (shared with the CLI verbs) ---
304        "registry.list" => {
305            let (_path, config) = tamias_registry_open::config_target(env.config_file.as_deref())?;
306            let items: Vec<Value> = config
307                .registries
308                .iter()
309                .map(|(name, reg)| tamias_registry_open::registry_json(name, reg))
310                .collect();
311            Ok(json!({"registries": items}))
312        }
313        "registry.get" => {
314            let name = str_field(&req, "name")?;
315            let reg = tamias_registry_open::get_registry(env.config_file.as_deref(), name)?;
316            Ok(tamias_registry_open::registry_json(name, &reg))
317        }
318        "registry.add" => {
319            let name = str_field(&req, "name")?;
320            let kind = str_field(&req, "kind")?;
321            let url = str_field(&req, "url")?;
322            let token_env = req
323                .get("token_env")
324                .and_then(Value::as_str)
325                .map(str::to_string);
326            let id = req.get("id").and_then(Value::as_str).map(str::to_string);
327            let path = tamias_registry_open::add_registry(
328                env.config_file.as_deref(),
329                name,
330                kind,
331                url,
332                token_env,
333                id,
334            )?;
335            Ok(json!({"name": name, "config": path.display().to_string()}))
336        }
337        "registry.remove" => {
338            let name = str_field(&req, "name")?;
339            let path = tamias_registry_open::remove_registry(env.config_file.as_deref(), name)?;
340            Ok(json!({"name": name, "config": path.display().to_string()}))
341        }
342        "search" => {
343            let query = str_field(&req, "query")?;
344            let flag = req.get("registry").and_then(Value::as_str);
345            let config = env_config(env);
346            // Same routing as `tamias search`: flag → TAMIAS_REGISTRY →
347            // search.default → publish.default → local default root —
348            // the shared `search` (`tamias-registry-open`).
349            let (found, _target) =
350                tamias_registry_open::search(&env.registry_root, Some(&config), flag, query)?;
351            Ok(json!({"results": found.iter().map(registry_entry_json).collect::<Vec<_>>()}))
352        }
353        "publish" => {
354            let dir = str_field(&req, "dir")?;
355            let flag = req.get("registry").and_then(Value::as_str);
356            let config = env_config(env);
357            // Same routing as `tamias publish`: flag → TAMIAS_REGISTRY →
358            // publish.default → local default root — the shared
359            // `publish_plan` resolution (`tamias-registry-open`).
360            // `"keep_includes":true` ships the raw payload — the
361            // default flattens `include:` bases read-only.
362            let keep_includes = req
363                .get("keep_includes")
364                .and_then(Value::as_bool)
365                .unwrap_or(false);
366            // `"keep":<n>` — per-invocation retention override (P-024);
367            // absent → `retention.keep` config.
368            let keep = req.get("keep").and_then(Value::as_u64);
369            let plan = tamias_registry_open::publish_plan(
370                Path::new(dir),
371                &env.registry_root,
372                Some(&config),
373                flag,
374                &[config.publish.default.as_deref()],
375                &env.store_root,
376                keep_includes,
377                keep,
378                &mut |_, _| {},
379            )?;
380            // `"sign":true` — ssh-keygen -Y sign the provenance anchors;
381            // the key resolves like the CLI's `publish --sign`.
382            let signature = if req.get("sign").and_then(Value::as_bool).unwrap_or(false) {
383                let key = tamias_sign::resolve_signing_key(Some(&config), None)?;
384                let policy = tamias_sign::SigningPolicy::resolve(Some(&config));
385                Some(tamias_sign::sign_package(
386                    &plan.root,
387                    &plan.manifest,
388                    &key,
389                    &policy.namespace,
390                )?)
391            } else {
392                None
393            };
394            let outcome = plan.publish(signature.as_deref())?;
395            // `pruned` — index entries the retention policy dropped
396            // (P-024); payloads stay.
397            let mut out = registry_entry_json(&outcome.entry);
398            out["pruned"] = json!(outcome.pruned);
399            Ok(out)
400        }
401        other => Err(Error::Failed(format!("unknown op '{other}'"))),
402    }
403}

405/// One registry index entry as a JSON object — the `search`/`publish`
406/// result shape (same fields the CLI's `--json` prints).
407fn registry_entry_json(entry: &tamias_registry::RegistryEntry) -> Value {
408    json!({
409        "label": entry.label(),
410        "name": entry.name,
411        "scope": entry.scope,
412        "version": entry.version,
413        "kind": entry.kind,
414        "digest": entry.digest,
415        "signed": entry.signature.is_some(),
416        "publisher": entry.publisher,
417        "file": entry.file,
418    })

436    /// Serializes tests that mutate process env (`TAMIAS_HOME`,
437    /// `TAMIAS_ALLOW_DRIVERS`, `TAMIAS_REGISTRY`, config env). `cargo
438    /// test` runs tests on threads in one process, so mutations must
439    /// be serialized *and* restored even on panic.
440    static DRVENV: std::sync::Mutex<()> = std::sync::Mutex::new(());
441
442    /// RAII guard for [`DRVENV`]: restores the process environment and
443    /// working directory on drop — panic-safe. Poison is ignored: an
444    /// unwinding sibling already restored everything via its `Drop`.
445    struct EnvGuard {
446        _lock: std::sync::MutexGuard<'static, ()>,
447        env: Vec<(std::ffi::OsString, std::ffi::OsString)>,
448        cwd: std::path::PathBuf,
449    }
450
451    impl EnvGuard {
452        fn new() -> Self {
453            let lock = DRVENV.lock().unwrap_or_else(|e| e.into_inner());
454            let env = std::env::vars_os().collect();
455            let cwd = std::env::current_dir().unwrap();
456            Self {
457                _lock: lock,
458                env,
459                cwd,
460            }
461        }
462    }
463
464    impl Drop for EnvGuard {
465        fn drop(&mut self) {
466            for (k, _) in std::env::vars_os() {
467                if !self.env.iter().any(|(sk, _)| *sk == k) {
468                    std::env::remove_var(&k);
469                }
470            }
471            for (k, v) in &self.env {
472                std::env::set_var(k, v);
473            }
474            let _ = std::env::set_current_dir(&self.cwd);
475        }
476    }
