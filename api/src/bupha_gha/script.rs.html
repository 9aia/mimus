1//! script — render expanded [`Job`]s into one bash program. Pure
2//! string building; the runner spawns it with `bash -c` in the
3//! worktree. Jobs run sequentially in topo order; matrix combos are
4//! sequential expanded jobs (parallel matrix is a documented
5//! deferral — each leg would need its own workspace like a real
6//! runner provides).
7
8use crate::plan::{BinOp, Cond, DynRef, Job, Step};
9use std::collections::HashMap;

11/// Everything the generated program needs from the run context.
12pub struct ScriptCtx<'a> {
13    /// Absolute worktree path (exported as BUPHA_WORKDIR).
14    pub workdir: &'a str,
15    pub sha: &'a str,
16    pub git_ref: &'a str,
17    /// "owner/name".
18    pub repository: &'a str,
19}

21/// Single-quote a literal for safe embedding in bash.
22pub(crate) fn sq(s: &str) -> String {
23    format!("'{}'", s.replace('\'', "'\\''"))
24}

26/// `cd` target: workdir + optional repo-relative working-directory.
27/// Deferred `${{ }}` output refs become `$(cat …)` lookups — the
28/// pieces are quoted so the result stays one word.
29fn cd_target(
30    workdir: &str,
31    wd: Option<&str>,
32    ji: usize,
33    job_ji: &HashMap<String, usize>,
34) -> Result<String, String> {
35    match wd {
36        Some(w) if !w.is_empty() => shq_exprs(&format!("{workdir}/{w}"), ji, job_ji),
37        _ => Ok(sq(workdir)),
38    }
39}

41/// `export` lines for an env list. Values may carry deferred output
42/// refs — each becomes a quoted `$(cat …)` piece inside the word.
43fn exports(
44    pairs: &[(String, String)],
45    ji: usize,
46    job_ji: &HashMap<String, usize>,
47) -> Result<String, String> {
48    pairs
49        .iter()
50        .map(|(k, v)| Ok(format!("export {k}={}", shq_exprs(v, ji, job_ji)?)))
51        .collect::<Result<Vec<_>, String>>()
52        .map(|lines| lines.join("\n"))
53}

55/// Pick a heredoc delimiter the script can't collide with.
56fn heredoc_delim(script: &str, base: &str) -> String {
57    let mut d = base.to_string();
58    while script.lines().any(|l| l == d) {
59        d.push('X');
60    }
61    d
62}

64/// The shell GitHub would use for a `run` step: `shell:` maps to a
65/// known template, default is bash with GH's flags.
66fn shell_cmd(shell: Option<&str>) -> Result<&'static str, String> {
67    // A `shell:` value can't carry a `$(cat …)` lookup — it names
68    // the interpreter, not a word. Deferred output refs here are a
69    // hard error rather than a silently literal `${{ … }}`.
70    if shell.map(|s| s.contains("${{")).unwrap_or(false) {
71        return Err("`shell:` cannot contain `${{ }}` expressions".to_string());
72    }
73    match shell {
74        None | Some("bash") => Ok("bash --noprofile --norc -eo pipefail"),
75        Some("sh") => Ok("sh -e"),
76        Some(other) => Err(format!(
77            "unsupported `shell: {other}` — v1 supports bash (default) and sh"
78        )),
79    }
80}

82/// Everything a condition needs at render time: the expanded-job
83/// index (for `steps.*` refs of the current job), the job's `needs`
84/// indices (for the status-check functions) and the id → index map
85/// (for `needs.*` refs).
86struct CondCx<'a> {
87    ji: usize,
88    needs: &'a [usize],
89    job_ji: &'a HashMap<String, usize>,
90}

92/// Compile a [`Cond`] to a bash and-or list whose exit status is
93/// the truth value (0 = true). Used directly in `if …; then`.
94fn cond_bool(c: &Cond, cx: &CondCx) -> Result<String, String> {
95    Ok(match c {
96        Cond::Lit(s) => format!("bupha_truthy {}", sq(s)),
97        Cond::Dyn(d) => format!("bupha_truthy {}", dyn_word(d, cx)?),
98        Cond::Call { name, args } => match name.as_str() {
99            "success" => format!("bupha_needs_ok{}", idx_args(cx.needs)),
100            "failure" => format!("bupha_needs_failed{}", idx_args(cx.needs)),
101            "cancelled" => format!("bupha_needs_cancelled{}", idx_args(cx.needs)),
102            "always" => ":".to_string(),
103            "contains" | "startswith" | "endswith" => {
104                let (a, b) = (&args[0], &args[1]);
105                // GHA's contains/startsWith/endsWith are
106                // case-insensitive — `${var,,}` lowercases.
107                let pat = match name.as_str() {
108                    "contains" => "*\"${_b_i,,}\"*",
109                    "startswith" => "\"${_b_i,,}\"*",
110                    _ => "*\"${_b_i,,}\"",
111                };
112                format!(
113                    "( _b_s={}; _b_i={}; [[ \"${{_b_s,,}}\" == {pat} ]] )",
114                    cond_word(a, cx)?,
115                    cond_word(b, cx)?
116                )
117            }
118            _ => return Err(format!("unsupported `if` function `{name}()`")),
119        },
120        Cond::Bin { op, l, r } => match op {
121            BinOp::And => format!("( {} ) && ( {} )", cond_bool(l, cx)?, cond_bool(r, cx)?),
122            BinOp::Or => format!("( {} ) || ( {} )", cond_bool(l, cx)?, cond_bool(r, cx)?),
123            BinOp::Eq | BinOp::Ne => {
124                let cmp = if *op == BinOp::Eq { "==" } else { "!=" };
125                format!(
126                    "( _b_a={}; _b_b={}; [[ \"${{_b_a,,}}\" {cmp} \"${{_b_b,,}}\" ]] )",
127                    cond_word(l, cx)?,
128                    cond_word(r, cx)?
129                )
130            }
131        },
132        Cond::Not(inner) => format!("! ( {} )", cond_bool(inner, cx)?),
133    })
134}

136/// Compile a [`Cond`] to a bash *word* evaluating to its string
137/// value. Boolean-valued nodes yield `true`/`false` — matching how
138/// GHA stringifies booleans in comparisons.
139fn cond_word(c: &Cond, cx: &CondCx) -> Result<String, String> {
140    Ok(match c {
141        Cond::Lit(s) => sq(s),
142        Cond::Dyn(d) => dyn_word(d, cx)?,
143        _ => format!(
144            "\"$(if {}; then echo true; else echo false; fi)\"",
145            cond_bool(c, cx)?
146        ),
147    })
148}

150/// ` needs` indices as ` 0 1 2` for the `bupha_needs_*` helpers.
151fn idx_args(needs: &[usize]) -> String {
152    needs.iter().map(|i| format!(" {i}")).collect::<String>()
153}
154
155/// A deferred [`DynRef`] as a quoted bash word.
156fn dyn_word(d: &DynRef, cx: &CondCx) -> Result<String, String> {
157    Ok(match d {
158        DynRef::NeedOutput { job, name } => output_lookup(
159            &format!("needs.{job}.outputs.{name}"),
160            cx.ji,
161            cx.job_ji,
162            true,
163        )?,
164        DynRef::StepOutput { step, name } => output_lookup(
165            &format!("steps.{step}.outputs.{name}"),
166            cx.ji,
167            cx.job_ji,
168            true,
169        )?,
170        DynRef::NeedResult { job } => {
171            format!(
172                "\"$(cat \"$BUPHA_GHA_TMP/jobresult/{}\" 2>/dev/null || :)\"",
173                job
174            )
175        }
176        DynRef::StepConclusion { step } => {
177            // Step `run:` scripts execute in a child `bash` fed by
178            // a heredoc — shell functions don't cross that boundary,
179            // so conclusions live in files like outputs do.
180            format!(
181                "\"$(cat \"$BUPHA_GHA_TMP/stepconcl/{}/{}\" 2>/dev/null || :)\"",
182                cx.ji, step
183            )
184        }
185    })
186}

188/// Render all jobs into one bash script. Job order is the input
189/// order (already topo-sorted per workflow by the caller).
190pub fn render(jobs: &[Job], ctx: &ScriptCtx) -> Result<String, String> {
191    // job id → every expanded index (matrix legs) — `needs.<job>.*`
192    // aggregates over all of them.
193    let mut job_idx: HashMap<String, Vec<usize>> = HashMap::new();
194    for (i, j) in jobs.iter().enumerate() {
195        job_idx.entry(j.id.clone()).or_default().push(i);
196    }
197    // job id → expanded-job index, last matrix leg wins (legs run in
198    // order, so the last to finish has the freshest outputs). This
199    // is how `needs.<job>.outputs.*` resolves to a concrete path.
200    let mut job_ji: HashMap<String, usize> = HashMap::new();
201    for (i, j) in jobs.iter().enumerate() {
202        job_ji.insert(j.id.clone(), i);
203    }
204    // Job-level `timeout-minutes` runs the body under a watchdog —
205    // monitor mode gives each background job its own process group
206    // so `kill -- -$pid` takes the whole tree down.
207    let any_job_timeout = jobs.iter().any(|j| j.timeout_seconds.is_some());
208
209    let mut out = String::with_capacity(4096);
210    out.push_str("#!/usr/bin/env bash\n");
211    out.push_str("# Generated by bupha gha runner — .github/workflows as CI.\n");
212    out.push_str("set -uo pipefail\n");
213    if any_job_timeout {
214        out.push_str("set -m\n");
215    }
216    out.push_str(&format!("export GITHUB_EVENT_NAME={}\n", sq("push")));
217    out.push_str(&format!("export GITHUB_SHA={}\n", sq(ctx.sha)));
218    out.push_str(&format!("export GITHUB_REF={}\n", sq(ctx.git_ref)));
219    out.push_str(&format!(
220        "export GITHUB_REF_NAME={}\n",
221        sq(short_ref(ctx.git_ref))
222    ));
223    out.push_str(&format!(
224        "export GITHUB_REPOSITORY={}\n",
225        sq(ctx.repository)
226    ));
227    out.push_str(&format!("export BUPHA_WORKDIR={}\n", sq(ctx.workdir)));
228    out.push_str("declare -a BUPHA_GHA_STATUS=()\n");
229    out.push_str("declare -A BUPHA_GHA_JOBIDX=()\n");
230    for (id, idxs) in &job_idx {
231        let list = idxs
232            .iter()
233            .map(usize::to_string)
234            .collect::<Vec<_>>()
235            .join(" ");
236        out.push_str(&format!("BUPHA_GHA_JOBIDX[{}]={}\n", sq(id), sq(&list)));
237    }
238    out.push_str("BUPHA_GHA_FAILS=0\n");
239    // Per-run scratch for the $GITHUB_* command files — the dialect
240    // equivalent of the runner's per-job temp dir. One tmp dir for
241    // the whole run; each job gets its own subdir so env-file writes
242    // scope to the job like on GitHub.
243    out.push_str("BUPHA_GHA_TMP=$(mktemp -d)\n");
244    out.push_str("export BUPHA_GHA_TMP\n");
245    out.push_str("mkdir -p \"$BUPHA_GHA_TMP/jobstatus\" \"$BUPHA_GHA_TMP/stepconcl\" \"$BUPHA_GHA_TMP/jobresult\"\n");
246    out.push_str("trap 'rm -rf \"$BUPHA_GHA_TMP\"' EXIT\n");
247    out.push_str("export BUPHA_GHA_OUTPUTS=\"$BUPHA_GHA_TMP/outputs.txt\"\n");
248    out.push_str(": > \"$BUPHA_GHA_OUTPUTS\"\n");
249    out.push_str(ABSORB_FNS);
250
251    for (i, j) in jobs.iter().enumerate() {
252        let mut label = format!("{} ({})", j.label, j.workflow);
253        if !j.runs_on.is_empty() {
254            label.push_str(&format!(" [runs-on: {}]", j.runs_on.join(", ")));
255        }
256        out.push_str(&format!(
257            "echo {}\n",
258            shq_exprs(&format!("gha: job '{label}' starting"), i, &job_ji)?
259        ));
260        if let Some(mp) = j.max_parallel {
261            out.push_str(&format!(
262                "echo {}\n",
263                shq_exprs(
264                    &format!("gha: job '{label}' max-parallel {mp} ignored — matrix legs run sequentially"),
265                    i,
266                    &job_ji
267                )?
268            ));
269        }
270
271        // The run gate: `if:` when present (it already carries the
272        // implicit success() when the author didn't name a status
273        // function), else the implicit all-needs-passed check.
274        let gate: Option<(String, &'static str)> = if let Some(cond) = &j.cond {
275            let cx = CondCx {
276                ji: i,
277                needs: &j.needs,
278                job_ji: &job_ji,
279            };
280            Some((cond_bool(cond, &cx)?, "if condition false"))
281        } else if !j.needs.is_empty() {
282            let conds = j
283                .needs
284                .iter()
285                .map(|&d| format!("[ \"${{BUPHA_GHA_STATUS[{d}]:-2}}\" -eq 0 ]"))
286                .collect::<Vec<_>>()
287                .join(" && ");
288            Some((conds, "needs did not pass"))
289        } else {
290            None
291        };
292        let fail_fast = j.fail_fast;
293
294        // Gate opening — three shapes: fail-fast cancel outermost,
295        // then the run gate, else unconditional.
296        if fail_fast {
297            out.push_str(&format!(
298                "if bupha_job_failed {}; then\n  bupha_set_status {i} 3\n  echo {}\n",
299                sq(&j.id),
300                shq_exprs(
301                    &format!("gha: job '{label}' cancelled — fail-fast"),
302                    i,
303                    &job_ji
304                )?
305            ));
306            if let Some((g, _)) = &gate {
307                out.push_str(&format!("elif {g}; then\n"));
308            } else {
309                out.push_str("else\n");
310            }
311        } else if let Some((g, _)) = &gate {
312            out.push_str(&format!("if {g}; then\n"));
313        }
314
315        // Job body — the same `( set -e; … )` + status + outputs
316        // block as before, optionally under a timeout watchdog.
317        let timed = j.timeout_seconds;
318        out.push_str("  (\n    set -e\n");
319        out.push_str(&format!("    export GITHUB_WORKFLOW={}\n", sq(&j.workflow)));
320        // $GITHUB_* file commands: point each at a per-job scratch
321        // file. Steps append with `>>`; absorb helpers fold them
322        // back after every step (env → exports, path → PATH,
323        // output → per-step files under stepouts/ for the
324        // steps.*/needs.* expression lookups).
325        out.push_str(&format!(
326            "    BUPHA_JOB_TMP=\"$BUPHA_GHA_TMP/job_{i}\"\n    mkdir -p \"$BUPHA_JOB_TMP\"\n    export RUNNER_TEMP=\"$BUPHA_JOB_TMP\"\n    export GITHUB_WORKSPACE={}\n    export GITHUB_ENV=\"$BUPHA_JOB_TMP/github_env\"\n    export GITHUB_OUTPUT=\"$BUPHA_JOB_TMP/github_output\"\n    export GITHUB_PATH=\"$BUPHA_JOB_TMP/github_path\"\n    export GITHUB_STEP_SUMMARY=\"$BUPHA_JOB_TMP/step_summary.md\"\n    : > \"$GITHUB_ENV\"; : > \"$GITHUB_OUTPUT\"; : > \"$GITHUB_PATH\"; : > \"$GITHUB_STEP_SUMMARY\"\n",
327            sq(ctx.workdir)
328        ));
329        if !j.env.is_empty() {
330            out.push_str(&format!(
331                "    {}\n",
332                exports(&j.env, i, &job_ji)?.replace('\n', "\n    ")
333            ));
334        }
335        for (si, s) in j.steps.iter().enumerate() {
336            out.push_str(&render_step(s, i, si, j, ctx, &job_ji)?);
337        }
338        if let Some(secs) = timed {
339            // Watchdog: the job body runs in a background process
340            // group (set -m); the killer sleeps N seconds then
341            // TERMs the whole group. `wait` still reports the real
342            // exit status — 143 when killed.
343            out.push_str(&format!(
344                "  ) &\n  bupha_pid=$!\n  ( sleep {secs}; touch \"$BUPHA_GHA_TMP/timeout_{i}\"; kill -TERM -- -\"$bupha_pid\" 2>/dev/null ) &\n  bupha_wd=$!\n  wait \"$bupha_pid\"\n  bupha_set_status {i} \"$?\"\n  kill \"$bupha_wd\" 2>/dev/null\n  wait \"$bupha_wd\" 2>/dev/null\n  if [ -f \"$BUPHA_GHA_TMP/timeout_{i}\" ]; then rm -f \"$BUPHA_GHA_TMP/timeout_{i}\"; echo {}; fi\n",
345                shq_exprs(
346                    &format!("gha: job '{label}' timed out"),
347                    i,
348                    &job_ji
349                )?
350            ));
351        } else {
352            out.push_str("  )\n");
353            out.push_str(&format!("  bupha_set_status {i} \"$?\"\n"));
354        }
355        // Job `outputs:` evaluate once the job's steps are done —
356        // steps.*.outputs files are already on disk, needs.*.outputs
357        // read the producers' jobouts/ dirs. Skipped/failed jobs
358        // publish nothing, matching GitHub.
359        let mut pass = String::new();
360        for (name, tmpl) in &j.outputs {
361            pass.push_str(&format!(
362                "    bupha_job_out {i} {} {}\n",
363                sq(name),
364                shq_exprs(tmpl, i, &job_ji)?
365            ));
366        }
367        pass.push_str(&format!(
368            "    echo {}\n",
369            shq_exprs(&format!("gha: job '{label}' passed"), i, &job_ji)?
370        ));
371        out.push_str(&format!(
372            "  if [ \"${{BUPHA_GHA_STATUS[{i}]}}\" -eq 0 ]; then\n{pass}  else\n    echo {}\n    BUPHA_GHA_FAILS=$((BUPHA_GHA_FAILS+1))\n  fi\n",
373            shq_exprs(&format!("gha: job '{label}' failed"), i, &job_ji)?,
374        ));
375        if let Some((_, reason)) = &gate {
376            out.push_str(&format!(
377                "else\n  bupha_set_status {i} 2\n  echo {}\nfi\n",
378                shq_exprs(
379                    &format!("gha: job '{label}' skipped — {reason}"),
380                    i,
381                    &job_ji
382                )?
383            ));
384        } else if fail_fast {
385            out.push_str("fi\n");
386        }
387        // Refresh needs.<id>.result now that this leg's status is
388        // final — downstream jobs and steps read the aggregate file.
389        // Legs of one id run adjacently, so the aggregate converges
390        // once the last leg settles.
391        if let Some(legs) = job_idx.get(&j.id) {
392            let list = legs
393                .iter()
394                .map(usize::to_string)
395                .collect::<Vec<_>>()
396                .join(" ");
397            out.push_str(&format!("bupha_agg_result {} {list}\n", sq(&j.id)));
398        }
399    }
400    out.push_str("if [ \"$BUPHA_GHA_FAILS\" -gt 0 ]; then\n  echo \"gha: workflow(s) failed\"\n  exit 1\nfi\necho \"gha: all jobs passed\"\n");
401    Ok(out)
402}

404/// The filesystem key a step's `$GITHUB_OUTPUT` writes land under —
405/// its `id:` when declared, else an unreachable auto name (only
406/// `id:`-bearing steps can be referenced by `steps.<id>.outputs.*`).
407fn step_key(id: Option<&str>, si: usize) -> String {
408    id.map(str::to_string)
409        .unwrap_or_else(|| format!("__auto{si}"))
410}
411
412/// Record a step's conclusion — `$BUPHA_GHA_TMP/stepconcl/<job>/<key>`
413/// holds `success`/`skipped`; `steps.<id>.conclusion` reads it back.
414/// Files, not shell state: `run:` scripts execute in a child `bash`
415/// fed by a heredoc and can't see parent functions or arrays.
416fn step_concl_write(ji: usize, key: &str, value: &str) -> String {
417    format!(
418        "    mkdir -p \"$BUPHA_GHA_TMP/stepconcl/{ji}\"; printf '%s' {value} > \"$BUPHA_GHA_TMP/stepconcl/{ji}/{key}\"\n",
419        value = sq(value)
420    )
421}

423fn render_step(
424    s: &Step,
425    ji: usize,
426    si: usize,
427    job: &Job,
428    ctx: &ScriptCtx,
429    job_ji: &HashMap<String, usize>,
430) -> Result<String, String> {
431    let mut out = String::new();
432    match s {
433        Step::Checkout { id, cond } => {
434            let key = step_key(id.as_deref(), si);
435            let gate = step_gate(cond, ji, job, job_ji)?;
436            if let Some(g) = &gate {
437                out.push_str(&format!("    if {g}; then\n"));
438            }
439            out.push_str("    (\n      set -eo pipefail\n");
440            out.push_str(&format!(
441                "      cd {}\n",
442                cd_target(ctx.workdir, None, ji, job_ji)?
443            ));
444            if bupha_core::is_valid_sha(ctx.sha) {
445                out.push_str(&format!(
446                    "      head_sha=$(git rev-parse HEAD)\n      case \"$head_sha\" in {}*) ;; *) echo \"gha: actions/checkout — HEAD $head_sha != pushed {}\"; exit 1;; esac\n",
447                    ctx.sha,
448                    ctx.sha,
449                ));
450            }
451            out.push_str(
452                "      echo \"gha: actions/checkout — worktree already at pushed sha\"\n    )\n",
453            );
454            out.push_str(&step_concl_write(ji, &key, "success"));
455            if gate.is_some() {
456                out.push_str(&format!(
457                    "    else\n{}    echo {}\n    fi\n",
458                    step_concl_write(ji, &key, "skipped"),
459                    shq_exprs(
460                        &format!(
461                            "gha: step '{}' skipped — if condition false",
462                            step_label(id.as_deref(), si)
463                        ),
464                        ji,
465                        job_ji
466                    )?
467                ));
468            }
469        }
470        Step::Run(r) => {
471            let shell = shell_cmd(r.shell.as_deref().or(job.shell.as_deref()))?;
472            let wd = r
473                .working_directory
474                .as_deref()
475                .or(job.working_directory.as_deref());
476            if wd.map(|w| w.starts_with('/')).unwrap_or(false) {
477                return Err(format!(
478                    "absolute working-directory `{wd:?}` is unsupported"
479                ));
480            }
481            let key = step_key(r.id.as_deref(), si);
482            let gate = step_gate(&r.cond, ji, job, job_ji)?;
483            if let Some(g) = &gate {
484                out.push_str(&format!("    if {g}; then\n"));
485            }
486            out.push_str("    (\n      set -eo pipefail\n");
487            out.push_str(&format!(
488                "      cd {}\n",
489                cd_target(ctx.workdir, wd, ji, job_ji)?
490            ));
491            if !r.env.is_empty() {
492                out.push_str(&format!(
493                    "      {}\n",
494                    exports(&r.env, ji, job_ji)?.replace('\n', "\n      ")
495                ));
496            }
497            let delim = heredoc_delim(&r.script, &format!("BUPHA_EOF_{ji}_{si}"));
498            // `timeout-minutes` wraps the step's shell — GNU
499            // timeout signals the child's whole process group, so
500            // grandchildren (cargo, npm, …) die with it.
501            let timeout = r
502                .timeout_seconds
503                .map(|s| format!("timeout --kill-after=5 {s} "))
504                .unwrap_or_default();
505            out.push_str(&format!("      {timeout}{shell} <<'{delim}'\n"));
506            out.push_str(&script_body(&r.script, ji, job_ji)?);
507            if !r.script.ends_with('\n') {
508                out.push('\n');
509            }
510            out.push_str(&format!("{delim}\n    )\n"));
511            // Fold the step's $GITHUB_* file writes back: env
512            // entries export for later steps, PATH lines prepend,
513            // outputs land as per-name files under stepouts/<job>/
514            // <step-id>/ — the store steps.*.outputs.* and job
515            // outputs: read. A failed step never reaches this —
516            // `set -e` ends the job first, matching the runner's
517            // stop-on-failure v1 semantics.
518            out.push_str("    bupha_absorb_env_file \"$GITHUB_ENV\"\n");
519            out.push_str("    bupha_absorb_path_file \"$GITHUB_PATH\"\n");
520            out.push_str(&format!(
521                "    bupha_absorb_output_file \"$GITHUB_OUTPUT\" {} {}\n",
522                ji,
523                sq(&key)
524            ));
525            out.push_str(&step_concl_write(ji, &key, "success"));
526            if gate.is_some() {
527                out.push_str(&format!(
528                    "    else\n{}    echo {}\n    fi\n",
529                    step_concl_write(ji, &key, "skipped"),
530                    shq_exprs(
531                        &format!(
532                            "gha: step '{}' skipped — if condition false",
533                            step_label(r.id.as_deref(), si)
534                        ),
535                        ji,
536                        job_ji
537                    )?
538                ));
539            }
540        }
541    }
542    Ok(out)
543}

551fn step_gate(
552    cond: &Option<Cond>,
553    ji: usize,
554    job: &Job,
555    job_ji: &HashMap<String, usize>,

571/// A deferred `${{ needs.*.outputs.* }}` / `${{ steps.*.outputs.* }}`
572/// ref → the bash that reads its value: `cat` of the file the
573/// producer's absorb wrote. `quoted` wraps the substitution so the
574/// result stays one word (env exports, echo args); unquoted leaves
575/// word-splitting to the step's own quoting, matching how GitHub
576/// substitutes the raw value into `run:` scripts.
577fn output_lookup(
578    inner: &str,
579    ji: usize,
580    job_ji: &HashMap<String, usize>,
581    quoted: bool,
582) -> Result<String, String> {
583    let sub = if let Some(rest) = inner.strip_prefix("needs.") {
584        if let Some((job, name)) = rest.split_once(".outputs.") {
585            if job.is_empty() || name.is_empty() {
586                return Err(format!(
587                    "unsupported deferred expression `${{{{ {inner} }}}}`"
588                ));
589            }
590            // Unknown job ids read an impossible path → empty, like
591            // an output the producer never declared.
592            let target = job_ji
593                .get(job)
594                .map(|i| i.to_string())
595                .unwrap_or_else(|| "_".to_string());
596            format!("$(cat \"$BUPHA_GHA_TMP/jobouts/{target}/{name}\" 2>/dev/null || :)")
597        } else if let Some(job) = rest.strip_suffix(".result") {
598            if job.is_empty() {
599                return Err(format!(
600                    "unsupported deferred expression `${{{{ {inner} }}}}`"
601                ));
602            }
603            needs_result_lookup(job)
604        } else {
605            return Err(format!(
606                "unsupported deferred expression `${{{{ {inner} }}}}`"
607            ));
608        }
609    } else if let Some(rest) = inner.strip_prefix("steps.") {
610        if let Some((sid, name)) = rest.split_once(".outputs.") {
611            if sid.is_empty() || name.is_empty() {
612                return Err(format!(
613                    "unsupported deferred expression `${{{{ {inner} }}}}`"
614                ));
615            }
616            format!("$(cat \"$BUPHA_GHA_TMP/stepouts/{ji}/{sid}/{name}\" 2>/dev/null || :)")
617        } else if let Some(sid) = rest.strip_suffix(".conclusion") {
618            if sid.is_empty() {
619                return Err(format!(
620                    "unsupported deferred expression `${{{{ {inner} }}}}`"
621                ));
622            }
623            format!("$(cat \"$BUPHA_GHA_TMP/stepconcl/{ji}/{sid}\" 2>/dev/null || :)")
624        } else {
625            return Err(format!(
626                "unsupported deferred expression `${{{{ {inner} }}}}`"
627            ));
628        }
629    } else {
630        return Err(format!(
631            "unsupported deferred expression `${{{{ {inner} }}}}`"
632        ));
633    };
634    Ok(if quoted { format!("\"{sub}\"") } else { sub })
635}

637/// `needs.<job>.result` as a `cat` of the aggregate file
638/// `bupha_agg_result` maintains — self-contained, so it works inside
639/// step `run:` scripts (child `bash`, no helper functions). An
640/// unknown job id reads an impossible path → empty.
641fn needs_result_lookup(job: &str) -> String {
642    format!("$(cat \"$BUPHA_GHA_TMP/jobresult/{job}\" 2>/dev/null || :)")
643}

645/// Turn an expanded template into a bash word for positions outside
646/// `run:` scripts (env exports, echo args, cd targets): literal
647/// stretches single-quoted, deferred output refs as quoted `$(cat …)`
648/// pieces — adjacent pieces concatenate into one word.
649fn shq_exprs(tpl: &str, ji: usize, job_ji: &HashMap<String, usize>) -> Result<String, String> {
650    let mut out = String::new();
651    let mut rest = tpl;
652    while let Some(start) = rest.find("${{") {
653        out.push_str(&sq(&rest[..start]));
654        let after = &rest[start + 3..];
655        let Some(end) = after.find("}}") else {
656            return Err("unclosed `${{` in rendered template".to_string());
657        };
658        out.push_str(&output_lookup(after[..end].trim(), ji, job_ji, true)?);
659        rest = &after[end + 2..];
660    }
661    out.push_str(&sq(rest));
662    Ok(out)
663}
664
665/// Rewrite a `run:` script's deferred output refs into `$(cat …)`.
666/// Literal text passes through verbatim — the quoted heredoc keeps
667/// it literal for the step's own shell, so the substitution lands
668/// exactly where GitHub's pre-compiled expression would have put
669/// the value (inside the user's own quoting).
670fn script_body(script: &str, ji: usize, job_ji: &HashMap<String, usize>) -> Result<String, String> {
671    let mut out = String::with_capacity(script.len());
672    let mut rest = script;
673    while let Some(start) = rest.find("${{") {
674        out.push_str(&rest[..start]);
675        let after = &rest[start + 3..];
676        let Some(end) = after.find("}}") else {
677            return Err("unclosed `${{` in rendered script".to_string());
678        };
679        out.push_str(&output_lookup(after[..end].trim(), ji, job_ji, false)?);
680        rest = &after[end + 2..];
681    }
682    out.push_str(rest);
683    Ok(out)
684}

693/// Bash helpers emitted once per generated program. They implement
694/// the GitHub env-file contract the runner's `absorb_env_files`
695/// implements in Rust: `NAME=value` lines, `NAME<<DELIM` heredocs,
696/// `#` comments; each file is truncated after folding so the next
697/// step starts clean. Names must be shell identifiers — anything
698/// else is skipped rather than eval'd.
699const ABSORB_FNS: &str = r#"
700bupha_ltrim() { local s="$1"; printf '%s' "${s#"${s%%[![:space:]]*}"}"; }
701bupha_rtrim() { local s="$1"; printf '%s' "${s%"${s##*[![:space:]]}"}"; }
702
703bupha_absorb_env_file() {
704  local f="$1" line name val delim l
705  [ -f "$f" ] || return 0
706  while IFS= read -r line || [ -n "$line" ]; do
707    case "$line" in ''|'#'*) continue ;; esac
708    if [[ "$line" == *'<<'* && "$line" != *=* ]]; then
709      name=$(bupha_rtrim "${line%%'<<'*}")
710      delim=$(bupha_rtrim "$(bupha_ltrim "${line#*'<<'}")")
711      delim="${delim//\'/}"
712      delim="${delim//\"/}"
713      val=""
714      while IFS= read -r l; do
715        if [ "$l" = "$delim" ]; then break; fi
716        val+="$l"$'\n'
717      done
718    elif [[ "$line" == *=* ]]; then
719      name=$(bupha_rtrim "${line%%=*}")
720      val="${line#*=}"
721    else
722      continue
723    fi
724    name=$(bupha_ltrim "$name")
725    case "$name" in
726      ''|[0-9]*|*[!A-Za-z0-9_]*) continue ;;
727    esac
728    export "$name=$val"
729  done < "$f"
730  : > "$f"
731}
732
733bupha_absorb_path_file() {
734  local f="$1" line
735  [ -f "$f" ] || return 0
736  local -a dirs=()
737  while IFS= read -r line || [ -n "$line" ]; do
738    line=$(bupha_rtrim "$line")
739    if [ -n "$line" ]; then dirs+=("$line"); fi
740  done < "$f"
741  # GitHub prepends each line in file order, so the LAST line ends
742  # up front-most; later steps' entries outrank earlier ones
743  # because this runs after every step.
744  local i
745  for ((i = 0; i < ${#dirs[@]}; i++)); do
746    PATH="${dirs[$i]}:$PATH"
747  done
748  : > "$f"
749}
750
751bupha_absorb_output_file() {
752  # $1 = the step's GITHUB_OUTPUT file, $2 = expanded-job index,
753  # $3 = step key (its `id:` or an auto name). Each NAME=VALUE /
754  # NAME<<DELIM entry becomes $BUPHA_GHA_TMP/stepouts/<job>/<step>/<name>
755  # — the store `steps.<id>.outputs.*` and job `outputs:` read — and
756  # the raw lines still append to the run's outputs.txt log.
757  local f="$1" jikey="$2" sid="$3"
758  [ -f "$f" ] || return 0
759  if [ -s "$f" ]; then
760    { printf '# step %s/%s\n' "$jikey" "$sid"; cat "$f"; } >> "$BUPHA_GHA_OUTPUTS"
761  fi
762  local dir="$BUPHA_GHA_TMP/stepouts/$jikey/$sid"
763  local line name val delim l
764  while IFS= read -r line || [ -n "$line" ]; do
765    case "$line" in ''|'#'*) continue ;; esac
766    if [[ "$line" == *'<<'* && "$line" != *=* ]]; then
767      name=$(bupha_rtrim "${line%%'<<'*}")
768      delim=$(bupha_rtrim "$(bupha_ltrim "${line#*'<<'}")")
769      delim="${delim//\'/}"
770      delim="${delim//\"/}"
771      val=""
772      while IFS= read -r l; do
773        if [ "$l" = "$delim" ]; then break; fi
774        val+="$l"$'\n'
775      done
776    elif [[ "$line" == *=* ]]; then
777      name=$(bupha_rtrim "${line%%=*}")
778      val="${line#*=}"
779    else
780      continue
781    fi
782    name=$(bupha_ltrim "$name")
783    # Path-component safety: the expression charset never produces
784    # `/`, but a hand-written GITHUB_OUTPUT could — skip rather than
785    # write outside stepouts/.
786    case "$name" in
787      ''|.|..|*[!A-Za-z0-9_.-]*) continue ;;
788    esac
789    mkdir -p "$dir"
790    printf '%s' "$val" > "$dir/$name"
791  done < "$f"
792  : > "$f"
793}

804# Record an expanded job's final status: $1 = index, $2 = exit
805# code. Mirrors GHA's conclusion vocabulary into
806# $BUPHA_GHA_TMP/jobstatus/<i> so `needs.*.result` resolves inside
807# step `run:` scripts too — those execute in a child `bash` that
808# can't see this process's arrays or functions.
809bupha_set_status() {
810  local w
811  BUPHA_GHA_STATUS[$1]=$2
812  case "$2" in
813    0) w=success ;;
814    2) w=skipped ;;
815    3) w=cancelled ;;
816    *) w=failure ;;
817  esac
818  printf '%s' "$w" > "$BUPHA_GHA_TMP/jobstatus/$1"
819}

822# GHA truthiness, close enough for CI: "", "0", "false" (any case)

825  case "$1" in
826    ''|0|[fF][aA][lL][sS][eE]|[nN][uU][lL][lL]) return 1 ;;
827    *) return 0 ;;
828  esac

874# Recompute needs.<id>.result after a job's legs settle: $1 = job
875# id, rest = expanded indices. Aggregation matches GHA — any
876# failure → failure, else any cancellation → cancelled, else all
877# skipped → skipped, else success — and lands in
878# $BUPHA_GHA_TMP/jobresult/<id> so `needs.*.result` resolves with a
879# plain `cat` even inside step `run:` scripts (child `bash`, no
880# access to this process's arrays or functions).
881bupha_agg_result() {
882  local id="$1"; shift
883  local i st seen=0 allskip=1 anyfail=0 anycancel=0 res
884  for i in "$@"; do
885    seen=1
886    st=$(cat "$BUPHA_GHA_TMP/jobstatus/$i" 2>/dev/null)
887    case "$st" in
888      success) allskip=0 ;;
889      ""|skipped) ;;
890      cancelled) anycancel=1; allskip=0 ;;
891      *) anyfail=1; allskip=0 ;;
892    esac
893  done
894  if [ "$seen" -eq 0 ]; then
895    res=""
896  elif [ "$anyfail" -eq 1 ]; then
897    res=failure
898  elif [ "$anycancel" -eq 1 ]; then
899    res=cancelled
900  elif [ "$allskip" -eq 1 ]; then
901    res=skipped
902  else
903    res=success
904  fi
905  printf '%s' "$res" > "$BUPHA_GHA_TMP/jobresult/$id"
906}
