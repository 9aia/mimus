182/// A parsed `if:` condition. `env.*`/`github.*`/`matrix.*` fold to
183/// [`Cond::Lit`] during expand; `needs.*`/`steps.*` stay deferred —
184/// the renderer turns them into lookups against the run's status
185/// arrays and output stores.
186#[derive(Debug)]
187pub enum Cond {
188    /// A folded scalar — truthiness applies (`""`, `"0"`, `"false"`,
189    /// `"null"` are falsy, matching GHA's loose rules closely
190    /// enough for CI conditions).
191    Lit(String),
192    /// A runtime-resolved reference.
193    Dyn(DynRef),
194    /// A builtin function call (name lowercased).
195    Call {
196        name: String,
197        args: Vec<Cond>,
198    },
199    Bin {
200        op: BinOp,
201        l: Box<Cond>,
202        r: Box<Cond>,
203    },
204    Not(Box<Cond>),
205}
206
207/// The deferred half of a condition — resolved by the renderer
208/// against the run's `jobstatus/`+`jobresult/`+`stepconcl/`+
209/// `stepouts/`+`jobouts/` stores under `$BUPHA_GHA_TMP`.
210#[derive(Debug)]
211pub enum DynRef {
212    /// `needs.<job>.outputs.<name>` — the producer job's published
213    /// output (last matrix leg wins, like job outputs generally).
214    NeedOutput { job: String, name: String },
215    /// `steps.<id>.outputs.<name>` — an earlier step's
216    /// `$GITHUB_OUTPUT` entry in the same job.
217    StepOutput { step: String, name: String },
218    /// `needs.<job>.result` — `success`/`failure`/`cancelled`/
219    /// `skipped` aggregated over the job's expanded legs.
220    NeedResult { job: String },
221    /// `steps.<id>.conclusion` — `success`/`skipped` (`failure` is
222    /// unobservable: a failed step ends the job first).
223    StepConclusion { step: String },
224}
225
226#[derive(Debug, Clone, Copy, PartialEq, Eq)]
227pub enum BinOp {
228    Eq,
229    Ne,
230    And,
231    Or,
232}

234impl Cond {
235    /// Does the tree contain a status-check function anywhere?
236    /// GHA implicitly ANDs `success()` onto an `if:` that lacks
237    /// one — the parser applies that wrap.
238    fn has_status_fn(&self) -> bool {
239        match self {
240            Cond::Lit(_) | Cond::Dyn(_) => false,
241            Cond::Call { name, .. } => matches!(
242                name.as_str(),
243                "success" | "failure" | "cancelled" | "always"
244            ),
245            Cond::Bin { l, r, .. } => l.has_status_fn() || r.has_status_fn(),
246            Cond::Not(inner) => inner.has_status_fn(),
247        }
248    }
249}

285/// Reject keys outside the v1 allowlist — unsupported syntax is a
286/// hard error, never a silent skip. `unsupported` names get a
287/// clearer message than truly unknown keys.
288fn check_keys(
289    m: &serde_norway::Mapping,
290    allowed: &[&str],
291    unsupported: &[&str],
292    what: &str,
293) -> Result<(), String> {
294    for (k, _) in m {
295        let key = k.as_str().unwrap_or("");
296        if unsupported.contains(&key) {
297            return Err(format!(
298                "{what}: unsupported key `{key}` (outside the gha v1 subset)"
299            ));
300        }
301        if !allowed.contains(&key) {
302            return Err(format!("{what}: unknown key `{key}`"));
303        }
304    }
305    Ok(())
306}

308fn string_list(v: &Value, what: &str) -> Result<Vec<String>, String> {
309    match v {
310        Value::String(s) => Ok(vec![s.clone()]),
311        Value::Sequence(seq) => {
312            if seq.is_empty() {
313                return Err(format!("{what}: empty list"));
314            }
315            seq.iter()
316                .map(|i| match i {
317                    Value::String(s) => Ok(s.clone()),
318                    _ => Err(format!("{what}: entries must be strings")),
319                })
320                .collect()
321        }
322        _ => Err(format!("{what}: expected a string or list of strings")),
323    }
324}

326fn env_map(m: &serde_norway::Mapping, what: &str) -> Result<Vec<(String, String)>, String> {
327    let mut out = Vec::with_capacity(m.len());
328    for (k, v) in m {
329        let key = k
330            .as_str()
331            .ok_or_else(|| format!("{what}: env names must be strings"))?;
332        let valid = key
333            .chars()
334            .next()
335            .map(|c| c.is_ascii_alphabetic() || c == '_')
336            .unwrap_or(false)
337            && key.chars().all(|c| c.is_ascii_alphanumeric() || c == '_');
338        if !valid {
339            return Err(format!("{what}: invalid env name `{key}`"));
340        }
341        // `KEY:` with no value is YAML null — GitHub coerces it to
342        // the empty string, so do the same.
343        let value = match v {
344            Value::Null => String::new(),
345            _ => scalar(v, &format!("{what}.{key}"))?,
346        };
347        out.push((key.to_string(), value));
348    }
349    Ok(out)
350}

356fn parse_push_filters(v: &Value, what: &str) -> Result<PushFilters, String> {
357    let m = match v {
358        Value::Null => return Ok(PushFilters::default()),
359        Value::Mapping(m) => m,
360        _ => return Err(format!("{what}: `on.push` must be a mapping or null")),
361    };
362    check_keys(
363        m,
364        &["branches", "branches-ignore", "paths", "paths-ignore"],
365        &["tags", "types", "workflows"],
366        what,
367    )?;
368    let mut f = PushFilters::default();
369    for (k, v) in m {
370        let key = k.as_str().unwrap_or("");
371        let list = string_list(v, &format!("{what}.{key}"))?;
372        match key {
373            "branches" => f.branches = Some(list),
374            "branches-ignore" => f.branches_ignore = Some(list),
375            "paths" => f.paths = Some(list),
376            "paths-ignore" => f.paths_ignore = Some(list),
377            _ => unreachable!(),
378        }
379    }
380    Ok(f)
381}

383/// `on:` may be a string, a list, or a map of trigger → config.
384/// Triggers other than `push` never fire here — ignored, not errors.
385fn parse_on(v: Option<&Value>, what: &str) -> Result<PushFilters, String> {
386    let Some(v) = v else {
387        return Err(format!("{what}: missing `on`"));
388    };
389    match v {
390        Value::String(s) => Ok(if s == "push" {
391            PushFilters::default()
392        } else {
393            PushFilters {
394                never: true,
395                ..PushFilters::default()
396            }
397        }),
398        Value::Sequence(seq) => {
399            let mut saw_push = false;
400            for item in seq {
401                match item.as_str() {
402                    Some("push") => saw_push = true,
403                    Some(_) => {}
404                    None => return Err(format!("{what}: `on` entries must be strings")),
405                }
406            }
407            Ok(PushFilters {
408                never: !saw_push,
409                ..PushFilters::default()
410            })
411        }
412        Value::Mapping(m) => {
413            let mut filters = PushFilters {
414                never: true,
415                ..PushFilters::default()
416            };
417            for (k, v) in m {
418                if k.as_str() == Some("push") {
419                    filters = parse_push_filters(v, &format!("{what}.on.push"))?;
420                }
421            }
422            Ok(filters)
423        }
424        _ => Err(format!("{what}: unsupported `on` shape ({})", kind(v))),
425    }
426}

428// ------------------------------------------------------------------
429// glob matching — `*`/`?` stay inside a path segment, `**` crosses `/`
430// ------------------------------------------------------------------
431
432fn glob_match(pat: &[u8], text: &[u8]) -> bool {
433    if pat.is_empty() {
434        return text.is_empty();
435    }
436    if pat.starts_with(b"**") {
437        return (0..=text.len()).any(|skip| glob_match(&pat[2..], &text[skip..]));
438    }
439    if text.is_empty() {
440        return false;
441    }
442    match pat[0] {
443        b'*' => (0..=text.len())
444            .take_while(|&skip| !text[..skip].contains(&b'/'))
445            .any(|skip| glob_match(&pat[1..], &text[skip..])),
446        b'?' => text[0] != b'/' && glob_match(&pat[1..], &text[1..]),
447        c => text[0] == c && glob_match(&pat[1..], &text[1..]),
448    }
449}
450
451fn any_glob(patterns: &[String], text: &str) -> bool {
452    patterns
453        .iter()
454        .any(|p| glob_match(p.as_bytes(), text.as_bytes()))
455}

464impl Workflow {
465    /// Does this workflow's `on.push` select the event?
466    pub fn matches_push(&self, ev: &Event) -> Result<bool, String> {
467        if self.filters.never {
468            return Ok(false);
469        }
470        let f = &self.filters;
471        let short = short_ref(ev.git_ref);
472        if let Some(pats) = &f.branches {
473            if ev.git_ref.is_empty() {
474                return Err("on.push.branches: cannot evaluate — event ref is unknown".into());
475            }
476            if !any_glob(pats, short) {
477                return Ok(false);
478            }
479        }
480        if let Some(pats) = &f.branches_ignore {
481            if ev.git_ref.is_empty() {
482                return Err(
483                    "on.push.branches-ignore: cannot evaluate — event ref is unknown".into(),
484                );
485            }
486            if any_glob(pats, short) {
487                return Ok(false);
488            }
489        }
490        if let Some(pats) = &f.paths {
491            if !ev.changed_files.iter().any(|f| any_glob(pats, f)) {
492                return Ok(false);
493            }
494        }
495        if let Some(pats) = &f.paths_ignore {
496            // Runs unless EVERY changed file matches an ignore
497            // pattern; an empty diff has nothing to ignore.
498            let all_ignored =
499                !ev.changed_files.is_empty() && ev.changed_files.iter().all(|f| any_glob(pats, f));
500            if all_ignored {
501                return Ok(false);
502            }
503        }
504        Ok(true)
505    }

507    /// Fan out matrix combos, interpolate `${{ }}`, resolve `needs`
508    /// and topo-order. `ctx` supplies the github.* facts.
509    pub fn expand(&self, ctx: GhaContext) -> Result<Vec<Job>, String> {
510        // Matrix fan-out, declaration order preserved.
511        let mut flat: Vec<(usize, Vec<(String, String)>)> = Vec::new();
512        for (ji, j) in self.jobs.iter().enumerate() {
513            for combo in matrix_combos(&j.matrix, &format!("jobs.{}", j.id))? {
514                flat.push((ji, combo));
515            }
516        }
517        // needs id → every expanded index of that job.
518        let mut by_id: HashMap<&str, Vec<usize>> = HashMap::new();
519        for (idx, (ji, _)) in flat.iter().enumerate() {
520            by_id
521                .entry(self.jobs[*ji].id.as_str())
522                .or_default()
523                .push(idx);
524        }
525        let mut jobs = Vec::with_capacity(flat.len());
526        for (ji, combo) in &flat {
527            jobs.push(self.build_job(&self.jobs[*ji], combo, &by_id, ctx)?);
528        }
529        topo_order(jobs)
530    }

532    fn build_job(
533        &self,
534        j: &RawJob,
535        combo: &[(String, String)],
536        by_id: &HashMap<&str, Vec<usize>>,
537        ctx: GhaContext,
538    ) -> Result<Job, String> {
539        let what = format!("jobs.{}", j.id);
540        let mut needs: Vec<usize> = j
541            .needs
542            .iter()
543            .flat_map(|n| by_id.get(n.as_str()).cloned().unwrap_or_default())
544            .collect();
545        needs.sort_unstable();
546        needs.dedup();
547
548        // env.* is unavailable while defining env — github.*/matrix.*
549        // only. Workflow env merges under job env.
550        let no_env = Eval {
551            ctx,
552            matrix: combo,
553            env: None,
554        };
555        let wf_env = interpolate_pairs(&self.env, &no_env, &format!("{what}.env"))?;
556        let job_env = merge_env(
557            &wf_env,
558            &interpolate_pairs(&j.env, &no_env, &format!("{what}.env"))?,
559        );
560        let eval = Eval {
561            ctx,
562            matrix: combo,
563            env: Some(&job_env),
564        };
565
566        // `if:` evaluates in job context — env.* sees the merged
567        // workflow+job env (step env doesn't exist yet), github.*/
568        // matrix.* fold, needs.*/steps.* defer to the renderer.
569        let cond = j
570            .cond
571            .as_deref()
572            .map(|raw| parse_cond(raw, &eval, &format!("{what}.if")))
573            .transpose()?;
574        let mut steps = Vec::with_capacity(j.steps.len());
575        for (si, s) in j.steps.iter().enumerate() {
576            steps.push(build_step(
577                s,
578                &eval,
579                &job_env,
580                ctx,
581                &format!("{what}.steps[{si}]"),
582            )?);
583        }
584        let defaults = Defaults {
585            working_directory: pick(
586                &j.defaults.working_directory,
587                &self.defaults.working_directory,
588            )
589            .map(|d| interpolate(d, &eval, &format!("{what}.defaults.run.working-directory")))
590            .transpose()?,
591            shell: pick(&j.defaults.shell, &self.defaults.shell)
592                .map(|s| interpolate(s, &eval, &format!("{what}.defaults.run.shell")))
593                .transpose()?,
594        };
595        // `outputs:` values evaluate in job context — env.* resolves
596        // now; needs.*/steps.* stay deferred for the renderer.
597        let outputs = j
598            .outputs
599            .iter()
600            .map(|(k, v)| {
601                Ok((
602                    k.clone(),
603                    interpolate(v, &eval, &format!("{what}.outputs.{k}"))?,
604                ))
605            })
606            .collect::<Result<Vec<_>, String>>()?;
607        Ok(Job {
608            workflow: self.name.clone(),
609            id: j.id.clone(),
610            label: match &j.name {
611                Some(n) => interpolate(n, &eval, &format!("{what}.name"))?,
612                None => j.id.clone(),
613            },
614            runs_on: j
615                .runs_on
616                .iter()
617                .map(|r| interpolate(r, &eval, &format!("{what}.runs-on")))
618                .collect::<Result<_, _>>()?,
619            needs,
620            env: job_env,
621            working_directory: defaults.working_directory,
622            shell: defaults.shell,
623            outputs,
624            cond,
625            timeout_seconds: j.timeout_seconds,
626            fail_fast: j.fail_fast,
627            max_parallel: j.max_parallel,
628            steps,
629        })
630    }
631}

637/// Last-wins merge of two env lists.
638fn merge_env(base: &[(String, String)], over: &[(String, String)]) -> Vec<(String, String)> {
639    let mut out = base.to_vec();
640    for (k, v) in over {
641        match out.iter_mut().find(|(ek, _)| ek == k) {
642            Some(slot) => slot.1 = v.clone(),
643            None => out.push((k.clone(), v.clone())),
644        }
645    }
646    out
647}

649/// Interpolate every value in an env list (env.* unavailable).
650fn interpolate_pairs(
651    raw: &[(String, String)],
652    eval: &Eval,
653    what: &str,
654) -> Result<Vec<(String, String)>, String> {
655    raw.iter()
656        .map(|(k, v)| Ok((k.clone(), interpolate(v, eval, &format!("{what}.{k}"))?)))
657        .collect()
658}

660/// Interpolation context for one expansion — github facts plus the
661/// matrix bindings and merged env visible at this point.
662pub struct Eval<'a> {
663    pub ctx: GhaContext<'a>,
664    pub matrix: &'a [(String, String)],
665    pub env: Option<&'a [(String, String)]>,
666}

668fn build_step(
669    s: &RawStep,
670    eval: &Eval,
671    job_env: &[(String, String)],
672    ctx: GhaContext,
673    what: &str,
674) -> Result<Step, String> {
675    match s {
676        RawStep::Checkout { id, cond } => Ok(Step::Checkout {
677            id: id.clone(),
678            cond: cond
679                .as_deref()
680                .map(|raw| parse_cond(raw, eval, &format!("{what}.if")))
681                .transpose()?,
682        }),
683        RawStep::RustToolchain {
684            id,
685            cond,
686            timeout_seconds,
687            with,
688        } => {
689            // Lower to a synthesized run step: interpolate the
690            // `with:` inputs, then emit the rustup script. The
691            // script itself contains no `${{ }}` — interpolation
692            // already happened — so it goes through verbatim.
693            let mut pairs = Vec::with_capacity(with.len());
694            for (k, vals) in with {
695                let mut out = Vec::with_capacity(vals.len());
696                for v in vals {
697                    out.push(interpolate(v, eval, &format!("{what}.with.{k}"))?);
698                }
699                pairs.push((k.clone(), out));
700            }
701            let script = rust_toolchain_script(&pairs, what)?;
702            Ok(Step::Run(RunStep {
703                id: id.clone(),
704                cond: cond
705                    .as_deref()
706                    .map(|raw| parse_cond(raw, eval, &format!("{what}.if")))
707                    .transpose()?,
708                timeout_seconds: *timeout_seconds,
709                shell: None,
710                working_directory: None,
711                env: Vec::new(),
712                script,
713            }))
714        }
715        RawStep::Run {
716            id,
717            cond,
718            timeout_seconds,
719            shell,
720            working_directory,
721            env,
722            script,
723        } => {
724            // Step env merges over job env; its own pairs are what
725            // the step block exports.
726            let step_own = interpolate_pairs(env, eval, &format!("{what}.env"))?;
727            let step_env = merge_env(job_env, &step_own);
728            let step_eval = Eval {
729                ctx,
730                matrix: eval.matrix,
731                env: Some(&step_env),
732            };
733            Ok(Step::Run(RunStep {
734                id: id.clone(),
735                cond: cond
736                    .as_deref()
737                    .map(|raw| parse_cond(raw, &step_eval, &format!("{what}.if")))
738                    .transpose()?,
739                timeout_seconds: *timeout_seconds,
740                shell: shell
741                    .as_ref()
742                    .map(|s| interpolate(s, &step_eval, &format!("{what}.shell")))
743                    .transpose()?,
744                working_directory: working_directory
745                    .as_ref()
746                    .map(|w| interpolate(w, &step_eval, &format!("{what}.working-directory")))
747                    .transpose()?,
748                env: step_own,
749                script: interpolate(script, &step_eval, &format!("{what}.run"))?,
750            }))
751        }
752    }
753}

755/// `dtolnay/rust-toolchain` lowered to bash: install the toolchain
756/// via rustup (minimal profile + requested components/targets),
757/// then make it win for later steps — the toolchain's bin dir goes
758/// on `$GITHUB_PATH` (last line front-most after absorb) and
759/// `RUSTUP_TOOLCHAIN` on `$GITHUB_ENV` so rustup proxies agree.
760/// With no `toolchain:` input the channel comes from
761/// `rust-toolchain.toml`/`rust-toolchain` in the worktree, else
762/// `stable` — mirroring the action.
763fn rust_toolchain_script(pairs: &[(String, Vec<String>)], what: &str) -> Result<String, String> {
764    let mut toolchain = None;
765    let mut components = Vec::new();
766    let mut targets = Vec::new();
767    for (k, vals) in pairs {
768        match k.as_str() {
769            "toolchain" => {
770                if vals.len() != 1 {
771                    return Err(format!("{what}.with.toolchain: expected a single value"));
772                }
773                toolchain = Some(vals[0].clone());
774            }
775            "components" | "targets" => {
776                let flat: Vec<String> = vals
777                    .iter()
778                    .flat_map(|v| {
779                        v.split(',')
780                            .map(str::trim)
781                            .filter(|s| !s.is_empty())
782                            .map(str::to_string)
783                            .collect::<Vec<_>>()
784                    })
785                    .collect();
786                if k == "components" {
787                    components = flat;
788                } else {
789                    targets = flat;
790                }
791            }
792            other => {
793                return Err(format!(
794                    "{what}.with.{other}: unsupported input for \
795                     dtolnay/rust-toolchain — supported: toolchain, \
796                     components, targets"
797                ))
798            }
799        }
800    }
801    let tc = toolchain.unwrap_or_default();
802    let mut script = String::new();
803    script.push_str("set -e\n");
804    script.push_str(&format!("tc={}\n", bash_sq(&tc)));
805    script.push_str(
806        "if [ -z \"$tc\" ]; then\n\
807         \x20 if [ -f rust-toolchain.toml ]; then\n\
808         \x20   tc=$(sed -n 's/.*channel[[:space:]]*=[[:space:]]*\"\\([^\"]*\\)\".*/\\1/p' rust-toolchain.toml | head -1)\n\
809         \x20 elif [ -f rust-toolchain ]; then\n\
810         \x20   tc=$(head -1 rust-toolchain | tr -d '[:space:]')\n\
811         \x20 fi\n\
812         \x20 tc=${tc:-stable}\n\
813         fi\n\
814         args=(toolchain install \"$tc\" --profile minimal)\n",
815    );
816    if !components.is_empty() {
817        script.push_str(&format!(
818            "args+=(--component {})\n",
819            bash_sq(&components.join(","))
820        ));
821    }
822    if !targets.is_empty() {
823        script.push_str(&format!(
824            "args+=(--target {})\n",
825            bash_sq(&targets.join(","))
826        ));
827    }
828    script.push_str(
829        "rustup \"${args[@]}\"\n\
830         tcbin=$(dirname \"$(rustup which --toolchain \"$tc\" cargo 2>/dev/null)\" 2>/dev/null || :)\n\
831         if [ -n \"$tcbin\" ] && [ -d \"$tcbin\" ]; then echo \"$tcbin\" >> \"$GITHUB_PATH\"; fi\n\
832         printf 'RUSTUP_TOOLCHAIN=%s\\n' \"$tc\" >> \"$GITHUB_ENV\"\n",
833    );
834    Ok(script)
835}

843/// Kahn topo sort, declaration order breaking ties. Indices in
844/// `needs` refer to the input vec; the output re-maps them.
845fn topo_order(jobs: Vec<Job>) -> Result<Vec<Job>, String> {
846    let n = jobs.len();
847    let mut slots: Vec<Option<Job>> = jobs.into_iter().map(Some).collect();
848    let mut out_pos = vec![usize::MAX; n];
849    let mut emitted = vec![false; n];
850    let mut order: Vec<usize> = Vec::with_capacity(n);
851    for _ in 0..n {
852        let mut progressed = false;
853        for i in 0..n {
854            if emitted[i] {
855                continue;
856            }
857            let ready = slots[i]
858                .as_ref()
859                .map(|j| j.needs.iter().all(|&d| emitted[d]))
860                .unwrap_or(false);
861            if !ready {
862                continue;
863            }
864            emitted[i] = true;
865            out_pos[i] = order.len();
866            order.push(i);
867            progressed = true;
868        }
869        if order.len() == n {
870            break;
871        }
872        if !progressed {
873            return Err("jobs: dependency cycle in `needs`".into());
874        }
875    }
876    let mut out = Vec::with_capacity(n);
877    for i in order {
878        if let Some(mut j) = slots[i].take() {
879            j.needs = j.needs.iter().map(|&d| out_pos[d]).collect();
880            j.needs.sort_unstable();
881            j.needs.dedup();
882            out.push(j);
883        }
884    }
885    Ok(out)
886}

892/// Replace every `${{ expr }}` in `tpl`. Anything that isn't a
893/// plain dotted path is an explicit error.
894pub fn interpolate(tpl: &str, eval: &Eval, what: &str) -> Result<String, String> {
895    let mut out = String::with_capacity(tpl.len());
896    let mut rest = tpl;
897    while let Some(start) = rest.find("${{") {
898        out.push_str(&rest[..start]);
899        let after = &rest[start + 3..];
900        let Some(end) = after.find("}}") else {
901            return Err(format!("{what}: unclosed `${{{{`"));
902        };
903        out.push_str(&resolve(after[..end].trim(), eval, what)?);
904        rest = &after[end + 2..];
905    }
906    out.push_str(rest);
907    Ok(out)
908}

910/// Is `inner` a runtime-deferred ref — `needs.<job>.outputs.<name>`,
911/// `steps.<id>.outputs.<name>`, `needs.<job>.result` or
912/// `steps.<id>.conclusion`? These resolve while the generated
913/// program runs (status arrays + output stores), so the `${{ … }}`
914/// survives expand verbatim for the renderer. The output split is
915/// at the FIRST `.outputs.`: the middle is the job/step id,
916/// everything after is the output name (dots allowed).
917fn is_deferred_ref(inner: &str) -> bool {
918    if is_output_ref(inner) {
919        return true;
920    }
921    for (prefix, suffix) in [("needs.", ".result"), ("steps.", ".conclusion")] {
922        if let Some(rest) = inner.strip_prefix(prefix) {
923            if let Some(id) = rest.strip_suffix(suffix) {
924                return !id.is_empty();
925            }
926        }
927    }
928    false
929}
930
931fn is_output_ref(inner: &str) -> bool {
932    for prefix in ["needs.", "steps."] {
933        if let Some(rest) = inner.strip_prefix(prefix) {
934            if let Some((head, name)) = rest.split_once(".outputs.") {
935                return !head.is_empty() && !name.is_empty();
936            }
937        }
938    }
939    false
940}

942/// Names that become path components in the renderer's output store
943/// (job `outputs:` keys, step `id:`s): `[A-Za-z0-9_.-]`, never `.`
944/// or `..`. Expression-derived names are already charset-checked by
945/// `resolve`; these YAML-side names need the same guarantee.
946fn safe_key(name: &str) -> bool {
947    !name.is_empty()
948        && name != "."
949        && name != ".."
950        && name
951            .chars()
952            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-'))
953}
954
955fn resolve(inner: &str, eval: &Eval, what: &str) -> Result<String, String> {
956    let bad = || {
957        format!(
958            "{what}: unsupported expression `${{{{ {inner} }}}}` — supported: \
959             env.*, github.{{sha,ref,ref_name,repository,event_name}}, matrix.*, \
960             needs.<job>.outputs.<name>, steps.<id>.outputs.<name>"
961        )
962    };
963    let dotted_ok = !inner.is_empty()
964        && inner
965            .chars()
966            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-'));
967    if !dotted_ok {
968        return Err(bad());
969    }
970    let (root, rest) = inner.split_once('.').unwrap_or((inner, ""));
971    match root {
972        "env" => {
973            if rest.is_empty() {
974                return Err(bad());
975            }
976            let Some(env) = eval.env else {
977                return Err(format!("{what}: `env.*` is not available in this position"));
978            };
979            // Undefined env expands empty, like GitHub.
980            Ok(env
981                .iter()
982                .rev()
983                .find(|(k, _)| k == rest)
984                .map(|(_, v)| v.clone())
985                .unwrap_or_default())
986        }
987        "github" => match rest {
988            "sha" => Ok(eval.ctx.sha.to_string()),
989            "ref" => Ok(eval.ctx.git_ref.to_string()),
990            "ref_name" => Ok(short_ref(eval.ctx.git_ref).to_string()),
991            "repository" => Ok(eval.ctx.repository.to_string()),
992            "event_name" => Ok(eval.ctx.event_name.to_string()),
993            _ => Err(format!(
994                "{what}: unsupported github context `{inner}` — supported: \
995                 sha, ref, ref_name, repository, event_name"
996            )),
997        },
998        "matrix" => {
999            if rest.is_empty() {
1000                return Err(bad());
1001            }
1002            eval.matrix
1003                .iter()
1004                .find(|(k, _)| k == rest)
1005                .map(|(_, v)| v.clone())
1006                .ok_or_else(|| format!("{what}: matrix has no `{rest}` binding"))
1007        }
1008        "needs" | "steps" => {
1009            // Runtime refs can't resolve at expand time — the values
1010            // are produced while the generated program runs. Keep the
1011            // `${{ … }}` source verbatim; the renderer turns it into
1012            // a lookup against the status arrays / output stores.
1013            // Everything else under these contexts stays unsupported.
1014            if is_deferred_ref(inner) {
1015                Ok(format!("${{{{ {inner} }}}}"))
1016            } else {
1017                Err(format!(
1018                    "{what}: unsupported expression `${{{{ {inner} }}}}` — \
1019                     under `needs`/`steps` only `.outputs.<name>`, \
1020                     `needs.*.result` and `steps.*.conclusion` refs are \
1021                     supported (steps.*.outcome and friends are not)"
1022                ))
1023            }
1024        }
1025        _ => Err(bad()),
1026    }
1027}

1033/// Parse + validate one workflow file. `source` names the file in
1034/// errors.
1035pub fn parse_workflow(source: &str, text: &str) -> Result<Workflow, String> {
1036    let v: Value =
1037        serde_norway::from_str(text).map_err(|e| format!("{source}: YAML parse error: {e}"))?;
1038    let top = as_map(&v, source)?;
1039    check_keys(
1040        top,
1041        &[
1042            "name",
1043            "run-name",
1044            "on",
1045            "env",
1046            "defaults",
1047            "jobs",
1048            "permissions",
1049        ],
1050        &[],
1051        source,
1052    )?;
1053    let name = get(top, "name")
1054        .map(|n| scalar(n, &format!("{source}.name")))
1055        .transpose()?
1056        .unwrap_or_else(|| {
1057            source
1058                .rsplit('/')
1059                .next()
1060                .unwrap_or(source)
1061                .trim_end_matches(".yml")
1062                .trim_end_matches(".yaml")
1063                .to_string()
1064        });
1065    let env = match get(top, "env") {
1066        Some(Value::Mapping(m)) => env_map(m, &format!("{source}.env"))?,
1067        Some(v) => return Err(format!("{source}.env: expected a mapping, got {}", kind(v))),
1068        None => Vec::new(),
1069    };
1070    let defaults = parse_defaults(get(top, "defaults"), &format!("{source}.defaults"))?;
1071    let filters = parse_on(get(top, "on"), &format!("{source}.on"))?;
1072    let jobs_v = get(top, "jobs").ok_or_else(|| format!("{source}: missing `jobs`"))?;
1073    let jobs_m = as_map(jobs_v, &format!("{source}.jobs"))?;
1074    if jobs_m.is_empty() {
1075        return Err(format!("{source}: `jobs` is empty"));
1076    }
1077    let mut jobs = Vec::with_capacity(jobs_m.len());
1078    for (k, jv) in jobs_m {
1079        let id = k
1080            .as_str()
1081            .ok_or_else(|| format!("{source}.jobs: job ids must be strings"))?
1082            .to_string();
1083        jobs.push(parse_job(
1084            as_map(jv, &format!("{source}.jobs.{id}"))?,
1085            &id,
1086            source,
1087        )?);
1088    }
1089    let ids: HashSet<&str> = jobs.iter().map(|j| j.id.as_str()).collect();
1090    for j in &jobs {
1091        for n in &j.needs {
1092            if !ids.contains(n.as_str()) {
1093                return Err(format!(
1094                    "{source}.jobs.{}: `needs` references unknown job `{n}`",
1095                    j.id
1096                ));
1097            }
1098        }
1099    }
1100    Ok(Workflow {
1101        name,
1102        env,
1103        defaults,
1104        filters,
1105        jobs,
1106    })
1107}

1109fn parse_defaults(v: Option<&Value>, what: &str) -> Result<Defaults, String> {

1135fn parse_job(m: &serde_norway::Mapping, id: &str, source: &str) -> Result<RawJob, String> {
1136    let what = format!("{source}.jobs.{id}");
1137    check_keys(
1138        m,
1139        &[
1140            "name",
1141            "if",
1142            "needs",
1143            "runs-on",
1144            "env",
1145            "defaults",
1146            "steps",
1147            "strategy",
1148            "permissions",
1149            "outputs",
1150            "timeout-minutes",
1151        ],
1152        // The v1 boundary, loudly: containers, services,
1153        // environment, concurrency, reusable-workflow calls.
1154        &[
1155            "services",
1156            "container",
1157            "environment",
1158            "concurrency",
1159            "continue-on-error",
1160            "uses",
1161            "secrets",
1162        ],
1163        &what,
1164    )?;
1165    let name = get(m, "name")
1166        .map(|n| scalar(n, &format!("{what}.name")))
1167        .transpose()?;
1168    let cond = get(m, "if")
1169        .map(|v| scalar(v, &format!("{what}.if")))
1170        .transpose()?;
1171    let timeout_seconds = get(m, "timeout-minutes")
1172        .map(|v| timeout_minutes(v, &format!("{what}.timeout-minutes")))
1173        .transpose()?;
1174    let needs = match get(m, "needs") {
1175        Some(v) => string_list(v, &format!("{what}.needs"))?,
1176        None => Vec::new(),
1177    };
1178    let runs_on = match get(m, "runs-on") {
1179        Some(v) => string_list(v, &format!("{what}.runs-on"))?,
1180        None => Vec::new(),
1181    };
1182    let env = match get(m, "env") {
1183        Some(Value::Mapping(em)) => env_map(em, &format!("{what}.env"))?,
1184        Some(v) => return Err(format!("{what}.env: expected a mapping, got {}", kind(v))),
1185        None => Vec::new(),
1186    };
1187    let defaults = parse_defaults(get(m, "defaults"), &format!("{what}.defaults"))?;
1188    let strategy = parse_strategy(get(m, "strategy"), &what)?;
1189    let outputs = match get(m, "outputs") {
1190        Some(Value::Mapping(om)) => {
1191            let mut out = Vec::with_capacity(om.len());
1192            for (k, v) in om {
1193                let key = k
1194                    .as_str()
1195                    .ok_or_else(|| format!("{what}.outputs: names must be strings"))?;
1196                if !safe_key(key) {
1197                    return Err(format!(
1198                        "{what}.outputs: invalid output name `{key}` — \
1199                         `[A-Za-z0-9_.-]` only"
1200                    ));
1201                }
1202                out.push((
1203                    key.to_string(),
1204                    scalar(v, &format!("{what}.outputs.{key}"))?,
1205                ));
1206            }
1207            out
1208        }
1209        Some(v) => {
1210            return Err(format!(
1211                "{what}.outputs: expected a mapping, got {}",
1212                kind(v)
1213            ))
1214        }
1215        None => Vec::new(),
1216    };
1217    let steps_v = get(m, "steps").ok_or_else(|| format!("{what}: missing `steps`"))?;
1218    let Value::Sequence(seq) = steps_v else {
1219        return Err(format!("{what}.steps: expected a sequence"));
1220    };
1221    if seq.is_empty() {
1222        return Err(format!("{what}.steps: empty"));
1223    }
1224    let mut steps = Vec::with_capacity(seq.len());
1225    let mut step_ids: HashSet<String> = HashSet::new();
1226    for (si, sv) in seq.iter().enumerate() {
1227        let sw = format!("{what}.steps[{si}]");
1228        let sm = as_map(sv, &sw)?;
1229        // `id:` is the steps.<id>.outputs.* handle — it becomes a
1230        // path component in the renderer's output store, so it gets
1231        // the same charset rule as job output names. Duplicate ids
1232        // are ambiguous, like on GitHub.
1233        let id = get(sm, "id")
1234            .map(|s| scalar(s, &format!("{sw}.id")))
1235            .transpose()?;
1236        if let Some(id) = &id {
1237            if !safe_key(id) {
1238                return Err(format!(
1239                    "{sw}.id: invalid step id `{id}` — `[A-Za-z0-9_.-]` only"
1240                ));
1241            }
1242            if !step_ids.insert(id.clone()) {
1243                return Err(format!("{sw}.id: duplicate step id `{id}`"));
1244            }
1245        }
1246        check_keys(
1247            sm,
1248            &[
1249                "id",
1250                "name",
1251                "if",
1252                "run",
1253                "uses",
1254                "with",
1255                "working-directory",
1256                "env",
1257                "shell",
1258                "timeout-minutes",
1259            ],
1260            &["continue-on-error"],
1261            &sw,
1262        )?;
1263        let cond = get(sm, "if")
1264            .map(|v| scalar(v, &format!("{sw}.if")))
1265            .transpose()?;
1266        let timeout_seconds = get(sm, "timeout-minutes")
1267            .map(|v| timeout_minutes(v, &format!("{sw}.timeout-minutes")))
1268            .transpose()?;
1269        let run = get(sm, "run");
1270        let uses = get(sm, "uses");
1271        match (run, uses) {
1272            (Some(_), Some(_)) => {
1273                return Err(format!("{sw}: step cannot have both `run` and `uses`"))
1274            }
1275            (None, None) => return Err(format!("{sw}: step needs `run` or `uses`")),
1276            (Some(rv), None) => {
1277                if get(sm, "with").is_some() {
1278                    return Err(format!("{sw}: `with` requires `uses`"));
1279                }
1280                let Value::String(script) = rv else {
1281                    return Err(format!("{sw}.run: expected a string"));
1282                };
1283                let env = match get(sm, "env") {
1284                    Some(Value::Mapping(em)) => env_map(em, &format!("{sw}.env"))?,
1285                    Some(v) => {
1286                        return Err(format!("{sw}.env: expected a mapping, got {}", kind(v)))
1287                    }
1288                    None => Vec::new(),
1289                };
1290                steps.push(RawStep::Run {
1291                    id,
1292                    cond,
1293                    timeout_seconds,
1294                    shell: get(sm, "shell")
1295                        .map(|s| scalar(s, &format!("{sw}.shell")))
1296                        .transpose()?,
1297                    working_directory: get(sm, "working-directory")
1298                        .map(|s| scalar(s, &format!("{sw}.working-directory")))
1299                        .transpose()?,
1300                    env,
1301                    script: script.clone(),
1302                });
1303            }
1304            (None, Some(uv)) => {
1305                let uses = scalar(uv, &format!("{sw}.uses"))?;
1306                match parse_uses(&uses, sm, &sw)? {
1307                    UsesKind::Checkout => steps.push(RawStep::Checkout { id, cond }),
1308                    UsesKind::RustToolchain(with) => steps.push(RawStep::RustToolchain {
1309                        id,
1310                        cond,
1311                        timeout_seconds,
1312                        with,
1313                    }),
1314                }
1315            }
1316        }
1317    }
1318    Ok(RawJob {
1319        id: id.to_string(),
1320        name,
1321        cond,
1322        needs,
1323        runs_on,
1324        env,
1325        defaults,
1326        matrix: strategy.matrix,
1327        fail_fast: strategy.fail_fast,
1328        max_parallel: strategy.max_parallel,
1329        timeout_seconds,
1330        outputs,
1331        steps,
1332    })
1333}

1335/// `timeout-minutes:` → whole seconds. GitHub accepts floats
1336/// (0.5 = 30s); non-positive or non-numeric values are errors.
1337fn timeout_minutes(v: &Value, what: &str) -> Result<u64, String> {
1338    let s = scalar(v, what)?;
1339    let mins: f64 = s
1340        .trim()
1341        .parse()
1342        .map_err(|_| format!("{what}: expected a number of minutes, got `{s}`"))?;
1343    if !mins.is_finite() || mins <= 0.0 {

1349enum UsesKind {
1350    Checkout,
1351    /// `with:` pairs normalized to value lists.
1352    RustToolchain(Vec<(String, Vec<String>)>),
1353}

1355/// `uses:` — `actions/checkout` is a verification no-op (the
1356/// worktree is already at the pushed sha) and
1357/// `dtolnay/rust-toolchain` lowers to rustup calls. Everything
1358/// else, `actions/setup-*` included, is an explicit error.
1359fn parse_uses(uses: &str, sm: &serde_norway::Mapping, what: &str) -> Result<UsesKind, String> {
1360    let unsupported = || {
1361        format!(
1362            "{what}: unsupported `uses: {uses}` — v1 supports \
1363             `actions/checkout` and `dtolnay/rust-toolchain` \
1364             (`actions/setup-*` is a later milestone)"
1365        )
1366    };
1367    let body = uses.split('@').next().unwrap_or(uses);
1368    if body.starts_with("docker://") || body.starts_with("./") || body.starts_with('/') {
1369        return Err(unsupported());
1370    }
1371    let is_checkout = body.eq_ignore_ascii_case("actions/checkout");
1372    let is_rust_toolchain = body.eq_ignore_ascii_case("dtolnay/rust-toolchain");
1373    if !is_checkout && !is_rust_toolchain {
1374        return Err(unsupported());
1375    }
1376    let with = match get(sm, "with") {
1377        Some(Value::Mapping(w)) => w,
1378        Some(_) => return Err(format!("{what}.with: expected a mapping")),
1379        None => {
1380            if is_checkout {
1381                return Ok(UsesKind::Checkout);
1382            }
1383            return Ok(UsesKind::RustToolchain(Vec::new()));
1384        }
1385    };
1386    if is_checkout {
1387        if get(with, "path").is_some() {
1388            return Err(format!(
1389                "{what}: `with.path` on actions/checkout is unsupported — \
1390                 the worktree root is fixed"
1391            ));
1392        }
1393        // Other checkout inputs (ref, fetch-depth, token, …)
1394        // don't change local semantics: the sha is checked out.
1395        return Ok(UsesKind::Checkout);
1396    }
1397    // dtolnay/rust-toolchain — normalize `with:` values to lists
1398    // (scalar → one entry, sequence of scalars → many).
1399    let mut out = Vec::with_capacity(with.len());
1400    for (k, v) in with {
1401        let key = k
1402            .as_str()
1403            .ok_or_else(|| format!("{what}.with: input names must be strings"))?;
1404        let vals = match v {
1405            Value::Sequence(seq) => seq
1406                .iter()
1407                .map(|i| scalar(i, &format!("{what}.with.{key}")))
1408                .collect::<Result<Vec<_>, _>>()?,
1409            _ => vec![scalar(v, &format!("{what}.with.{key}"))?],
1410        };
1411        out.push((key.to_string(), vals));
1412    }
1413    Ok(UsesKind::RustToolchain(out))
1414}

1416struct Strategy {
1417    matrix: Vec<(String, Vec<String>)>,
1418    /// GHA default: fail-fast is on when a matrix exists.
1419    fail_fast: bool,
1420    max_parallel: Option<u32>,
1421}
1422
1423fn parse_strategy(v: Option<&Value>, what: &str) -> Result<Strategy, String> {
1424    let Some(v) = v else {
1425        return Ok(Strategy {
1426            matrix: Vec::new(),
1427            fail_fast: false,
1428            max_parallel: None,
1429        });
1430    };
1431    let m = as_map(v, &format!("{what}.strategy"))?;
1432    check_keys(
1433        m,
1434        &["matrix", "fail-fast", "max-parallel"],
1435        &[],
1436        &format!("{what}.strategy"),
1437    )?;
1438    let mut fail_fast = None;
1439    let mut max_parallel = None;
1440    if let Some(fv) = get(m, "fail-fast") {
1441        fail_fast = Some(match fv {
1442            Value::Bool(b) => *b,
1443            Value::String(s) => s == "true",
1444            _ => {
1445                return Err(format!(
1446                    "{what}.strategy.fail-fast: expected a bool, got {}",
1447                    kind(fv)
1448                ))
1449            }
1450        });
1451    }
1452    if let Some(pv) = get(m, "max-parallel") {
1453        let s = scalar(pv, &format!("{what}.strategy.max-parallel"))?;
1454        let n: u32 = s.parse().map_err(|_| {
1455            format!("{what}.strategy.max-parallel: expected a positive integer, got `{s}`")
1456        })?;
1457        if n == 0 {
1458            return Err(format!("{what}.strategy.max-parallel: must be >= 1"));
1459        }
1460        max_parallel = Some(n);
1461    }
1462    let Some(mv) = get(m, "matrix") else {
1463        return Ok(Strategy {
1464            matrix: Vec::new(),
1465            fail_fast: false,
1466            max_parallel,
1467        });
1468    };
1469    let mm = as_map(mv, &format!("{what}.strategy.matrix"))?;
1470    let mut matrix = Vec::with_capacity(mm.len());
1471    for (k, vv) in mm {
1472        let key = k
1473            .as_str()
1474            .ok_or_else(|| format!("{what}.strategy.matrix: keys must be strings"))?;
1475        let Value::Sequence(seq) = vv else {
1476            return Err(format!(
1477                "{what}.strategy.matrix.{key}: expected a list — \
1478                 `include`/`exclude` are not supported"
1479            ));
1480        };
1481        if seq.is_empty() {
1482            return Err(format!("{what}.strategy.matrix.{key}: empty list"));
1483        }
1484        let vals = seq
1485            .iter()
1486            .map(|i| scalar(i, &format!("{what}.strategy.matrix.{key}")))
1487            .collect::<Result<Vec<_>, _>>()?;
1488        matrix.push((key.to_string(), vals));
1489    }
1490    // fail-fast only has meaning with a matrix; GHA defaults it on.
1491    Ok(Strategy {
1492        fail_fast: fail_fast.unwrap_or(true),
1493        matrix,
1494        max_parallel,
1495    })
1496}

1498/// Cartesian product of matrix axes; no matrix → one empty combo.
1499fn matrix_combos(
1500    axes: &[(String, Vec<String>)],
1501    what: &str,
1502) -> Result<Vec<Vec<(String, String)>>, String> {
1503    let mut combos: Vec<Vec<(String, String)>> = vec![Vec::new()];
1504    for (name, vals) in axes {
1505        let mut next = Vec::with_capacity(combos.len() * vals.len());
1506        for c in &combos {
1507            for v in vals {
1508                let mut c = c.clone();
1509                c.push((name.clone(), v.clone()));
1510                next.push(c);
1511            }
1512        }
1513        combos = next;
1514    }
1515    if combos.is_empty() {
1516        return Err(format!("{what}: matrix produced zero combinations"));
1517    }
1518    Ok(combos)
1519}

1521// ------------------------------------------------------------------
1522// `if:` expression parsing — recursive descent over a small GHA
1523// expression subset, then context folding via `Eval`
1524// ------------------------------------------------------------------
1525
1526#[derive(Debug, Clone, PartialEq)]
1527enum Tok {
1528    LParen,
1529    RParen,
1530    Not,
1531    And,
1532    Or,
1533    Eq,
1534    Ne,
1535    Comma,
1536    /// 'single' or "double" quoted literal contents.
1537    Str(String),
1538    /// Bare word — dotted path, call name, number, true/false/null.
1539    Word(String),
1540}
1541
1542fn lex_cond(src: &str, what: &str) -> Result<Vec<Tok>, String> {
1543    let b = src.as_bytes();
1544    let mut i = 0;
1545    let mut out = Vec::new();
1546    while i < b.len() {
1547        let c = b[i] as char;
1548        if c.is_whitespace() {
1549            i += 1;
1550            continue;
1551        }
1552        match c {
1553            '(' => {
1554                out.push(Tok::LParen);
1555                i += 1;
1556            }
1557            ')' => {
1558                out.push(Tok::RParen);
1559                i += 1;
1560            }
1561            '!' => {
1562                if b.get(i + 1) == Some(&b'=') {
1563                    out.push(Tok::Ne);
1564                    i += 2;
1565                } else {
1566                    out.push(Tok::Not);
1567                    i += 1;
1568                }
1569            }
1570            '&' => {
1571                if b.get(i + 1) == Some(&b'&') {
1572                    out.push(Tok::And);
1573                    i += 2;
1574                } else {
1575                    return Err(format!("{what}: bare `&` in `if` expression"));
1576                }
1577            }
1578            '|' => {
1579                if b.get(i + 1) == Some(&b'|') {
1580                    out.push(Tok::Or);
1581                    i += 2;
1582                } else {
1583                    return Err(format!("{what}: bare `|` in `if` expression"));
1584                }
1585            }
1586            '=' => {
1587                if b.get(i + 1) == Some(&b'=') {
1588                    out.push(Tok::Eq);
1589                    i += 2;
1590                } else {
1591                    return Err(format!("{what}: bare `=` in `if` expression"));
1592                }
1593            }
1594            ',' => {
1595                out.push(Tok::Comma);
1596                i += 1;
1597            }
1598            '\'' | '"' => {
1599                let quote = b[i];
1600                i += 1;
1601                let start = i;
1602                while i < b.len() && b[i] != quote {
1603                    i += 1;
1604                }
1605                if i >= b.len() {
1606                    return Err(format!("{what}: unterminated string in `if`"));
1607                }
1608                out.push(Tok::Str(src[start..i].to_string()));
1609                i += 1;
1610            }
1611            _ => {
1612                let start = i;
1613                while i < b.len() {
1614                    let c = b[i] as char;
1615                    if c.is_whitespace()
1616                        || matches!(c, '(' | ')' | '!' | '&' | '|' | '=' | ',' | '\'' | '"')
1617                    {
1618                        break;
1619                    }
1620                    i += 1;
1621                }
1622                if start == i {
1623                    return Err(format!("{what}: unexpected character `{c}` in `if`"));
1624                }
1625                out.push(Tok::Word(src[start..i].to_string()));
1626            }
1627        }
1628    }
1629    Ok(out)
1630}

1632/// Pre-fold AST — paths and calls still raw; `fold` resolves
1633/// env/github/matrix and builds [`Cond`].
1634enum PExpr {
1635    Str(String),
1636    Path(String),
1637    Call {
1638        name: String,
1639        args: Vec<PExpr>,
1640    },
1641    Bin {
1642        op: BinOp,
1643        l: Box<PExpr>,
1644        r: Box<PExpr>,
1645    },
1646    Not(Box<PExpr>),
1647}
1648
1649struct CondParser<'a> {
1650    toks: &'a [Tok],
1651    pos: usize,
1652    what: &'a str,
1653}
1654
1655impl CondParser<'_> {
1656    fn peek(&self) -> Option<&Tok> {
1657        self.toks.get(self.pos)
1658    }
1659
1660    fn next(&mut self) -> Option<Tok> {
1661        let t = self.toks.get(self.pos).cloned();
1662        if t.is_some() {
1663            self.pos += 1;
1664        }
1665        t
1666    }
1667
1668    fn err(&self, msg: &str) -> String {
1669        format!("{}: {msg} in `if` expression", self.what)
1670    }
1671
1672    /// or := and ('||' and)*
1673    fn parse_or(&mut self) -> Result<PExpr, String> {
1674        let mut l = self.parse_and()?;
1675        while matches!(self.peek(), Some(Tok::Or)) {
1676            self.next();
1677            let r = self.parse_and()?;
1678            l = PExpr::Bin {
1679                op: BinOp::Or,
1680                l: Box::new(l),
1681                r: Box::new(r),
1682            };
1683        }
1684        Ok(l)
1685    }
1686
1687    /// and := eq ('&&' eq)*
1688    fn parse_and(&mut self) -> Result<PExpr, String> {
1689        let mut l = self.parse_eq()?;
1690        while matches!(self.peek(), Some(Tok::And)) {
1691            self.next();
1692            let r = self.parse_eq()?;
1693            l = PExpr::Bin {
1694                op: BinOp::And,
1695                l: Box::new(l),
1696                r: Box::new(r),
1697            };
1698        }
1699        Ok(l)
1700    }
1701
1702    /// eq := unary (('=='|'!=') unary)*
1703    fn parse_eq(&mut self) -> Result<PExpr, String> {
1704        let mut l = self.parse_unary()?;
1705        loop {
1706            let op = match self.peek() {
1707                Some(Tok::Eq) => BinOp::Eq,
1708                Some(Tok::Ne) => BinOp::Ne,
1709                _ => break,
1710            };
1711            self.next();
1712            let r = self.parse_unary()?;
1713            l = PExpr::Bin {
1714                op,
1715                l: Box::new(l),
1716                r: Box::new(r),
1717            };
1718        }
1719        Ok(l)
1720    }
1721
1722    /// unary := '!' unary | primary
1723    fn parse_unary(&mut self) -> Result<PExpr, String> {
1724        if matches!(self.peek(), Some(Tok::Not)) {
1725            self.next();
1726            return Ok(PExpr::Not(Box::new(self.parse_unary()?)));
1727        }
1728        self.parse_primary()
1729    }
1730
1731    fn parse_primary(&mut self) -> Result<PExpr, String> {
1732        match self.next() {
1733            Some(Tok::LParen) => {
1734                let e = self.parse_or()?;
1735                match self.next() {
1736                    Some(Tok::RParen) => Ok(e),
1737                    _ => Err(self.err("missing `)`")),
1738                }
1739            }
1740            Some(Tok::Str(s)) => Ok(PExpr::Str(s)),
1741            Some(Tok::Word(w)) => {
1742                if matches!(self.peek(), Some(Tok::LParen)) {
1743                    self.parse_call(w)
1744                } else {
1745                    Ok(PExpr::Path(w))
1746                }
1747            }
1748            Some(t) => Err(self.err(&format!("unexpected token {t:?}"))),
1749            None => Err(self.err("unexpected end of expression")),
1750        }
1751    }
1752
1753    /// Word already consumed, `(` is next — a function call.
1754    fn parse_call(&mut self, word: String) -> Result<PExpr, String> {
1755        let name = word.to_lowercase();
1756        self.next(); // consume '('
1757        let mut args = Vec::new();
1758        if !matches!(self.peek(), Some(Tok::RParen)) {
1759            loop {
1760                args.push(self.parse_or()?);
1761                match self.next() {
1762                    Some(Tok::Comma) => {}
1763                    Some(Tok::RParen) => break,
1764                    _ => return Err(self.err("expected `,` or `)` in call")),
1765                }
1766            }
1767        } else {
1768            self.next();
1769        }
1770        Ok(PExpr::Call { name, args })
1771    }
1772}

1774/// Parse + fold one `if:` value. The raw text may be a bare
1775/// expression or a single `${{ … }}` wrapper (mixed text errors).
1776/// env/github/matrix fold via `eval`; needs/steps become `DynRef`s;
1777/// an `if` without a status-check function gets `success()` ANDed
1778/// on, like GitHub.
1779fn parse_cond(raw: &str, eval: &Eval, what: &str) -> Result<Cond, String> {
1780    let text = raw.trim();
1781    let inner = if text.starts_with("${{") {
1782        if !text.ends_with("}}") || text.matches("${{").count() != 1 {
1783            return Err(format!(
1784                "{what}: `if` must be a single `${{{{ … }}}}` expression or a bare expression"
1785            ));
1786        }
1787        text[3..text.len() - 2].trim()
1788    } else if text.contains("${{") {
1789        return Err(format!(
1790            "{what}: `${{{{` in `if` must wrap the whole expression"
1791        ));
1792    } else {
1793        text
1794    };
1795    if inner.is_empty() {
1796        return Err(format!("{what}: empty `if` expression"));
1797    }
1798    let toks = lex_cond(inner, what)?;
1799    let mut p = CondParser {
1800        toks: &toks,
1801        pos: 0,
1802        what,
1803    };
1804    let expr = p.parse_or()?;
1805    if p.pos != toks.len() {
1806        return Err(format!("{what}: trailing tokens in `if` expression"));
1807    }
1808    let mut cond = fold_cond(&expr, eval, what)?;
1809    if !cond.has_status_fn() {
1810        cond = Cond::Bin {
1811            op: BinOp::And,
1812            l: Box::new(Cond::Call {
1813                name: "success".into(),
1814                args: Vec::new(),
1815            }),
1816            r: Box::new(cond),
1817        };
1818    }
1819    Ok(cond)
1820}
1821
1822fn fold_cond(e: &PExpr, eval: &Eval, what: &str) -> Result<Cond, String> {
1823    Ok(match e {
1824        PExpr::Str(s) => Cond::Lit(s.clone()),
1825        PExpr::Path(w) => fold_path(w, eval, what)?,
1826        PExpr::Not(inner) => Cond::Not(Box::new(fold_cond(inner, eval, what)?)),
1827        PExpr::Bin { op, l, r } => Cond::Bin {
1828            op: *op,
1829            l: Box::new(fold_cond(l, eval, what)?),
1830            r: Box::new(fold_cond(r, eval, what)?),
1831        },
1832        PExpr::Call { name, args } => {
1833            let arity = match name.as_str() {
1834                "success" | "failure" | "cancelled" | "always" => 0,
1835                "contains" | "startswith" | "endswith" => 2,
1836                _ => {
1837                    return Err(format!(
1838                        "{what}: unsupported function `{name}()` in `if` — \
1839                         supported: success, failure, cancelled, always, \
1840                         contains, startsWith, endsWith"
1841                    ))
1842                }
1843            };
1844            if args.len() != arity {
1845                return Err(format!(
1846                    "{what}: `{name}()` expects {arity} argument(s), got {}",
1847                    args.len()
1848                ));
1849            }
1850            Cond::Call {
1851                name: name.clone(),
1852                args: args
1853                    .iter()
1854                    .map(|a| fold_cond(a, eval, what))
1855                    .collect::<Result<_, _>>()?,
1856            }
1857        }
1858    })
1859}

1861/// Fold a bare word: literals (numbers, true/false/null), env/
1862/// github/matrix paths → `Cond::Lit`, needs/steps refs → `DynRef`.
1863fn fold_path(w: &str, eval: &Eval, what: &str) -> Result<Cond, String> {
1864    let bad = || {
1865        format!(
1866            "{what}: unsupported expression `{w}` in `if` — supported: \
1867             literals, env.*, github.{{sha,ref,ref_name,repository,event_name}}, \
1868             matrix.*, needs.<job>.{{result,outputs.<name>}}, \
1869             steps.<id>.{{conclusion,outputs.<name>}}"
1870        )
1871    };
1872    // Literals first — `true`/`false`/`null` and bare numbers.
1873    if w == "true" {
1874        return Ok(Cond::Lit("true".into()));
1875    }
1876    if w == "false" {
1877        return Ok(Cond::Lit("false".into()));
1878    }
1879    if w == "null" {
1880        return Ok(Cond::Lit(String::new()));
1881    }
1882    if !w.is_empty()
1883        && w.chars()
1884            .all(|c| c.is_ascii_digit() || c == '-' || c == '+' || c == '.')
1885        && w.chars().any(|c| c.is_ascii_digit())
1886    {
1887        return Ok(Cond::Lit(w.to_string()));
1888    }
1889    let (root, rest) = w.split_once('.').unwrap_or((w, ""));
1890    match root {
1891        "env" => {
1892            if rest.is_empty() {
1893                return Err(bad());
1894            }
1895            let Some(env) = eval.env else {
1896                return Err(format!("{what}: `env.*` is not available in this position"));
1897            };
1898            Ok(Cond::Lit(
1899                env.iter()
1900                    .rev()
1901                    .find(|(k, _)| k == rest)
1902                    .map(|(_, v)| v.clone())
1903                    .unwrap_or_default(),
1904            ))
1905        }
1906        "github" => {
1907            let v = match rest {
1908                "sha" => eval.ctx.sha,
1909                "ref" => eval.ctx.git_ref,
1910                "ref_name" => short_ref(eval.ctx.git_ref),
1911                "repository" => eval.ctx.repository,
1912                "event_name" => eval.ctx.event_name,
1913                _ => return Err(bad()),
1914            };
1915            Ok(Cond::Lit(v.to_string()))
1916        }
1917        "matrix" => {
1918            if rest.is_empty() {
1919                return Err(bad());
1920            }
1921            eval.matrix
1922                .iter()
1923                .find(|(k, _)| k == rest)
1924                .map(|(_, v)| Cond::Lit(v.clone()))
1925                .ok_or_else(|| format!("{what}: matrix has no `{rest}` binding in `if`"))
1926        }
1927        "needs" => {
1928            // `.outputs.` wins over `.result` — same precedence as
1929            // `is_deferred_ref` (an output name may itself end in
1930            // `.result`).
1931            if let Some((job, name)) = rest.split_once(".outputs.") {
1932                if !job.is_empty() && !name.is_empty() {
1933                    return Ok(Cond::Dyn(DynRef::NeedOutput {
1934                        job: job.to_string(),
1935                        name: name.to_string(),
1936                    }));
1937                }
1938                return Err(bad());
1939            }
1940            if let Some(job) = rest.strip_suffix(".result") {
1941                if job.is_empty() {
1942                    return Err(bad());
1943                }
1944                return Ok(Cond::Dyn(DynRef::NeedResult {
1945                    job: job.to_string(),
1946                }));
1947            }
1948            Err(bad())
1949        }
1950        "steps" => {
1951            if let Some((id, name)) = rest.split_once(".outputs.") {
1952                if !id.is_empty() && !name.is_empty() {
1953                    return Ok(Cond::Dyn(DynRef::StepOutput {
1954                        step: id.to_string(),
1955                        name: name.to_string(),
1956                    }));
1957                }
1958                return Err(bad());
1959            }
1960            if let Some(id) = rest.strip_suffix(".conclusion") {
1961                if id.is_empty() {
1962                    return Err(bad());
1963                }
1964                return Ok(Cond::Dyn(DynRef::StepConclusion {
1965                    step: id.to_string(),
1966                }));
1967            }
1968            Err(bad())
1969        }
1970        _ => Err(bad()),
1971    }
1972}
