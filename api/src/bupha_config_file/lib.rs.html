1//! File impl of the config port — **discovery rides scarab-config**
2//! (P-017): [`scarab_config::Discovery::for_app`] conventions —
3//! explicit path → `$BUPHA_CONFIG` → `./bupha.*` →
4//! `~/.config/bupha/bupha.*`, YAML preferred within each directory.
5//! The per-syntax adapters are gone (`bupha-config-toml`/`-yaml`
6//! deleted); parsing dispatches through [`scarab_config::ConfigFormat`]
7//! impls local to this crate, preserving the old adapter semantics:
8//! `.yaml`/`.yml` parse as YAML (serde_norway transcode), **every**
9//! other extension — including extensionless and unknown — parses as
10//! TOML (the `is_yaml` back-compat rule for explicit paths), and
11//! parsed `Config`s get `~` expanded in local-path fields
12//! (`Config::expand_tilde`).
13//!
14//! [`load_file`] stays the entry point for callers that enumerate
15//! config files themselves (e.g. peregrin-node's
16//! `~/.config/bupha/bundles/` ci-job fleet scan).
17
18use std::path::{Path, PathBuf};
19
20use bupha_core::Config;
21use scarab_config::{format_for, ConfigFormat, Discovery};
22use scarab_config_file::FileSource as ScarabFileSource;
23
24/// The search-miss error — kept verbatim from the pre-scarab
25/// dispatcher (a named path that doesn't exist reports this, not
26/// scarab's "config file not found").
27const NOT_FOUND: &str =
28    "no bupha.{yaml,yml,toml} found (search: $BUPHA_CONFIG, ., ~/.config/bupha/)";
29
30/// File config source — resolves + loads the `bupha.{yaml,yml,toml}`
31/// config. `path` is the explicit `--config`-style override; `None`
32/// runs the standard search (`$BUPHA_CONFIG` → `./bupha.*` →
33/// `~/.config/bupha/`).
34pub struct FileSource {
35    /// Explicit `--config` path, or None for the default search.
36    pub path: Option<String>,
37}
38
39/// TOML syntax — `toml::from_str` + `Config::expand_tilde`, exactly
40/// what `bupha-config-toml`'s `TomlSource` did. `render` is
41/// unsupported: `Config` is deserialize-only, and nothing writes
42/// config files.
43struct Toml;
44
45impl ConfigFormat<Config> for Toml {
46    fn extensions(&self) -> &'static [&'static str] {
47        &["toml"]
48    }
49
50    fn parse(&self, text: &str) -> Result<Config, String> {
51        let mut cfg: Config = toml::from_str(text).map_err(|e| e.to_string())?;
52        cfg.expand_tilde();
53        Ok(cfg)
54    }
55
56    fn render(&self, _config: &Config) -> Result<String, String> {
57        Err("bupha config render unsupported".into())
58    }
59}
60
61/// YAML syntax — `serde_norway` transcode (`Value` → `Config`, the
62/// same `Config` shape as TOML) + `Config::expand_tilde`, exactly
63/// what `bupha-config-yaml`'s `YamlSource` did. Claims `yaml` before
64/// `yml` so `bupha.yaml` wins over `bupha.yml` on ties.
65struct Yaml;
66
67impl ConfigFormat<Config> for Yaml {
68    fn extensions(&self) -> &'static [&'static str] {
69        &["yaml", "yml"]
70    }
71
72    fn parse(&self, text: &str) -> Result<Config, String> {
73        let v: serde_norway::Value = serde_norway::from_str(text).map_err(|e| e.to_string())?;
74        let mut cfg: Config = serde_norway::from_value(v).map_err(|e| e.to_string())?;
75        cfg.expand_tilde();
76        Ok(cfg)
77    }
78
79    fn render(&self, _config: &Config) -> Result<String, String> {
80        Err("bupha config render unsupported".into())
81    }
82}
83
84/// The format set in preference order — YAML first (preferred),
85/// TOML last (permanent fallback), matching the old `NAMES` table.
86fn formats() -> Vec<Box<dyn ConfigFormat<Config>>> {
87    vec![Box::new(Yaml), Box::new(Toml)]
88}
89
90/// The explicit/named path: `--config`-style `path` first, then
91/// `$BUPHA_CONFIG`. No empty filtering — an empty value names a
92/// path that can't exist, which the miss error reports (the old
93/// dispatcher's exact behavior).
94fn named_path(source: &FileSource) -> Option<String> {
95    source
96        .path
97        .clone()
98        .or_else(|| std::env::var("BUPHA_CONFIG").ok())
99}
100
101/// Load one config file at an already-resolved path — extension
102/// dispatch through [`format_for`], with every unclaimed extension
103/// (extensionless, `.json`, …) falling back to TOML, the `is_yaml`
104/// rule verbatim. For callers that enumerate config files
105/// themselves (e.g. the `~/.config/bupha/bundles/` fleet dir a plane
106/// scans).
107///
108/// # Errors
109/// Read/parse/validation failures carry a readable message.
110pub fn load_file(path: &Path) -> Result<Config, String> {
111    let text =
112        std::fs::read_to_string(path).map_err(|e| format!("read {}: {e}", path.display()))?;
113    let owned = formats();
114    let refs: Vec<&dyn ConfigFormat<Config>> = owned.iter().map(|f| &**f).collect();
115    // `.yaml`/`.yml` are claimed by `Yaml`; everything else — the old
116    // `is_yaml` else-arm — parses as TOML.
117    let format = format_for::<Config>(path, &refs).unwrap_or(&Toml);
118    format
119        .parse(&text)
120        .map_err(|e| format!("parse {}: {e}", path.display()))
121}
122
123impl FileSource {
124    /// The scarab discovery source for the default search —
125    /// [`Discovery::for_app`] gives `$BUPHA_CONFIG` + `[.,
126    /// ~/.config/bupha]`; `explicit` carries `path`.
127    fn discovery_source(&self) -> ScarabFileSource<Config> {
128        let mut src = ScarabFileSource::with_formats(Discovery::for_app("bupha"), formats());
129        src.explicit = self.path.clone();
130        src
131    }
132
133    /// Load the resolved config.
134    ///
135    /// # Errors
136    /// A named path (`--config`/`$BUPHA_CONFIG`) that doesn't exist
137    /// reports the search-miss error; discovery misses and
138    /// read/parse failures carry readable messages.
139    pub fn load(&self) -> Result<Config, String> {
140        // Named path wins over discovery — handled here (not via
141        // `locate`) so a missing named file keeps the historical
142        // search-miss error and `exists()`-semantics (a directory
143        // named like a config still reaches the read arm).
144        if let Some(p) = named_path(self) {
145            let path = PathBuf::from(p);
146            return if path.exists() {
147                load_file(&path)
148            } else {
149                Err(NOT_FOUND.into())
150            };
151        }
152        let path = self.discovery_source().locate()?;
153        load_file(&path)
154    }
155}
