1//! File impl of `ConfigSource` — resolves which config file to load:
2//! explicit path → $BUPHA_CONFIG → ./bupha.* → ~/.config/bupha/bupha.*;
3//! at each search step YAML wins when both formats exist. Syntax is
4//! dispatched by extension to the per-syntax adapters:
5//! `.yaml`/`.yml` → `bupha-config-yaml`, everything else →
6//! `bupha-config-toml` (back-compat for explicit/extensionless paths).
7
8use bupha_config::ConfigSource;
9use bupha_config_toml::TomlSource;
10use bupha_config_yaml::YamlSource;
11use std::path::{Path, PathBuf};
12
13pub struct FileSource {
14    /// Explicit `--config` path, or None for the default search.
15    pub path: Option<String>,
16}
17
18/// File names tried, in order, inside each searched directory —
19/// YAML first (preferred), TOML last (permanent fallback).
20const NAMES: [&str; 3] = ["bupha.yaml", "bupha.yml", "bupha.toml"];
21
22/// Which syntax a config path carries — `.yaml`/`.yml` parse as
23/// YAML, everything else as TOML (back-compat for explicit paths).
24fn is_yaml(p: &Path) -> bool {
25    matches!(
26        p.extension().and_then(|e| e.to_str()),
27        Some("yaml") | Some("yml")
28    )
29}
30
31/// Load one config file at an already-resolved path — the same
32/// extension dispatch [`FileSource`] applies per candidate. For
33/// callers that enumerate config files themselves (e.g. the
34/// `~/.config/bupha/bundles/` fleet dir a plane scans).
35///
36/// # Errors
37/// Read/parse/validation failures carry a readable message.
38pub fn load_file(path: &Path) -> Result<bupha_core::Config, String> {
39    if is_yaml(path) {
40        YamlSource {
41            path: path.to_path_buf(),
42        }
43        .load()
44    } else {
45        TomlSource {
46            path: path.to_path_buf(),
47        }
48        .load()
49    }
50}
51
52impl ConfigSource for FileSource {
53    fn load(&self) -> Result<bupha_core::Config, String> {
54        let candidates: Vec<PathBuf> = match self
55            .path
56            .clone()
57            .or_else(|| std::env::var("BUPHA_CONFIG").ok())
58        {
59            Some(p) => vec![PathBuf::from(p)],
60            None => {
61                let home = std::env::var("HOME")
62                    .map(PathBuf::from)
63                    .unwrap_or(".".into())
64                    .join(".config/bupha");
65                let mut v = Vec::with_capacity(NAMES.len() * 2);
66                for n in NAMES {
67                    v.push(PathBuf::from(n));
68                }
69                for n in NAMES {
70                    v.push(home.join(n));
71                }
72                v
73            }
74        };
75        for p in candidates {
76            if p.exists() {
77                return load_file(&p);
78            }
79        }
80        Err("no bupha.{yaml,yml,toml} found (search: $BUPHA_CONFIG, ., ~/.config/bupha/)".into())
81    }
82}
