1//! mimus — spots the cuckoo eggs in your repo. Composition root:
2//! parses mimus.toml, wires checker adapters, runs subcommands.

11const EXAMPLE_TOML: &str = include_str!("../../../examples/mimus.toml.example");

13#[derive(Parser)]
14#[command(
15    name = "mimus",
16    version,
17    about = "Spots the cuckoo eggs in your repo — claim verification: docs-as-tests, memory audit"
18)]
19struct Cli {
20    /// Config path (else $MIMUS_CONFIG, ./mimus.toml,
21    /// ~/.config/mimus/mimus.toml).
22    #[arg(long, global = true)]
23    config: Option<String>,
24
25    /// Output format: human | json.
26    #[arg(long, global = true, default_value = "human")]
27    format: String,
28
29    #[command(subcommand)]
30    cmd: Cmd,
31}

33#[derive(Subcommand)]
34enum Cmd {
35    /// Detect sources -> mimus.toml + rules/ scaffold.
36    Init,
37    /// Verify claims. Mechanical tier by default.
38    Check {
39        /// Include the semantic tier (advisory only, never gates).
40        #[arg(long)]
41        all: bool,
42        /// Never execute documented commands.
43        #[arg(long)]
44        no_cmd: bool,
45    },
46    /// Trust-score .kuda/ cards vs source.
47    Audit,
48    /// Full violations list with evidence.
49    Report,
50    /// Checker/config/kuda-index health.
51    Doctor,
52    /// Emit CLI reference markdown (docs pipeline).
53    #[command(hide = true, name = "gen-cli-docs")]
54    GenCliDocs,
55}
56
57fn main() {
58    let cli = Cli::parse();
59    let code = match &cli.cmd {
60        Cmd::Init => init(&cli),
61        Cmd::Check { all, no_cmd } => check(&cli, *all, *no_cmd),
62        Cmd::Report => report(&cli),
63        Cmd::Audit => audit(&cli),
64        Cmd::Doctor => doctor(&cli),
65        Cmd::GenCliDocs => {
66            println!("{}", clap_markdown::help_markdown::<Cli>());
67            0
68        }
69    };
70    std::process::exit(code);
71}

73fn root(cli: &Cli) -> PathBuf {
74    cli.config
75        .as_deref()
76        .map(Path::new)
77        .and_then(Path::parent)
78        .map(Path::to_path_buf)
79        .unwrap_or_else(|| ".".into())
80}
81
82fn init(cli: &Cli) -> i32 {
83    let dir = root(cli);
84    let toml_path = dir.join("mimus.toml");
85    if toml_path.exists() {
86        eprintln!("mimus.toml already exists — not overwriting");
87        return 1;
88    }
89    if let Err(e) = std::fs::write(&toml_path, EXAMPLE_TOML) {
90        eprintln!("write {}: {e}", toml_path.display());
91        return 1;
92    }
93    let rules = dir.join("rules");
94    if let Err(e) = std::fs::create_dir_all(&rules) {
95        eprintln!("mkdir rules: {e}");
96        return 1;
97    }
98    println!("wrote {} + rules/ — run `mimus check`", toml_path.display());
99    0
100}

109/// Extract claims from configured doc sources and run checkers.
110fn check(cli: &Cli, _all: bool, _no_cmd: bool) -> i32 {
111    let cfg = match load(cli) {
112        Ok(c) => c,
113        Err(e) => {
114            eprintln!("{e}");
115            return 2;
116        }
117    };
118    let root = root(cli);
119    let mut claims = Vec::new();
120    for pat in &cfg.sources.docs {
121        for p in mimus_rules::matching_files(&root, pat) {
122            if let Ok(text) = std::fs::read_to_string(&p) {
123                claims.extend(mimus_claims::extract(&text, &p));
124            }
125        }
126    }
127    let mut checkers: Vec<Box<dyn Checker>> = vec![
128        Box::new(mimus_check_fs::FsChecker::new(&root)),
129        Box::new(mimus_check_cmd::CmdChecker::new()),
130    ];
131    if let Some(idx) = &cfg.check.kuda_index {
132        checkers.push(Box::new(mimus_check_symbol::SymbolChecker::new(idx)));
133    }
134    let mut violations: Vec<Violation> = mimus_check::run(&checkers, &claims)
135        .filter_map(|r| match r {
136            Ok(v) => Some(v),
137            Err(e) => {
138                eprintln!("checker: {e}");
139                None
140            }
141        })
142        .collect();
143    // data-driven rules
144    if let Ok(rules) = mimus_rules::load_rules(&root.join(&cfg.rules.dir)) {
145        for rule in &rules {
146            for f in mimus_rules::matching_files(&root, &rule.files) {
147                if let Ok(text) = std::fs::read_to_string(&f) {
148                    violations.extend(mimus_rules::eval_rule(rule, &f, &text));
149                }
150            }
151        }
152    }
153    emit(&violations, &cli.format);
154    if violations.iter().any(|v| v.severity == Severity::Error) {
155        1
156    } else {
157        0
158    }
159}

161fn report(cli: &Cli) -> i32 {
162    check(cli, true, false)
163}

165/// Trust-score `.kuda/` cards: verify files exist + provenance
166/// hashes match current content. Advisory — reports, never gates.
167fn audit(cli: &Cli) -> i32 {
168    let root = root(cli);
169    let cards = mimus_rules::matching_files(&root, ".kuda/modules/*.yaml");
170    if cards.is_empty() {
171        println!("no .kuda cards — run `kuda sync` first");
172        return 0;
173    }
174    let (mut fresh, mut drifted, mut contradicted) = (0, 0, 0);
175    for p in cards {
176        let Ok(v) = serde_yaml::from_str::<serde_yaml::Value>(
177            &std::fs::read_to_string(&p).unwrap_or_default(),
178        ) else {
179            continue;
180        };
181        let name = v["name"].as_str().unwrap_or("?").to_owned();
182        let files: Vec<String> = v["provenance"]["files"]
183            .as_sequence()
184            .map(|s| s.iter().filter_map(|f| f.as_str().map(str::to_owned)).collect())
185            .unwrap_or_default();
186        let hashes: Vec<String> = v["provenance"]["hashes"]
187            .as_sequence()
188            .map(|s| s.iter().filter_map(|f| f.as_str().map(str::to_owned)).collect())
189            .unwrap_or_default();
190        let mut score = "fresh";
191        for (f, h) in files.iter().zip(&hashes) {
192            let fp = root.join(f);
193            if !fp.exists() {
194                score = "contradicted";
195                break;
196            }
197            let cur = std::process::Command::new("git")
198                .arg("-C")
199                .arg(&root)
200                .args(["hash-object"])
201                .arg(f)
202                .output()
203                .ok()
204                .filter(|o| o.status.success())
205                .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_owned())
206                .unwrap_or_default();
207            if cur != *h {
208                score = "drifted";
209            }
210        }
211        match score {
212            "contradicted" => contradicted += 1,
213            "drifted" => drifted += 1,
214            _ => fresh += 1,
215        }
216        println!("{score:<12} {name}");
217    }
218    println!("{fresh} fresh, {drifted} drifted, {contradicted} contradicted");
219    0
220}

222fn emit(violations: &[Violation], format: &str) {
223    if format == "json" {
224        println!("{}", serde_json::to_string_pretty(violations).unwrap_or_default());
225        return;
226    }
227    if violations.is_empty() {
228        println!("no violations — {} claims checked", "all");
229        return;
230    }
231    for v in violations {
232        println!(
233            "{}:{} [{}] `{}` — {}{}",
234            v.claim.source.display(),
235            v.claim.line,
236            match v.severity {
237                Severity::Error => "error",
238                Severity::Warn => "warn",
239            },
240            v.claim.text,
241            v.evidence,
242            v.fix.as_ref().map(|f| format!(" (fix: {f})")).unwrap_or_default(),
243        );
244    }
245    println!("{} violation(s)", violations.len());
246}

248fn doctor(cli: &Cli) -> i32 {
249    println!("mimus doctor");
250    match load(cli) {
251        Ok(c) => println!(
252            "  config   ok  (mechanical: {}, semantic: {}, rules: {})",
253            c.check.mechanical,
254            c.check.semantic,
255            c.rules.dir.display()
256        ),
257        Err(e) => println!("  config   --  {e}"),
258    }
259    println!("  adapters    fs, cmd (symbol/llm pending kuda schema + provider)");
260    0
