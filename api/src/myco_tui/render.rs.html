1//! Drawing — tab bar on top, board list filling the frame, status
2//! bar at the bottom. Preview mode swaps the list for the item
3//! detail pane (pm's `Enter` full-pane preview). Everything degrades
4//! per frame: rows truncate with `…` on narrow screens, chrome bars
5//! yield to the list on short ones, and `?` folds into columns.
6

15const ACCENT: Color = Color::Indexed(69); // pm's nav blue
16const MUTED: Color = Color::Indexed(241);

18/// Chrome budget — which fixed rows the screen can afford, so the
19/// list keeps a real viewport on short terminals. `(tabs, status)`:
20/// tabs cost a row at height ≥3, the status bar only at height ≥5
21/// (at 4 rows it yields to a three-row viewport).
22fn chrome(height: u16) -> (bool, bool) {
23    (height >= 3, height >= 5)
24}
25
26/// One full frame. Also records the tab/board geometry on `m` so
27/// the next mouse event can hit-test against where rows landed.
28pub fn draw(f: &mut Frame, m: &mut Model) {
29    let area = f.area();
30    let (show_tabs, show_status) = chrome(area.height);
31    let mut constraints = Vec::with_capacity(3);
32    if show_tabs {
33        constraints.push(Constraint::Length(1));
34    }
35    constraints.push(Constraint::Min(0));
36    if show_status {
37        constraints.push(Constraint::Length(1));
38    }
39    let chunks = Layout::vertical(constraints).split(area);
40
41    let mut next = 0;
42    if show_tabs {
43        draw_tabs(f, m, chunks[next]);
44        m.tabs_area = Some(chunks[next]);
45        next += 1;
46    } else {
47        m.tabs_area = None;
48    }
49    match m.mode {
50        Mode::Preview => draw_preview(f, m, chunks[next]),
51        _ => draw_board(f, m, chunks[next]),
52    }
53    if show_status {
54        draw_status(f, m, chunks[next + 1]);
55    }
56    if m.mode == Mode::Help {
57        draw_help(f, area);
58    }
59    if m.mode == Mode::Confirm {
60        draw_confirm(f, area, m);
61    }
62}

64/// Tab captions for `width` columns — full names when they fit,
65/// initials when they don't (the `1`/`2`/`h`/`l` binds still work).
66fn tab_labels(width: usize) -> Vec<String> {
67    let full: Vec<String> = Board::ALL
68        .iter()
69        .map(|b| format!(" {} ", b.label()))
70        .collect();
71    // leading space + one separator per tab
72    let need: usize = full.iter().map(|s| s.chars().count()).sum::<usize>() + full.len() + 1;
73    if need <= width {
74        full
75    } else {
76        Board::ALL
77            .iter()
78            .map(|b| b.label()[..1].to_string())
79            .collect()
80    }
81}

83fn draw_tabs(f: &mut Frame, m: &Model, area: Rect) {
84    let mut spans = vec![Span::raw(" ")];
85    for (b, label) in Board::ALL.iter().zip(tab_labels(area.width as usize)) {
86        let style = if *b == m.board {
87            Style::default().fg(Color::White).bg(ACCENT)
88        } else {
89            Style::default().fg(MUTED)
90        };
91        spans.push(Span::styled(label, style));
92        spans.push(Span::raw(" "));
93    }
94    f.render_widget(Paragraph::new(Line::from(spans)), area);
95}

97/// Click hit-test for the tab row: which tab does column `x` land
98/// on in a `width`-wide bar? The leading and separator columns
99/// belong to no tab. Mirrors the span layout `draw_tabs` emits.
100pub fn tab_hit(width: u16, x: u16) -> Option<usize> {
101    let mut col = 1u16; // leading space
102    for (i, label) in tab_labels(width as usize).iter().enumerate() {
103        let w = label.chars().count() as u16;
104        if (col..col + w).contains(&x) {
105            return Some(i);
106        }
107        col += w + 1; // the separator after each tab
108    }
109    None
110}

112/// Fit `spans` into `width` columns; when they overflow, keep the
113/// longest prefix that leaves one column for a muted `…`. Widths are
114/// char counts — wide glyphs may still let ratatui clip the tail.
115fn ellipsize<'a>(spans: Vec<Span<'a>>, width: usize) -> Vec<Span<'a>> {
116    let total: usize = spans.iter().map(|s| s.content.chars().count()).sum();
117    if total <= width {
118        return spans;
119    }
120    if width == 0 {
121        return Vec::new();
122    }
123    let keep = width - 1; // reserve a column for the ellipsis
124    let mut out = Vec::new();
125    let mut used = 0;
126    for span in spans {
127        let n = span.content.chars().count();
128        if used + n <= keep {
129            used += n;
130            out.push(span);
131        } else {
132            let take = keep - used;
133            if take > 0 {
134                let cut: String = span.content.chars().take(take).collect();
135                out.push(Span::styled(cut, span.style));
136            }
137            break;
138        }
139    }
140    out.push(Span::styled("…", Style::default().fg(MUTED)));
141    out
142}

144fn draw_board(f: &mut Frame, m: &mut Model, area: Rect) {
145    // the highlight symbol eats two columns of every row
146    let row_w = (area.width as usize).saturating_sub(2);
147    let items: Vec<ListItem> = m
148        .rows
149        .iter()
150        .map(|r| match r {
151            Row::Header(name) => ListItem::new(Line::from(ellipsize(
152                vec![Span::styled(
153                    format!("## {name}"),
154                    Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
155                )],
156                row_w,
157            ))),
158            Row::Item { item, depth, repo } => {
159                let indent = "  ".repeat(*depth);
160                let mut spans = vec![Span::raw(indent)];
161                // index-sourced items carry `repo:ID` — the same
162                // tag `myco view` cards emit
163                if let Some(r) = repo {
164                    spans.push(Span::styled(format!("{r}:"), Style::default().fg(MUTED)));
165                    if !m.show_ids {
166                        spans.push(Span::raw(" "));
167                    }
168                }
169                if m.show_ids {
170                    spans.push(Span::styled(
171                        item.id.to_string(),
172                        Style::default().fg(ACCENT),
173                    ));
174                    spans.push(Span::raw(" "));
175                }
176                spans.push(Span::raw(item.title.clone()));
177                let mut meta = format!(" {}", item.status);
178                if !matches!(item.priority, myco_core::Priority::Medium) {
179                    meta += &format!(" pri:{}", item.priority.token());
180                }
181                for l in &item.labels {
182                    meta += &format!(" #{l}");
183                }
184                spans.push(Span::styled(meta, Style::default().fg(MUTED)));
185                ListItem::new(Line::from(ellipsize(spans, row_w)))
186            }
187        })
188        .collect();
189
190    let list = List::new(items)
191        .highlight_style(Style::default().bg(Color::Indexed(238)))
192        .highlight_symbol("▸ ");
193    let mut state = ListState::default().with_selected(Some(m.sel));
194    f.render_stateful_widget(list, area, &mut state);
195    // remember where the rows landed for mouse hit-testing
196    m.board_area = area;
197    m.viewport_top = state.offset();
198}

200fn draw_preview(f: &mut Frame, m: &mut Model, area: Rect) {
201    m.link_hits.clear();
202    let Some((item, body, repo)) = m.preview_content() else {
203        return;
204    };
205    let width = area.width as usize;
206    let mut lines = vec![
207        Line::from(Span::styled(
208            item.id.to_string(),
209            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
210        )),
211        Line::from(Span::styled(
212            item.title.clone(),
213            Style::default().add_modifier(Modifier::BOLD),
214        )),
215        Line::from(""),
216        Line::from(vec![
217            Span::styled("status ", Style::default().fg(MUTED)),
218            Span::raw(item.status.to_string()),
219        ]),
220        Line::from(vec![
221            Span::styled("pri    ", Style::default().fg(MUTED)),
222            Span::raw(item.priority.token().to_string()),
223        ]),
224    ];
225    if let Some(r) = &repo {
226        lines.push(Line::from(vec![
227            Span::styled("repo   ", Style::default().fg(MUTED)),
228            Span::raw(r.clone()),
229        ]));
230    }
231    if let Some(o) = &item.owner {
232        lines.push(Line::from(vec![
233            Span::styled("owner  ", Style::default().fg(MUTED)),
234            Span::raw(o.to_string()),
235        ]));
236    }
237    if !item.labels.is_empty() {
238        lines.push(Line::from(vec![
239            Span::styled("labels ", Style::default().fg(MUTED)),
240            Span::raw(
241                item.labels
242                    .iter()
243                    .map(|l| format!("#{l}"))
244                    .collect::<Vec<_>>()
245                    .join(" "),
246            ),
247        ]));
248    }
249    for (key, val) in [
250        ("epic   ", item.epic.clone()),
251        ("ms     ", item.milestone.clone()),
252        ("due    ", item.due.map(myco_core::date::render)),
253        ("est    ", item.estimate.map(|e| e.to_string())),
254        ("cycle  ", item.cycle.clone()),
255    ] {
256        if let Some(v) = val {
257            lines.push(Line::from(vec![
258                Span::styled(key, Style::default().fg(MUTED)),
259                Span::raw(v),
260            ]));
261        }
262    }
263    if let Some(d) = item.created {
264        lines.push(Line::from(vec![
265            Span::styled("created", Style::default().fg(MUTED)),
266            Span::raw(format!(" {d}")),
267        ]));
268    }
269    // The related block is the clickable zone — each dst id gets
270    // its own line and a hit rect. `rel_len` bounds the chrome we
271    // must keep single-line so buffer indices map to screen rows.
272    let rel_start = lines.len();
273    let rel = related_lines(m, &item, repo.as_deref());
274    let rel_len = rel_start + rel.len();
275    let mut link_meta: Vec<(usize, myco_core::ItemId, Option<String>)> = Vec::new();
276    for (i, (line, link)) in rel.into_iter().enumerate() {
277        if let Some((id, lr)) = link {
278            link_meta.push((rel_start + i, id, lr));
279        }
280        lines.push(line);
281    }
282    if !link_meta.is_empty() {
283        // Everything above and inside the related block must be
284        // exactly one visual row per buffer line, or the hit rows
285        // drift. Fit (truncate, never wrap) the chrome lines.
286        for line in lines.iter_mut().take(rel_len) {
287            *line = fit_line(std::mem::take(line), width);
288        }
289    }
290    if !item.children.is_empty() {
291        lines.push(Line::from(""));
292        lines.push(Line::from(Span::styled(
293            "children",
294            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
295        )));
296        for c in &item.children {
297            lines.push(Line::from(format!("  {}", myco_ops::items::compact(c))));
298        }
299    }
300    if let Some(note) = &item.note {
301        lines.push(Line::from(""));
302        lines.push(Line::from(Span::styled(
303            "note",
304            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
305        )));
306        lines.push(Line::from(note.clone()));
307    }
308    // File-backed items render their body (pm's note preview).
309    if let Some(body) = body {
310        lines.push(Line::from(""));
311        for l in body.lines() {
312            let style = if l.starts_with('#') {
313                Style::default().fg(ACCENT).add_modifier(Modifier::BOLD)
314            } else {
315                Style::default()
316            };
317            lines.push(Line::from(Span::styled(l.to_string(), style)));
318        }
319    }
320    // wrap rather than clip long titles/notes on narrow screens;
321    // the pane scrolls (`j`/`k`/… in Preview mode), clamped to the
322    // content so `G` lands on the last line
323    let scroll = (m.preview_scroll as usize).min(lines.len().saturating_sub(1)) as u16;
324    // Record where each link landed — buffer index minus scroll is
325    // the visual row because the chrome above was fit to one line.
326    for (idx, id, lr) in link_meta {
327        let y = area.y as isize + idx as isize - scroll as isize;
328        if y >= area.y as isize && y < area.bottom() as isize {
329            m.link_hits.push(LinkHit {
330                rect: Rect {
331                    x: area.x,
332                    y: y as u16,
333                    width: area.width,
334                    height: 1,
335                },
336                id,
337                repo: lr,
338            });
339        }
340    }
341    f.render_widget(
342        Paragraph::new(lines)
343            .wrap(Wrap { trim: false })
344            .scroll((scroll, 0)),
345        area,
346    );
347}

349/// Truncate a line to `width` columns — same `…` tail as
350/// `ellipsize`, for lines that must stay one visual row.
351fn fit_line(line: Line<'static>, width: usize) -> Line<'static> {
352    Line::from(ellipsize(line.spans, width))
353}

355/// A related-section row: the rendered line plus the link a click
356/// on it resolves (`(item id, repo hint)`), `None` for chrome.
357type RelatedRow = (Line<'static>, Option<(myco_core::ItemId, Option<String>)>);
358
359/// The preview's "related" section — edges in/out grouped by
360/// kind (P-005, the I-019 related-files view). Each dst gets its
361/// own line; item ids are styled as links and returned with the
362/// `(id, repo)` a click resolves. `other` resolves to
363/// `ID title` when the neighbor is loaded; a `handoff` self edge
364/// shows the file path instead of the item's own id.
365fn related_lines(m: &Model, item: &myco_core::Item, repo: Option<&str>) -> Vec<RelatedRow> {
366    let related = m.related(item.id);
367    if related.is_empty() {
368        return Vec::new();
369    }
370    let mut out = vec![
371        (Line::from(""), None),
372        (
373            Line::from(Span::styled(
374                "related",
375                Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
376            )),
377            None,
378        ),
379    ];
380    for kind in myco_core::EdgeKind::ALL {
381        let dsts: Vec<myco_core::ItemId> = related
382            .iter()
383            .filter(|(k, _)| *k == kind)
384            .map(|(_, other)| *other)
385            .collect();
386        if dsts.is_empty() {
387            continue;
388        }
389        for (i, other) in dsts.iter().enumerate() {
390            // kind label on the group's first line only — the rest
391            // indent under it
392            let label = if i == 0 {
393                format!("  {:<13}", kind.name())
394            } else {
395                " ".repeat(15)
396            };
397            let (text, link) = if kind == myco_core::EdgeKind::Handoff {
398                (format!("handoffs/{other}.md"), None)
399            } else {
400                let title = m.item_title(*other).filter(|t| !t.is_empty());
401                let text = match title {
402                    Some(t) => format!("{other} {t}"),
403                    None => other.to_string(),
404                };
405                (text, Some((*other, repo.map(str::to_owned))))
406            };
407            let style = if link.is_some() {
408                Style::default().fg(ACCENT)
409            } else {
410                Style::default()
411            };
412            out.push((
413                Line::from(vec![
414                    Span::styled(label, Style::default().fg(MUTED)),
415                    Span::styled(text, style),
416                ]),
417                link,
418            ));
419        }
420    }
421    out
422}

424/// Normal-mode hint sets, widest first — `pick_hints` takes the
425/// first that fits so narrow terminals degrade instead of cropping.
426const HINTS: [&str; 3] = [
427    "j/k move · tab boards · space done · x cancel · D defer · p promote · u/U undo·redo · alt+↑/↓ reorder · a add · / filter · : cmd · enter/→ open · ← back · E edit · R reload · ? help · q quit",
428    "j/k move · space done · u/U undo·redo · a add · / filter · : cmd · enter/→ · ← back · E edit · ? help · q quit",
429    "? help · q quit",
430];

432/// The hint line for `mode` that fits `width` columns (including the
433/// leading space). The minimal set always wins as the fallback.
434fn pick_hints(mode: Mode, width: usize) -> &'static str {
435    match mode {
436        Mode::Add => "enter add · esc cancel",
437        Mode::Filter => "enter keep · esc clear",
438        Mode::Command => "enter run · esc cancel",
439        Mode::Preview => "j/k scroll · ←/esc back",
440        Mode::Help => "esc/?/q close",
441        Mode::Confirm => "y confirm · n/esc cancel",
442        Mode::Normal => HINTS
443            .iter()
444            .copied()
445            .find(|h| h.chars().count() < width)
446            .unwrap_or(HINTS[HINTS.len() - 1]),
447    }
448}

450fn draw_status(f: &mut Frame, m: &Model, area: Rect) {
451    if matches!(m.mode, Mode::Add | Mode::Filter | Mode::Command) {
452        let (prompt, buf) = match m.mode {
453            Mode::Add => ("add> ", &m.input),
454            Mode::Filter => ("/ ", &m.filter),
455            _ => (":", &m.cmdline),
456        };
457        let line = Line::from(vec![
458            Span::styled(prompt, Style::default().fg(ACCENT)),
459            Span::raw(buf.clone()),
460            Span::styled("█", Style::default().fg(ACCENT)),
461        ]);
462        f.render_widget(Paragraph::new(line), area);
463        return;
464    }
465    // The status message eats columns first; hints get what remains.
466    let status_w = if m.status.is_empty() {
467        0
468    } else {
469        m.status.chars().count() + 2
470    };
471    let hints = pick_hints(m.mode, (area.width as usize).saturating_sub(status_w));
472    let line = if m.status.is_empty() {
473        Line::from(Span::styled(
474            format!(" {hints}"),
475            Style::default().fg(MUTED),
476        ))
477    } else {
478        Line::from(vec![
479            Span::styled(format!(" {} ", m.status), Style::default().fg(Color::White)),
480            Span::styled(format!(" {hints}"), Style::default().fg(MUTED)),
481        ])
482    };
483    f.render_widget(Paragraph::new(line), area);
484}

486/// A `width`×`height` rect centered in `area`, clamped to fit.
487fn centered(area: Rect, width: u16, height: u16) -> Rect {
488    let width = width.min(area.width);
489    let height = height.min(area.height);
490    Rect {
491        x: area.x + (area.width - width) / 2,
492        y: area.y + (area.height - height) / 2,
493        width,
494        height,
495    }
496}

549/// Break `lines` into side-by-side columns: as many as the screen
550/// height needs (one when everything fits vertically), capped by
551/// what the width affords. `inner` is the box content area.
552fn help_columns(lines: &[Line<'static>], area: Rect) -> Vec<Vec<Line<'static>>> {
553    let inner_h = area.height.saturating_sub(2).max(1) as usize;
554    let inner_w = area.width.saturating_sub(2).max(1) as usize;
555    let col_w = lines.iter().map(Line::width).max().unwrap_or(0) + 3;
556    let fits = (inner_w / col_w.max(1)).max(1); // columns the width allows
557    let needs = lines.len().div_ceil(inner_h).max(1); // columns the height wants
558    let n = fits.min(needs);
559    let per = lines.len().div_ceil(n);
560    lines.chunks(per).map(<[Line]>::to_vec).collect()
561}

563fn draw_help(f: &mut Frame, area: Rect) {
564    let cols = help_columns(&help_lines(), area);
565    let col_ws: Vec<u16> = cols
566        .iter()
567        .map(|c| c.iter().map(Line::width).max().unwrap_or(0) as u16 + 3)
568        .collect();
569    let width = col_ws.iter().sum::<u16>() + 2;
570    let height = cols.iter().map(Vec::len).max().unwrap_or(0) as u16 + 2;
571    let popup = centered(area, width, height);
572    f.render_widget(Clear, popup);
573    let block = Block::default()
574        .borders(Borders::ALL)
575        .border_style(Style::default().fg(ACCENT))
576        .title(Span::styled(" keys ", Style::default().fg(ACCENT)));
577    let inner = block.inner(popup);
578    f.render_widget(block, popup);
579    let rects = Layout::horizontal(col_ws.iter().map(|w| Constraint::Length(*w))).split(inner);
580    for (lines, rect) in cols.iter().zip(rects.iter()) {
581        f.render_widget(Paragraph::new(lines.clone()), *rect);
582    }
583}

585/// The confirm popup — a centered box holding the pending prompt;
586/// the `y`/`n` hint also shows in the status bar via `pick_hints`.
587fn draw_confirm(f: &mut Frame, area: Rect, m: &Model) {
588    let Some(p) = &m.confirm else {
589        return;
590    };
591    let width = (p.prompt.chars().count() as u16 + 4).min(area.width);
592    let popup = centered(area, width, 5);
593    f.render_widget(Clear, popup);
594    let block = Block::default()
595        .borders(Borders::ALL)
596        .border_style(Style::default().fg(ACCENT))
597        .title(Span::styled(" confirm ", Style::default().fg(ACCENT)));
598    let inner = block.inner(popup);
599    f.render_widget(block, popup);

716    /// Draw one frame and return the buffer as text.
717    fn screen(m: &mut Model, w: u16, h: u16) -> String {
718        let backend = TestBackend::new(w, h);
719        let mut term = Terminal::new(backend).unwrap();
720        term.draw(|f| draw(f, m)).unwrap();
721        let buf = term.backend().buffer().clone();
722        let mut s = String::new();
723        for y in 0..buf.area.height {
724            for x in 0..buf.area.width {
725                s.push_str(buf[(x, y)].symbol());
726            }
727            s.push('\n');
728        }
729        s
730    }

743                    .epic("e1")

745                    .due(myco_core::date::parse("2026-12-01").unwrap())
746                    .estimate(5)
747                    .cycle("sprint-12")
748                    .child(t(2))
749                    .blocks(ItemId::new(ItemKind::Task, 3))
750                    .blocks(ItemId::new(ItemKind::Task, 99)), // unloaded neighbor
751                None,

755            l.put_todo(t(3).blocks(ItemId::new(ItemKind::Task, 1)), None);

759                ItemId::new(ItemKind::Task, 1),

762                    actor: "devin".parse().unwrap(),
763                    session: None,
764                    note: "done".into(),
765                },
766            )
767            .unwrap();
768        // reload (not just load_files) so the link graph sees the
769        // handoff file — `related` feeds the preview's edge list
770        m.reload();
771        m
772    }
