2//! tamias-registry-open — the shared registry-source layer.
3//!
4//! Resolves a registry value — a `registries` config **name** first,
5//! then a raw dir/http ref — into a [`Registry`] impl, and owns the
6//! config-file CRUD behind `tamias registry add|remove|get|list` and
7//! the `tamias-serve` `registry.*` ops. Both `tamias-cli` and
8//! `tamias-serve` consume this so name resolution, limit lookup, and
9//! config writes can't diverge (same role `tamias-install` plays for
10//! the install pipeline).
11

22/// The effective byte limits for a publish/search target — `value`
23/// doubles as the registry name when it matches a `registries` key
24/// (raw dir/http refs simply miss the lookup and get global + package
25/// rules). `None` config → built-in defaults.
26#[must_use]
27pub fn effective_limits(
28    config: Option<&Config>,
29    value: Option<&str>,
30    package: Option<&str>,
31) -> tamias_config::EffectiveLimits {
32    config.map_or_else(
33        || Config::default().effective_limits(value, package),
34        |c| c.effective_limits(value, package),
35    )
36}

38/// The retention `keep` count for a publish/search target — same
39/// name-or-ref resolution as [`effective_limits`]. `keep_override`
40/// (e.g. `publish --keep`) wins over the config chain; the result is
41/// `None` when nothing sets it → pruning disabled.
42#[must_use]
43pub fn effective_retention(
44    config: Option<&Config>,
45    value: Option<&str>,
46    package: Option<&str>,
47    keep_override: Option<u64>,
48) -> Option<u64> {
49    keep_override.or_else(|| config.and_then(|c| c.effective_retention(value, package)))
50}

52/// The `--registry`/`"registry"` value `publish`/`search` resolve:
53/// explicit flag → `TAMIAS_REGISTRY` env (wins over config, mirroring
54/// the store/cache env-wins rule) → the config routing defaults, in
55/// order → `None` (the local default root). `config_defaults` is e.g.
56/// `[search.default, publish.default]` for `search`.
57#[must_use]
58pub fn registry_target(flag: Option<&str>, config_defaults: &[Option<&str>]) -> Option<String> {
59    flag.map(str::to_string)
60        .or_else(|| {
61            std::env::var("TAMIAS_REGISTRY")
62                .ok()
63                .filter(|v| !v.is_empty())
64        })
65        .or_else(|| {
66            config_defaults
67                .iter()
68                .flatten()
69                .next()
70                .map(|s| (*s).to_string())
71        })
72}

74/// Open a registry: `value` is a `registries` config **name** first,
75/// then a raw dir/http ref ([`parse_registry_ref`]). `None` → the
76/// env / default local root at `registry_root`. Callers resolve the
77/// routing defaults (`publish.default`/`search.default`,
78/// `TAMIAS_REGISTRY` env) into `value` before calling — env wins over
79/// config, flag over env.
80///
81/// `package` (the manifest name on publish) selects per-package limit
82/// overrides; the resolved [`tamias_config::EffectiveLimits`] are handed
83/// to the adapter (`with_max_upload`), which enforces them — the
84/// packed-tarball size probe runs post-pack/pre-upload inside
85/// `publish`, so an over-limit payload never reaches the wire or the
86/// index.
87/// `keep` is the per-invocation retention override (`publish --keep`)
88/// — `None` resolves `retention.keep` from config (global →
89/// per-registry → per-package, see [`effective_retention`]). It only
90/// matters on the publish path; search-built registries carry it
91/// inertly.
92pub fn open_registry(
93    registry_root: &Path,
94    config: Option<&Config>,
95    value: Option<&str>,
96    package: Option<&str>,
97    keep: Option<u64>,
98) -> Result<Box<dyn Registry>, Error> {
99    let Some(value) = value else {
100        return Ok(Box::new(
101            LocalRegistry::new(registry_root)
102                .with_max_upload(effective_limits(config, None, package).max_upload)
103                .with_retention(effective_retention(config, None, package, keep)),
104        ));
105    };
106    if let Some(named) = config.and_then(|c| c.registries.get(value)) {
107        return open_named_registry(registry_root, value, named, config, package, keep);
108    }
109    let eff = effective_limits(config, Some(value), package);
110    let retention = effective_retention(config, Some(value), package, keep);
111    Ok(match parse_registry_ref(value) {
112        RegistryRef::Local(path) => Box::new(
113            LocalRegistry::new(path)
114                .with_max_upload(eff.max_upload)
115                .with_retention(retention),
116        ),
117        RegistryRef::Http(base) => Box::new(
118            http_registry(&base, None, value)?
119                .with_max_upload(eff.max_upload)
120                .with_retention(retention),
121        ),
122    })
123}

125/// Build the `Registry` impl for a `registries` entry — explicit
126/// `kind` wins, url-scheme inference is the fallback
127/// ([`tamias_config::Registry::effective_kind`]). `http`/`cf` share
128/// the `HttpRegistry` client; auth is `token_env` (an env var *name*)
129/// → `TAMIAS_REGISTRY_TOKEN` → none.
130fn open_named_registry(
131    registry_root: &Path,
132    name: &str,
133    reg: &ConfigRegistry,
134    config: Option<&Config>,
135    package: Option<&str>,
136    keep: Option<u64>,
137) -> Result<Box<dyn Registry>, Error> {
138    let eff = effective_limits(config, Some(name), package);
139    let retention = effective_retention(config, Some(name), package, keep);
140    match reg.effective_kind().as_str() {
141        "local" => {
142            let root = reg
143                .url
144                .clone()
145                .map(PathBuf::from)
146                .unwrap_or_else(|| registry_root.to_path_buf());
147            Ok(Box::new(
148                LocalRegistry::new(root)
149                    .with_max_upload(eff.max_upload)
150                    .with_retention(retention),
151            ))
152        }
153        "http" | "cf" => {
154            let base = reg.url.clone().ok_or_else(|| {
155                Error::Failed(format!("registry '{name}': kind http/cf needs a url"))
156            })?;
157            http_registry(&base, reg.token_env.as_deref(), name).map(|r| {
158                Box::new(r.with_max_upload(eff.max_upload).with_retention(retention))
159                    as Box<dyn Registry>
160            })
161        }
162        other => {
163            eprintln!(
164                "tamias: warning — registry '{name}': unknown kind '{other}'; inferring from url"
165            );
166            match parse_registry_ref(reg.url.as_deref().unwrap_or_default()) {
167                RegistryRef::Local(path) => Ok(Box::new(
168                    LocalRegistry::new(path)
169                        .with_max_upload(eff.max_upload)
170                        .with_retention(retention),
171                )),
172                RegistryRef::Http(base) => http_registry(&base, reg.token_env.as_deref(), name)
173                    .map(|r| {
174                        Box::new(r.with_max_upload(eff.max_upload).with_retention(retention))
175                            as Box<dyn Registry>
176                    }),
177            }
178        }
179    }
180}

182/// An http(s) registry with bearer auth — `token_env` names the env
183/// var holding the token (the var's *name* lives in config; the value
184/// is read at connect time, never stored). `TAMIAS_REGISTRY_TOKEN` is
185/// the fallback when `token_env` is unset or its var is empty.
186///
187/// Returns the concrete [`HttpRegistry`] (not the trait object) so
188/// callers can layer on the resolved upload cap.
189fn http_registry(base: &str, token_env: Option<&str>, name: &str) -> Result<HttpRegistry, Error> {
190    let base = base.trim_end_matches('/');
191    if !(base.starts_with("http://") || base.starts_with("https://")) {
192        return Err(Error::Failed(format!(
193            "registry '{name}': url '{base}' is not http(s)"
194        )));
195    }
196    let token = token_env
197        .and_then(|var| match std::env::var(var) {
198            Ok(value) if !value.is_empty() => Some(value),
199            _ => {
200                eprintln!("tamias: warning — registry '{name}': ${var} is unset; no token");
201                None
202            }
203        })
204        .or_else(|| {
205            std::env::var("TAMIAS_REGISTRY_TOKEN")
206                .ok()
207                .filter(|t| !t.is_empty())
208        });
209    let mut registry = HttpRegistry::new(base);
210    if let Some(token) = token {
211        registry = registry.with_token(token);
212    }
213    Ok(registry)
214}

216/// `search <query> [registry]` — the shared routing + open + query
217/// behind `tamias search`, `tamias serve`'s `search` op, and `tamias
218/// ui`'s Registry pane: `registry_target(flag, [search.default,
219/// publish.default])` → `open_registry` → `registry.search`. Returns
220/// `(results, resolved_target)` — `None` target means the local
221/// default root.
222///
223/// # Errors
224/// Fails on registry resolution or the search itself.
225pub fn search(
226    registry_root: &Path,
227    config: Option<&Config>,
228    flag: Option<&str>,
229    query: &str,
230) -> Result<(Vec<RegistryEntry>, Option<String>), Error> {
231    let target = registry_target(
232        flag,
233        &[
234            config.and_then(|c| c.search.default.as_deref()),
235            config.and_then(|c| c.publish.default.as_deref()),
236        ],
237    );
238    let registry = open_registry(registry_root, config, target.as_deref(), None, None)?;
239    let results = registry.search(query)?;
240    Ok((results, target))
241}

243/// A resolved `publish <dir> [registry]` — the payload root, loaded
244/// manifest, routed target, and opened registry (named-registry +
245/// per-package limits applied). Surfaces render their own progress
246/// around [`publish_plan`] and finish with [`PublishPlan::publish`].
247///
248/// When the manifest carries `include:` and `keep_includes` is false,
249/// [`publish_plan`] flattens bases read-only via
250/// [`tamias_publish::compose`] — `root` then points at the *composed*
251/// scratch payload (kept alive by `_composed`) so signing and packing
252/// both cover the flattened tree. The manifest is unchanged —
253/// `include:` stays as provenance.
254pub struct PublishPlan {
255    /// `payload_root(dir)` — or the composed scratch dir when
256    /// `include:` was flattened. Holds `tamias.yaml` either way.
257    pub root: PathBuf,
258    /// The validated package manifest (unmodified — `include:` kept).
259    pub manifest: Manifest,
260    /// The opened registry — [`PublishPlan::publish`] goes through it.
261    pub registry: Box<dyn Registry>,
262    /// The resolved target (`--registry` flag → `TAMIAS_REGISTRY` →
263    /// `config_defaults`); `None` → the local default root.
264    pub target: Option<String>,
265    /// Keeps the flatten scratch dir alive for the plan's lifetime.
266    _composed: Option<tamias_publish::Composed>,
267}
268
269impl PublishPlan {
270    /// Upload the packed payload — `registry.publish(&root, &manifest,
271    /// signature)`. `signature` is the armored `ssh-keygen -Y` signature
272    /// over the payload's provenance anchors (`tamias publish --sign`);
273    /// `None` publishes unsigned. Returns the [`PublishOutcome`] —
274    /// `pruned` lists the older versions the retention policy dropped
275    /// from the index (P-024).
276    ///
277    /// # Errors
278    /// Propagates pack/publish failures from the registry impl.
279    pub fn publish(&self, signature: Option<&str>) -> Result<PublishOutcome, Error> {
280        self.registry.publish(&self.root, &self.manifest, signature)
281    }
282}

284/// `publish <dir> [registry]` resolution — the shared head of
285/// `tamias publish`, `tamias serve`'s `publish` op, and `tamias ui`'s
286/// Author pane: `payload_root` → `Manifest::load` → `include:`
287/// flatten (unless `keep_includes`) → `registry_target` →
288/// `open_registry`. `store_root` is the install store the flatten
289/// resolves bases against — read-only, publish never installs.
290/// `report` fires after the manifest loads and the target resolves but
291/// *before* the registry opens, so each surface keeps its own
292/// "publishing …" line in its own position (the CLI prints post-open;
293/// the TUI logs pre-open).
294///
295/// `keep` is the per-invocation retention override (`publish --keep`,
296/// serve's `"keep"` param) — `None` resolves `retention.keep` from
297/// config inside [`open_registry`].
298///
299/// # Errors
300/// Fails on a missing/invalid manifest, an unresolvable `include:`
301/// base, or registry resolution.
302#[allow(clippy::too_many_arguments)]
303pub fn publish_plan(
304    dir: &Path,
305    registry_root: &Path,
306    config: Option<&Config>,
307    flag: Option<&str>,
308    config_defaults: &[Option<&str>],
309    store_root: &Path,
310    keep_includes: bool,
311    keep: Option<u64>,
312    report: &mut dyn FnMut(&Manifest, Option<&str>),
313) -> Result<PublishPlan, Error> {
314    let root = payload_root(dir)?;
315    let manifest = Manifest::load(&root)?;
316    let target = registry_target(flag, config_defaults);
317    report(&manifest, target.as_deref());
318    // Flatten `include:` bases read-only — the composed payload is
319    // what gets signed and packed; `root` swaps to the scratch dir.
320    let (root, composed) = if keep_includes || manifest.include.is_empty() {
321        (root, None)
322    } else {
323        let composed = tamias_publish::compose(&root, store_root)?;
324        (composed.dir.clone(), Some(composed))
325    };
326    let registry = open_registry(
327        registry_root,
328        config,
329        target.as_deref(),
330        Some(&manifest.name),
331        keep,
332    )?;
333    Ok(PublishPlan {
334        root,
335        manifest,
336        registry,
337        target,
338        _composed: composed,
339    })
340}

342/// The user-level config dir — `$XDG_CONFIG_HOME/tamias`, falling back
343/// to `~/.config/tamias` (HOME, then USERPROFILE). `None` when no home
344/// can be determined.
345fn user_config_dir() -> Option<PathBuf> {
346    if let Some(xdg) = std::env::var_os("XDG_CONFIG_HOME") {
347        return Some(PathBuf::from(xdg).join("tamias"));
348    }
349    std::env::var_os("HOME")
350        .or_else(|| std::env::var_os("USERPROFILE"))
351        .map(|home| PathBuf::from(home).join(".config").join("tamias"))
352}

374/// The config file registry management verbs read/write — `file`,
375/// then `TAMIAS_CONFIG`, then `./tamias.{yml,yaml,toml}` discovery. No
376/// file anywhere → `./tamias.yml` + defaults (created on write). The
377/// returned [`Config`] is *that file's own* — never the merged read
378/// view — so writes can't flatten a lower layer's entries into the
379/// upper file, and a bare `registry add` in an empty dir still creates
380/// a project `tamias.yml` rather than silently editing the global
381/// config (pass `--file ~/.config/tamias/tamias.toml` for that).
382pub fn config_target(file: Option<&Path>) -> Result<(PathBuf, Config), Error> {
383    let formats: [&dyn ConfigFormat; 2] = [&YmlFormat, &TomlFormat];
384    let explicit = file
385        .map(Path::to_path_buf)
386        .or_else(|| std::env::var_os(CONFIG_ENV).map(PathBuf::from));
387    if let Some(path) = explicit {
388        let config = if path.is_file() {
389            tamias_config::load_file(&path, &formats)?
390        } else {
391            Config::default()
392        };
393        return Ok((path, config));
394    }
395    Ok(match tamias_config::load_in(Path::new("."), &formats)? {
396        Some(found) => found,
397        None => (PathBuf::from("tamias.yml"), Config::default()),
398    })
399}

401/// The effective config for read-side resolution (`search`/`publish`
402/// routing, `install`'s download cap): **layered** —
403///
404/// - `file`/`TAMIAS_CONFIG` → that file alone (explicit means
405///   explicit; no layering);
406/// - otherwise the user-level config (`~/.config/tamias/tamias.*`) is
407///   the base and a cwd `tamias.*` file overlays it per-field
408///   ([`Config::overlay`]) — project wins, global fills the gaps;
409/// - neither → defaults.
410///
411/// Unlike the CLI's warn-and-ignore reader this propagates parse
412/// errors — callers that want CLI semantics catch and warn.
413pub fn load_config(file: Option<&Path>) -> Result<Config, Error> {
414    let formats: [&dyn ConfigFormat; 2] = [&YmlFormat, &TomlFormat];
415    let explicit = file
416        .map(Path::to_path_buf)
417        .or_else(|| std::env::var_os(CONFIG_ENV).map(PathBuf::from));
418    if let Some(path) = explicit {
419        return Ok(if path.is_file() {
420            tamias_config::load_file(&path, &formats)?
421        } else {
422            Config::default()
423        });
424    }
425    let mut merged = user_config(&formats)?
426        .map(|(_, config)| config)
427        .unwrap_or_default();
428    if let Some((_, project)) = tamias_config::load_in(Path::new("."), &formats)? {
429        merged.overlay(&project);
430    }
431    Ok(merged)
432}

434/// Render `config` back to `path`'s format and write it — the only
435/// write path for `add_registry`/`remove_registry`.
436fn write_config(path: &Path, config: &Config) -> Result<(), Error> {
437    let formats: [&dyn ConfigFormat; 2] = [&YmlFormat, &TomlFormat];
438    let format = tamias_config::format_for(path, &formats)
439        .ok_or_else(|| Error::Failed(format!("no config format for {}", path.display())))?;
440    let text = format.render(config)?;
441    std::fs::write(path, text).map_err(Error::io(format!("write {}", path.display())))
442}

461/// Validate a `registry add --kind` value — `local` | `http` | `cf`
462/// (case-insensitive), canonicalized to the lowercase spelling stored
463/// in config.
464fn registry_kind(kind: &str) -> Result<&'static str, Error> {
465    match kind.to_lowercase().as_str() {
466        "local" => Ok("local"),
467        "http" => Ok("http"),
468        "cf" => Ok("cf"),
469        other => Err(Error::Failed(format!(
470            "unknown registry kind '{other}' (expected local, http, or cf)"
471        ))),
472    }
473}

475/// Validate a `registry add --id` value — must be usable as a ref
476/// label (`[a-zA-Z0-9][a-zA-Z0-9_.-]*`, same charset the registry-ref
477/// grammar claims), so a uuid or any slug-safe string works.
478fn registry_id(id: &str) -> Result<&str, Error> {
479    let valid = !id.is_empty()
480        && id.chars().next().is_some_and(|c| c.is_ascii_alphanumeric())
481        && id
482            .chars()
483            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-'));
484    if valid {
485        Ok(id)
486    } else {
487        Err(Error::Failed(format!(
488            "invalid registry id '{id}' — expected [a-zA-Z0-9][a-zA-Z0-9_.-]*"
489        )))
490    }
491}

493/// `registry add` — validate kind/url/id, refuse duplicates, render the
494/// config back through its own format. Returns the config path
495/// written. http/cf need an http(s) url; local takes a dir (or
496/// anything — the explicit kind wins over url-scheme inference at
497/// resolve time). `id` is the optional stable identifier registry-scoped
498/// install refs (`<id>:<name>@<version>`) match before the map key.
499///
500/// # Errors
501/// Fails on an unknown kind, a non-http(s) url for http/cf, a bad or
502/// duplicate id, a duplicate name, or a config read/write error.
503pub fn add_registry(
504    file: Option<&Path>,
505    name: &str,
506    kind: &str,
507    url: &str,
508    token_env: Option<String>,
509    id: Option<String>,
510) -> Result<PathBuf, Error> {
511    let kind = registry_kind(kind)?;
512    if kind != "local" {
513        match parse_registry_ref(url) {
514            RegistryRef::Http(_) => {}
515            RegistryRef::Local(_) => {
516                return Err(Error::Failed(format!(
517                    "registry '{name}': kind {kind} needs an http(s) url, got '{url}'"
518                )));
519            }
520        }
521    }
522    let id = id.as_deref().map(registry_id).transpose()?;
523    let (path, mut config) = config_target(file)?;
524    if config.registries.contains_key(name) {
525        return Err(Error::Failed(format!("registry '{name}' already exists")));
526    }
527    if let Some(id) = id {
528        if config
529            .registries
530            .values()
531            .any(|reg| reg.id.as_deref() == Some(id))
532        {
533            return Err(Error::Failed(format!("registry id '{id}' already exists")));
534        }
535    }
536    config.registries.insert(
537        name.to_string(),
538        ConfigRegistry {
539            id: id.map(str::to_string),
540            kind: Some(kind.to_string()),
541            url: Some(url.to_string()),
542            token_env,
543            limits: Default::default(),
544            retention: Default::default(),
545        },
546    );
547    write_config(&path, &config)?;
548    Ok(path)
549}

551/// `registry remove` — drop a named registry; unknown names error.
552/// Returns the config path written.
553///
554/// # Errors
555/// Fails when `name` isn't configured or the config can't be written.
556pub fn remove_registry(file: Option<&Path>, name: &str) -> Result<PathBuf, Error> {
557    let (path, mut config) = config_target(file)?;
558    if config.registries.remove(name).is_none() {
559        return Err(Error::Failed(format!("registry '{name}' not found")));
560    }
561    write_config(&path, &config)?;
562    Ok(path)
563}

565/// `registry get` — one named registry's config entry, resolved
566/// against the *effective* (layered) config — a registry declared in
567/// the user-level file answers even inside a project that doesn't
568/// redeclare it.
569///
570/// # Errors
571/// Fails when `name` isn't configured or the config can't be read.
572pub fn get_registry(file: Option<&Path>, name: &str) -> Result<ConfigRegistry, Error> {

574    config

588    #[test]
589    fn add_get_list_remove_round_trip() {
590        let tmp = tempfile::tempdir().unwrap();
591        let file = tmp.path().join("tamias.yml");
592        // add creates the file (defaults + the new entry)
593        let path = add_registry(
594            Some(&file),
595            "mine",
596            "local",
597            "/tmp/reg",
598            Some("MY_TOKEN".into()),
599            None,
600        )
601        .unwrap();
602        assert_eq!(path, file);
603        assert!(file.is_file());
604        // duplicate name errors
605        assert!(add_registry(Some(&file), "mine", "local", "/x", None, None).is_err());
606        // get returns the entry; list sees it; token_env is a name only
607        let reg = get_registry(Some(&file), "mine").unwrap();
608        assert_eq!(reg.effective_kind(), "local");
609        assert_eq!(reg.token_env.as_deref(), Some("MY_TOKEN"));
610        let (_p, config) = config_target(Some(&file)).unwrap();
611        assert_eq!(config.registries.len(), 1);
612        let json = registry_json("mine", &reg);
613        assert_eq!(json["name"], "mine");
614        assert_eq!(json["token_env"], "MY_TOKEN");
615        // remove drops it; removing again errors
616        remove_registry(Some(&file), "mine").unwrap();
617        assert!(get_registry(Some(&file), "mine").is_err());
618        assert!(remove_registry(Some(&file), "mine").is_err());
619    }

676    #[test]
677    fn open_registry_resolves_name_then_ref() {
678        let tmp = tempfile::tempdir().unwrap();
679        let reg_root = tmp.path().join("reg");
680        let mut config = Config::default();
681        config.registries.insert(
682            "mine".into(),
683            ConfigRegistry {
684                kind: Some("local".into()),
685                url: Some(reg_root.display().to_string()),
686                ..Default::default()
687            },
688        );
689        // by name
690        let r = open_registry(&reg_root, Some(&config), Some("mine"), None, None).unwrap();
691        assert!(r.entries().unwrap().is_empty());
692        // raw local ref
693        let raw = reg_root.display().to_string();
694        let r = open_registry(&reg_root, Some(&config), Some(&raw), None, None).unwrap();
695        assert!(r.entries().unwrap().is_empty());
696        // unknown name falls through to raw-ref parse → local dir
697        let r = open_registry(
698            &reg_root,
699            Some(&config),
700            Some("/nonexistent-xyz"),
701            None,
702            None,
703        )
704        .unwrap();
705        // entries() on a missing dir errors or is empty — either way open succeeded
706        let _ = r.entries();
707        // None → default root
708        let r = open_registry(&reg_root, None, None, None, None).unwrap();
709        let _ = r.entries();
710    }

712    #[test]
713    fn registry_target_order() {
714        // env var may be set by the environment — only assert the
715        // flag-wins and config-fallback arms that don't depend on it.
716        assert_eq!(
717            registry_target(Some("flag"), &[Some("cfg")]).as_deref(),
718            Some("flag")
719        );
720        if std::env::var_os("TAMIAS_REGISTRY").is_none() {
721            assert_eq!(
722                registry_target(None, &[Some("cfg")]).as_deref(),
723                Some("cfg")
724            );
725            assert_eq!(
726                registry_target(None, &[None, Some("pub")]).as_deref(),
727                Some("pub")
728            );
729            assert_eq!(registry_target(None, &[None]), None);
730        }
731    }
