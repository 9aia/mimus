2//! tamias-publish — author-side packing: a payload dir + manifest → a
3//! `.tgz` (the registry stores these; the http source fetches them).
4//!
5//! `include:` handling (PL-0007 / P-004): by default [`pack`]
6//! **flattens** — each base resolves *read-only* (installed store
7//! entry → its payload dir; a ref already installed → the entry whose
8//! `source_ref` matches; an existing fs path → resolved in place) and
9//! [`tamias_core::overlay`] composes them under the local payload in a
10//! scratch dir that is packed instead. Publish never installs
11//! anything: an unresolvable base fails naming it. `keep_includes`
12//! ships the raw payload dir for consumers that compose themselves.
13//!
14//! Flattening is **not transitive**: a base contributes whatever
15//! payload is stored/installed — a store entry is already composed if
16//! it had `include:` at its own install; an fs-dir base's own
17//! `include:` is *not* re-resolved (same rule as install's
18//! store-name path). The packed manifest is unchanged — `include:`
19//! stays as provenance.
20

32/// How [`pack`] treats a manifest's `include:` list.
33#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
34pub enum Includes {
35    /// Resolve bases read-only and pack the composed payload — the
36    /// default for `publish`/`bundle pack`.
37    #[default]
38    Flatten,
39    /// Ship the raw payload dir; consumers compose the bases
40    /// themselves (`--keep-includes`).
41    Keep,
42}

44/// The composed payload [`compose`] produced — `dir` holds the
45/// overlay result (manifest pinned to the local package's), `_scratch`
46/// keeps the tempdir alive until the caller is done packing/uploading.
47pub struct Composed {
48    /// The composed payload root — pass to `pack_tgz`/`registry.publish`.
49    pub dir: PathBuf,
50    /// Owns the scratch dir; dropping it deletes the composed tree.
51    _scratch: tempfile::TempDir,
52}
53
54/// Compose `root`'s payload over its manifest's `include:` bases into
55/// a fresh scratch dir — the read-only half of payload composition.
56/// Callers check `manifest.include.is_empty()` first; an empty list
57/// still produces a valid composed copy.
58///
59/// # Errors
60/// Fails on an unresolvable/ambiguous base or I/O errors while
61/// overlaying. Never installs, fetches, or mutates the store.
62pub fn compose(root: &Path, store_root: &Path) -> Result<Composed, Error> {
63    let manifest = Manifest::load(root)?;
64    let store = Store::new(store_root);
65    let entries = store.entries()?;
66    let mut bases = Vec::with_capacity(manifest.include.len());
67    // Scratch dirs for fs-tarball bases must outlive the overlay —
68    // collect them so they drop only after `overlay` returns.
69    let mut tarball_scratch = Vec::new();
70    for text in &manifest.include {
71        bases.push(resolve_base(&entries, text, &mut tarball_scratch)?);
72    }
73    let scratch = tempfile::tempdir().map_err(Error::io("create include-compose scratch"))?;
74    let base_refs: Vec<&Path> = bases.iter().map(PathBuf::as_path).collect();
75    overlay(&base_refs, root, scratch.path())?;
76    pin_manifest(root, scratch.path())?;
77    Ok(Composed {
78        dir: scratch.path().to_path_buf(),
79        _scratch: scratch,
80    })
81}

83/// One include entry → one base payload dir, read-only. Order:
84/// installed store name → store entry whose `source_ref` is the entry
85/// verbatim (an already-installed ref) → an existing fs path (dir or
86/// tarball, resolved in place — never installed). Anything else fails
87/// naming the base.
88fn resolve_base(
89    entries: &[StoreEntry],
90    text: &str,
91    tarball_scratch: &mut Vec<tempfile::TempDir>,
92) -> Result<PathBuf, Error> {
93    // 1. Store name — label / `@scope/name` / bare `name`.
94    if let Some(entry) = find_store_entry(entries, text)? {
95        return Ok(entry.dir);
96    }
97    // `@scope/name` is a name, never a path.
98    if text.starts_with('@') {
99        return Err(unresolvable(text));
100    }
101    // 2. A ref that's already installed — provenance match.
102    if let Some(entry) = entries.iter().find(|e| e.record.source_ref == text) {
103        return Ok(entry.dir.clone());
104    }
105    // 3. Locally resolvable fs path — dir or tarball, in place.
106    if FsRefKind.claims(text) {
107        let path = expand_path(Path::new(text));
108        if path.is_dir() {
109            return payload_root(&path);
110        }
111        if path.is_file() {
112            return untar_base(&path, tarball_scratch);
113        }
114    }
115    Err(unresolvable(text))
116}

118/// Unpack a tarball base into a scratch dir and return its payload
119/// root. The [`tempfile::TempDir`] is pushed onto `scratch` so it
120/// lives until the overlay completes.
121fn untar_base(path: &Path, scratch: &mut Vec<tempfile::TempDir>) -> Result<PathBuf, Error> {
122    let name = path
123        .file_name()
124        .and_then(|n| n.to_str())
125        .unwrap_or_default();
126    let dir = tempfile::tempdir().map_err(Error::io("create include-tarball scratch"))?;
127    let file = fs::File::open(path).map_err(Error::io(format!("open {}", path.display())))?;
128    if name.ends_with(".tgz") || name.ends_with(".tar.gz") {
129        untar_gz(file, dir.path())?;
130    } else if name.ends_with(".tar") {
131        untar(file, dir.path())?;
132    } else {
133        return Err(Error::Failed(format!(
134            "unsupported include tarball '{}' — expected .tgz/.tar.gz/.tar",
135            path.display()
136        )));
137    }
138    let root = payload_root(dir.path())?;
139    scratch.push(dir);
140    Ok(root)
141}

143/// The publish-side failure — names the base and says why it can't be
144/// flattened (publish never installs; `--keep-includes` is the out).
145fn unresolvable(text: &str) -> Error {
146    Error::Failed(format!(
147        "include '{text}' can't be resolved for publish — it is not an installed \
148         package or ref, and publish never installs; install it first, point \
149         include: at an installed name/ref, or republish with --keep-includes"
150    ))
151}

153/// Pin the composed payload's manifest to the local package's manifest
154/// file, byte-for-byte — same rule as install's compose: the top
155/// package's manifest is authoritative (`include:` stays for
156/// provenance); stray manifest spellings a base contributed are
157/// removed. See `tamias_install`'s `include::pin_manifest` — duplicated
158/// here because `tamias-publish` must not depend on the install
159/// pipeline.
160fn pin_manifest(local: &Path, dest: &Path) -> Result<(), Error> {
161    let Some(src) = Manifest::path_in(local) else {
162        return Ok(());
163    };
164    let file_name = src
165        .file_name()
166        .map(|n| n.to_string_lossy().into_owned())
167        .unwrap_or_default();
168    for candidate in MANIFEST_NAMES {
169        let target = dest.join(candidate);
170        if candidate == file_name {
171            fs::copy(&src, &target)
172                .map_err(Error::io(format!("pin manifest {}", target.display())))?;
173        } else if target.exists() {
174            fs::remove_file(&target).map_err(Error::io(format!(
175                "remove stray manifest {}",
176                target.display()
177            )))?;
178        }
179    }
180    Ok(())
181}

183/// Pack `dir` into a `.tgz` — `<name>-<version>.tgz` by default.
184///
185/// `includes` selects `include:` handling: [`Includes::Flatten`] (the
186/// default) resolves bases read-only against `store_root` and packs
187/// the composed payload; [`Includes::Keep`] ships the raw dir.
188///
189/// # Errors
190/// Fails when `dir` has no manifest, a base can't be resolved, or on
191/// I/O error.
192pub fn pack(
193    dir: &Path,
194    out: Option<&Path>,
195    includes: Includes,
196    store_root: &Path,
197) -> Result<PathBuf, Error> {
198    let root = payload_root(dir)?;
199    let manifest = Manifest::load(&root)?;
200    let default_name = match &manifest.version {
201        Some(v) => format!("{}-{v}.tgz", manifest.name),
202        None => format!("{}.tgz", manifest.name),
203    };
204    let out = out.map_or_else(|| PathBuf::from(&default_name), Path::to_path_buf);
205    if includes == Includes::Flatten && !manifest.include.is_empty() {
206        let composed = compose(&root, store_root)?;
207        return pack_tgz(&composed.dir, &out);
208    }
209    pack_tgz(&root, &out)
210}

219    /// Serializes tests that mutate process env/cwd — `cargo test`
220    /// runs tests on threads in one process, so mutations must be
221    /// serialized *and* restored even on panic.
222    static ENV: std::sync::Mutex<()> = std::sync::Mutex::new(());
223
224    /// RAII guard for [`ENV`]: restores the process environment and
225    /// working directory on drop — panic-safe. Poison is ignored: an
226    /// unwinding sibling already restored everything via its `Drop`.
227    struct EnvGuard {
228        _lock: std::sync::MutexGuard<'static, ()>,
229        env: Vec<(std::ffi::OsString, std::ffi::OsString)>,
230        cwd: std::path::PathBuf,
231    }
232
233    impl EnvGuard {
234        fn new() -> Self {
235            let lock = ENV.lock().unwrap_or_else(|e| e.into_inner());
236            let env = std::env::vars_os().collect();
237            let cwd = std::env::current_dir().unwrap();
238            Self {
239                _lock: lock,
240                env,
241                cwd,
242            }
243        }
244    }
245
246    impl Drop for EnvGuard {
247        fn drop(&mut self) {
248            for (k, _) in std::env::vars_os() {
249                if !self.env.iter().any(|(sk, _)| *sk == k) {
250                    std::env::remove_var(&k);
251                }
252            }
253            for (k, v) in &self.env {
254                std::env::set_var(k, v);
255            }
256            let _ = std::env::set_current_dir(&self.cwd);
257        }
258    }

260    fn store_entry(root: &Path, name: &str, files: &[(&str, &str)]) -> PathBuf {
261        let dir = root.join(name);
262        std::fs::create_dir_all(&dir).unwrap();
263        std::fs::write(
264            dir.join("tamias.yaml"),
265            format!("name: {}\nversion: 1.0.0\n", name.replace('/', "-")),
266        )
267        .unwrap();
268        for (f, c) in files {
269            std::fs::write(dir.join(f), c).unwrap();
270        }
271        dir
272    }

417    /// File list inside a `.tgz` — entries are stored under a single
418    /// top-level dir; we compare suffixes.
419    fn tgz_files(tgz: &Path) -> Vec<String> {
420        let mut names = Vec::new();
421        // untar_gz writes to a dir — use a scratch and walk it
422        let dir = tempfile::tempdir().unwrap();
423        untar_gz(fs::File::open(tgz).unwrap(), dir.path()).unwrap();
424        for rel in tamias_core::digest::payload_files(dir.path()).unwrap() {
425            names.push(rel.to_string_lossy().replace('\\', "/"));
426        }
427        names
428    }
