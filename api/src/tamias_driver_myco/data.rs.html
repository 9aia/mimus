1//! data — one read of a repo's `project/` worktree per wire op.
2//!
3//! The driver is a *reader*: every `view`/`query` resolves the
4//! target project, takes the shared worktree lock for the span of
5//! the load (same discipline as `myco status`), and drops it. No
6//! mutations, no index — `myco serve` stays the write/embedding
7//! surface; this crate only needs what the TUI rendered.
8//!
9//! Project resolution order (first hit wins):
10//!
11//! 1. Request params — `dir` (a `project/` worktree path) or
12//!    `root` (a repo root; `<root>/project` is tried), e.g.
13//!    `pavo view myco:board --param dir=/path/to/repo/project`.
14//! 2. `$MYCO_DIR` — treated as `dir`.
15//! 3. cwd discovery — `myco_ops::discover` walks up from the
16//!    driver's cwd for a qualifying `project/` worktree (the same
17//!    rules `myco` verbs use, including `myco.toml`'s
18//!    `project_dir`).
19//!
20//! When nothing resolves, [`Snapshot::error`] carries the reason
21//! and every query degrades to an `{"error": …}` payload — the
22//! wire stays `ok:true` so hosts render the failure inline.
23

34/// The config impl ops calls share — the scarab-backed TOML
35/// loader `myco-ops` owns, same as the CLI's `ops::CONFIG`.
36pub const CONFIG: myco_ops::config::ConfigLoader = myco_ops::config::TOML;

38/// `MdStore` rooted at a worktree path, resolving item layout
39/// through the repo's kind table — identical to the CLI's
40/// `ops::store_at` (kept local: that helper lives in the `myco`
41/// bin crate, which a library consumer can't link).
42fn store_at(worktree: &Path, kinds: &KindTable) -> Result<MdStore, String> {
43    let root = camino::Utf8PathBuf::from_path_buf(worktree.to_path_buf())
44        .map_err(|p| format!("{}: non-utf8 path", p.display()))?;
45    Ok(MdStore::with_kinds(root, kinds.clone()))
46}

48/// One handoff file's row for the `handoffs` view.
49#[derive(Debug, Clone)]
50pub struct HandoffRow {
51    /// The item the file records handoffs for.
52    pub id: ItemId,
53    /// Entry count.
54    pub entries: usize,
55    /// Lives under `handoffs/archive/`.
56    pub archived: bool,
57    /// Latest entry's `YYYY-MM-DD`.
58    pub updated: Option<String>,
59    /// Latest entry's actor.
60    pub actor: Option<String>,
61    /// Latest entry's note, truncated for table display.
62    pub note: Option<String>,
63}

65/// Everything the views need, loaded once per op.
66pub struct Snapshot {
67    /// The resolved project — `None` when discovery/open failed.
68    pub project: Option<discover::Project>,
69    /// TODO.md + BACKLOG.md.
70    pub ledger: Option<Ledger>,
71    /// Live file-backed items (issues/proposals/plans/adrs +
72    /// user kinds), metas + bodies.
73    pub files: Vec<ItemFile>,
74    /// The live link graph — edges + title lookup, live *and*
75    /// archived items (same source `item show` uses).
76    pub graph: Option<LiveGraph>,
77    /// `handoffs/*.md` summaries.
78    pub handoffs: Vec<HandoffRow>,
79    /// The repo's kind table (names/pads for user kinds).
80    pub kinds: KindTable,
81    /// Per-source load problems — surfaced, never fatal.
82    pub warnings: Vec<String>,
83    /// Why there is no project at all.
84    pub error: Option<String>,
85}
86
87impl Default for Snapshot {
88    fn default() -> Self {
89        Snapshot {
90            project: None,
91            ledger: None,
92            files: Vec::new(),
93            graph: None,
94            handoffs: Vec::new(),
95            kinds: myco_core::kind::prelude_table().clone(),
96            warnings: Vec::new(),
97            error: None,
98        }
99    }
100}
101
102impl Snapshot {
103    /// The empty/degraded snapshot — `error` says why.
104    fn failed(error: String) -> Self {
105        Snapshot {
106            error: Some(error),
107            ..Snapshot::default()
108        }
109    }
110
111    /// The `{"error": …}` payload every query emits when the
112    /// snapshot couldn't load — the degradation marker pavo
113    /// renderers already understand.
114    pub fn error_payload(&self) -> serde_json::Value {
115        serde_json::json!({"error": self.error.clone().unwrap_or_else(|| "no project".into())})
116    }
117
118    /// Is there a usable project?
119    pub fn ok(&self) -> bool {
120        self.error.is_none() && self.project.is_some()
121    }
122}

124/// `params` → `(dir, root)` overrides. `dir` wins over `root`
125/// when both are given (mirrors the CLI globals' precedence).
126fn overrides(
127    params: &serde_json::Map<String, serde_json::Value>,
128) -> (Option<PathBuf>, Option<PathBuf>) {
129    let dir = params
130        .get("dir")
131        .or_else(|| params.get("project"))
132        .and_then(|v| v.as_str())
133        .map(PathBuf::from);
134    let root = params
135        .get("root")
136        .and_then(|v| v.as_str())
137        .map(PathBuf::from);
138    (dir, root)
139}

141/// Resolve + load. `params` are the wire op's caller params —
142/// `dir`/`root`/`project` steer discovery, the rest pass through
143/// to the query resolvers untouched.
144pub fn load(params: &serde_json::Map<String, serde_json::Value>) -> Snapshot {
145    let (mut dir, root) = overrides(params);
146    if dir.is_none() && root.is_none() {
147        if let Ok(d) = std::env::var("MYCO_DIR") {
148            if !d.is_empty() {
149                dir = Some(PathBuf::from(d));
150            }
151        }
152    }
153    let dir = dir.as_deref();
154    let root = root.as_deref();
155
156    let (proj, locked) = match items::open_read_with(dir, root, CONFIG, store_at) {
157        Ok(pair) => pair,
158        Err(e) => return Snapshot::failed(e),
159    };
160
161    let mut snap = Snapshot {
162        project: Some(proj),
163        kinds: locked.kinds().clone(),
164        ..Snapshot::default()
165    };
166
167    // Ledger docs — a corrupt TODO.md/BACKLOG.md degrades the
168    // board, not the whole snapshot.
169    match items::load_ledger(&locked) {
170        Ok(ledger) => snap.ledger = Some(ledger),
171        Err(e) => snap.warnings.push(format!("ledger: {e}")),
172    }
173
174    // File-backed items, live scope only — `archive/` stays out of
175    // listings (the TUI's FILES board didn't show it either); the
176    // graph still sees archived edges via `live_graph`.
177    for spec in locked.kinds().file_backed() {
178        match locked.list_items(spec.id_kind()) {
179            Ok(mut v) => snap.files.append(&mut v),
180            Err(e) => snap
181                .warnings
182                .push(format!("{} files: {e}", spec.dir_name())),
183        }
184    }
185
186    // Handoff files — id from the filename, latest entry for the
187    // table's `updated`/`actor`/`note` columns.
188    match locked.list_handoff_files() {
189        Ok(hs) => {
190            for h in hs {
191                let mut row = HandoffRow {
192                    id: h.id,
193                    entries: 0,
194                    archived: h.archived,
195                    updated: None,
196                    actor: None,
197                    note: None,
198                };
199                match locked.read_handoffs(h.id) {
200                    Ok(entries) => {
201                        row.entries = entries.len();
202                        if let Some(last) = entries.last() {
203                            row.updated = Some(last.date.to_string());
204                            row.actor = Some(last.actor.to_string());
205                            let note: String = last.note.chars().take(120).collect();
206                            row.note = Some(if last.note.chars().count() > 120 {
207                                format!("{note}…")
208                            } else {
209                                note
210                            });
211                        }
212                    }
213                    Err(e) => snap.warnings.push(format!("handoffs/{}: {e}", h.id)),
214                }
215                snap.handoffs.push(row);
216            }
217        }
218        Err(e) => snap.warnings.push(format!("handoffs: {e}")),
219    }
220
221    // The link graph — live read, no index (see `links::live_graph`
222    // for why display surfaces derive rather than persist).
223    let (graph, gw) = links::live_graph(&locked);
224    snap.graph = Some(graph);
225    snap.warnings.extend(gw);
226
227    snap
228}
