1//! Engine ops — `cli.<verb>` wire queries dispatch into the `myco`
2//! engine lib through this binary's own `run` verb: each call
3//! re-execs `tamias-driver-myco run <argv>` with the request's repo
4//! as cwd, so `myco::dispatch` writes to the child's captured
5//! stdout and never touches the serve loop's ndjson stream — and
6//! every mutation keeps the engine's own write-path semantics
7//! (per-verb commit, `Actor:` trailer, worktree lock).
8//!
9//! `ops` answers the ops manifest as JSON — derived from the
10//! engine's own [`myco::registry::VERBS`] table, so a verb can't
11//! exist on the CLI and be missing here. Unbounded verbs
12//! (`serve`, `ui`) are deliberately *not* ops — they never
13//! return, so they live in `surfaces` instead: surface drivers
14//! (`tamias-driver-cli` / `-mcp`, the pavo-driver-kit `surface`
15//! layer) expose them as streaming subcommands.
16//!
17//! Wire params carry each verb's declared param schema: `arg` =
18//! positional, `opt` = `--name <v>`, `flag` = `--name` — the
19//! contract `pavo-driver-kit`'s surface layer renders args from.
20

23use myco::registry::{self, On, PKind, Verb};
24use pavo_protocol::wire::{QueryResult, Response};
25use serde_json::{json, Map, Value};

27/// Unbounded surfaces — verbs that never return, so they can
28/// never be request/response ops. The CLI/MCP surfaces stream
29/// them via `run` passthrough instead.
30const SURFACES: &[&str] = &["serve", "ui"];
31
32fn is_surface(name: &str) -> bool {
33    SURFACES.contains(&name)
34}
35
36/// The ops table — every CLI verb minus hidden and surfaces.
37fn ops() -> impl Iterator<Item = &'static Verb> {
38    registry::VERBS
39        .iter()
40        .filter(|v| v.cli.is_some() && !v.hidden && !is_surface(v.name))
41}
42
43/// `Param` → the manifest's typed param — `arg`/`opt`/`flag` as
44/// `pavo-driver-kit`'s surface layer spells them. Wire-only params
45/// (`Spec` patches) can't ride argv, so they stay out — surfaces
46/// only offer what `run` can pass.
47fn param_json(p: &registry::Param) -> Value {
48    json!({
49        "name": p.name,
50        "kind": match p.kind {
51            PKind::Pos => "arg",
52            PKind::Opt => "opt",
53            PKind::Flag => "flag",
54        },
55        "doc": p.help,
56    })
57}
58
59/// The `ops` query/`ops` argv verb payload — self-describing so
60/// hosts and agents discover the engine surface without argv help.
61/// `json`/`no_commit`/`dir`/`root` are the engine's globals —
62/// accepted by every op, listed once on the manifest itself.
63pub fn manifest() -> Value {
64    json!({
65        "engine": "myco",
66        "invoke": "query `cli.<verb>` with the listed params, or argv `tamias-driver-myco run <myco argv>`",
67        "globals": [
68            {"name": "json", "kind": "flag", "doc": "Emit JSON (accepted everywhere)."},
69            {"name": "no_commit", "kind": "flag", "doc": "Skip the per-verb commit (also suppressed by $MYCO_NO_COMMIT)."},
70            {"name": "dir", "kind": "opt", "doc": "Project worktree to act on, overriding discovery."},
71            {"name": "root", "kind": "opt", "doc": "Repo root to act on (the worktree is `<root>/project`)."},
72        ],
73        "ops": ops().map(|v| json!({
74            "query": format!("cli.{}", v.name),
75            "doc": v.about,
76            "params": v.params()
77                .filter(|p| p.on != On::Wire)
78                .map(param_json)
79                .collect::<Vec<_>>(),
80            "mutates": v.mutates,
81        })).collect::<Vec<_>>(),
82        "surfaces": registry::VERBS.iter()
83            .filter(|v| v.cli.is_some() && is_surface(v.name))
84            .map(|v| v.name)
85            .collect::<Vec<_>>(),
86    })
87}

89/// Is `query` an engine op (`cli.<verb>`)? Returns the verb.
90pub fn op_of(query: &str) -> Option<&'static Verb> {
91    query
92        .strip_prefix("cli.")
93        .and_then(registry::find)
94        .filter(|v| v.cli.is_some() && !v.hidden && !is_surface(v.name))
95}

97fn bool_param(params: &Map<String, Value>, key: &str) -> bool {
98    params.get(key).and_then(Value::as_bool).unwrap_or(false)
99}
100
101/// A wire scalar → its argv token (numbers/bools stringify —
102/// same read `Args::from_json` gives them).
103fn scalar(v: &Value) -> Option<String> {
104    match v {
105        Value::String(s) => Some(s.clone()),
106        Value::Number(n) => Some(n.to_string()),
107        Value::Bool(b) => Some(b.to_string()),
108        _ => None,
109    }
110}
111
112/// All values of a param — arrays flatten, scalars wrap.
113fn values(v: &Value) -> Vec<String> {
114    match v {
115        Value::Array(xs) => xs.iter().filter_map(scalar).collect(),
116        other => scalar(other).into_iter().collect(),
117    }
118}

120/// `cli.<verb>` params → myco argv (no argv[0] — `run` prepends
121/// it). Globals go first; the dotted verb name splits into the
122/// group path (`todo.add` → `todo add`). Declared params map by
123/// kind — positionals land last, in schema order.
124fn argv(v: &Verb, params: &Map<String, Value>) -> Vec<String> {
125    let mut out: Vec<String> = Vec::new();
126    if bool_param(params, "json") {
127        out.push("--json".into());
128    }
129    if bool_param(params, "no_commit") || bool_param(params, "no-commit") {
130        out.push("--no-commit".into());
131    }
132    // `dir` wins over its `project` alias; `root` is separate —
133    // same precedence `data::load` resolves the project by.
134    let dir = params
135        .get("dir")
136        .or_else(|| params.get("project"))
137        .and_then(Value::as_str);
138    if let Some(d) = dir {
139        out.extend(["--dir".into(), d.into()]);
140    }
141    if let Some(r) = params.get("root").and_then(Value::as_str) {
142        out.extend(["--root".into(), r.into()]);
143    }
144    out.extend(v.name.split('.').map(String::from));
145    let mut positional: Vec<String> = Vec::new();
146    for p in v.params() {
147        if p.on == On::Wire {
148            continue;
149        }
150        // canonical underscore name, dashed alias tolerated
151        let raw = params
152            .get(p.name)
153            .or_else(|| params.get(p.name.replace('_', "-").as_str()));
154        let Some(raw) = raw else { continue };
155        match p.kind {
156            PKind::Pos => positional.extend(values(raw)),
157            PKind::Flag => {
158                if raw.as_bool().unwrap_or(false) {
159                    out.push(format!("--{}", p.name.replace('_', "-")));
160                }
161            }
162            PKind::Opt => {
163                for s in values(raw) {
164                    out.push(format!("--{}", p.name.replace('_', "-")));
165                    out.push(s);
166                }
167            }
168        }
169    }
170    out.extend(positional);
171    out
172}

174/// The child's working dir — the steering params first
175/// (`dir`/`project`/`root`), then `$MYCO_DIR`, then the driver's
176/// own cwd (pavo inherits the user's cwd, so a bare `cli.status`
177/// inside repo X reads X's ledger).
178fn workdir(params: &Map<String, Value>) -> PathBuf {
179    for key in ["dir", "project", "root"] {
180        if let Some(s) = params.get(key).and_then(Value::as_str) {
181            return PathBuf::from(s);
182        }
183    }
184    if let Ok(d) = std::env::var("MYCO_DIR") {
185        if !d.is_empty() {
186            return PathBuf::from(d);
187        }
188    }
189    std::env::current_dir().unwrap_or_else(|_| PathBuf::from("."))
190}

192/// Run one engine op by name: re-exec `run` in a child whose cwd
193/// is the request's repo, and hand back `{op, argv, code, stdout,
194/// stderr}` as the query payload. `ok:false` is reserved for
195/// spawn failures — a nonzero `code` is the engine's own exit
196/// status (e.g. a `check` finding), honest data. The bare-name
197/// entry the embedded surfaces use (`OpSpec.name` is
198/// `cli.`-stripped); the wire path goes `op_of` → `run_verb`.
199#[cfg(feature = "mcp")]
200pub fn run_op(op: &str, params: &Map<String, Value>) -> Response {
201    let Some(v) = op_of(&format!("cli.{}", op.strip_prefix("cli.").unwrap_or(op))) else {
202        return Response::err(format!("unknown op: {op}"));
203    };
204    run_verb(v, params)
205}
206
207/// [`run_op`] with the verb already resolved — the `cli.<verb>`
208/// wire-query path lands here too.
209pub fn run_verb(v: &Verb, params: &Map<String, Value>) -> Response {
210    let args = argv(v, params);
211    let out = std::env::current_exe().and_then(|exe| {
212        std::process::Command::new(exe)
213            .arg("run")
214            .args(&args)
215            .current_dir(workdir(params))
216            .output()
217    });
218    let data = match out {
219        Ok(o) => json!({
220            "op": v.name,
221            "argv": args,
222            "code": o.status.code().unwrap_or(-1),
223            "stdout": String::from_utf8_lossy(&o.stdout),
224            "stderr": String::from_utf8_lossy(&o.stderr),
225        }),
226        Err(e) => return Response::err(format!("run {}: {e}", v.name)),
227    };
228    Response::ok_typed(&QueryResult {
229        query: format!("cli.{}", v.name),
230        data,
231    })
232}

234#[cfg(test)]
235mod tests {
236    #![allow(clippy::unwrap_used)]
237    use super::*;
238    use serde_json::json;
239
240    #[test]
241    fn op_of_accepts_cli_prefix_only() {
242        assert_eq!(op_of("cli.status").unwrap().name, "status");
243        assert_eq!(op_of("cli.todo.add").unwrap().name, "todo.add");
244        assert!(op_of("cli.serve").is_none()); // surface, not an op
245        assert!(op_of("cli.ui").is_none());
246        assert!(op_of("cli.gen-cli-docs").is_none()); // hidden
247        assert!(op_of("status").is_none()); // view query, not an op
248        assert!(op_of("cli.bogus").is_none());
249    }
250
251    #[test]
252    fn argv_maps_params_to_myco_flags() {
253        let v = registry::find("todo.add").unwrap();
254        let mut p = Map::new();
255        p.insert("title".into(), json!("fix it"));
256        p.insert("pri".into(), json!("high"));
257        p.insert("label".into(), json!(["a", "b"]));
258        p.insert("json".into(), json!(true));
259        p.insert("root".into(), json!("/repo"));
260        assert_eq!(
261            argv(v, &p),
262            vec![
263                "--json", "--root", "/repo", "todo", "add", "--pri", "high", "--label", "a",
264                "--label", "b", "fix it",
265            ]
266        );
267    }
268
269    #[test]
270    fn argv_skips_wire_only_and_absent_params() {
271        let v = registry::find("view").unwrap();
272        let mut p = Map::new();
273        p.insert("name".into(), json!("board"));
274        p.insert("spec".into(), json!({"limit": 5})); // wire-only — can't ride argv
275        assert_eq!(argv(v, &p), vec!["view", "board"]);
276    }
277
278    #[test]
279    fn manifest_lists_every_op_as_cli_query() {
280        let m = manifest();
281        let ops = m["ops"].as_array().unwrap();
282        for v in registry::VERBS
283            .iter()
284            .filter(|v| v.cli.is_some() && !v.hidden && !is_surface(v.name))
285        {
286            assert!(
287                ops.iter()
288                    .any(|o| o["query"] == json!(format!("cli.{}", v.name))),
289                "{} missing from ops manifest",
290                v.name
291            );
292        }
293        assert!(m["surfaces"].as_array().unwrap().contains(&json!("serve")));
294        for o in ops {
295            for p in o["params"].as_array().unwrap() {
296                assert!(["arg", "opt", "flag"].contains(&p["kind"].as_str().unwrap()));
297            }
298        }
299    }
