2//! tamias-driver — the content **driver** host: discovery + dispatch of
3//! `tamias-driver-<type>` executables, with **no product knowledge** in
4//! tamias.
5//!
6//! A **driver** is an executable `tamias-driver-<type>` on `PATH` or in
7//! `<tamias-home>/drivers/`. tamias reads a package's `type:`, resolves its
8//! driver, and runs:
9//!
10//! ```text
11//! tamias-driver-<type> <verb> --dir <dir> --name <name> [--config <file>]
12//! ```
13//!
14//! `verb ∈ {install, deploy, uninstall}` for the lifecycle dispatch;
15//! drivers may declare further provider verbs (`health`, `hooks`, …)
16//! in `capabilities.verbs`, reached through [`dispatch_with_args`] —
17//! `tamias driver exec`. A driver may also answer `capabilities`.
18//!
19//! **`config-schema` — a reserved verb by convention.** A driver that
20//! consumes a `--config` file *may* implement `config-schema`: dispatch
21//! it and the driver prints a JSON Schema (draft 2020-12) on stdout
22//! describing that file's format, exit 0. Drivers without one exit
23//! non-zero with a clear stderr message — callers treat that as
24//! "unsupported" and surface it, never a crash. The standard
25//! `--dir`/`--name`/`--config` flags are appended like any verb, but
26//! the schema should be instance-agnostic. Fetch it with
27//! [`config_schema`] or `tamias driver config-schema <pkg>`.
28//!
29//! Validating a config *against* the schema is deliberately
30//! driver-side, not engine-side: tamias ships no JSON Schema validator,
31//! so a driver that wants validation implements it as an ordinary
32//! provider verb — `tamias driver exec <pkg> config validate
33//! --config <file>` is the intended shape.
34//!
35//! **Operator config overrides (T-026).** Drivers that *generate*
36//! config on `deploy`/`install` wipe hand edits every run. tamias owns
37//! the persistence half in [`overrides`]: drop-in fragments under
38//! `<config>/tamias/deploy.d/<kind>/<name>/` (`tamias deploy --set
39//! key=value` edits the `cli.yaml` fragment), injected into every
40//! dispatch as [`overrides::OVERRIDES_ENV`]. A driver that generates
41//! config merges the dir's fragments over the generated doc before
42//! writing — [`tamias_core::merge_values`] gives the same deep-merge
43//! `include:` overlays use.
44//! Installing a product drops its driver into
45//! `<tamias-home>/drivers/` (from the payload's `drivers/` dir), so tamias
46//! finds it. The engine never links a product crate.
47//!
48//! **Bundled drivers.** A payload's `drivers/` dir ships ready-to-run
49//! driver files:
50//!
51//! - `drivers/<name>` — portable drivers (interpreted scripts like
52//!   `#!/usr/bin/env python3` ship as source).
53//! - `drivers/bin/<target-triple>/<name>` — compiled drivers, one dir per
54//!   rustc target triple (`x86_64-unknown-linux-gnu`, …). At install time
55//!   only the dir matching [`target_triple`] is copied; a package that
56//!   ships platform dirs but none for this host fails the install with a
57//!   clear error rather than silently dropping the driver.
58//!
59//! Both forms land flat in `<tamias-home>/drivers/` (executable bit set),
60//! stay inside the payload digest, and are consent-gated — see
61//! [`drivers_allowed`].
62

76/// A driver verb (`install`/`deploy`/`uninstall`) can legitimately take a

96/// `<tamias-home>` — `$TAMIAS_HOME` → `$XDG_DATA_HOME/tamias` →
97/// `~/.local/share/tamias` → `./.tamias`.
98///
99/// The no-env fallback is `./.tamias` — the same project-local home the
100/// CLI's `store_root()`/`cache_root()`/`registry_root()` fall back to
101/// (`./.tamias/store`, `./.tamias/fetch`, `./.tamias/registry`), so the
102/// drivers dir lands at `./.tamias/drivers` beside them. (`./tamias`
103/// would collide with a `tamias` source checkout or package dir.)
104#[must_use]
105pub fn tamias_home() -> PathBuf {
106    std::env::var_os("TAMIAS_HOME")
107        .map(PathBuf::from)
108        .or_else(|| std::env::var_os("XDG_DATA_HOME").map(|d| PathBuf::from(d).join("tamias")))
109        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share/tamias")))
110        .unwrap_or_else(|| PathBuf::from("./.tamias"))
111}

119/// Find the driver for a content type — `tamias-driver-<kind>` on `PATH`,
120/// then `<tamias-home>/drivers/`. Kinds outside the manifest charset
121/// ([`tamias_core::is_valid_kind`]) never resolve — a hand-built store
122/// record can't smuggle `../` into the driver-name join.
123#[must_use]
124pub fn find_driver(kind: &str) -> Option<PathBuf> {
125    if !tamias_core::is_valid_kind(kind) {
126        return None;
127    }
128    let name = format!("tamias-driver-{kind}");
129    if let Some(path) = lookup(&name) {
130        return Some(path);
131    }
132    // Otherwise map via capabilities: the first driver whose `types` include `kind`.
133    discovered_drivers()
134        .into_iter()
135        .find(|(_, path)| driver_handles(path, kind))
136        .map(|(_, path)| path)
137}
138
139/// `<name>` on `PATH`, then `<tamias-home>/drivers/`.
140fn lookup(name: &str) -> Option<PathBuf> {
141    let on_path = std::env::var_os("PATH").and_then(|path| {
142        std::env::split_paths(&path)
143            .map(|dir| dir.join(name))
144            .find(|candidate| candidate.is_file())
145    });
146    on_path.or_else(|| {
147        let candidate = drivers_dir().join(name);
148        candidate.is_file().then_some(candidate)
149    })
150}
151
152/// Whether `driver` reports handling `kind` (its `capabilities` `types`).
153fn driver_handles(driver: &Path, kind: &str) -> bool {
154    capabilities(driver)
155        .ok()
156        .and_then(|value| value.get("types").cloned())
157        .and_then(|t| t.as_array().cloned())
158        .is_some_and(|a| a.iter().any(|x| x.as_str() == Some(kind)))
159}

161/// Run a driver's `capabilities` probe and return the JSON it prints —
162/// the same bounded call [`driver_handles`] makes, exposed for the
163/// `tamias drivers` listing. `Ok` only on a zero exit *and* parseable
164/// JSON; anything else is an [`Error`] the caller renders as "unreachable".
165///
166/// # Errors
167/// Propagates spawn/timeout/parse failures.
168pub fn capabilities(driver: &Path) -> Result<serde_json::Value, Error> {
169    let mut cmd = std::process::Command::new(driver);
170    cmd.arg("capabilities");
171    let out = output_with_timeout(&mut cmd, CAPABILITIES_TIMEOUT)
172        .map_err(|e| Error::Failed(format!("{} capabilities: {e}", driver.display())))?;
173    if !out.status.success() {
174        return Err(Error::Failed(format!(
175            "{} capabilities exited {}",
176            driver.display(),
177            out.status
178        )));
179    }
180    serde_json::from_slice(&out.stdout)
181        .map_err(|e| Error::Failed(format!("{} capabilities: bad JSON: {e}", driver.display())))
182}

184/// Every discovered driver as `(type, path)` — for `doctor` and
185/// `tamias drivers`.
186#[must_use]
187pub fn discovered_drivers() -> Vec<(String, PathBuf)> {
188    let mut found: Vec<(String, PathBuf)> = Vec::new();
189    let mut scan = |dir: &Path| {
190        let Some(read) = std::fs::read_dir(dir).ok() else {
191            return;
192        };
193        for entry in read.flatten() {
194            let path = entry.path();
195            let kind = path
196                .file_name()
197                .and_then(|n| n.to_str())
198                .and_then(|n| n.strip_prefix("tamias-driver-"));
199            // `file_type` never follows symlinks — a linked driver isn't
200            // dispatchable (mirrors register_bundled_drivers).
201            let is_file = entry.file_type().is_ok_and(|t| t.is_file());
202            if let Some(kind) = kind {
203                if is_file && !found.iter().any(|(k, _)| k == kind) {
204                    found.push((kind.to_string(), path));
205                }
206            }
207        }
208    };
209    for dir in std::env::var_os("PATH")
210        .map(|p| std::env::split_paths(&p).collect::<Vec<_>>())
211        .unwrap_or_default()
212    {
213        scan(&dir);
214    }
215    scan(&drivers_dir());
216    found.sort();
217    found
218}

220/// Dispatch a verb to the content type's driver. Returns whether a driver
221/// ran (`false` = nothing to do — a `files` package, or no driver present).
222///
223/// # Errors
224/// Fails when a driver is found but exits non-zero.
225pub fn dispatch(
226    kind: &str,
227    verb: &str,
228    dir: &Path,
229    name: &str,
230    config: Option<&Path>,
231) -> Result<bool, Error> {
232    dispatch_with_timeout(kind, verb, dir, name, config, &[], DISPATCH_TIMEOUT)
233        .map(|ran| ran.is_some())
234}

236/// [`dispatch`] plus arbitrary trailing argv — the `tamias driver exec`
237/// passthrough for provider verbs beyond the lifecycle three
238/// (`hooks register`, `health`, …). `args` land after the standard
239/// `--dir`/`--name`/`--config` flags. On success the driver's captured
240/// stdout/stderr come back for the caller to render — provider verbs
241/// answer with status JSON the operator needs to see. `Ok(None)` = no
242/// driver found (same `false` as [`dispatch`]).
243///
244/// # Errors
245/// Fails when a driver is found but exits non-zero.
246pub fn dispatch_with_args(
247    kind: &str,
248    verb: &str,
249    dir: &Path,
250    name: &str,
251    config: Option<&Path>,
252    args: &[String],
253) -> Result<Option<DriverOutput>, Error> {
254    dispatch_with_timeout(kind, verb, dir, name, config, args, DISPATCH_TIMEOUT)
255}

257/// Fetch a kind driver's `config-schema` — the reserved verb that
258/// prints a JSON Schema (draft 2020-12) describing the `--config` file
259/// the driver consumes. Bounded like `capabilities` (a probe, not a
260/// lifecycle verb). `Ok(None)` = no driver found; a driver that doesn't
261/// implement the verb exits non-zero and surfaces as [`Error`] with its
262/// stderr — "unsupported" is a normal outcome, not a crash.
263///
264/// # Errors
265/// Fails when a driver is found but exits non-zero or prints non-JSON.
266pub fn config_schema(
267    kind: &str,
268    dir: &Path,
269    name: &str,
270    config: Option<&Path>,
271) -> Result<Option<serde_json::Value>, Error> {
272    let Some((stdout, _)) = dispatch_with_timeout(
273        kind,
274        "config-schema",
275        dir,
276        name,
277        config,
278        &[],
279        CAPABILITIES_TIMEOUT,
280    )?
281    else {
282        return Ok(None);
283    };
284    serde_json::from_slice(&stdout)
285        .map(Some)
286        .map_err(|e| Error::Failed(format!("config-schema: bad JSON: {e}")))
287}

289/// [`dispatch`] with an explicit bound — tests shrink it; production
290/// callers get [`DISPATCH_TIMEOUT`]. `args` is the trailing argv
291/// [`dispatch_with_args`] appends.
292fn dispatch_with_timeout(
293    kind: &str,
294    verb: &str,
295    dir: &Path,
296    name: &str,
297    config: Option<&Path>,
298    args: &[String],
299    timeout: Duration,
300) -> Result<Option<DriverOutput>, Error> {
301    // Defense in depth: manifests validate `type:`/`kind` at parse, but
302    // store records and direct callers can bypass that — refuse loudly
303    // rather than resolving `tamias-driver-<kind>` for a hostile kind.
304    if !tamias_core::is_valid_kind(kind) {
305        return Err(Error::Failed(format!("invalid driver kind {kind:?}")));
306    }
307    let Some(driver) = find_driver(kind) else {
308        return Ok(None);
309    };
310    let mut cmd = std::process::Command::new(&driver);
311    cmd.arg(verb).arg("--dir").arg(dir).arg("--name").arg(name);
312    // Operator overrides: when the package has a `deploy.d/<kind>/<name>/`
313    // drop-in dir, hand it to the driver — config-generating drivers
314    // merge it over the generated doc (T-026). `name` here is the bare
315    // record name the driver sees on argv.
316    if let Some(dir) = overrides::env_dir(kind, name) {
317        cmd.env(overrides::OVERRIDES_ENV, dir);
318    }
319    if let Some(config) = config {
320        cmd.arg("--config").arg(config);
321    }
322    cmd.args(args);
323    let out = output_with_timeout(&mut cmd, timeout)
324        .map_err(|e| Error::Failed(format!("run driver {}: {e}", driver.display())))?;
325    if !out.status.success() {
326        return Err(Error::Failed(format!(
327            "driver {} {verb} failed: {}",
328            driver.display(),
329            String::from_utf8_lossy(&out.stderr).trim()
330        )));
331    }
332    Ok(Some((out.stdout, out.stderr)))
333}

342/// Names of the driver files a payload ships under `drivers/` — regular
343/// files only, mirroring `register_bundled_drivers`'s filter, plus the
344/// platform bins under `drivers/bin/<triple>/` (reported as
345/// `bin/<triple>/<name>`). Lets callers detect (and gate) bundled drivers
346/// *before* copying or dispatching them — including binaries for other
347/// platforms, which are still bundled executable content.
348///
349/// # Errors
350/// Propagates read failures.
351pub fn bundled_drivers(payload_root: &Path) -> Result<Vec<String>, Error> {
352    let src = payload_root.join("drivers");
353    if !src.is_dir() {
354        return Ok(Vec::new());
355    }
356    let mut names = Vec::new();
357    for entry in std::fs::read_dir(&src).map_err(Error::io(format!("read {}", src.display())))? {
358        let entry = entry.map_err(Error::io("read driver entry"))?;
359        let file_type = entry
360            .file_type()
361            .map_err(Error::io(format!("stat {}", entry.path().display())))?;
362        let file_name = entry.file_name();
363        let Some(name) = file_name.to_str() else {
364            continue;
365        };
366        // `file_type` never follows symlinks — a linked `bin/` dir can't
367        // smuggle outside paths into the listing (mirrors registration).
368        if file_type.is_file() {
369            names.push(name.to_string());
370        } else if name == "bin" && file_type.is_dir() {
371            // `drivers/bin/<triple>/<name>` — compiled driver payloads.
372            names.extend(platform_bin_names(&entry.path())?);
373        }
374    }
375    names.sort();
376    Ok(names)
377}

379/// `bin/<triple>/<name>` labels for every regular file one level under
380/// each `drivers/bin/<triple>/` dir — all platforms, not just this host's.
381fn platform_bin_names(bin: &Path) -> Result<Vec<String>, Error> {
382    let mut names = Vec::new();
383    let Ok(triples) = std::fs::read_dir(bin) else {
384        return Ok(names);
385    };
386    for triple in triples {
387        let triple = triple.map_err(Error::io("read driver bin entry"))?;
388        if !triple
389            .file_type()
390            .map_err(Error::io(format!("stat {}", triple.path().display())))?
391            .is_dir()
392        {
393            continue;
394        }
395        let Some(triple_name) = triple.file_name().to_str().map(str::to_string) else {
396            continue;
397        };
398        for file in std::fs::read_dir(triple.path())
399            .map_err(Error::io(format!("read {}", triple.path().display())))?
400        {
401            let file = file.map_err(Error::io("read driver bin file"))?;
402            let is_file = file
403                .file_type()
404                .map_err(Error::io(format!("stat {}", file.path().display())))?
405                .is_file();
406            if is_file {
407                if let Some(name) = file.file_name().to_str() {
408                    names.push(format!("bin/{triple_name}/{name}"));
409                }
410            }
411        }
412    }
413    Ok(names)
414}

416/// Copy any bundled drivers a payload ships into `<tamias-home>/drivers/`
417/// so tamias discovers them: top-level `drivers/<name>` files as-is, plus
418/// the `drivers/bin/<target-triple>/` dir matching this host (see
419/// [`target_triple`]). Platform bins for other triples are skipped; a
420/// payload that ships platform dirs but none for this host is an error —
421/// the package intended a runnable compiled driver and can't provide one.
422/// Returns action lines.
423///
424/// # Errors
425/// Propagates read/copy failures, and the no-matching-platform error.
426pub fn register_bundled_drivers(payload_root: &Path) -> Result<Vec<String>, Error> {
427    let src = payload_root.join("drivers");
428    if !src.is_dir() {
429        return Ok(Vec::new());
430    }
431    let dest = drivers_dir();
432    std::fs::create_dir_all(&dest).map_err(Error::io(format!("create {}", dest.display())))?;
433    let mut actions = Vec::new();
434    let mut bin = None;
435    for entry in std::fs::read_dir(&src).map_err(Error::io(format!("read {}", src.display())))? {
436        let entry = entry.map_err(Error::io("read driver entry"))?;
437        // `file_type` never follows symlinks — a payload can't install a
438        // linked "driver" that resolves outside the payload.
439        let file_type = entry
440            .file_type()
441            .map_err(Error::io(format!("stat {}", entry.path().display())))?;
442        if file_type.is_dir() && entry.file_name() == "bin" {
443            bin = Some(entry.path());
444            continue;
445        }
446        if !file_type.is_file() {
447            continue;
448        }
449        let path = entry.path();
450        let name = path.file_name().unwrap_or_default();
451        let to = dest.join(name);
452        std::fs::copy(&path, &to).map_err(Error::io(format!("copy {}", path.display())))?;
453        set_executable(&to);
454        actions.push(format!("driver: {}", to.display()));
455    }
456    if let Some(bin) = bin {
457        actions.extend(register_platform_bins(&bin, &dest)?);
458    }
459    Ok(actions)
460}

462/// The `drivers/bin/<target-triple>/` payload: copy the dir matching this
463/// host into `dest` (flat, executable). Errors when platform dirs exist
464/// but none match — listing what the package does ship.
465fn register_platform_bins(bin: &Path, dest: &Path) -> Result<Vec<String>, Error> {
466    let mut triples = Vec::new();
467    for entry in std::fs::read_dir(bin).map_err(Error::io(format!("read {}", bin.display())))? {
468        let entry = entry.map_err(Error::io("read driver bin entry"))?;
469        if entry
470            .file_type()
471            .map_err(Error::io(format!("stat {}", entry.path().display())))?
472            .is_dir()
473        {
474            if let Some(name) = entry.file_name().to_str() {
475                triples.push(name.to_string());
476            }
477        }
478    }
479    triples.sort();
480    if triples.is_empty() {
481        return Ok(Vec::new());
482    }
483    let current = target_triple();
484    if !triples.iter().any(|t| t == current) {
485        return Err(Error::Failed(format!(
486            "package ships driver binaries for {} but not this platform ({current})",
487            triples.join(", ")
488        )));
489    }
490    let mut actions = Vec::new();
491    let platform = bin.join(current);
492    for entry in
493        std::fs::read_dir(&platform).map_err(Error::io(format!("read {}", platform.display())))?
494    {
495        let entry = entry.map_err(Error::io("read driver bin file"))?;
496        if !entry
497            .file_type()
498            .map_err(Error::io(format!("stat {}", entry.path().display())))?
499            .is_file()
500        {
501            continue;
502        }
503        let path = entry.path();
504        let name = path.file_name().unwrap_or_default();
505        let to = dest.join(name);
506        std::fs::copy(&path, &to).map_err(Error::io(format!("copy {}", path.display())))?;
507        set_executable(&to);
508        actions.push(format!("driver: {}", to.display()));
509    }
510    Ok(actions)
511}

513/// Best-effort exec bit on unix — mirrors the previous inline block.
514fn set_executable(path: &Path) {
515    #[cfg(unix)]
516    {
517        use std::os::unix::fs::PermissionsExt;
518        let _ = std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o755));
519    }
520    #[cfg(not(unix))]
521    {
522        let _ = path;
523    }
524}

533    /// Serializes tests that mutate process env (`TAMIAS_HOME`,
534    /// `TAMIAS_ALLOW_DRIVERS`, `PATH`). `cargo test` runs tests on
535    /// threads in one process, so mutations must be serialized *and*
536    /// restored even on panic.
537    static ENV: std::sync::Mutex<()> = std::sync::Mutex::new(());
538
539    /// RAII guard for [`ENV`]: restores the process environment and
540    /// working directory on drop — panic-safe. Poison is ignored: an
541    /// unwinding sibling already restored everything via its `Drop`.
542    struct EnvGuard {
543        _lock: std::sync::MutexGuard<'static, ()>,
544        env: Vec<(std::ffi::OsString, std::ffi::OsString)>,
545        cwd: std::path::PathBuf,
546    }
547
548    impl EnvGuard {
549        fn new() -> Self {
550            let lock = ENV.lock().unwrap_or_else(|e| e.into_inner());
551            let env = std::env::vars_os().collect();
552            let cwd = std::env::current_dir().unwrap();
553            Self {
554                _lock: lock,
555                env,
556                cwd,
557            }
558        }
559    }
560
561    impl Drop for EnvGuard {
562        fn drop(&mut self) {
563            for (k, _) in std::env::vars_os() {
564                if !self.env.iter().any(|(sk, _)| *sk == k) {
565                    std::env::remove_var(&k);
566                }
567            }
568            for (k, v) in &self.env {
569                std::env::set_var(k, v);
570            }
571            let _ = std::env::set_current_dir(&self.cwd);
572        }
573    }

692        // Driver answers `config-schema` with JSON.

702        let err = config_schema("csbad", tmp.path(), "n", None).unwrap_err();
703        assert!(err.to_string().contains("config-schema failed"), "{err}");
704        // Driver prints non-JSON → Err.

706        assert!(config_schema("csjunk", tmp.path(), "n", None).is_err());

710    #[test]
711    fn capability_and_error_paths() {
712        let _guard = EnvGuard::new();
713        let tmp = tempfile::tempdir().unwrap();
714        std::env::set_var("TAMIAS_HOME", tmp.path().join("home"));
715        let dd = drivers_dir();
716        // capabilities exits 1 → not matched
717        write_driver(&dd, "tamias-driver-failcap", "#!/bin/sh\nexit 1\n");
718        assert!(find_driver("nomatch").is_none());
719        // capabilities returns non-json → not matched
720        write_driver(&dd, "tamias-driver-junk", "#!/bin/sh\necho nope\n");
721        assert!(find_driver("nomatch2").is_none());
722        // a capability match maps a type to a differently-named driver
723        write_driver(
724            &dd,
725            "tamias-driver-ok",
726            "#!/bin/sh\nif [ \"$1\" = capabilities ]; then echo '{\"types\":[\"thetype\"]}'; fi\n",
727        );
728        assert!(find_driver("thetype").is_some());
729        // dispatch: driver exits 1 → Err
730        write_driver(&dd, "tamias-driver-bad", "#!/bin/sh\nexit 1\n");
731        assert!(dispatch("bad", "install", tmp.path(), "n", None).is_err());
732        // the public capabilities probe: json on success, Err otherwise
733        let caps = capabilities(&dd.join("tamias-driver-ok")).unwrap();
734        assert_eq!(caps["types"], serde_json::json!(["thetype"]));
735        assert!(capabilities(&dd.join("tamias-driver-junk")).is_err());
736        assert!(capabilities(&dd.join("tamias-driver-failcap")).is_err());
737        assert!(capabilities(&dd.join("tamias-driver-bad")).is_err());
738        write_driver(&dd, "tamias-driver-boom", "#!/bin/sh\nexit 1\n");
739        assert!(dispatch("boom", "install", tmp.path(), "n", None).is_err());
740        assert!(discovered_drivers().iter().any(|(k, _)| k == "ok"));
741        std::env::remove_var("TAMIAS_HOME");
742    }

827    fn platform_bins_select_current_triple() {
828        let _guard = EnvGuard::new();
829        let tmp = tempfile::tempdir().unwrap();
830        std::env::set_var("TAMIAS_HOME", tmp.path().join("home"));
831        let payload = tmp.path().join("payload");
832        let current = target_triple();
833        // a triple that can never be this host's (rustc has no such target)
834        let foreign = "not-a-real-target-triple";
835        assert_ne!(current, foreign);
836        let cur = payload.join("drivers/bin").join(current);
837        let oth = payload.join("drivers/bin").join(foreign);
838        std::fs::create_dir_all(&cur).unwrap();
839        std::fs::create_dir_all(&oth).unwrap();
840        std::fs::write(cur.join("tamias-driver-cur"), "#!/bin/sh\nexit 0\n").unwrap();
841        std::fs::write(oth.join("tamias-driver-oth"), "#!/bin/sh\nexit 0\n").unwrap();
842        // interpreted driver alongside the bins still registers
843        std::fs::create_dir_all(payload.join("drivers")).unwrap();
844        std::fs::write(payload.join("drivers/tamias-driver-script"), "#!/bin/sh\n").unwrap();
845        let actions = register_bundled_drivers(&payload).unwrap();
846        assert_eq!(actions.len(), 2);
847        assert!(drivers_dir().join("tamias-driver-cur").is_file());
848        assert!(drivers_dir().join("tamias-driver-script").is_file());
849        // the foreign-platform binary is never copied
850        assert!(!drivers_dir().join("tamias-driver-oth").exists());
851        // exec bit set on the registered bin
852        #[cfg(unix)]
853        {
854            use std::os::unix::fs::PermissionsExt;
855            let mode = std::fs::metadata(drivers_dir().join("tamias-driver-cur"))
856                .unwrap()
857                .permissions()
858                .mode();
859            assert_eq!(mode & 0o111, 0o111);
860        }
861        // consent detection sees both forms, labeled by platform
862        let shipped = bundled_drivers(&payload).unwrap();
863        assert!(shipped.contains(&"tamias-driver-script".to_string()));
864        assert!(shipped.contains(&format!("bin/{current}/tamias-driver-cur")));
865        assert!(shipped.contains(&format!("bin/{foreign}/tamias-driver-oth")));
866        std::env::remove_var("TAMIAS_HOME");
867    }

942        );

944        // No drop-in dir → no env injected (the probe still runs; it
945        // just records an empty variable).
946        assert!(dispatch("ovr", "deploy", tmp.path(), "pkg", None).unwrap());
947        assert_eq!(std::fs::read_to_string(&probe).unwrap(), "");
948        // With `<root>/ovr/pkg/` present the driver sees the dir.
949        let dir = root.join("ovr").join("pkg");
