2//! tamias-provider — the provider lifecycle shared by `tamias` (CLI
3//! verbs) and `tamias ui` (the Services pane). Everything here used to
4//! live private in `tamias-cli`: resolving an installed package's
5//! `provider:` block into unit/task names, host-backend state probes
6//! (systemd `systemctl` / Windows `schtasks` / macOS `launchctl` via
7//! [`ServiceBackend`]),
8//! deploy/uninstall teardown (driver verb + declarative units +
9//! shared-unit survivor re-render), the foreground `run` exec spec
10//! (env_file → env → workdir, the same order the unit renders), and
11//! the journal unit list `logs` tails.
12//!
13//! Nothing shells out to the `tamias` binary and nothing re-parses
14//! CLI output: callers get typed rows/specs/reports and do their own
15//! rendering (println, ratatui, serve JSON).
16

28/// One provider service as the Services surfaces show it — package
29/// label, service name, unit base name (`<provider>-<service>`),
30/// `systemctl is-enabled` state when systemd answers, and the declared
31/// `exec` line.
32#[derive(Clone, Debug, PartialEq, Eq)]
33pub struct ServiceRow {
34    /// Store label of the declaring package (`@scope/name@version`).
35    pub package: String,
36    /// `provider.services[].name`.
37    pub service: String,
38    /// Unit base name (no `.service` suffix).
39    pub unit: String,
40    /// `systemctl --user is-enabled` output; `None` when systemd can't
41    /// answer (no user bus, timeout).
42    pub state: Option<String>,
43    /// The declared `exec` command line.
44    pub exec: String,
45}

119/// The per-user service backend for this host — systemd `--user`
120/// units on Linux/Unix, Task Scheduler per-user tasks on Windows,
121/// launchd `gui/<uid>` agents on macOS. All render the same
122/// [`ServiceSpec`] from the `tamias-deploy` port; [`ServiceBackend::new`]
123/// picks by `cfg!` once, so callers never branch on platform
124/// themselves.

126    /// systemd user units — [`LocalDeploy`].

128    /// Task Scheduler tasks — [`WindowsDeploy`].
129    Tasks(WindowsDeploy),
130    /// launchd LaunchAgents — [`LaunchdDeploy`].
131    Launchd(LaunchdDeploy),
132}

315/// Windows `log_stream`: one task → `schtasks /query /tn <name> /v
316/// /fo list`; several → the same queries `&`-chained under
317/// `cmd.exe /c` (names are validated — no quotes — before embedding).
318fn windows_log_stream(names: &[String]) -> Result<(String, Vec<String>), Error> {
319    if names.is_empty() {
320        return Err(Error::Failed("logs: no tasks given".into()));
321    }
322    for name in names {
323        // Validates the name for both the single- and multi-task paths.
324        tamias_deploy_windows::query_args(name)?;
325    }
326    if names.len() == 1 {
327        return Ok((
328            "schtasks".into(),
329            tamias_deploy_windows::query_args(&names[0])?,
330        ));
331    }
332    let chain = names
333        .iter()
334        .map(|n| format!("schtasks /query /tn \"{n}\" /v /fo list"))
335        .collect::<Vec<_>>()
336        .join(" & ");
337    Ok(("cmd.exe".into(), vec!["/d".into(), "/c".into(), chain]))
338}

340/// Best-effort `schtasks /query /tn <unit> /fo csv /nh` — bounded at
341/// 3s; the CSV status field (`Ready`/`Running`/…) is the state,
342/// stderr text the fallback when the task can't be queried.
343#[cfg(windows)]
344fn windows_unit_state(unit: &str) -> Option<String> {
345    let args = tamias_deploy_windows::query_csv_args(unit).ok()?;
346    let mut cmd = std::process::Command::new("schtasks");
347    cmd.args(&args);
348    let out = output_with_timeout(&mut cmd, Duration::from_secs(3)).ok()?;
349    let text = String::from_utf8_lossy(&out.stdout);
350    if let Some(state) = tamias_deploy_windows::csv_status(&text) {
351        return Some(state);
352    }
353    let err = String::from_utf8_lossy(&out.stderr).trim().to_string();
354    (!err.is_empty()).then_some(err)
355}

357/// The provider block of an installed package's manifest — `None`
358/// when the payload carries no manifest or no `provider:` (a plain
359/// package has nothing declarative to deploy).
360fn provider_of(entry: &StoreEntry) -> Result<Option<Manifest>, Error> {
361    if Manifest::path_in(&entry.dir).is_none() {
362        return Ok(None);
363    }
364    let manifest = entry.manifest()?;
365    Ok(manifest.provider.is_some().then_some(manifest))
366}

368/// The provider display/identity base — `provider.name`, else the
369/// scope-qualified package name (shared by [`unit_name`] and
370/// [`list`]).
371#[must_use]
372pub fn provider_base(manifest: &Manifest) -> String {
373    manifest
374        .provider
375        .as_ref()
376        .and_then(|p| p.name.clone())
377        .unwrap_or_else(|| match &manifest.scope {
378            Some(scope) => format!("{scope}-{}", manifest.name),
379            None => manifest.name.clone(),
380        })

416/// Best-effort host-backend state probe — `systemctl --user
417/// is-enabled <unit>` on Linux/Unix, `schtasks /query` on Windows,
418/// `launchctl print gui/<uid>/<label>` on macOS; bounded at 3s, `None`
419/// when the backend can't answer (no user bus, missing `schtasks`/
420/// `launchctl`, timeout). Exit statuses don't map cleanly to "enabled"
421/// on any backend, so the state is read from stdout (stderr as
422/// fallback), not the exit status.
423#[must_use]
424pub fn unit_state(unit: &str) -> Option<String> {
425    #[cfg(windows)]
426    return windows_unit_state(unit);
427    #[cfg(target_os = "macos")]
428    return launchd_unit_state(unit);
429    #[cfg(not(any(windows, target_os = "macos")))]
430    return systemd_unit_state(unit);
431}
432
433/// `systemctl --user is-enabled <unit>` probe — `is-enabled` exits
434/// non-zero for disabled/missing units, so the state is read from
435/// stdout (stderr as fallback), not the exit status.
436#[cfg(not(any(windows, target_os = "macos")))]
437fn systemd_unit_state(unit: &str) -> Option<String> {
438    let mut cmd = std::process::Command::new("systemctl");
439    cmd.args(["--user", "is-enabled", unit]);
440    let out = output_with_timeout(&mut cmd, Duration::from_secs(3)).ok()?;
441    let state = String::from_utf8_lossy(&out.stdout).trim().to_string();
442    if state.is_empty() {
443        let err = String::from_utf8_lossy(&out.stderr).trim().to_string();
444        return (!err.is_empty()).then_some(err);
445    }
446    Some(state)
447}

449/// `launchctl print gui/<uid>/<label>` probe — `print` exits non-zero
450/// when the label isn't loaded; on success the dump's `state = …`
451/// line (running/stopped/not running) is the state, stderr text the
452/// fallback.
453#[cfg(target_os = "macos")]
454fn launchd_unit_state(unit: &str) -> Option<String> {
455    let target = tamias_deploy_launchd::service_target(unit).ok()?;
456    let mut cmd = std::process::Command::new("launchctl");
457    cmd.arg("print").arg(&target);
458    let out = output_with_timeout(&mut cmd, Duration::from_secs(3)).ok()?;
459    let text = String::from_utf8_lossy(&out.stdout);
460    if let Some(state) = text.lines().find_map(|l| l.trim().strip_prefix("state = ")) {
461        return Some(state.trim().to_string());
462    }
463    if out.status.success() {
464        return Some("loaded".into());
465    }
466    let err = String::from_utf8_lossy(&out.stderr).trim().to_string();
467    (!err.is_empty()).then_some(err)
468}

470/// Every installed package's provider view, in store order —
471/// `provider list`'s data source. Packages without a `provider:`
472/// block come back with `provider: None` (callers filter for
473/// `--all`-style output).
474///
475/// # Errors
476/// Propagates store listing failures; per-entry manifest problems
477/// degrade to `provider: None` (same posture as the CLI).
478pub fn list(store_root: &Path) -> Result<Vec<ProviderPackage>, Error> {
479    let mut packages = Vec::new();
480    for entry in Store::new(store_root).entries()? {
481        let manifest = entry.manifest().ok();
482        let provider = manifest.and_then(|m| {
483            let p = m.provider.as_ref()?;
484            let base = provider_base(&m);
485            let services = p
486                .services
487                .iter()
488                .map(|s| {
489                    let unit = unit_name(&m, &s.name);
490                    ServiceRow {
491                        package: entry.label(),
492                        service: s.name.clone(),
493                        state: unit_state(&unit),
494                        unit,
495                        exec: s.exec.clone(),
496                    }
497                })
498                .collect();
499            Some(ProviderView {
500                base,
501                services,
502                ingress: p.ingress.clone(),
503                verbs: p.verbs.clone(),
504                health: p.health.as_ref().map(|h| h.cmd.clone()),
505            })
506        });
507        packages.push(ProviderPackage {
508            label: entry.label(),
509            dir: entry.dir.clone(),
510            provider,
511        });
512    }
513    Ok(packages)
514}

529/// The unit base names an installed package's provider declares —
530/// `logs`' unit list. Errors when the package has no `provider:`
531/// block or declares no services.
532///
533/// # Errors
534/// Store/manifest failures, or "no provider services".
535pub fn units_of(entry: &StoreEntry) -> Result<Vec<String>, Error> {
536    let Some(manifest) = provider_of(entry)? else {
537        return Err(Error::Failed(format!(
538            "{}: no provider: block",
539            entry.label()
540        )));
541    };
542    let units: Vec<String> = manifest
543        .provider
544        .as_ref()
545        .map(|p| {
546            p.services
547                .iter()
548                .map(|s| unit_name(&manifest, &s.name))
549                .collect()
550        })
551        .unwrap_or_default();
552    if units.is_empty() {
553        return Err(Error::Failed(format!(
554            "{}: provider declares no services",
555            entry.label()
556        )));
557    }
558    Ok(units)
559}

561/// `deploy <name>` — the installed package's provider deploy step:
562/// the kind driver's `deploy` verb *and*, independently, every
563/// `provider.services[]` entry rendered as a per-user systemd unit.
564/// Units are written + daemon-reloaded but not enabled.
565///
566/// # Errors
567/// Store lookup, driver dispatch, spec render, or unit write failures.
568pub fn deploy(store_root: &Path, name: &str, config: Option<&Path>) -> Result<DeployReport, Error> {
569    let entry = Store::new(store_root).find(name)?;
570    let kind = entry.record.kind.as_deref().unwrap_or("files");
571    let mut report = DeployReport::default();
572    let driver_ran =
573        tamias_driver::dispatch(kind, "deploy", &entry.dir, &entry.record.name, config)?;
574    if driver_ran {
575        report.lines.push(format!("deployed {}", entry.label()));
576    }
577    let mut deployed = false;
578    if let Some(manifest) = provider_of(&entry)? {
579        let deploy = ServiceBackend::new();
580        if let Some(provider) = &manifest.provider {
581            for service in &provider.services {
582                let spec = service_spec(&entry, &manifest, service)?;
583                let path = deploy.install(&spec)?;
584                deployed = true;
585                report.lines.push(format!(
586                    "unit {} → {} (enable: tamias service enable {})",
587                    spec.name,
588                    path.display(),
589                    spec.name
590                ));
591            }
592        }
593        if deployed {
594            // Best-effort: a missing user bus must not fail the deploy —
595            // the unit files are already on disk.
596            let _ = deploy.daemon_reload();
597        }
598        // Ingress hooks stay manual: `public_url` fronts a tunnel the
599        // operator confirms first — auto-registering forge hooks at
600        // deploy time would point repos at whatever URL happens to be
601        // declared.
602        let hooks: usize = manifest
603            .provider
604            .as_ref()
605            .map(|p| p.ingress.iter().map(|i| i.hooks.len()).sum())
606            .unwrap_or(0);
607        if hooks > 0 {
608            report.warnings.push(format!(
609                "{} ingress hook(s) declared — review with `tamias driver exec {} hooks register --dry-run`, then drop --dry-run to register",
610                hooks,
611                entry.label()
612            ));
613        }
614    }
615    if !driver_ran && !deployed {
616        report.lines.push(format!(
617            "{}: nothing to deploy (type {kind})",
618            entry.label()
619        ));
620    }
621    Ok(report)
622}

624/// Of `names` (unit base names a just-uninstalled package declared),
625/// the ones *other* still-installed providers also declare — unit →
626/// `(spec, declaring package label)`, first declarer in sorted store
627/// order wins. A `None` spec means the survivor's `exec` won't parse:
628/// the unit is left untouched. Store/manifest read failures degrade
629/// to "not declared" so a broken sibling never blocks an uninstall.
630fn surviving_units(
631    store_root: &Path,
632    names: &BTreeSet<String>,
633    warnings: &mut Vec<String>,
634) -> std::collections::BTreeMap<String, Option<(ServiceSpec, String)>> {
635    let mut out = std::collections::BTreeMap::new();
636    if names.is_empty() {
637        return out;
638    }
639    let entries = match Store::new(store_root).entries() {
640        Ok(entries) => entries,
641        Err(e) => {
642            warnings.push(format!("could not rescan store for shared units: {e}"));
643            return out;
644        }
645    };
646    for entry in entries {
647        let manifest = match provider_of(&entry) {
648            Ok(Some(manifest)) => manifest,
649            Ok(None) => continue,
650            Err(e) => {
651                warnings.push(format!(
652                    "skipping {} in shared-unit scan: {e}",
653                    entry.label()
654                ));
655                continue;
656            }
657        };
658        let Some(provider) = manifest.provider.as_ref() else {
659            continue;
660        };
661        for service in &provider.services {
662            let unit = unit_name(&manifest, &service.name);
663            if !names.contains(&unit) || out.contains_key(&unit) {
664                continue;
665            }
666            match service_spec(&entry, &manifest, service) {
667                Ok(spec) => {
668                    out.insert(unit, Some((spec, entry.label())));
669                }
670                Err(e) => {
671                    warnings.push(format!(
672                        "{} service '{}': bad exec for shared unit {unit}: {e}",
673                        entry.label(),
674                        service.name
675                    ));
676                    out.insert(unit, None);
677                }
678            }
679        }
680    }
681    out
682}

684/// `uninstall <name>` — provider teardown + drop the store entry:
685/// driver `uninstall` first (a failing driver keeps the store entry),
686/// then declarative units are removed unless a *surviving* installed
687/// provider still declares the same unit name — that unit is
688/// re-rendered from the survivor's spec instead.
689///
690/// The driver `uninstall` verb is **version-aware** (T-029): drivers
691/// key shared operator state (`~/.config/<product>/…`) on the bare
692/// `--name`, so the verb runs only when the entry being dropped is the
693/// last installed one sharing that name — removing one of several
694/// versions must not tear down state the survivors still use. The
695/// store entry itself is always dropped; shared units already follow
696/// the same rule via [`surviving_units`].
697///
698/// # Errors
699/// Store lookup, driver dispatch, or store removal failures.
700pub fn uninstall(store_root: &Path, name: &str) -> Result<DeployReport, Error> {
701    let store = Store::new(store_root);
702    let entry = store.find(name)?;
703    let kind = entry.record.kind.as_deref().unwrap_or("files");
704    let mut report = DeployReport::default();
705    // Capture the provider spec before teardown — the store entry is
706    // dropped below, and the manifest is needed both to remove this
707    // package's units and to tell whether a *surviving* provider still
708    // declares them (unit names can be shared: several packages may
709    // render the same `<provider>-<service>` unit).
710    let manifest = provider_of(&entry)?;
711    let unit_names: Vec<String> = manifest
712        .as_ref()
713        .map(|m| {
714            m.provider
715                .as_ref()
716                .map(|p| p.services.iter().map(|s| unit_name(m, &s.name)).collect())
717                .unwrap_or_default()
718        })
719        .unwrap_or_default();
720    // Teardown runs before the entry is dropped: a failing driver keeps
721    // the store entry so the uninstall can be retried instead of
722    // leaking provider-side resources under a false "uninstalled". The
723    // verb itself is version-aware (T-029): while another same-name
724    // install remains, the driver is not dispatched at all — its
725    // name-keyed shared state (`~/.config/<product>/…`) belongs to the
726    // survivors.
727    let siblings = store.siblings_of(&entry)?;
728    if siblings.is_empty() {
729        if !tamias_driver::dispatch(kind, "uninstall", &entry.dir, &entry.record.name, None)?
730            && kind != "files"
731        {
732            report.warnings.push(format!(
733                "no driver for type '{kind}'; provider resources may remain"
734            ));
735        }
736    } else {
737        report.warnings.push(format!(
738            "driver uninstall for {} skipped — sibling installs remain: {}",
739            entry.label(),
740            siblings
741                .iter()
742                .map(StoreEntry::label)
743                .collect::<Vec<_>>()
744                .join(", ")
745        ));
746    }
747    let record = store.remove(name)?;
748    // Declarative provider services are tamias-native — remove their
749    // units too, unless a surviving installed provider still declares
750    // the same unit name. Units are best-effort per name: a failure
751    // warns but doesn't fail the uninstall — the store entry is
752    // already dropped.
753    let deploy = ServiceBackend::new();
754    let survivors = surviving_units(
755        store_root,
756        &unit_names.iter().cloned().collect::<BTreeSet<_>>(),
757        &mut report.warnings,
758    );
759    for unit in &unit_names {
760        match survivors.get(unit) {
761            Some(Some((spec, label))) => match deploy.install(spec) {
762                Ok(path) => report.lines.push(format!(
763                    "unit {unit} is shared — re-rendered from {label} → {}",
764                    path.display()
765                )),
766                Err(e) => report
767                    .warnings
768                    .push(format!("could not refresh shared unit {unit}: {e}")),
769            },
770            Some(None) => report.warnings.push(format!(
771                "unit {unit} is still declared but its surviving spec is invalid; left untouched"
772            )),
773            None => {
774                if let Err(e) = deploy.uninstall(unit) {
775                    report
776                        .warnings
777                        .push(format!("could not remove unit {unit}: {e}"));
778                }
779            }
780        }
781    }
782    if !unit_names.is_empty() {
783        // Best-effort: a missing user bus must not fail the uninstall —
784        // the unit files are already written/removed.
785        let _ = deploy.daemon_reload();
786    }
787    report
788        .lines
789        .push(format!("uninstalled {}@{}", record.name, record.resolved));
790    Ok(report)
791}

793/// Strip one layer of matching surrounding quotes from an env_file
794/// value — systemd unquotes `KEY="a b"`/`KEY='a b'` the same way.
795#[must_use]
796pub fn unquote_env(value: &str) -> String {
797    let bytes = value.as_bytes();
798    if bytes.len() >= 2
799        && ((bytes[0] == b'"' && bytes[bytes.len() - 1] == b'"')
800            || (bytes[0] == b'\'' && bytes[bytes.len() - 1] == b'\''))
801    {
802        return value[1..value.len() - 1].to_string();
803    }
804    value.to_string()
805}

807/// `run <pkg> [service] [-- args…]`'s resolved exec — the provider
808/// service's argv (plus `extra`), env (`env_file` parsed as `K=V`
809/// lines, inline `env` overriding — the same order systemd applies),
810/// and workdir (`~` expanded, else the payload dir). This is the
811/// no-systemd counterpart of the rendered unit.
812///
813/// # Errors
814/// Store lookup, missing `provider:`/service name, bad `exec` split,
815/// or unreadable `env_file`.
816pub fn exec_spec(
817    store_root: &Path,
818    name: &str,
819    service: Option<&str>,
820    extra: &[String],
821) -> Result<ExecSpec, Error> {
822    let entry = Store::new(store_root).find(name)?;
823    let manifest = provider_of(&entry)?.ok_or_else(|| {
824        Error::Failed(format!(
825            "{}: no provider: block — `run` needs provider.services",
826            entry.label()
827        ))
828    })?;
829    let provider = manifest
830        .provider
831        .as_ref()
832        .ok_or_else(|| Error::Failed(format!("{}: no provider: block", entry.label())))?;
833    let names = || {
834        provider
835            .services
836            .iter()
837            .map(|s| s.name.as_str())
838            .collect::<Vec<_>>()
839            .join(", ")
840    };
841    let svc = match service {
842        Some(s) => provider
843            .services
844            .iter()
845            .find(|x| x.name == s)
846            .ok_or_else(|| {
847                Error::Failed(format!(
848                    "{}: no provider service '{s}' (have: {})",
849                    entry.label(),
850                    names()
851                ))
852            })?,
853        None => match provider.services.as_slice() {
854            [] => {
855                return Err(Error::Failed(format!(
856                    "{}: provider declares no services",
857                    entry.label()
858                )));
859            }
860            [one] => one,
861            many => {
862                return Err(Error::Failed(format!(
863                    "{}: provider declares {} services — pass one: {}",
864                    entry.label(),
865                    many.len(),
866                    names()
867                )));
868            }
869        },
870    };
871    let mut argv = split_exec(&svc.exec)?;
872    argv.extend(extra.iter().cloned());
873    // env_file first, inline env overrides — systemd's order.
874    let mut vars: Vec<(String, String)> = Vec::new();
875    if let Some(env_file) = &svc.env_file {
876        let path = PathBuf::from(expand_home(env_file));
877        // The unit renders `EnvironmentFile=-` (optional) — a missing
878        // file is not an error here either.
879        if path.is_file() {
880            let text = std::fs::read_to_string(&path)
881                .map_err(Error::io(format!("read {}", path.display())))?;
882            for line in text.lines() {
883                let line = line.trim();
884                if line.is_empty() || line.starts_with('#') {
885                    continue;
886                }
887                match line.split_once('=') {
888                    Some((k, v)) => vars.push((k.trim().to_string(), unquote_env(v.trim()))),
889                    None => {
890                        eprintln!(
891                            "tamias: warning — ignoring malformed env_file line in {}",
892                            path.display()
893                        );
894                    }
895                }
896            }
897        }
898    }
899    vars.extend(svc.env.iter().map(|(k, v)| (k.clone(), v.clone())));
900    let workdir = svc
901        .workdir
902        .as_deref()
903        .map(expand_home)
904        .map(PathBuf::from)
905        .unwrap_or_else(|| entry.dir.clone());
906    Ok(ExecSpec {
907        service: svc.name.clone(),
908        argv,
909        env: vars,
910        workdir,
911    })
912}

914/// A host-backend lifecycle verb on a unit/task base name —
915/// `tamias service enable|disable` equivalents plus start/stop/
916/// restart for the interactive surfaces.
917///
918/// # Errors
919/// [`Error::Failed`] on an invalid unit name; propagates the backend
920/// failure.
921pub fn unit_action(unit: &str, action: UnitAction) -> Result<(), Error> {
922    let deploy = ServiceBackend::new();
923    match action {
924        UnitAction::Start => deploy.start(unit),
925        UnitAction::Stop => deploy.stop(unit),
926        UnitAction::Restart => deploy.restart(unit),
927        UnitAction::Enable => deploy.enable(unit),
928        UnitAction::Disable => deploy.disable(unit),
929    }
930}

1038                }
1039            }

1041                std::env::set_var(k, v);
1042            }
1043        }
