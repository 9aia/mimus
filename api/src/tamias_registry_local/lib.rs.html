19/// A dir-backed registry.
20#[derive(Clone, Debug)]
21pub struct LocalRegistry {
22    root: PathBuf,
23    /// Upload cap enforced in [`Registry::publish`] — `Some` rejects a
24    /// packed `.tgz` larger than the configured `limits.max_upload`.
25    /// `None` (the default) leaves direct dir writes uncapped; the
26    /// `registry serve` HTTP surface enforces its own cap regardless.
27    max_upload: Option<u64>,
28    /// Index retention (P-024) — `Some(keep)` prunes the published
29    /// package's older index entries beyond the `keep` most recent
30    /// semver versions on every publish. `None` (the default) keeps
31    /// every version. Payload blobs are never pruned.
32    retention: Option<u64>,
33}
34
35impl LocalRegistry {
36    /// A registry under `root` (created lazily on first publish).
37    #[must_use]
38    pub fn new(root: impl Into<PathBuf>) -> Self {
39        Self {
40            root: root.into(),
41            max_upload: None,
42            retention: None,
43        }
44    }
45
46    /// Cap publishes at `bytes` — the configured `max_upload`.
47    #[must_use]
48    pub fn with_max_upload(mut self, bytes: u64) -> Self {
49        self.max_upload = Some(bytes);
50        self
51    }
52
53    /// Retain only the `keep` most recent semver versions per package
54    /// in the index (P-024) — `None` disables pruning. The resolved
55    /// `retention.keep` from config (or `publish --keep`).
56    #[must_use]
57    pub fn with_retention(mut self, keep: Option<u64>) -> Self {
58        self.retention = keep;
59        self
60    }
61
62    /// The root dir.
63    #[must_use]
64    pub fn root(&self) -> &Path {
65        &self.root
66    }
67
68    fn index_path(&self) -> PathBuf {
69        self.root.join("index.json")
70    }

73impl Registry for LocalRegistry {
74    fn entries(&self) -> Result<Vec<RegistryEntry>, Error> {
75        let path = self.index_path();
76        if !path.is_file() {
77            return Ok(Vec::new());
78        }
79        let text =
80            fs::read_to_string(&path).map_err(Error::io(format!("read {}", path.display())))?;
81        serde_json::from_str(&text)
82            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))
83    }
84
85    fn publish(
86        &self,
87        payload_root: &Path,
88        manifest: &Manifest,
89        signature: Option<&str>,
90    ) -> Result<PublishOutcome, Error> {
91        // The manifest may not have been validated upstream — every field
92        // embedded in the blob path must be a single safe component.
93        if !is_valid_name(&manifest.name) {
94            return Err(Error::Failed(format!(
95                "unsafe package name '{}'",
96                manifest.name
97            )));
98        }
99        if let Some(scope) = &manifest.scope {
100            if !is_valid_name(scope) {
101                return Err(Error::Failed(format!("unsafe scope '{scope}'")));
102            }
103        }
104        let scoped = match &manifest.scope {
105            Some(scope) => format!("@{scope}/{}", manifest.name),
106            None => manifest.name.clone(),
107        };
108        let version = manifest
109            .version
110            .clone()
111            .unwrap_or_else(|| "0.0.0".to_string());
112        if !is_valid_version(&version) {
113            return Err(Error::Failed(format!(
114                "unsafe version '{version}' — expected [A-Za-z0-9][A-Za-z0-9._+-]* with no '..'"
115            )));
116        }
117        let rel = format!("{scoped}/{version}.tgz");
118        let out = self.root.join(&rel);
119        let parent = out.parent().unwrap_or(Path::new("."));
120        fs::create_dir_all(parent).map_err(Error::io(format!("create {}", parent.display())))?;
121        // Blob lands atomically — a crash mid-pack leaves a stray
122        // `*.tmp`, never a truncated `.tgz` at the canonical path.
123        pack_tgz_atomic(payload_root, &out)?;
124        // The size probe runs post-pack/pre-index: an over-limit blob is
125        // removed before it can be served or referenced.
126        if let Some(max) = self.max_upload {
127            let size = fs::metadata(&out)
128                .map_err(Error::io(format!("stat {}", out.display())))?
129                .len();
130            if size > max {
131                let _ = fs::remove_file(&out);
132                return Err(Error::Failed(format!(
133                    "payload tarball {} is {size} bytes — exceeds the {max}-byte upload limit",
134                    out.display()
135                )));
136            }
137        }
138        let entry = RegistryEntry {
139            name: manifest.name.clone(),
140            scope: manifest.scope.clone(),
141            version: manifest.version.clone(),
142            kind: manifest.kind.clone(),
143            digest: dir_digest(payload_root)?,
144            signature: signature.map(str::to_string),
145            publisher: None,
146            file: rel,
147        };
148        // The index read-modify-write is serialized across processes —
149        // concurrent publishes can't lose each other's entries.
150        fs::create_dir_all(&self.root)
151            .map_err(Error::io(format!("create {}", self.root.display())))?;
152        let _lock = IndexLock::acquire(&self.root)?;
153        let mut entries = self.entries()?;
154        entries.retain(|e| {
155            !(e.name == entry.name && e.scope == entry.scope && e.version == entry.version)
156        });
157        entries.push(entry.clone());
158        entries
159            .sort_by(|a, b| (&a.scope, &a.name, &a.version).cmp(&(&b.scope, &b.name, &b.version)));
160        // Retention (P-024) — index-entry prune only; the `.tgz`
161        // blobs of pruned versions stay on disk.
162        let pruned = self
163            .retention
164            .map(|keep| {
165                apply_retention(
166                    &mut entries,
167                    usize::try_from(keep).unwrap_or(usize::MAX),
168                    &entry,
169                )
170            })
171            .unwrap_or_default();
172        let text = serde_json::to_string_pretty(&entries)
173            .map_err(|e| Error::Failed(format!("serialize index: {e}")))?;
174        write_atomic(&self.index_path(), text.as_bytes())?;
175        Ok(PublishOutcome { entry, pruned })
176    }
177}

179/// `index.lock` — an atomic-create (`create_new`) lockfile serializing
180/// the registry index read-modify-write. A lock older than
181/// [`STALE_AFTER`] is treated as abandoned (holder crashed) and stolen;
182/// `create_new` still guarantees a single winner of the steal race.
183/// Residual: a holder legitimately keeping the lock past
184/// [`STALE_AFTER`] (a multi-minute index update) could have its lock
185/// stolen — index updates here are sub-second, so this is theoretical.
186struct IndexLock {
187    path: PathBuf,
188}
189
190/// A lockfile untouched for this long is considered stale.
191const STALE_AFTER: Duration = Duration::from_secs(30);
192/// How long `acquire` waits for a live holder before giving up.
193const LOCK_TIMEOUT: Duration = Duration::from_secs(10);
194
195impl IndexLock {
196    /// Take `root/index.lock`, waiting out live holders.
197    ///
198    /// # Errors
199    /// Fails on I/O errors or when the lock stays held past
200    /// [`LOCK_TIMEOUT`].
201    fn acquire(root: &Path) -> Result<Self, Error> {
202        let path = root.join("index.lock");
203        let deadline = Instant::now() + LOCK_TIMEOUT;
204        loop {
205            match fs::OpenOptions::new()
206                .write(true)
207                .create_new(true)
208                .open(&path)
209            {
210                Ok(mut f) => {
211                    use std::io::Write;
212                    let _ = writeln!(f, "{}", std::process::id());
213                    return Ok(Self { path });
214                }
215                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
216                    if lock_stale(&path) {
217                        let _ = fs::remove_file(&path);
218                        continue;
219                    }
220                    if Instant::now() >= deadline {
221                        return Err(Error::Failed(format!(
222                            "registry index locked ({})",
223                            path.display()
224                        )));
225                    }
226                    std::thread::sleep(Duration::from_millis(20));
227                }
228                Err(e) => return Err(Error::io(format!("lock {}", path.display()))(e)),
229            }
230        }
231    }
232}
233
234impl Drop for IndexLock {
235    fn drop(&mut self) {
236        let _ = fs::remove_file(&self.path);
237    }
238}

240/// Whether `path`'s mtime is older than [`STALE_AFTER`] — or unreadable,
241/// in which case removing it is harmless (the create race decides).
242fn lock_stale(path: &Path) -> bool {
243    fs::metadata(path)
244        .and_then(|m| m.modified())
245        .map(|t| t.elapsed().is_ok_and(|age| age > STALE_AFTER))
246        .unwrap_or(true)
247}

249/// Default PUT cap — mirrors `tamias-config`'s `DEFAULT_MAX_UPLOAD`
250/// (512 MiB); bigger blobs can't be installed anyway.

253/// Per-read timeout on a connection — an idle client can't stall the
254/// single-threaded accept loop (each `read` gets this long).
255const READ_TIMEOUT: Duration = Duration::from_secs(30);

257/// Env var that turns on bearer auth for the dev server — mirrors the
258/// client's `TAMIAS_REGISTRY_TOKEN`. Unset = open (dev default).
259const SERVE_TOKEN_ENV: &str = "TAMIAS_REGISTRY_SERVE_TOKEN";

261/// Serve `root` over a tiny blocking HTTP surface until the process stops.
262/// `GET /<key>` reads a blob; `PUT /<key>` writes one. The registry root is
263/// the blob namespace (paths sanitized — no traversal).
264///
265/// **Dev surface, not a real registry.** Anyone who can reach the port can
266/// read and overwrite every blob — including `index.json` (the `PUT /`
267/// sanitize maps the root to it, and `HttpRegistry` publishes through this
268/// protocol, so it stays allowed). Bind loopback, or set
269/// `TAMIAS_REGISTRY_SERVE_TOKEN` to require `Authorization: Bearer <token>`.
270///
271/// # Errors
272/// Propagates the bind failure; per-request errors are logged, not fatal.
273pub fn serve_http(addr: &str, root: &Path) -> Result<(), Error> {
274    serve_http_with_limits(addr, root, MAX_PUT)
275}
276
277/// [`serve_http`] with an explicit upload cap — the configured
278/// `limits.max_upload` (flag > config > this default).
279///
280/// # Errors
281/// Propagates the bind failure; per-request errors are logged, not fatal.
282pub fn serve_http_with_limits(addr: &str, root: &Path, max_upload: u64) -> Result<(), Error> {
283    let token = std::env::var(SERVE_TOKEN_ENV)
284        .ok()
285        .filter(|t| !t.is_empty());
286    serve_http_token(addr, root, token.as_deref(), max_upload)
287}

289/// [`serve_http_with_limits`] with an explicit bearer token — `Some`
290/// requires `Authorization: Bearer <token>` on every request. Split out
291/// so tests don't race on the process env.
292fn serve_http_token(
293    addr: &str,
294    root: &Path,
295    token: Option<&str>,
296    max_upload: u64,
297) -> Result<(), Error> {
298    let listener = std::net::TcpListener::bind(addr).map_err(Error::io(format!("bind {addr}")))?;
299    fs::create_dir_all(root).map_err(Error::io(format!("create {}", root.display())))?;
300    eprintln!(
301        "tamias registry serving {} at http://{addr}/ — dev server: \
302         unauthenticated blob store (set {SERVE_TOKEN_ENV} for bearer auth)",
303        root.display()
304    );
305    // `--addr 0.0.0.0` (or any non-loopback bind) with no bearer token
306    // exposes the writable store to the network — say so loudly.
307    let exposed = addr
308        .to_socket_addrs()
309        .map(|mut addrs| addrs.any(|a| !a.ip().is_loopback()))
310        .unwrap_or(false);
311    if exposed && token.is_none() {
312        eprintln!(
313            "warning: registry is bound to a non-loopback address ({addr}) with \
314             no {SERVE_TOKEN_ENV} — remote clients can read AND write the \
315             entire registry (every blob, including index.json)"
316        );
317    }
318    for stream in listener.incoming() {
319        match stream {
320            Ok(mut stream) => {
321                if let Err(e) = handle(&mut stream, root, token, max_upload) {
322                    eprintln!("tamias registry: {e}");
323                }
324            }
325            Err(e) => eprintln!("tamias registry: accept: {e}"),
326        }
327    }
328    Ok(())
329}

331fn handle(
332    stream: &mut std::net::TcpStream,
333    root: &Path,
334    token: Option<&str>,
335    max_upload: u64,
336) -> Result<(), Error> {
337    use std::io::{BufRead, BufReader, Read};
338
339    // Bound every read on this connection — headers AND body — so an
340    // idle or slow-loris client can't park the accept loop.
341    stream
342        .set_read_timeout(Some(READ_TIMEOUT))
343        .map_err(Error::io("set read timeout"))?;
344    let clone = stream.try_clone().map_err(Error::io("clone stream"))?;
345    let mut reader = BufReader::new(clone);
346    let mut request_line = String::new();
347    reader
348        .read_line(&mut request_line)
349        .map_err(Error::io("read request"))?;
350    let mut parts = request_line.split_whitespace();
351    let method = parts.next().unwrap_or("").to_string();
352    let raw_path = parts.next().unwrap_or("/").to_string();
353    let mut content_length = 0u64;
354    let mut authorization = String::new();
355    loop {
356        let mut line = String::new();
357        reader
358            .read_line(&mut line)
359            .map_err(Error::io("read header"))?;
360        if line == "\r\n" || line == "\n" || line.is_empty() {
361            break;
362        }
363        // Match the header name case-insensitively but keep the VALUE
364        // verbatim — bearer tokens are case-sensitive.
365        let lower = line.to_ascii_lowercase();
366        if let Some(v) = lower.strip_prefix("content-length:") {
367            content_length = v.trim().parse().unwrap_or(0);
368        } else if lower.starts_with("authorization:") {
369            authorization = line["authorization:".len()..].trim().to_string();
370        }
371    }
372    // Optional bearer auth — `Some(token)` means the operator opted in.
373    // `get(..)` slicing: a hostile non-ASCII header can't panic the loop.
374    if let Some(token) = token {
375        let ok = authorization
376            .get(..7)
377            .is_some_and(|s| s.eq_ignore_ascii_case("bearer "))
378            && authorization.get(7..).is_some_and(|t| t == token);
379        if !ok {
380            return respond(stream, 401, b"unauthorized");
381        }
382    }
383    let Some(key) = sanitize_key(&raw_path) else {
384        return respond(stream, 400, b"bad path");
385    };
386    match method.as_str() {
387        "GET" => match fs::read(root.join(&key)) {
388            Ok(bytes) => respond(stream, 200, &bytes),
389            Err(_) => respond(stream, 404, b"not found"),
390        },
391        "PUT" => {
392            // Reject the declared length BEFORE allocating — a huge
393            // Content-Length must not become a huge vec.
394            if content_length > max_upload {
395                return respond(stream, 413, b"payload too large");
396            }
397            let mut body = vec![0u8; content_length as usize];
398            reader
399                .read_exact(&mut body)
400                .map_err(Error::io("read body"))?;
401            let path = root.join(&key);
402            let parent = path.parent().unwrap_or(Path::new("."));
403            fs::create_dir_all(parent).map_err(Error::io("create blob dir"))?;
404            fs::write(&path, &body).map_err(Error::io(format!("write {}", path.display())))?;
405            respond(stream, 200, b"ok")
406        }
407        _ => respond(stream, 405, b"method not allowed"),
408    }
409}

411/// A request path → a safe relative key (no traversal; `/` → `index.json`).
412fn sanitize_key(raw: &str) -> Option<PathBuf> {
413    let trimmed = raw.split('?').next().unwrap_or(raw).trim_start_matches('/');
414    let mut rel = PathBuf::new();
415    for component in Path::new(trimmed).components() {
416        match component {
417            std::path::Component::Normal(p) => rel.push(p),
418            std::path::Component::CurDir => {}
419            _ => return None,
420        }
421    }
422    if rel.as_os_str().is_empty() {
423        rel.push("index.json");
424    }
425    Some(rel)
426}

428fn respond(stream: &mut std::net::TcpStream, code: u16, body: &[u8]) -> Result<(), Error> {
429    use std::io::Write;
430    let reason = match code {
431        200 => "OK",
432        400 => "Bad Request",
433        401 => "Unauthorized",
434        404 => "Not Found",
435        405 => "Method Not Allowed",
436        413 => "Payload Too Large",
437        _ => "Error",
438    };
439    let head = format!(
440        "HTTP/1.1 {code} {reason}\r\nContent-Length: {}\r\nContent-Type: application/octet-stream\r\nConnection: close\r\n\r\n",
441        body.len()
442    );
443    stream
444        .write_all(head.as_bytes())
445        .map_err(Error::io("write head"))?;
446    stream.write_all(body).map_err(Error::io("write body"))?;
447    stream.flush().map_err(Error::io("flush"))?;
448    Ok(())
449}
