2//! tamias-registry-cf — a registry over an HTTP blob store: the Cloudflare
3//! Worker (`tamias-deploy-cf` ships `worker/index.js`) backed by R2, or any host speaking the same
4//! protocol (`GET /<key>`, `PUT /<key>`). The **client** maintains the
5//! `index.json` (read-modify-write); the server is a dumb, token-guarded
6//! blob store. Auth is a bearer token — from `token_env` / cuora, never
7//! inline.
8
9#![cfg_attr(test, allow(clippy::unwrap_used))]

12use std::sync::atomic::{AtomicU64, Ordering};

17/// Largest `.tgz` blob a publish will PUT by default — mirrors
18/// `tamias-config`'s `DEFAULT_MAX_UPLOAD` (512 MiB);
19/// [`HttpRegistry::with_max_upload`] overrides with the configured
20/// `limits.max_upload`. Bigger payloads can't be installed anyway.
21const DEFAULT_MAX_UPLOAD: u64 = 512 * 1024 * 1024;
22
23/// Largest response body a blob GET buffers — mirrors the fetch-side
24/// `DEFAULT_MAX_DOWNLOAD` in `tamias-source-http` (512 MiB). The client
25/// only GETs `index.json` today (capped tighter by `MAX_INDEX`); this
26/// guards any future blob read.
27const DEFAULT_MAX_DOWNLOAD: u64 = 512 * 1024 * 1024;
28
29/// `index.json` is a small catalog — cap its GET well below the blob
30/// limit so a hostile/buggy store can't OOM the client.
31const MAX_INDEX: u64 = 16 * 1024 * 1024;
32
33/// A registry over an http(s) blob store.
34pub struct HttpRegistry {
35    base: String,
36    agent: ureq::Agent,
37    token: Option<String>,
38    /// Upload cap checked against the packed tarball before the PUT —
39    /// the configured `limits.max_upload` when the caller threads it
40    /// through. The server stays authoritative regardless.
41    max_upload: u64,
42    /// Index retention (P-024) — `Some(keep)` prunes the published
43    /// package's older index entries beyond the `keep` most recent
44    /// semver versions on every publish (client-side: this client owns
45    /// the index read-modify-write; the Worker is a dumb blob store).
46    /// `None` (the default) keeps every version. Payload blobs are
47    /// never pruned.
48    retention: Option<u64>,
49}
50
51impl HttpRegistry {
52    /// A registry at `base` (trailing `/` trimmed).
53    #[must_use]
54    pub fn new(base: impl Into<String>) -> Self {
55        Self {
56            base: base.into().trim_end_matches('/').to_string(),
57            agent: ureq::Agent::new_with_config(
58                ureq::Agent::config_builder()
59                    .http_status_as_error(false)
60                    .timeout_connect(Some(std::time::Duration::from_secs(15)))
61                    .timeout_global(Some(std::time::Duration::from_secs(300)))
62                    .build(),
63            ),
64            token: None,
65            max_upload: DEFAULT_MAX_UPLOAD,
66            retention: None,
67        }
68    }
69
70    /// A bearer token for auth (from `token_env`/cuora).
71    #[must_use]
72    pub fn with_token(mut self, token: impl Into<String>) -> Self {
73        self.token = Some(token.into());
74        self
75    }
76
77    /// Cap publishes at `bytes` — the configured `limits.max_upload`.
78    /// Best-effort client preflight; the server enforces its own limit.
79    #[must_use]
80    pub fn with_max_upload(mut self, bytes: u64) -> Self {
81        self.max_upload = bytes;
82        self
83    }
84
85    /// Retain only the `keep` most recent semver versions per package
86    /// in the index (P-024) — `None` disables pruning. The resolved
87    /// `retention.keep` from config (or `publish --keep`).
88    #[must_use]
89    pub fn with_retention(mut self, keep: Option<u64>) -> Self {
90        self.retention = keep;
91        self
92    }
93
94    /// Build a registry resolved from env: `TAMIAS_REGISTRY_TOKEN` is the
95    /// bearer token.
96    #[must_use]
97    pub fn from_env(base: impl Into<String>) -> Self {
98        let mut registry = Self::new(base);
99        if let Ok(token) = std::env::var("TAMIAS_REGISTRY_TOKEN") {
100            registry = registry.with_token(token);
101        }
102        registry
103    }
104
105    /// `GET <key>` → the body, or `None` when the store has no such key
106    /// (HTTP 404). Every other failure — transport error, 5xx, auth
107    /// rejection, unreadable body — is an `Err`. Callers must never
108    /// treat a failed read as "the key is absent": for `index.json` that
109    /// would let a transient outage masquerade as an empty registry (and
110    /// a publish could then overwrite the whole catalog).
111    fn get(&self, key: &str) -> Result<Option<Vec<u8>>, Error> {
112        let url = format!("{}/{key}", self.base);
113        let mut req = self.agent.get(&url);
114        if let Some(token) = &self.token {
115            req = req.header("authorization", &format!("Bearer {token}"));
116        }
117        let mut resp = req
118            .call()
119            .map_err(|e| Error::Failed(format!("GET {url}: {e}")))?;
120        let status = resp.status();
121        if status == ureq::http::StatusCode::NOT_FOUND {
122            return Ok(None);
123        }
124        if !status.is_success() {
125            return Err(Error::Failed(format!("GET {url}: HTTP {status}")));
126        }
127        // Bound the body — `index.json` is small, blobs are tarballs.
128        let limit = if key == "index.json" {
129            MAX_INDEX
130        } else {
131            DEFAULT_MAX_DOWNLOAD
132        };
133        resp.body_mut()
134            .with_config()
135            .limit(limit)
136            .read_to_vec()
137            .map(Some)
138            .map_err(|e| Error::Failed(format!("read {url}: {e}")))
139    }
140
141    /// `PUT <key>` — the body is streamed (`AsSendBody`), so a `.tgz`
142    /// publish never buffers the whole tarball in memory.
143    fn put(&self, key: &str, body: impl ureq::AsSendBody) -> Result<(), Error> {
144        let url = format!("{}/{key}", self.base);
145        let mut req = self.agent.put(&url);
146        if let Some(token) = &self.token {
147            req = req.header("authorization", &format!("Bearer {token}"));
148        }
149        let resp = req
150            .send(body)
151            .map_err(|e| Error::Failed(format!("PUT {url}: {e}")))?;
152        if !resp.status().is_success() {
153            return Err(Error::Failed(format!("PUT {url}: HTTP {}", resp.status())));
154        }
155        Ok(())
156    }
157}

159/// A per-call-unique temp path for the transient publish tarball, so
160/// concurrent publishes in one process don't clobber each other's file.
161fn publish_tmp_path(version: &str) -> std::path::PathBuf {
162    static SEQ: AtomicU64 = AtomicU64::new(0);
163    let seq = SEQ.fetch_add(1, Ordering::Relaxed);
164    let nanos = std::time::SystemTime::now()
165        .duration_since(std::time::UNIX_EPOCH)
166        .map_or(0, |d| d.as_nanos());
167    // The version is validated before this is called, but keep the temp
168    // filename single-component regardless — records/manifests can be
169    // hand-edited.
170    let safe: String = version
171        .chars()
172        .map(|c| {
173            if c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-' | '+') {
174                c
175            } else {
176                '_'
177            }
178        })
179        .collect();
180    std::env::temp_dir().join(format!(
181        "tamias-pub-{}-{nanos}-{seq}-{safe}.tgz",
182        std::process::id()
183    ))
184}

186impl Registry for HttpRegistry {
187    fn entries(&self) -> Result<Vec<RegistryEntry>, Error> {
188        // A cleartext index fetch is unauthenticated and modifiable in
189        // transit — the payload digests it records are only as
190        // trustworthy as this GET (signatures mitigate only when
191        // `allowed_signers` is configured). Warn per call: installs
192        // aren't hot loops.
193        if self.base.starts_with("http://") {
194            eprintln!(
195                "warning: fetching registry index {}/index.json over plain http — \
196                 the index is unauthenticated and modifiable in transit",
197                self.base
198            );
199        }
200        // Only a genuine 404 (no index yet — first publish) yields an
201        // empty list; any other read failure propagates so `publish`
202        // never PUTs an index built from a failed read.
203        match self.get("index.json")? {
204            Some(bytes) => serde_json::from_slice(&bytes)
205                .map_err(|e| Error::Failed(format!("parse index: {e}"))),
206            None => Ok(Vec::new()),
207        }
208    }
209
210    fn publish(
211        &self,
212        payload_root: &Path,
213        manifest: &Manifest,
214        signature: Option<&str>,
215    ) -> Result<PublishOutcome, Error> {
216        // The manifest may not have been validated upstream — every field
217        // embedded in the blob key/URL must be a single safe component.
218        if !is_valid_name(&manifest.name) {
219            return Err(Error::Failed(format!(
220                "unsafe package name '{}'",
221                manifest.name
222            )));
223        }
224        if let Some(scope) = &manifest.scope {
225            if !is_valid_name(scope) {
226                return Err(Error::Failed(format!("unsafe scope '{scope}'")));
227            }
228        }
229        let scoped = match &manifest.scope {
230            Some(scope) => format!("@{scope}/{}", manifest.name),
231            None => manifest.name.clone(),
232        };
233        let version = manifest
234            .version
235            .clone()
236            .unwrap_or_else(|| "0.0.0".to_string());
237        if !is_valid_version(&version) {
238            return Err(Error::Failed(format!(
239                "unsafe version '{version}' — expected [A-Za-z0-9][A-Za-z0-9._+-]* with no '..'"
240            )));
241        }
242        let rel = format!("{scoped}/{version}.tgz");
243        let tmp = publish_tmp_path(&version);
244        pack_tgz(payload_root, &tmp)?;
245        // Stream the tarball — `File`'s AsSendBody sets Content-Length
246        // from metadata, so the whole .tgz never lands in memory. Reject
247        // oversized payloads before the PUT: the configured
248        // `limits.max_upload` client-side, the server's own cap
249        // authoritatively.
250        let file =
251            std::fs::File::open(&tmp).map_err(Error::io(format!("open {}", tmp.display())))?;
252        let size = file
253            .metadata()
254            .map_err(Error::io(format!("stat {}", tmp.display())))?
255            .len();
256        if size > self.max_upload {
257            let _ = std::fs::remove_file(&tmp);
258            return Err(Error::Failed(format!(
259                "payload tarball {} is {size} bytes — exceeds the {}-byte upload limit",
260                tmp.display(),
261                self.max_upload
262            )));
263        }
264        self.put(&rel, file)?;
265        let _ = std::fs::remove_file(&tmp);
266
267        let entry = RegistryEntry {
268            name: manifest.name.clone(),
269            scope: manifest.scope.clone(),
270            version: manifest.version.clone(),
271            kind: manifest.kind.clone(),
272            digest: dir_digest(payload_root)?,
273            signature: signature.map(str::to_string),
274            publisher: None,
275            file: rel,
276        };
277        let mut entries = self.entries()?;
278        entries.retain(|e| {
279            !(e.name == entry.name && e.scope == entry.scope && e.version == entry.version)
280        });
281        entries.push(entry.clone());
282        entries
283            .sort_by(|a, b| (&a.scope, &a.name, &a.version).cmp(&(&b.scope, &b.name, &b.version)));
284        // Retention (P-024) — index-entry prune only; pruned `.tgz`
285        // blobs stay in the store.
286        let pruned = self
287            .retention
288            .map(|keep| {
289                apply_retention(
290                    &mut entries,
291                    usize::try_from(keep).unwrap_or(usize::MAX),
292                    &entry,
293                )
294            })
295            .unwrap_or_default();
296        let index = serde_json::to_vec_pretty(&entries)
297            .map_err(|e| Error::Failed(format!("serialize index: {e}")))?;
298        self.put("index.json", &index)?;
299        Ok(PublishOutcome { entry, pruned })
300    }
301}

310    /// Serializes tests that mutate process env — `cargo test` runs
311    /// tests on threads in one process, so mutations must be
312    /// serialized *and* restored even on panic.
313    static ENV: std::sync::Mutex<()> = std::sync::Mutex::new(());
314
315    /// RAII guard for [`ENV`]: restores the process environment and
316    /// working directory on drop — panic-safe. Poison is ignored: an
317    /// unwinding sibling already restored everything via its `Drop`.
318    struct EnvGuard {
319        _lock: std::sync::MutexGuard<'static, ()>,
320        env: Vec<(std::ffi::OsString, std::ffi::OsString)>,
321        cwd: std::path::PathBuf,
322    }
323
324    impl EnvGuard {
325        fn new() -> Self {
326            let lock = ENV.lock().unwrap_or_else(|e| e.into_inner());
327            let env = std::env::vars_os().collect();
328            let cwd = std::env::current_dir().unwrap();
329            Self {
330                _lock: lock,
331                env,
332                cwd,
333            }
334        }
335    }
336
337    impl Drop for EnvGuard {
338        fn drop(&mut self) {
339            for (k, _) in std::env::vars_os() {
340                if !self.env.iter().any(|(sk, _)| *sk == k) {
341                    std::env::remove_var(&k);
342                }
343            }
344            for (k, v) in &self.env {
345                std::env::set_var(k, v);
346            }
347            let _ = std::env::set_current_dir(&self.cwd);
348        }
349    }

582        let seen_srv = seen.clone();
583        std::thread::spawn(move || {
584            while let Ok((mut s, _)) = listener.accept() {
585                let mut buf = Vec::new();
586                let mut chunk = [0u8; 8192];
587                let header_end = loop {
588                    let n = s.read(&mut chunk).unwrap_or(0);
589                    if n == 0 {
590                        break buf.len();
591                    }
592                    buf.extend_from_slice(&chunk[..n]);
593                    if let Some(i) = buf.windows(4).position(|w| w == b"\r\n\r\n") {
594                        break i + 4;
595                    }
596                };
597                let head = String::from_utf8_lossy(&buf[..header_end]).to_string();
598                let first = head.lines().next().unwrap_or_default().to_string();
599                // Drain the request body so the response isn't cut short.
600                let content_length: usize = head
601                    .lines()
602                    .find_map(|l| {
603                        l.to_ascii_lowercase()
604                            .strip_prefix("content-length:")
605                            .and_then(|v| v.trim().parse().ok())
606                    })
607                    .unwrap_or(0);
608                let mut left = content_length.saturating_sub(buf.len() - header_end);
609                while left > 0 {
610                    let n = s.read(&mut chunk).unwrap_or(0);
611                    if n == 0 {
612                        break;
613                    }
614                    left = left.saturating_sub(n);
615                }
616                seen_srv.lock().unwrap().push(first.clone());
617                let body = if first.starts_with("GET") {
618                    b"HTTP/1.1 500 X\r\nContent-Length: 0\r\nConnection: close\r\n\r\n".as_slice()
619                } else {
620                    b"HTTP/1.1 200 OK\r\nContent-Length: 0\r\nConnection: close\r\n\r\n".as_slice()
621                };
622                let _ = s.write_all(body);
623                let _ = s.flush();
624            }
625        });
626
627        let src = tempfile::tempdir().unwrap();
628        std::fs::write(src.path().join("tamias.yaml"), "name: x\nversion: 1.0.0\n").unwrap();
629        let manifest = Manifest::load(src.path()).unwrap();
630        let registry = HttpRegistry::new(format!("http://127.0.0.1:{port}"));
631        assert!(registry.publish(src.path(), &manifest, None).is_err());
632        let reqs = seen.lock().unwrap();
633        assert!(
634            reqs.iter().any(|r| r.starts_with("GET /index.json")),
635            "{reqs:?}"
636        );
637        assert!(
638            !reqs.iter().any(|r| r.starts_with("PUT /index.json")),
639            "failed index read must not be overwritten: {reqs:?}"
640        );
641    }
642}
