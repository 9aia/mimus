3//! tamias-source-registry — the **registry-scoped install ref** kind:
4//! `<label>:<name>@<version>` (and `<label>:@scope/<name>@<version>`
5//! for scoped packages).
6//!
7//! `<label>` resolves against the *effective* config's `registries`
8//! map — an entry's `id` (uuid) first, then its map key (slug) — via
9//! [`tamias_registry_open::load_config`] (user-level `~/.config/tamias`
10//! layered under the cwd `tamias.*`). Only `http`/`cf` kind registries
11//! are installable; `local` (unserved dir) registries error with a
12//! pointer at `tamias registry serve`.
13//!
14//! The kind parses straight into a [`CfSpec`] — the same spec the
15//! `cf:` arm produces — so plan/fetch (index lookup → digest +
16//! signature verification → tarball fetch) runs through `CfSource`
17//! unchanged, and provenance records the verbatim `<label>:…` ref.
18//!
19//! `claims` is deliberately cheap — no config IO: any
20//! `<slug>:`-shaped prefix that isn't a reserved kind prefix is
21//! claimed, and `parse` does the config lookup (unknown label → hard
22//! error, never an fs-path fallback).
23

30/// Ref-kind prefixes that always win over a registry label — mirrors
31/// the claims of the kinds registered before this one (`git:`, `cf:`,
32/// `http://`, `https://`) plus the scheme spellings that must never be
33/// misread as a label (`http:`, `ssh:`, `fs:`, `file:`).
34const RESERVED_LABELS: &[&str] = &["git", "cf", "http", "https", "ssh", "fs", "file"];
35
36/// The `<registry>:<name>[@<version>]` ref kind.
37///
38/// [`RegistryRefKind::new`] resolves the effective config at parse
39/// time (`load_config`); [`RegistryRefKind::with_config`] pins a config
40/// for tests and embedders.
41#[derive(Clone, Debug, Default)]
42pub struct RegistryRefKind {
43    config: Option<Config>,
44}
45
46impl RegistryRefKind {
47    /// Resolve registries through the effective layered config.
48    #[must_use]
49    pub fn new() -> Self {
50        Self { config: None }
51    }
52
53    /// Resolve registries through `config` instead of discovery.
54    #[must_use]
55    pub fn with_config(config: Config) -> Self {
56        Self {
57            config: Some(config),
58        }
59    }
60}

62impl RefKind for RegistryRefKind {
63    /// Cheap shape check — no config IO. Claims any `<slug>:` prefix
64    /// (slug = `[a-zA-Z0-9][a-zA-Z0-9_.-]*`, `:` before any `/`, longer
65    /// than one char so Windows drive letters like `C:\…` can't
66    /// false-positive) that isn't a reserved kind prefix.
67    fn claims(&self, input: &str) -> bool {
68        let Some(colon) = input.find(':') else {
69            return false;
70        };
71        let label = &input[..colon];
72        // A '/' before the ':' means the colon sits inside a path —
73        // `a/b:c` is not `<label>:<rest>`.
74        if label.contains('/') || label.len() < 2 || !valid_label(label) {
75            return false;
76        }
77        !RESERVED_LABELS.contains(&label.to_ascii_lowercase().as_str())
78    }
79
80    /// Parse `<label>:<name>@<version>` (or `<label>:@scope/<name>@<version>`)
81    /// into a [`CfSpec`] — the config lookup happens here, so an
82    /// unknown label is a hard error.
83    ///
84    /// # Errors
85    /// Fails on a missing `@<version>`, a malformed name, an unknown
86    /// registry label, a `local`-kind registry, or a registry without
87    /// a url.
88    fn parse(&self, input: &str) -> Result<Spec, Error> {
89        let (label, rest) = input
90            .split_once(':')
91            .ok_or_else(|| Error::Failed(format!("bad registry ref '{input}'")))?;
92        let bad = || {
93            Error::Failed(format!(
94                "bad registry ref '{input}' — expected <registry>:<name>@<version>"
95            ))
96        };
97        // `@<version>` is required for v1 — a bare name never silently
98        // resolves "latest".
99        let (name_part, version) = rest.rsplit_once('@').ok_or_else(bad)?;
100        if version.is_empty() || name_part.is_empty() {
101            return Err(bad());
102        }
103        // `@scope/<name>` keeps the npm-style scoped spelling; anything
104        // else is a bare name. A stray '@' inside either part is
105        // malformed.
106        let (scope, name) = match name_part.strip_prefix('@') {
107            Some(scoped) => {
108                let (scope, name) = scoped.split_once('/').ok_or_else(bad)?;
109                (Some(scope), name)
110            }
111            None => (None, name_part),
112        };
113        if name.is_empty()
114            || scope.is_some_and(str::is_empty)
115            || name.contains('@')
116            || scope.is_some_and(|s| s.contains('@'))
117        {
118            return Err(bad());
119        }
120        let config = match &self.config {
121            Some(config) => config.clone(),
122            None => load_config(None)?,
123        };
124        let registry = find_registry(&config, label)
125            .ok_or_else(|| Error::Failed(format!("unknown registry '{label}'")))?;
126        let kind = registry.effective_kind();
127        match kind.as_str() {
128            "http" | "cf" => {}
129            "local" => {
130                return Err(Error::Failed(format!(
131                    "registry '{label}' is kind local — serve it with \
132                     `tamias registry serve` or use a cf:/http: ref"
133                )));
134            }
135            other => {
136                return Err(Error::Failed(format!(
137                    "registry '{label}': unknown kind '{other}' (expected http or cf)"
138                )));
139            }
140        }
141        let base_url = registry
142            .url
143            .clone()
144            .ok_or_else(|| Error::Failed(format!("registry '{label}': kind {kind} needs a url")))?;
145        Ok(Box::new(CfSpec {
146            base_url,
147            address: label.to_string(),
148            name: name.to_string(),
149            scope: scope.map(str::to_string),
150            version: version.to_string(),
151            token_env: registry.token_env.clone(),
152        }))
153    }
154}

156/// `<slug>` charset — `[a-zA-Z0-9][a-zA-Z0-9_.-]*` (the same shape
157/// `claims` enforces).
158fn valid_label(label: &str) -> bool {
159    let mut chars = label.chars();
160    chars.next().is_some_and(|c| c.is_ascii_alphanumeric())
161        && chars.all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '.' | '-'))
162}
163
164/// Resolve `label` against the effective config — `id` (uuid) first,
165/// then the `registries` map key (slug). Id wins on collision.
166fn find_registry<'a>(config: &'a Config, label: &str) -> Option<&'a ConfigRegistry> {
167    config
168        .registries
169        .values()
170        .find(|reg| reg.id.as_deref() == Some(label))
171        .or_else(|| config.registries.get(label))
172}

179    fn config_with(entries: &[(&str, ConfigRegistry)]) -> Config {
180        let mut config = Config::default();
181        for (name, reg) in entries {
182            config.registries.insert((*name).to_string(), reg.clone());
183        }
184        config

195    #[test]
196    fn claims_shape_without_io() {
197        let kind = RegistryRefKind::new();
198        assert!(kind.claims("9aia:cicada-demo@0.1.0"));
199        assert!(kind.claims("9aia:@scope/pkg@1.0.0"));
200        assert!(kind.claims("my-reg_1.2:x@1"));
201        // reserved prefixes are never registry labels
202        for input in [
203            "git:x",
204            "cf:a/b/x@1",
205            "http://x/y.tgz",
206            "https://x/y.tgz",
207            "ssh:host/x",
208            "fs:/x",
209            "file:///x",
210            "HTTP://x",
211        ] {
212            assert!(!kind.claims(input), "{input}");
213        }
214        // Windows drive letters + single-char labels can't false-positive
215        assert!(!kind.claims("C:\\pkg"));
216        assert!(!kind.claims("C:/pkg"));
217        assert!(!kind.claims("c:x@1"));
218        // no colon / colon after a slash / bad slug charset
219        assert!(!kind.claims("./x:y"));
220        assert!(!kind.claims("a/b:c@1"));
221        assert!(!kind.claims("-bad:x@1"));
222        assert!(!kind.claims(".bad:x@1"));
223        assert!(!kind.claims("no-colon"));
224        assert!(!kind.claims("x"));
225    }

227    #[test]
228    fn parse_resolves_by_name_and_id() {
229        let config = config_with(&[
230            ("9aia", http_registry("https://reg.example")),
231            (
232                "renamed",
233                ConfigRegistry {
234                    id: Some("018f3c2e-7b1a-7c4d-9e5f-2a6b8c0d1e2f".into()),
235                    ..http_registry("https://id.example")
236                },
237            ),
238        ]);
239        let kind = RegistryRefKind::with_config(config);
240
241        // by slug
242        let spec = kind.parse("9aia:cicada-demo@0.1.0").unwrap();
243        let cf = spec.downcast::<CfSpec>().unwrap();
244        assert_eq!(cf.base_url, "https://reg.example");
245        assert_eq!(cf.address, "9aia");
246        assert_eq!(cf.name, "cicada-demo");
247        assert_eq!(cf.version, "0.1.0");
248        assert_eq!(cf.scope, None);
249        assert_eq!(cf.token_env, None);
250
251        // by id — and id wins over a slug collision
252        let spec = kind
253            .parse("018f3c2e-7b1a-7c4d-9e5f-2a6b8c0d1e2f:@scope/pkg@1.0.0")
254            .unwrap();
255        let cf = spec.downcast::<CfSpec>().unwrap();
