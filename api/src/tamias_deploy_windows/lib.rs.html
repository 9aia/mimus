2//! tamias-deploy-windows — the `deploy` port's Windows adapter: manage a
3//! **per-user scheduled task** through `schtasks.exe` (Task Scheduler).
4//! No admin needed: tasks register under the interactive user with
5//! `RunLevel=LeastPrivilege`.
6//!
7//! A [`ServiceSpec`] renders to two artifacts under the task dir
8//! (`%LOCALAPPDATA%\tamias\tasks` by default):
9//!
10//! - `<name>.cmd` — a generated batch wrapper that applies the spec's
11//!   environment (`env_file` via `for /f`, inline `env` as `set`
12//!   lines, inline winning — systemd's order) and `cd /d`s to
13//!   `workdir` before exec'ing the argv. Task Scheduler actions
14//!   cannot carry env vars, so the wrapper is how env reaches the
15//!   process.
16//! - `<name>.xml` — the task definition imported via
17//!   `schtasks /create /xml … /f`: a `LogonTrigger` (the honest analog
18//!   of a systemd *user* unit starting at login), `InteractiveToken`
19//!   principal, and settings tuned for a long-running service
20//!   (`ExecutionTimeLimit=PT0S` — the 72h default would kill daemons;
21//!   battery blocks off; `RestartOnFailure` mapped from `restart`).
22//!
23//! ## Field mapping vs. the systemd backend
24//!
25//! | `ServiceSpec` | systemd | this backend |
26//! | --- | --- | --- |
27//! | `name` | `<name>.service` | task name (`/tn`) |
28//! | `description` | `Description=` | `<RegistrationInfo><Description>` |
29//! | `exec` | `ExecStart=` argv | wrapper `.cmd` last line (cmd quoting) |
30//! | `env` | `Environment=` | `set "K=V"` lines in the wrapper |
31//! | `env_file` | `EnvironmentFile=-` | `for /f` loader in the wrapper — `K=V`/`#` only; quoted values keep their quotes (systemd strips them) |
32//! | `workdir` | `WorkingDirectory=` | `<WorkingDirectory>` + `cd /d` in the wrapper |
33//! | `restart`/`restart_sec` | `Restart=`/`RestartSec=` | `<RestartOnFailure>` — see gaps |
34//! | `after` | `After=`/`Wants=` | **ignored** — Task Scheduler has no unit ordering |
35//!
36//! ## Honest gaps
37//!
38//! - **Restart count is capped.** systemd's `on-failure`/`always`
39//!   retries forever; Task Scheduler's `RestartOnFailure` takes a
40//!   `Count` of 1–255 — we emit `255` and document the cap.
41//!   `restart="always"` also can't restart on a *clean* exit(0) —
42//!   `RestartOnFailure` only fires on failure, so `always` degrades to
43//!   on-failure semantics. `restart_sec` maps to `Interval`, clamped
44//!   up to the schema minimum of `PT1M`.
45//! - **`env_file` is read by `for /f`**, not by the OS: only `K=V`
46//!   lines and `#` comments are understood; systemd-isms like
47//!   `export K=V`, quoting, and variable expansion are not.
48//! - **No journal.** `status`/`logs` are best-effort `schtasks /query`
49//!   output — Task Scheduler keeps no per-task log; capture service
50//!   logs inside the payload (or a `provider.health.cmd`) instead.
51//! - **`stop` is `schtasks /end`** — it terminates the process tree
52//!   root, not necessarily grandchildren the way systemd's cgroup
53//!   kill does.
54//! - Real `schtasks.exe` execution is `#[cfg(windows)]`-gated
55//!   (`SchtasksRunner` only exists there); everywhere else
56//!   [`WindowsDeploy::new`]
57//!   installs an unavailable runner so verbs fail loudly instead of
58//!   shelling out. Tests inject a [`Runner`] recorder on any platform.
59

69/// The seam that runs host commands (`schtasks`) — injected for tests.
70pub trait Runner: Send + Sync {
71    /// Run `program args…`, erroring on a non-zero exit.
72    ///
73    /// # Errors
74    /// Spawn/exit failures.
75    fn run(&self, program: &str, args: &[&str]) -> Result<(), Error>;
76}

78/// The real runner — shells out to `schtasks.exe`. Only exists on
79/// Windows: off-Windows there is no `schtasks` to run, and
80/// [`WindowsDeploy::new`] never installs this runner there.
81#[cfg(windows)]
82#[derive(Clone, Copy, Debug, Default)]
83pub struct SchtasksRunner;
84
85#[cfg(windows)]
86impl Runner for SchtasksRunner {
87    fn run(&self, program: &str, args: &[&str]) -> Result<(), Error> {
88        let out = Command::new(program)
89            .args(args)
90            .output()
91            .map_err(Error::io(format!("{program} {}", args.join(" "))))?;
92        if !out.status.success() {
93            return Err(Error::Failed(format!(
94                "{program} {}: {}",
95                args.join(" "),
96                String::from_utf8_lossy(&out.stderr).trim()
97            )));
98        }
99        Ok(())
100    }
101}

170    /// Write the wrapper `.cmd` + task `.xml`, then
171    /// `schtasks /create /tn <name> /xml <file> /f` (registers or
172    /// overwrites). Returns the task XML path.
173    ///
174    /// # Errors
175    /// [`Error::Failed`] on an invalid name/spec field, a `.cmd`
176    /// quoting failure, an empty `exec` argv, or a write/`schtasks`
177    /// failure. On non-Windows the default runner fails at the
178    /// `schtasks` step — the rendered files are still left on disk.
179    pub fn install(&self, spec: &ServiceSpec) -> Result<PathBuf, Error> {
180        check_task_name(&spec.name)?;
181        check_task_fields(spec)?;
182        std::fs::create_dir_all(&self.task_dir)
183            .map_err(Error::io(format!("create {}", self.task_dir.display())))?;
184        let wrapper = self.wrapper_path(&spec.name)?;
185        std::fs::write(&wrapper, render_wrapper(spec)?)
186            .map_err(Error::io(format!("write {}", wrapper.display())))?;
187        let xml = self.xml_path(&spec.name)?;
188        std::fs::write(&xml, render_task_xml(spec, &wrapper))
189            .map_err(Error::io(format!("write {}", xml.display())))?;
190        let args = create_args(&spec.name, &xml)?;
191        let refs: Vec<&str> = args.iter().map(String::as_str).collect();
192        self.runner.run("schtasks", &refs)?;
193        Ok(xml)
194    }

278    /// Best-effort "logs" over one or more tasks — Task Scheduler
279    /// keeps no per-task log, so this runs `schtasks /query` per task
280    /// name (`lines`/`follow` are accepted for signature parity and
281    /// ignored). Returns an error on an empty name list.
282    ///
283    /// # Errors
284    /// [`Error::Failed`] on an empty/invalid name list; propagates the
285    /// first runner failure.
286    pub fn logs_with(&self, names: &[String], _lines: u32, _follow: bool) -> Result<(), Error> {
287        if names.is_empty() {
288            return Err(Error::Failed("logs: no tasks given".into()));
289        }
290        for name in names {
291            self.status(name)?;
292        }
293        Ok(())
294    }

297    /// must not fail uninstall) + remove the wrapper `.cmd` and task

304        let _ = self.schtasks(&delete_args(name)?);
305        for path in [self.xml_path(name)?, self.wrapper_path(name)?] {
306            if path.exists() {
307                std::fs::remove_file(&path)
308                    .map_err(Error::io(format!("remove {}", path.display())))?;
309            }
310        }
311        Ok(())
312    }

424/// Pull the status field out of `query_csv_args` output — CSV columns
425/// are `TaskName,Next Run Time,Status`; quoted fields, status last.
426/// Returns `None` on empty/unparseable output (a line without a comma
427/// is an error message, not CSV).
428#[must_use]
429pub fn csv_status(output: &str) -> Option<String> {
430    let line = output
431        .lines()
432        .find(|l| !l.trim().is_empty() && l.contains(','))?;
433    let last = line.rsplit(',').next()?.trim().trim_matches('"').trim();
434    (!last.is_empty()).then(|| last.to_string())
435}

437/// A valid task name: non-empty, no path separators or Task
438/// Scheduler-hostile characters (`\ / : * ? " < > |`), no control
439/// chars, no `..`, no leading/trailing whitespace. Spaces are legal —
440/// the name travels as a single argv element, never through a shell.
441fn check_task_name(name: &str) -> Result<(), Error> {
442    let ok = !name.is_empty()
443        && name == name.trim()
444        && name != "."
445        && name != ".."
446        && !name.contains("..")
447        && name
448            .chars()
449            .all(|c| !c.is_control() && !"\\/:*?\"<>|".contains(c));
450    if ok {
451        Ok(())
452    } else {
453        Err(Error::Failed(format!(
454            "invalid task name '{name}' (no separators, '..', or \\ / : * ? \" < > |)"
455        )))
456    }
457}

459/// Reject spec fields that could break out of the `.cmd` wrapper or
460/// the task XML — a `\n`/`\r`/NUL lands as extra batch lines or XML
461/// content; a `"` in an env value or path terminates the `set "…"`
462/// quoting early.
463fn check_task_fields(spec: &ServiceSpec) -> Result<(), Error> {
464    if spec.exec.is_empty() {
465        return Err(Error::Failed(format!(
466            "task '{}' has an empty exec argv",
467            spec.name
468        )));
469    }
470    let bad = |what: &str, v: &str| {
471        if v.contains(['\n', '\r', '\0']) {
472            Some(Error::Failed(format!(
473                "task {what} contains a line break or NUL"
474            )))
475        } else {
476            None
477        }
478    };
479    let quoted = |what: &str, v: &str| {
480        if v.contains('"') {
481            Some(Error::Failed(format!(
482                "task {what} contains a double quote (unsupported in the .cmd wrapper)"
483            )))
484        } else {
485            None
486        }
487    };
488    for (what, v) in [
489        ("name", spec.name.as_str()),
490        ("description", spec.description.as_str()),
491    ] {
492        if let Some(e) = bad(what, v) {
493            return Err(e);
494        }
495    }
496    for (i, arg) in spec.exec.iter().enumerate() {
497        if let Some(e) = bad(&format!("exec argv[{i}]"), arg) {
498            return Err(e);
499        }
500    }
501    for (what, v) in [
502        ("env_file", spec.env_file.as_deref().unwrap_or("")),
503        ("workdir", spec.workdir.as_deref().unwrap_or("")),
504    ] {
505        // Check the *rendered* (post-`~`-expansion) value — that's what
506        // lands in the wrapper/XML.
507        let expanded = expand_home(v);
508        if let Some(e) = bad(what, &expanded).or_else(|| quoted(what, &expanded)) {
509            return Err(e);
510        }
511    }
512    if let Some(e) = bad("restart", spec.restart.as_deref().unwrap_or("")) {
513        return Err(e);
514    }
515    for (k, v) in &spec.env {
516        if let Some(e) = bad("env key", k)
517            .or_else(|| bad("env value", v))
518            .or_else(|| quoted("env key", k))
519            .or_else(|| quoted("env value", v))
520        {
521            return Err(e);
522        }
523    }
524    // `spec.after` is intentionally not validated: no Windows backend
525    // renders it (documented gap — no ordering primitive).
526    Ok(())
527}

529/// Escape `%` for batch-file context — `%%` collapses to a literal `%`
530/// when cmd.exe parses a `.cmd`, so user data can't smuggle `%VAR%`
531/// expansions into the wrapper.
532fn pct(s: &str) -> String {
533    s.replace('%', "%%")
534}
535
536/// Quote one argv element for the wrapper's final command line —
537/// cmd.exe rules, not POSIX: `%` is percent-escaped; an arg with
538/// whitespace or cmd metacharacters is wrapped in `"…"`; a literal
539/// `"` cannot be represented and is an error (documented gap).
540///
541/// # Errors
542/// [`Error::Failed`] when `arg` contains a double quote.
543fn cmd_quote(arg: &str) -> Result<String, Error> {
544    if arg.contains('"') {
545        return Err(Error::Failed(format!(
546            "exec argument {arg:?} contains a double quote (unsupported by cmd.exe)"
547        )));
548    }
549    let esc = pct(arg);
550    if esc
551        .chars()
552        .any(|c| c.is_whitespace() || "&|<>^()".contains(c))
553    {
554        Ok(format!("\"{esc}\""))
555    } else {
556        Ok(esc)
557    }
558}

560/// Render the wrapper `.cmd` — env first (`env_file` via `for /f`,
561/// then inline `env` overriding, systemd's order), `cd /d` to
562/// `workdir`, then the quoted argv, then `exit /b %errorlevel%`.
563///
564/// # Errors
565/// [`Error::Failed`] on an empty `exec` argv or an argument
566/// [`cmd_quote`] rejects.
567pub fn render_wrapper(spec: &ServiceSpec) -> Result<String, Error> {
568    let mut cmd = String::new();
569    cmd.push_str("@echo off\r\n");
570    cmd.push_str("setlocal\r\n");
571    cmd.push_str(&format!(
572        "rem tamias task '{}' — generated; do not edit\r\n",
573        spec.name.replace(['\r', '\n'], " ")
574    ));
575    if !spec.description.is_empty() {
576        cmd.push_str(&format!(
577            "rem {}\r\n",
578            spec.description.replace(['\r', '\n'], " ")
579        ));
580    }
581    if let Some(env_file) = &spec.env_file {
582        // `for /f` over a quoted path needs `usebackq`; `eol=#` skips
583        // comments; `tokens=1,* delims==` splits K=V keeping '=' in V.
584        cmd.push_str(&format!(
585            "for /F \"usebackq eol=# tokens=1,* delims==\" %%A in (\"{}\") do set \"%%A=%%B\"\r\n",
586            pct(&expand_home(env_file))
587        ));
588    }
589    for (k, v) in &spec.env {
590        cmd.push_str(&format!("set \"{}={}\"\r\n", pct(k), pct(v)));
591    }
592    if let Some(workdir) = &spec.workdir {
593        cmd.push_str(&format!("cd /d \"{}\"\r\n", pct(&expand_home(workdir))));
594    }
595    let mut argv = spec.exec.iter();
596    let Some(prog) = argv.next() else {
597        return Err(Error::Failed(format!(
598            "task '{}' has an empty exec argv",
599            spec.name
600        )));
601    };
602    let mut line = cmd_quote(&expand_home(prog))?;
603    for arg in argv {
604        line.push(' ');
605        line.push_str(&cmd_quote(&expand_home(arg))?);
606    }
607    cmd.push_str(&line);
608    cmd.push_str("\r\nexit /b %errorlevel%\r\n");
609    Ok(cmd)
610}

612/// XML-escape element text (`&`, `<`, `>` — quotes are legal in
613/// content and stay literal).
614fn xml_escape(s: &str) -> String {
615    s.replace('&', "&amp;")
616        .replace('<', "&lt;")
617        .replace('>', "&gt;")
618}

620/// The restart delay as an ISO-8601 duration — the schema floor is
621/// `PT1M`, so sub-minute `restart_sec` clamps up.
622fn restart_interval(restart_sec: Option<u32>) -> String {
623    let secs = restart_sec.unwrap_or(5).max(60);
624    format!("PT{secs}S")
625}
626
627/// Whether the spec asks for restart-on-failure — mirrors systemd's
628/// default (`on-failure` when unset); `"no"` disables, anything else
629/// (including `"always"`, which can only approximate) enables.
630fn restart_on_failure(spec: &ServiceSpec) -> bool {
631    spec.restart.as_deref() != Some("no")
632}

634/// Render the Task Scheduler task XML — a `LogonTrigger`
635/// (per-user service ≈ start at logon), `InteractiveToken` +
636/// `LeastPrivilege` principal (no admin), service-tuned settings
637/// (`ExecutionTimeLimit=PT0S` — the 72h default would kill daemons —
638/// battery blocks off, `RestartOnFailure` from `restart`/
639/// `restart_sec`), and an `Exec` action running `cmd.exe /d /c` on the
640/// wrapper `.cmd`.
641///
642/// # Errors
643/// Never fails on spec content — callers run [`check_task_fields`]
644/// first; paths are XML-escaped.
645#[must_use]
646pub fn render_task_xml(spec: &ServiceSpec, wrapper: &Path) -> String {
647    let mut xml = String::new();
648    xml.push_str("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\r\n");
649    xml.push_str("<Task version=\"1.2\" xmlns=\"http://schemas.microsoft.com/windows/2004/02/mit/task\">\r\n");
650    xml.push_str("  <RegistrationInfo>\r\n");
651    xml.push_str(&format!(
652        "    <Description>{}</Description>\r\n",
653        xml_escape(&spec.description)
654    ));
655    xml.push_str("  </RegistrationInfo>\r\n");
656    xml.push_str("  <Triggers>\r\n");
657    xml.push_str("    <LogonTrigger>\r\n      <Enabled>true</Enabled>\r\n    </LogonTrigger>\r\n");
658    xml.push_str("  </Triggers>\r\n");
659    xml.push_str("  <Principals>\r\n    <Principal id=\"Author\">\r\n");
660    xml.push_str("      <LogonType>InteractiveToken</LogonType>\r\n");
661    xml.push_str("      <RunLevel>LeastPrivilege</RunLevel>\r\n");
662    xml.push_str("    </Principal>\r\n  </Principals>\r\n");
663    xml.push_str("  <Settings>\r\n");
664    xml.push_str("    <MultipleInstancesPolicy>IgnoreNew</MultipleInstancesPolicy>\r\n");
665    xml.push_str("    <DisallowStartIfOnBatteries>false</DisallowStartIfOnBatteries>\r\n");
666    xml.push_str("    <StopIfGoingOnBatteries>false</StopIfGoingOnBatteries>\r\n");
667    xml.push_str("    <AllowHardTerminate>true</AllowHardTerminate>\r\n");
668    xml.push_str("    <StartWhenAvailable>true</StartWhenAvailable>\r\n");
669    xml.push_str("    <RunOnlyIfNetworkAvailable>false</RunOnlyIfNetworkAvailable>\r\n");
670    xml.push_str("    <Enabled>true</Enabled>\r\n");
671    xml.push_str("    <Hidden>false</Hidden>\r\n");
672    // PT0S = no time limit — the schema default (72h) would kill
673    // long-running provider services.
674    xml.push_str("    <ExecutionTimeLimit>PT0S</ExecutionTimeLimit>\r\n");
675    xml.push_str("    <Priority>7</Priority>\r\n");
676    if restart_on_failure(spec) {
677        xml.push_str("    <RestartOnFailure>\r\n");
678        xml.push_str(&format!(
679            "      <Interval>{}</Interval>\r\n",
680            restart_interval(spec.restart_sec)
681        ));
682        // systemd retries unboundedly; the schema caps Count at 255.
683        xml.push_str("      <Count>255</Count>\r\n");
684        xml.push_str("    </RestartOnFailure>\r\n");
685    }
686    xml.push_str("  </Settings>\r\n");
687    xml.push_str("  <Actions Context=\"Author\">\r\n    <Exec>\r\n");
688    xml.push_str("      <Command>cmd.exe</Command>\r\n");
689    xml.push_str(&format!(
690        "      <Arguments>/d /c \"{}\"</Arguments>\r\n",
691        xml_escape(&wrapper.to_string_lossy())
692    ));
693    if let Some(workdir) = &spec.workdir {
694        xml.push_str(&format!(
695            "      <WorkingDirectory>{}</WorkingDirectory>\r\n",
696            xml_escape(&expand_home(workdir))
697        ));
698    }
699    xml.push_str("    </Exec>\r\n  </Actions>\r\n</Task>\r\n");
700    xml
701}

703/// `%LOCALAPPDATA%\tamias\tasks` → `%USERPROFILE%\AppData\Local\…` →
704/// `$HOME/.local/share/…` → `./.tamias/tasks`.
705fn task_dir() -> PathBuf {
706    std::env::var_os("LOCALAPPDATA")
707        .map(|d| PathBuf::from(d).join("tamias/tasks"))
708        .or_else(|| {
709            std::env::var_os("USERPROFILE")
710                .map(|h| PathBuf::from(h).join("AppData/Local/tamias/tasks"))
711        })
712        .or_else(|| {
713            std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share/tamias/tasks"))
714        })
715        .unwrap_or_else(|| PathBuf::from(".tamias/tasks"))
716}
