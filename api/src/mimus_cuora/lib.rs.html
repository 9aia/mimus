1//! Cuora adapter of `Secrets` — resolves names through the `cuora`
2//! CLI, never by linking vault internals: env vars stay the
3//! interface (cuora ADR-0001). `get` → `cuora get cuora://{ns}/{name}
4//! --json`; `set` → `cuora put` with the value on **stdin** (never
5//! argv — cuora's own rule: args land in history and `/proc`).
6//!
7//! cuora's audit ndjson rides the inherited stderr through to the
8//! caller — every mimus lookup is a materialization on the wire.
9
10use mimus_secret::Secrets;
11use std::io::{Read, Write};
12use std::process::{Command, Stdio};
13
14pub struct CuoraSecrets {
15    /// The cuora binary — default `cuora` (PATH resolution).
16    pub bin: String,
17    /// Ref namespace: `get("ANTHROPIC_API_KEY")` resolves
18    /// `cuora://{ns}/ANTHROPIC_API_KEY`.
19    pub ns: String,
20    /// Env context forwarded as `--env`; None = cuora's own
21    /// resolution (`CUORA_ENV` → `dev`).
22    pub env: Option<String>,
23}
24
25impl Default for CuoraSecrets {
26    fn default() -> Self {
27        Self {
28            bin: "cuora".into(),
29            ns: "mimus".into(),
30            env: None,
31        }
32    }
33}
34
35impl CuoraSecrets {
36    fn rref(&self, name: &str) -> String {
37        format!("cuora://{}/{name}", self.ns)
38    }
39
40    fn cmd(&self) -> Command {
41        let mut c = Command::new(&self.bin);
42        // Attribution for cuora's audit events — a claimed label,
43        // not identity (spoofable, like every local actor claim).
44        // Honor a pre-rename ARCA_ACTOR export too.
45        if std::env::var_os("CUORA_ACTOR").is_none() && std::env::var_os("ARCA_ACTOR").is_none() {
46            c.env("CUORA_ACTOR", "mimus");
47        }
48        if let Some(e) = &self.env {
49            c.args(["--env", e]);
50        }
51        c
52    }
53}
54
55impl Secrets for CuoraSecrets {
56    fn name(&self) -> &str {
57        "cuora"
58    }
59
60    fn available(&self) -> bool {
61        Command::new(&self.bin)
62            .arg("--version")
63            .stdout(Stdio::null())
64            .stderr(Stdio::null())
65            .status()
66            .map(|s| s.success())
67            .unwrap_or(false)
68    }
69
70    fn get(&self, name: &str) -> Option<String> {
71        let mut child = self
72            .cmd()
73            .args(["get", &self.rref(name), "--json"])
74            .stdout(Stdio::piped())
75            .stderr(Stdio::inherit()) // audit events pass through
76            .spawn()
77            .ok()?;
78        let mut out = String::new();
79        child.stdout.take()?.read_to_string(&mut out).ok()?;
80        if !child.wait().ok()?.success() {
81            return None;
82        }
83        serde_json::from_str::<serde_json::Value>(&out)
84            .ok()?
85            .get("value")?
86            .as_str()
87            .filter(|v| !v.is_empty())
88            .map(str::to_string)
89    }
90
91    fn set(&self, name: &str, value: &str) -> std::io::Result<()> {
92        let mut child = self
93            .cmd()
94            .args(["put", &self.rref(name)])
95            .stdin(Stdio::piped())
96            .stderr(Stdio::inherit())
97            .spawn()?;
98        child.stdin.as_mut().unwrap().write_all(value.as_bytes())?;
99        if child.wait().map(|s| s.success()).unwrap_or(false) {
100            Ok(())
101        } else {
102            Err(std::io::Error::other("cuora put failed"))
103        }
104    }
105}
