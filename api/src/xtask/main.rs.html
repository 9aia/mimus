1//! xtask — repo chores. `cargo xtask <verb>` via the alias in
2//! `.cargo/config.toml`.
3//!
4//! - `check`    — `cargo fmt --check` + `cargo clippy -D warnings`
5//! - `test`     — `cargo nextest run` when installed, else `cargo test`
6//! - `coverage` — `cargo llvm-cov` (HTML → target/llvm-cov/html);
7//!   extra args pass through, e.g. `--fail-under-lines 100`
8//! - `install`  — release-build `tamias-driver-myco` → ~/.cargo/bin
9//!   (the `myco` command itself is the tamias pack's generated
10//!   bins[] shim — `cargo xtask pack` + `tamias install
11//!   ./libraries/myco`)
12//! - `pack`     — stage libraries/myco driver payload
13//!   (release tamias-driver-myco → drivers/bin/<triple>/)
14//! - `site`     — build the docs site (docs/ worktree, VitePress)
15
16use std::process::{Command, ExitStatus};
17
18fn run(args: &[&str]) -> ExitStatus {
19    let empty: &[&str] = &[];
20    let (prog, rest) = args.split_first().unwrap_or((&"true", empty));
21    Command::new(prog).args(rest).status().unwrap_or_else(|e| {
22        eprintln!("xtask: failed to spawn {prog}: {e}");
23        std::process::exit(1)
24    })
25}
26
27/// `run` with a working directory — for commands that must execute
28/// inside a child tree (the docs/ worktree).
29fn run_in(dir: &str, args: &[&str]) -> ExitStatus {
30    let empty: &[&str] = &[];
31    let (prog, rest) = args.split_first().unwrap_or((&"true", empty));
32    Command::new(prog)
33        .args(rest)
34        .current_dir(dir)
35        .status()
36        .unwrap_or_else(|e| {
37            eprintln!("xtask: failed to spawn {prog}: {e}");
38            std::process::exit(1)
39        })
40}

42fn have(tool: &str) -> bool {
43    Command::new(tool)
44        .arg("--version")
45        .stdout(std::process::Stdio::null())
46        .stderr(std::process::Stdio::null())
47        .status()
48        .map(|s| s.success())
49        .unwrap_or(false)
50}

52fn have_cargo_sub(sub: &str) -> bool {
53    Command::new("cargo")
54        .args([sub, "--version"])
55        .stdout(std::process::Stdio::null())
56        .stderr(std::process::Stdio::null())
57        .status()
58        .map(|s| s.success())
59        .unwrap_or(false)
60}

62fn check() -> i32 {
63    let ok = run(&["cargo", "fmt", "--all", "--", "--check"]).success()
64        && run(&[
65            "cargo",
66            "clippy",
67            "--workspace",
68            "--all-targets",
69            "--",
70            "-D",
71            "warnings",
72        ])
73        .success();
74    if ok {
75        0
76    } else {
77        1
78    }
79}

81fn test() -> i32 {
82    if have_cargo_sub("nextest") {
83        run(&["cargo", "nextest", "run", "--workspace"])
84    } else {
85        run(&["cargo", "test", "--workspace"])
86    }
87    .code()
88    .unwrap_or(1)
89}

91fn coverage(extra: &[String]) -> i32 {
92    if !have_cargo_sub("llvm-cov") {
93        eprintln!("cargo-llvm-cov not installed — cargo install cargo-llvm-cov");
94        return 1;
95    }
96    let mut args = vec!["cargo", "llvm-cov"];
97    if have_cargo_sub("nextest") {
98        args.push("nextest");
99        args.extend(["--workspace", "--no-tests=pass"]);
100    } else {
101        args.push("--workspace");
102    }
103    if extra.is_empty() {
104        args.push("--html");
105    }
106    let owned: Vec<&str> = extra.iter().map(String::as_str).collect();
107    args.extend(owned);
108    run(&args).code().unwrap_or(1)
109}

111fn install() -> i32 {
112    // There is no `myco` cargo bin — the command ships as the
113    // tamias pack (`cargo xtask pack` + `tamias install
114    // ./libraries/myco`); the driver serves both the pavo view
115    // surface and the `cli` bins[] shim passthrough. Installing the
116    // driver binary alone keeps `tamias-driver-myco` resolvable on
117    // PATH for dev; the pack install registers it under
118    // <tamias-home>/drivers/ durably.
119    if !run(&["cargo", "build", "--release", "-p", "myco-driver"]).success() {
120        return 1;
121    }
122    let src = target_dir().join("release").join("tamias-driver-myco");
123    let home = match std::env::var("HOME") {
124        Ok(h) => h,
125        Err(_) => {
126            eprintln!(
127                "xtask: $HOME unset — copy {} onto PATH yourself",
128                src.display()
129            );
130            return 1;
131        }
132    };
133    let dst = std::path::Path::new(&home)
134        .join(".cargo/bin")
135        .join("tamias-driver-myco");
136    if let Err(e) = std::fs::copy(&src, &dst) {
137        eprintln!("xtask: copy {} → {}: {e}", src.display(), dst.display());
138        return 1;
139    }
140    println!("installed tamias-driver-myco → {}", dst.display());
141    println!("install the myco command (generated bins[] shim):");
142    println!("  cargo xtask pack && tamias install --allow-drivers ./libraries/myco");
143    0
144}

146/// Effective target dir — `CARGO_TARGET_DIR` when set, else the
147/// workspace `target/` (keeps `pack`/`install` correct under a
148/// shared target dir without needing cargo metadata parsing).
149fn target_dir() -> std::path::PathBuf {
150    std::env::var_os("CARGO_TARGET_DIR")
151        .map(std::path::PathBuf::from)
152        .unwrap_or_else(|| std::path::PathBuf::from("target"))
153}
154
155/// Host target triple — the `drivers/bin/<triple>/` selector
156/// tamias uses for compiled driver payloads.
157fn host_triple() -> String {
158    let out = Command::new("rustc")
159        .args(["-vV"])
160        .output()
161        .ok()
162        .filter(|o| o.status.success())
163        .and_then(|o| String::from_utf8(o.stdout).ok())
164        .and_then(|v| {
165            v.lines()
166                .find_map(|l| l.strip_prefix("host: ").map(str::to_string))
167        })
168        .unwrap_or_else(|| "x86_64-unknown-linux-gnu".to_string());
169    out
170}

172/// `cargo xtask pack` — stage the tamias pack payload (T-006):
173/// release-build `tamias-driver-myco` and drop it under
174/// `libraries/myco/drivers/bin/<host-triple>/` — the
175/// bundled-driver dir `tamias install` registers into
176/// `<tamias-home>/drivers/`. Gitignored on purpose: the binary is a
177/// build artifact; the committed pack is just `tamias.yaml` + this
178/// staging verb.
179fn pack() -> i32 {
180    if !run(&["cargo", "build", "--release", "-p", "myco-driver"]).success() {
181        return 1;
182    }
183    let triple = host_triple();
184    let dest_dir = std::path::Path::new("libraries/myco/drivers/bin").join(&triple);
185    if let Err(e) = std::fs::create_dir_all(&dest_dir) {
186        eprintln!("xtask: create {}: {e}", dest_dir.display());
187        return 1;
188    }
189    let src = target_dir().join("release").join("tamias-driver-myco");
190    let dest = dest_dir.join("tamias-driver-myco");
191    if let Err(e) = std::fs::copy(&src, &dest) {
192        eprintln!("xtask: copy {} → {}: {e}", src.display(), dest.display());
193        return 1;
194    }
195    #[cfg(unix)]
196    {
197        use std::os::unix::fs::PermissionsExt;
198        let _ = std::fs::set_permissions(&dest, std::fs::Permissions::from_mode(0o755));
199    }
200    println!("pack staged: {}", dest.display());
201    println!();
202    println!("install the myco command (generated bins[] shim):");
203    println!("  tamias install --allow-drivers ./libraries/myco");
204    println!(
205        "  # or a pinned ref: tamias install 'git:9aia/myco//libraries/myco#main' --allow-drivers"
206    );
207    0
208}

210fn site() -> i32 {
211    if !std::path::Path::new("docs/package.json").exists() {
212        eprintln!("no docs/ worktree — `git worktree add docs docs` first");
213        return 1;
214    }
215    // run inside docs/ — works for bun and npm alike (`bun --cwd`
216    // isn't a `bun run` flag, so flag plumbing gets fragile)
217    let pkg = if have("bun") { "bun" } else { "npm" };
218    if !run_in("docs", &[pkg, "install"]).success() {
219        eprintln!("{pkg} install failed in docs/ — fix deps before building");
220        return 1;
221    }
222    run_in("docs", &[pkg, "run", "docs:build"])
223        .code()
224        .unwrap_or(1)
225}

227fn main() {
228    let mut args = std::env::args().skip(1);
229    let cmd = args.next().unwrap_or_default();
230    let rest: Vec<String> = args.collect();
231    let code = match cmd.as_str() {
232        "check" => check(),
233        "test" => test(),
234        "coverage" => coverage(&rest),
235        "install" => install(),
236        "pack" => pack(),
237        "site" => site(),
238        other => {
239            eprintln!(
240                "usage: cargo xtask <check|test|coverage|install|pack|site>\n  unknown verb: {other:?}"
241            );
242            2
243        }
244    };
245    std::process::exit(code);
246}
