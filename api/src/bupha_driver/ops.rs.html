1//! ops — the tamias **content-type** verbs of `tamias-driver-bupha`.
2//!
3//! tamias discovers `tamias-driver-<type>` executables and runs
4//! `<verb> --dir <payload> --name <pkg> [--config <f>]` for the
5//! lifecycle three (`install`/`deploy`/`uninstall`) and any provider
6//! verb the driver's `capabilities` advertises (`host`, `hooks`,
7//! `health`, `status`, `logs`, `config-schema`). This module is that surface —
8//! the pavo verbs (`manifest`/`serve`/`capabilities`) stay in
9//! `main.rs`/`lib.rs`; `main.rs` dispatches on the subcommand, and
10//! tamias verbs always carry `--dir`/`--name` while pavo verbs never
11//! do, so the two surfaces can't collide.
12//!
13//! Config model: the bundle payload ships `fleet.toml` — a complete
14//! `bupha_core::Config` doc (every `[[repo]]`, `[runner]`, `[status]`,
15//! `[queue]`, …) *except* the two fields the manifest owns:
16//! `tunnel.public_url` and `trigger.listen`, which `deploy` injects
17//! from `provider.ingress[]` (the same values `hooks register`
18//! points GitHub at — one source of truth, no drift). `deploy`
19//! writes the merged doc to `~/.config/bupha/bundles/<name>.toml`;
20//! the provider units' `tamias-driver-bupha host|worker --config
21//! <that path>` consume it (`host` runs the watcher engine
22//! in-process via `bupha-watch`, `worker` polls `ci-job` entities
23//! on the `[worker] node` — both through the same merged config;
24//! ADR-0006: units exec the driver binary, not a cargo-installed
25//! `bupha`). The `host`/`hooks`/`provider.ingress[]` path is
26//! **standalone/dev mode** (ADR-0007): production webhook/forge
27//! ingress is a peregrin campaign listener that normalizes
28//! deliveries through bupha's `Trigger` semantics and enqueues
29//! `ci-job` via `ci.enqueue`, consumed by the `worker` service —
30//! use the listener verbs only where no peregrin runs.
31//! Per-bundle files beat
32//! merging into `bupha.yaml`: the config
33//! loader (`bupha-config-file`) has no include mechanism and a
34//! merge-writer would have to track provenance to stay idempotent —
35//! a dedicated file is trivially regenerable and `uninstall` is a
36//! delete.
37//!
38//! Secrets are env-var NAMES only (`secret_env`, `token_env`) —
39//! values resolve env → cuora → OS keyring → `~/.config/bupha/env`
40//! (cuora canonical per T-056), the same chain `bupha` itself uses,
41//! and are never printed or written.
42

62/// Parsed driver flags — the standard `--dir`/`--name`/`--config`
63/// triple plus the provider-verb extras. Positional args survive for
64/// `hooks <action>`.
65#[derive(Debug, Default)]
66pub struct Flags {
67    /// Payload dir tamias passes (`--dir`).
68    pub dir: Option<PathBuf>,
69    /// Package name (`--name`).
70    pub name: Option<String>,
71    /// `--config` override — deploy's output path when given.
72    pub config: Option<PathBuf>,
73    /// `hooks --dry-run` — print the plan, touch nothing.
74    pub dry_run: bool,
75    /// `logs --lines N`.
76    pub lines: u32,
77    /// `hooks --api-base` — forge REST base override.
78    pub api_base: Option<String>,
79    /// `hooks --token-env` — forge-token secret NAME override.
80    pub token_env: Option<String>,
81    /// `worker --node` — hypha/peregrin node base URL override
82    /// (`[worker] node` / `BUPHA_WORKER_NODE` are the config/env
83    /// fallbacks).
84    pub node: Option<String>,
85    /// `worker --node-token-env` — secret NAME holding the node's
86    /// bearer credential (`[worker] node_token_env` fallback).
87    pub node_token_env: Option<String>,
88    /// Positional args (the `hooks` action).
89    pub positional: Vec<String>,
90}

92/// Parse `--flag value` / `--flag=value` argv. Unknown flags are an
93/// error — a typo'd flag must not silently no-op a lifecycle verb.
94pub fn parse_flags(args: &[String]) -> Result<Flags, String> {
95    let mut f = Flags {
96        lines: 200,
97        ..Default::default()
98    };
99    let mut i = 0;
100    while i < args.len() {
101        let arg = &args[i];
102        let (flag, inline) = match arg.split_once('=') {
103            Some((f, v)) => (f, Some(v.to_string())),
104            None => (arg.as_str(), None),
105        };
106        let take = |i: &mut usize| -> Result<String, String> {
107            if let Some(v) = inline.clone() {
108                return Ok(v);
109            }
110            *i += 1;
111            args.get(*i)
112                .cloned()
113                .ok_or_else(|| format!("{flag}: missing value"))
114        };
115        match flag {
116            "--dir" => f.dir = Some(PathBuf::from(take(&mut i)?)),
117            "--name" => f.name = Some(take(&mut i)?),
118            "--config" => f.config = Some(PathBuf::from(take(&mut i)?)),
119            "--api-base" => f.api_base = Some(take(&mut i)?),
120            "--token-env" => f.token_env = Some(take(&mut i)?),
121            "--node" => f.node = Some(take(&mut i)?),
122            "--node-token-env" => f.node_token_env = Some(take(&mut i)?),
123            "--lines" => {
124                f.lines = take(&mut i)?
125                    .parse()
126                    .map_err(|_| "--lines: expected a number".to_string())?
127            }
128            "--dry-run" => f.dry_run = true,
129            _ if arg.starts_with("--") => return Err(format!("unknown flag {arg}")),
130            _ => f.positional.push(arg.clone()),
131        }
132        i += 1;
133    }
134    Ok(f)
135}

137/// `--dir`/`--name` are mandatory on every tamias verb — the dispatch
138/// contract guarantees them, but a hand-run driver gets a clear error.
139fn required(f: &Flags, verb: &str) -> Result<(PathBuf, String), String> {
140    let dir = f
141        .dir
142        .clone()
143        .ok_or_else(|| format!("{verb}: --dir is required"))?;
144    let name = f
145        .name
146        .clone()
147        .ok_or_else(|| format!("{verb}: --name is required"))?;
148    Ok((dir, name))
149}

206/// Load `tamias.{yaml,yml,toml}` from the payload dir — the first
207/// name present wins, parsed by extension like `bupha-config-file`.
208fn load_manifest(dir: &Path) -> Result<Manifest, String> {
209    for name in MANIFEST_NAMES {
210        let path = dir.join(name);
211        if !path.is_file() {
212            continue;
213        }
214        let text =
215            std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", path.display()))?;
216        return if name.ends_with(".toml") {
217            toml::from_str(&text).map_err(|e| format!("{name}: {e}"))
218        } else {
219            let v: serde_norway::Value =
220                serde_norway::from_str(&text).map_err(|e| format!("{name}: {e}"))?;
221            serde_norway::from_value(v).map_err(|e| format!("{name}: {e}"))
222        };
223    }
224    Err(format!(
225        "{}: no {} found",
226        dir.display(),
227        MANIFEST_NAMES.join("/")
228    ))
229}

231/// The provider block, or a clear error naming the missing piece.
232fn provider<'a>(manifest: &'a Manifest, dir: &Path) -> Result<&'a Provider, String> {
233    manifest.provider.as_ref().ok_or_else(|| {
234        format!(
235            "{}: no provider: block — nothing to deploy/probe",
236            dir.display()
237        )
238    })
239}

241/// First ingress entry with a non-empty `public_url` — the public
242/// endpoint hooks register against and `tunnel.public_url` mirrors.
243fn ingress_url(provider: &Provider) -> Option<&Ingress> {
244    provider
245        .ingress
246        .iter()
247        .find(|i| !i.public_url.trim().is_empty())
248}

250/// GitHub forge hooks across every ingress entry, deduped by repo in
251/// declaration order — the repos `hooks register` manages and
252/// `deploy webhook` would have.
253fn github_repos(provider: &Provider) -> Vec<String> {
254    let mut repos = Vec::new();
255    for hook in provider
256        .ingress
257        .iter()
258        .flat_map(|i| i.hooks.iter())
259        .filter(|h| h.forge.eq_ignore_ascii_case("github"))
260    {
261        if !repos.contains(&hook.repo) {
262            repos.push(hook.repo.clone());
263        }
264    }
265    repos
266}

270/// `~/.config/bupha` — the same resolution `bupha`'s `cfg_dir()` uses:
271/// `$XDG_CONFIG_HOME/bupha` → `$HOME/.config/bupha` → `./.config/bupha`.
272fn config_dir() -> PathBuf {
273    let base = std::env::var("XDG_CONFIG_HOME")
274        .ok()
275        .filter(|s| !s.is_empty())
276        .map(PathBuf::from)
277        .or_else(|| {
278            std::env::var("HOME")
279                .ok()
280                .filter(|s| !s.is_empty())
281                .map(|h| PathBuf::from(h).join(".config"))
282        })
283        .unwrap_or_else(|| PathBuf::from(".").join(".config"));
284    base.join("bupha")
285}

287/// `<config>/bundles/<name>.toml` — where `deploy` writes the
288/// operator config; `--config` overrides the whole path.
289fn bundle_config_path(name: &str, config: Option<&Path>) -> PathBuf {
290    config.map_or_else(
291        || {
292            // Package names are path-safe already; sanitize anyway so a
293            // scoped/weird `--name` can't escape the bundles dir.
294            let safe: String = name
295                .chars()
296                .map(|c| {
297                    if c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-') {
298                        c
299                    } else {
300                        '_'
301                    }
302                })
303                .collect();
304            config_dir().join("bundles").join(format!("{safe}.toml"))
305        },
306        Path::to_path_buf,
307    )
308}

310/// `<dir>/fleet.toml` — the payload's operator-config source.
311fn fleet_path(dir: &Path) -> PathBuf {
312    dir.join(FLEET_FILE)
313}

315fn read_fleet(dir: &Path) -> Result<String, String> {
316    let path = fleet_path(dir);
317    std::fs::read_to_string(&path).map_err(|e| {
318        format!(
319            "{}: {e} — the bundle must ship {FLEET_FILE}",
320            path.display()
321        )
322    })
323}

325/// The fleet doc as a TOML table — deploy merges the manifest-owned
326/// fields into it; hooks/status read it back as a `Config`.
327fn fleet_table(dir: &Path) -> Result<toml::Table, String> {
328    let text = read_fleet(dir)?;
329    match toml::from_str::<toml::Value>(&text) {
330        Ok(toml::Value::Table(t)) => Ok(t),
331        Ok(_) => Err(format!("{FLEET_FILE}: top level must be a table")),
332        Err(e) => Err(format!("{FLEET_FILE}: {e}")),
333    }
334}
335
336/// The fleet doc parsed as a full `bupha_core::Config` — used by
337/// `hooks` for repo/secret names and by `deploy` for validation.
338fn fleet_config(dir: &Path) -> Result<bupha_core::Config, String> {
339    let text = read_fleet(dir)?;
340    let mut cfg: bupha_core::Config =
341        toml::from_str(&text).map_err(|e| format!("{FLEET_FILE}: {e}"))?;
342    cfg.expand_tilde();
343    Ok(cfg)
344}

346/// Navigate/create nested tables in a `toml::Table` — errors when the
347/// fleet already uses the key for a non-table value.
348fn table_mut<'a>(root: &'a mut toml::Table, key: &str) -> Result<&'a mut toml::Table, String> {
349    let entry = root
350        .entry(key)
351        .or_insert_with(|| toml::Value::Table(toml::Table::new()));
352    entry
353        .as_table_mut()
354        .ok_or_else(|| format!("{FLEET_FILE}: '{key}' must be a table"))
355}

357/// `deploy` — write the operator config
358/// `tamias-driver-bupha host|worker --config` consumes: `fleet.toml` with
359/// `tunnel.public_url` and
360/// `trigger.listen`/`kind` filled from `provider.ingress[]` (the
361/// manifest wins those fields — they're the same values the forge
362/// hooks point at). Regeneration is idempotent: the file is wholly
363/// rewritten from payload + manifest every time.
364///
365/// # Errors
366/// Fails on a missing manifest/provider/fleet, an invalid merged
367/// config (`Config::watched` validation), or a write failure.
368pub fn deploy(dir: &Path, name: &str, config: Option<&Path>) -> Result<String, String> {
369    let manifest = load_manifest(dir)?;
370    let provider = provider(&manifest, dir)?;
371    let mut fleet = fleet_table(dir)?;
372
373    if let Some(ingress) = ingress_url(provider) {
374        let url = ingress.public_url.trim().trim_end_matches('/').to_string();
375        if !url.is_empty() {
376            {
377                let tunnel = table_mut(&mut fleet, "tunnel")?;
378                tunnel.insert("public_url".into(), toml::Value::String(url));
379            }
380            if let Some(listen) = ingress.listen.as_deref().filter(|l| !l.is_empty()) {
381                {
382                    let trigger = table_mut(&mut fleet, "trigger")?;
383                    trigger.insert("listen".into(), toml::Value::String(listen.to_string()));
384                    // A declared public ingress means the forge webhook
385                    // source — unless the fleet already picked a kind
386                    // (e.g. a sturn-only worker config shipped as fleet).
387                    trigger
388                        .entry("kind")
389                        .or_insert_with(|| toml::Value::String("github".into()));
390                }
391                // cloudflared/direct tunnels forward to the listener —
392                // keep local_url consistent with the declared bind.
393                let tunnel = table_mut(&mut fleet, "tunnel")?;
394                tunnel
395                    .entry("local_url")
396                    .or_insert_with(|| toml::Value::String(format!("http://{listen}")));
397            }
398        }
399    }
400
401    let text = toml::to_string(&toml::Value::Table(fleet))
402        .map_err(|e| format!("{FLEET_FILE}: serialize: {e}"))?;
403    // Validate the merged doc exactly as `host --config` will
404    // load it — a bad fleet must fail here, not at unit start.
405    let mut cfg: bupha_core::Config =
406        toml::from_str(&text).map_err(|e| format!("merged config: {e}"))?;
407    cfg.expand_tilde();
408    cfg.watched().map_err(|e| format!("merged config: {e}"))?;
409
410    let path = bundle_config_path(name, config);
411    if let Some(parent) = path.parent() {
412        std::fs::create_dir_all(parent).map_err(|e| format!("{}: {e}", parent.display()))?;
413    }
414    let body = format!(
415        "# Generated by tamias-driver-bupha deploy — do not edit.\n\
416         # Source: {} fleet.toml + tamias.yaml provider.ingress.\n\
417         # Regenerate: tamias deploy {name} (or `tamias driver exec {name} deploy`).\n\
418         {text}",
419        dir.display()
420    );
421    std::fs::write(&path, body).map_err(|e| format!("{}: {e}", path.display()))?;
422    Ok(format!(
423        "wrote operator config {} ({} repo entries)",
424        path.display(),
425        cfg.repos.len()
426    ))
427}

429/// `uninstall` — remove the config `deploy` wrote. `--dir` is
430/// accepted for dispatch parity and ignored; `--config`/`--name`
431/// resolve the same path deploy used.
432pub fn uninstall(name: &str, config: Option<&Path>) -> String {
433    let path = bundle_config_path(name, config);
434    if path.is_file() {
435        let _ = std::fs::remove_file(&path);
436    }
437    format!("removed operator config {}", path.display())
438}

450/// `~/.config/bupha/env` — the dotenv leg of the secret chain (same
451/// file the provider unit's `env_file` loads).
452fn env_file() -> PathBuf {
453    config_dir().join("env")
454}
455
456/// Secret resolution chain, mirroring `bupha`'s own `secret()`
457/// (T-056 — cuora canonical): env var → cuora broker
458/// (`cuora://bupha/<NAME>` via the `cuora` CLI) → OS keyring →
459/// `~/.config/bupha/env` (both LEGACY fallbacks). Values are used
460/// in-flight only — never printed, never written to configs.
461fn secret(name: &str) -> Option<String> {
462    use bupha_secret::Secrets;
463    std::env::var(name)
464        .ok()
465        .filter(|v| !v.is_empty())
466        .or_else(|| bupha_secret_cuora::CuoraSecrets::default().get(name))
467        .or_else(|| bupha_keyring::KeyringSecrets::default().get(name))
468        .or_else(|| bupha_dotenv::DotenvSecrets { path: env_file() }.get(name))
469}
470
471fn missing_secret(name: &str) -> String {
472    format!("{name} required (env, cuora, keyring or ~/.config/bupha/env)")
473}

477/// One concrete hook target: a github-hook repo + the public URL
478/// bupha's `/hook` route hangs off.
479struct HookTarget {
480    repo: String,
481    /// `provider.ingress[].public_url` — `register_webhook` appends
482    /// `/hook` itself.
483    public_url: String,
484    /// Declared event names (bupha triggers on `push` only).
485    events: Vec<String>,
486    /// The repo's webhook-secret env NAME (fleet `secret_env`, else
487    /// the trigger global).
488    secret_env: String,
489    /// The forge REST base.
490    api_base: String,
491}
492
493/// Resolve hook targets: manifest `provider.ingress[].hooks[]`
494/// (github only) for the repo/URL set, `fleet.toml` for each repo's
495/// `secret_env` override and the global defaults. Non-github forges
496/// stay declared data — bupha has no forge impl for them.
497fn hook_targets(dir: &Path, api_base: Option<&str>) -> Result<Vec<HookTarget>, String> {
498    let manifest = load_manifest(dir)?;
499    let provider = provider(&manifest, dir)?;
500    let url = ingress_url(provider)
501        .map(|i| i.public_url.trim().trim_end_matches('/').to_string())
502        .filter(|u| !u.is_empty())
503        .ok_or_else(|| {
504            format!(
505                "{}: provider.ingress declares no public_url — nothing to hook",
506                dir.display()
507            )
508        })?;
509    let repos = github_repos(provider);
510    if repos.is_empty() {
511        return Err(format!(
512            "{}: provider.ingress declares no github hooks — nothing to manage",
513            dir.display()
514        ));
515    }
516    // Fleet supplies per-repo secret_env + the global fallbacks.
517    // Missing fleet → every repo verifies with the global secret,
518    // same as a config without overrides.
519    let fleet = fleet_config(dir).ok();
520    let global_secret = fleet
521        .as_ref()
522        .map(|c| c.trigger.secret_env.clone())
523        .unwrap_or_else(|| "BUPHA_WEBHOOK_SECRET".into());
524    let default_api = fleet
525        .as_ref()
526        .map(|c| c.trigger.api_base.clone())
527        .unwrap_or_else(|| "https://api.github.com".into());
528    let base = api_base.unwrap_or(&default_api).to_string();
529    let mut targets = Vec::new();
530    for ingress in &provider.ingress {
531        for hook in ingress
532            .hooks
533            .iter()
534            .filter(|h| h.forge.eq_ignore_ascii_case("github"))
535        {
536            let secret_env = fleet
537                .as_ref()
538                .and_then(|c| c.repos.iter().find(|r| r.name == hook.repo))
539                .and_then(|r| r.secret_env.clone())
540                .unwrap_or_else(|| global_secret.clone());
541            let events = if hook.events.is_empty() {
542                vec!["push".to_string()]
543            } else {
544                hook.events.clone()
545            };
546            let t = HookTarget {
547                repo: hook.repo.clone(),
548                public_url: url.clone(),
549                events,
550                secret_env,
551                api_base: base.clone(),
552            };
553            if !targets
554                .iter()
555                .any(|o: &HookTarget| o.repo == t.repo && o.public_url == t.public_url)
556            {
557                targets.push(t);
558            }
559        }
560    }
561    Ok(targets)
562}

564/// `hooks register|uninstall` — upsert/remove the GitHub webhooks
565/// the manifest's `provider.ingress[].hooks[]` declare, via
566/// `bupha_github`'s REST path (exact-URL match, idempotent).
567///
568/// **Standalone mode** (ADR-0007): this registers hooks at the
569/// bupha webhook listener — the fallback ingress for deployments
570/// with no peregrin. Production ingress is a peregrin campaign
571/// listener enqueuing `ci-job` via `ci.enqueue`; don't point
572/// production webhooks here.
573///
574/// Register needs the forge token (`--token-env`/`[runner]
575/// token_env`, default `BUPHA_TOKEN`) and each repo's webhook
576/// secret by NAME; unregister needs only the token. `--dry-run`
577/// prints the plan and resolves nothing.
578///
579/// # Errors
580/// Fails on an unreadable manifest/fleet, missing secrets, or forge
581/// API errors. Returns the report text (driver-verb output).
582pub fn hooks(f: &Flags, register: bool) -> Result<String, String> {
583    let verb = if register { "register" } else { "unregister" };
584    let (dir, _name) = required(f, "hooks")?;
585    let targets = hook_targets(&dir, f.api_base.as_deref())?;
586    if f.dry_run {
587        return Ok(targets
588            .iter()
589            .map(|t| {
590                format!(
591                    "{}: would {} {}/hook events=[{}] secret={} (name — value never shown)",
592                    t.repo,
593                    verb,
594                    t.public_url,
595                    t.events.join(","),
596                    t.secret_env
597                )
598            })
599            .collect::<Vec<_>>()
600            .join("\n"));
601    }
602    let fleet = fleet_config(&dir).ok();
603    let token_name = f
604        .token_env
605        .clone()
606        .or_else(|| fleet.as_ref().map(|c| c.runner.token_env.clone()))
607        .unwrap_or_else(|| "BUPHA_TOKEN".into());
608    let token = secret(&token_name).ok_or_else(|| missing_secret(&token_name))?;
609    let mut lines = Vec::with_capacity(targets.len());
610    for t in &targets {
611        let msg = if register {
612            // bupha triggers on push deliveries only — a manifest
613            // declaring more gets them noted, not registered.
614            if t.events.iter().any(|e| e != "push") {
615                eprintln!(
616                    "warning: {} declares events [{}] — bupha registers [push] only",
617                    t.repo,
618                    t.events.join(",")
619                );
620            }
621            let wh = secret(&t.secret_env).ok_or_else(|| missing_secret(&t.secret_env))?;
622            bupha_github::register_webhook(&token, &t.repo, &t.public_url, &wh, &t.api_base)
623                .map_err(|e| format!("webhook {}: {e}", t.repo))?
624        } else {
625            bupha_github::unregister_webhook(&token, &t.repo, &t.public_url, &t.api_base)
626                .map_err(|e| format!("webhook {}: {e}", t.repo))?
627        };
628        lines.push(msg);
629    }
630    Ok(lines.join("\n"))
631}

635/// `systemctl --user <args>` — stdout trimmed, else stderr trimmed,
636/// else `None` when systemd can't answer (no user bus, missing
637/// binary). Exit status ignored: `is-active`/`is-enabled` exit
638/// non-zero for inactive/disabled/not-found states.
639fn systemctl(args: &[&str]) -> Option<String> {
640    let out = std::process::Command::new("systemctl")
641        .arg("--user")
642        .args(args)
643        .output()
644        .ok()?;
645    let text = String::from_utf8_lossy(&out.stdout).trim().to_string();
646    if text.is_empty() {
647        let err = String::from_utf8_lossy(&out.stderr).trim().to_string();
648        return (!err.is_empty()).then_some(err);
649    }
650    Some(text)
651}

653/// The provider's unit base names — `<provider-or-package>-<service>`
654/// `.service`, the same derivation `tamias deploy` renders
655/// (`tamias-provider::unit_name`).
656fn provider_units(manifest: &Manifest, fallback_name: &str) -> Vec<(String, String)> {
657    let Some(provider) = &manifest.provider else {
658        return Vec::new();
659    };
660    let base = provider
661        .name
662        .clone()
663        .unwrap_or_else(|| match &manifest.scope {
664            Some(scope) => format!("{scope}-{}", manifest.name),
665            None if manifest.name.is_empty() => fallback_name.to_string(),
666            None => manifest.name.clone(),
667        });
668    provider
669        .services
670        .iter()
671        .map(|s| (s.name.clone(), format!("{base}-{}.service", s.name)))
672        .collect()
673}

675/// `health` — probe each ingress `listen`+`path` with the webhook
676/// `ping` (`POST <listen><path>` + `X-GitHub-Event: ping` → `200
677/// pong`), falling back to `systemctl --user is-active` per provider
678/// unit when no ingress declares both. Prints a JSON status object.
679///
680/// # Errors
681/// Fails when the manifest can't be read or nothing is probeable.
682pub fn health(dir: &Path, name: &str) -> Result<String, String> {
683    let manifest = load_manifest(dir)?;
684    let provider = provider(&manifest, dir)?;
685    let agent = ureq::Agent::config_builder()
686        .timeout_global(Some(std::time::Duration::from_secs(5)))
687        .build()
688        .new_agent();
689    let mut probes = Vec::new();
690    for ingress in &provider.ingress {
691        let (Some(listen), path) = (&ingress.listen, ingress.path.as_deref()) else {
692            continue;
693        };
694        // bupha-webhook answers only /hook — an undeclared path
695        // probes the real route, a declared non-/hook path still
696        // gets probed as declared (and reports the 404 honestly).
697        let path = path.unwrap_or(HOOK_PATH);
698        let url = format!("http://{listen}{path}");
699        let result = agent
700            .post(&url)
701            .header("x-github-event", "ping")
702            .header("User-Agent", "bupha")
703            .send_empty()
704            .map_or_else(
705                |e| (false, e.to_string()),
706                |mut resp| {
707                    let status = resp.status().as_u16();
708                    let body = resp
709                        .body_mut()
710                        .read_to_string()
711                        .unwrap_or_else(|_| String::new());
712                    (status == 200, format!("{status} {}", body.trim()))
713                },
714            );
715        probes.push(json!({
716            "kind": "hook-ping",
717            "url": url,
718            "ok": result.0,
719            "detail": result.1,
720        }));
721    }
722    if probes.is_empty() {
723        for (service, unit) in provider_units(&manifest, name) {
724            let state = systemctl(&["is-active", &unit]).unwrap_or_else(|| "unknown".into());
725            probes.push(json!({
726                "kind": "unit",
727                "service": service,
728                "unit": unit,
729                "ok": state == "active",
730                "detail": state,
731            }));
732        }
733    }
734    if probes.is_empty() {
735        return Err(format!(
736            "{name}: provider declares no ingress listeners or services — nothing to probe"
737        ));
738    }
739    let ok = probes.iter().all(|p| p["ok"].as_bool() == Some(true));
740    serde_json::to_string(&json!({"name": name, "ok": ok, "probes": probes}))
741        .map_err(|e| e.to_string())
742}

744/// `status` — per-unit `systemctl --user is-active`/`is-enabled` as a
745/// JSON object; units systemd can't answer report `"unknown"`.
746///
747/// # Errors
748/// Fails when the manifest can't be read or declares no services.
749pub fn status(dir: &Path, name: &str) -> Result<String, String> {
750    let manifest = load_manifest(dir)?;
751    provider(&manifest, dir)?;
752    let units = provider_units(&manifest, name);
753    if units.is_empty() {
754        return Err(format!(
755            "{name}: provider declares no services — nothing to report"
756        ));
757    }
758    let rows: Vec<Value> = units
759        .iter()
760        .map(|(service, unit)| {
761            let active = systemctl(&["is-active", unit]).unwrap_or_else(|| "unknown".into());
762            let enabled = systemctl(&["is-enabled", unit]).unwrap_or_else(|| "unknown".into());
763            json!({
764                "service": service,
765                "unit": unit,
766                "active": active,
767                "enabled": enabled,
768                "ok": active == "active",
769            })
770        })
771        .collect();
772    let ok = rows.iter().all(|r| r["ok"].as_bool() == Some(true));
773    serde_json::to_string(&json!({"name": name, "ok": ok, "units": rows}))
774        .map_err(|e| e.to_string())
775}

777/// `logs` — `journalctl --user -u <unit>` over every provider unit,
778/// newest `lines` entries, no follow (dispatch is bounded — a `-f`
779/// would just be killed). Returns the journal text.
780///
781/// # Errors
782/// Fails when the manifest can't be read or declares no services.
783pub fn logs(dir: &Path, name: &str, lines: u32) -> Result<String, String> {
784    use std::fmt::Write as _;
785    let manifest = load_manifest(dir)?;
786    provider(&manifest, dir)?;
787    let units = provider_units(&manifest, name);
788    if units.is_empty() {
789        return Err(format!(
790            "{name}: provider declares no services — nothing to journal"
791        ));
792    }
793    let n = lines.max(1).to_string();
794    let mut out = String::new();
795    for (_, unit) in &units {
796        let _ = writeln!(out, "── {unit} ──");
797        let journal = std::process::Command::new("journalctl")
798            .args(["--user", "-u", unit, "-n", &n, "--no-pager"])
799            .output()
800            .map_err(|e| format!("journalctl {unit}: {e}"))?;
801        out.push_str(&String::from_utf8_lossy(&journal.stdout));
802        if !journal.status.success() {
803            out.push_str(&String::from_utf8_lossy(&journal.stderr));
804        }
805    }
806    Ok(out)
807}

923/// `host` — engine-host mode: run the watcher pipeline
924/// **in-process** so deployed provider units exec this driver
925/// binary instead of a cargo-installed `bupha` (ADR-0006: no
926/// user-facing bupha CLI on the host path; bupha T-052).
927///
928/// Loads the operator config `deploy` wrote — `--config <path>`,
929/// else `~/.config/bupha/bundles/<name>.toml` (`--name` required
930/// then) — and blocks in `bupha_watch::watch`: trigger listener +
931/// queues + runner + status sinks, identical to `bupha watch`.
932/// `--dir` is accepted (tamias's dispatch contract) but unused —
933/// the payload's `fleet.toml` only matters at `deploy` time.
934///
935/// # Errors
936/// Fails on a missing `--config`/`--name` or an unloadable config;
937/// once running, config errors inside `watch` exit the process
938/// like the CLI's did.
939pub fn host(f: &Flags) -> Result<String, String> {
940    let path = match f.config.clone() {
941        Some(p) => p,
942        None => bundle_config_path(
943            f.name
944                .as_deref()
945                .ok_or_else(|| "host: --config or --name is required".to_string())?,
946            None,
947        ),
948    };
949    use bupha_config::ConfigSource;
950    let cfg = bupha_config_file::FileSource {
951        path: Some(path.to_string_lossy().to_string()),
952    }
953    .load()
954    .map_err(|e| format!("host: {}: {e}", path.display()))?;
955    // Blocks until the sources stop (e2e shutdown hook, signal) —
956    // the unit's Restart= policy owns the lifecycle from there.
957    bupha_watch::watch(std::sync::Arc::new(cfg));
958    Ok(String::new())
959}

961/// `worker` — the `ci-job` consumption loop (P-002 §4.1): poll
962/// `kind="ci-job"` entities on a hypha/peregrin node, `ci.claim`
963/// the pending ones this box's `[worker] labels` cover, run them
964/// through the same `bupha-watch` pipeline `host` uses (worktree →
965/// runner → status sink), `ci.renew` on the run ticker,
966/// `ci.complete` with the result. This is the **second service** a
967/// fleet deployment runs alongside `host` — the watcher ingests
968/// pushes (standalone mode) while workers execute jobs the
969/// entity plane hands out; in peregrin-hosted mode the plane lives
970/// on the node and `host` isn't needed at all.
971///
972/// Node endpoint resolution: `--node` → `BUPHA_WORKER_NODE` →
973/// `[worker] node` in the operator config. Credential: `--node-token-env`
974/// / `[worker] node_token_env` names a secret resolved through the
975/// usual chain (env → cuora → keyring → `~/.config/bupha/env`);
976/// `BUPHA_NODE_TOKEN` is a direct-token escape hatch; neither set =
977/// unauthenticated (dev nodes).
978///
979/// # Errors
980/// Fails on a missing `--config`/`--name`, an unloadable config, a
981/// missing node endpoint, or an unresolvable token NAME; once
982/// running, poll errors back off and retry.
983pub fn worker(f: &Flags) -> Result<String, String> {
984    let path = match f.config.clone() {
985        Some(p) => p,
986        None => bundle_config_path(
987            f.name
988                .as_deref()
989                .ok_or_else(|| "worker: --config or --name is required".to_string())?,
990            None,
991        ),
992    };
993    use bupha_config::ConfigSource;
994    use bupha_watch::entity_worker::JobExec as _;
995    let cfg = bupha_config_file::FileSource {
996        path: Some(path.to_string_lossy().to_string()),
997    }
998    .load()
999    .map_err(|e| format!("worker: {}: {e}", path.display()))?;
1000
1001    let node = f
1002        .node
1003        .clone()
1004        .or_else(|| {
1005            std::env::var("BUPHA_WORKER_NODE")
1006                .ok()
1007                .filter(|s| !s.is_empty())
1008        })
1009        .or_else(|| cfg.worker.node.clone())
1010        .ok_or_else(|| {
1011            "worker: no node endpoint — set [worker] node, --node, or BUPHA_WORKER_NODE".to_string()
1012        })?;
1013    let token =
1014        match f
1015            .node_token_env
1016            .clone()
1017            .or_else(|| cfg.worker.node_token_env.clone())
1018        {
1019            Some(name) => Some(bupha_watch::secret(&name).ok_or_else(|| {
1020                format!("worker: {name} required (env, cuora, keyring or env file)")
1021            })?),
1022            None => std::env::var("BUPHA_NODE_TOKEN")
1023                .ok()
1024                .filter(|s| !s.is_empty()),
1025        };
1026
1027    let api = bupha_watch::entity_worker::EntityJobApi::http(&node, token)
1028        .map_err(|e| format!("worker: {e}"))?;
1029    let filter = bupha_watch::entity_worker::ClaimFilter::from_config(&cfg)
1030        .map_err(|e| format!("worker: config: {e}"))?;
1031    let exec = bupha_watch::entity_worker::WatchExec::new(std::sync::Arc::new(cfg));
1032    // Same e2e shutdown hook as `host` — the loop
1033    // checks it between jobs (a run in flight finishes first).
1034    let shutdown = || {
1035        std::env::var("BUPHA_WATCH_SHUTDOWN_FILE")
1036            .ok()
1037            .is_some_and(|f| Path::new(&f).exists())
1038    };
1039    println!(
1040        "ci-job worker polling entities on {node} ({})",
1041        exec.worker()
1042    );
1043    bupha_watch::entity_worker::run(&filter, &api, &exec, &shutdown);
1044    Ok(String::new())
1045}

1049/// Run one tamias verb. `verb` is the subcommand `main.rs` already
1050/// matched; `args` is the remaining argv (`--dir … --name …` plus
1051/// verb flags). Prints the verb's report to stdout; errors go to
1052/// stderr with a non-zero exit — the tamias dispatch contract.
1053pub fn run(verb: &str, args: &[String]) -> i32 {
1054    let result = match verb {
1055        "install" => parse_flags(args).and_then(|f| {
1056            required(&f, "install")?;
1057            Ok(install().to_string())
1058        }),
1059        "deploy" => parse_flags(args).and_then(|f| {
1060            let (dir, name) = required(&f, "deploy")?;
1061            deploy(&dir, &name, f.config.as_deref())
1062        }),
1063        "uninstall" => parse_flags(args).and_then(|f| {
1064            let (_dir, name) = required(&f, "uninstall")?;
1065            Ok(uninstall(&name, f.config.as_deref()))
1066        }),
1067        "hooks" => parse_flags(args).and_then(|f| {
1068            let action = f
1069                .positional
1070                .first()
1071                .map(String::as_str)
1072                .ok_or_else(|| "hooks: expected `register` or `unregister`".to_string())?;
1073            match action {
1074                "register" => hooks(&f, true),
1075                "unregister" | "uninstall" => hooks(&f, false),
1076                other => Err(format!("hooks: unknown action '{other}'")),
1077            }
1078        }),
1079        "health" => parse_flags(args).and_then(|f| {
1080            let (dir, name) = required(&f, "health")?;
1081            health(&dir, &name)
1082        }),
1083        "status" => parse_flags(args).and_then(|f| {
1084            let (dir, name) = required(&f, "status")?;
1085            status(&dir, &name)
1086        }),
1087        "logs" => parse_flags(args).and_then(|f| {
1088            let (dir, name) = required(&f, "logs")?;
1089            logs(&dir, &name, f.lines)
1090        }),
1091        "host" => parse_flags(args).and_then(|f| host(&f)),
1092        "worker" => parse_flags(args).and_then(|f| worker(&f)),
1093        "config-schema" => {
1094            // Instance-agnostic probe — flags accepted, ignored.
1095            match config_schema() {
1096                Ok(s) => {
1097                    println!("{s}");
1098                    return 0;
1099                }
1100                Err(e) => Err(e),
1101            }
1102        }
1103        other => Err(format!("unknown verb '{other}'")),
1104    };
1105    match result {
1106        Ok(output) => {
1107            if !output.is_empty() {
1108                println!("{output}");
1109            }
1110            0
1111        }
1112        Err(e) => {
1113            eprintln!("tamias-driver-bupha: {e}");
1114            1
1115        }
1116    }
1117}

1334                r#"
