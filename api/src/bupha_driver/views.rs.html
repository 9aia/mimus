1//! views — the driver's manifest: every ViewSpec plus the query
2//! resolvers that feed them.
3//!
4//! Read path: every query resolves a [`crate::data::Plane`] (params
5//! `base`/`config`, else the watcher's own config discovery) and
6//! reads `history.jsonl` / `queue.json` / the systemd units — see
7//! `data.rs`. A failed read degrades to an `{"error": …}` or
8//! `{"note": …}` payload inside an `ok:true` response, matching the
9//! sibling drivers' contract.
10//!
11//! Surface mapping:
12//!
13//! | pavo view       | data                                        |
14//! |-----------------|---------------------------------------------|
15//! | `bupha:status`  | plane digest — base, queue kind, counts, unit states |
16//! | `bupha:runs`    | `history.jsonl` tail (`repo`/`limit` params)  |
17//! | `bupha:queue`   | `queue.json` pending/running per key          |
18//! | `bupha:repos`   | latest history record per repo/job key        |
19//! | `bupha:service` | `systemctl --user` state of watch/worker units |
20

30// Node/manifest construction goes through pavo-driver-kit — the
31// constructors emit the same ViewNode/QueryBinding structs the
32// hand-rolled literals did.
33
34fn col(title: &str, children: Vec<ViewNode>) -> ViewNode {
35    kit::column(kit::base().title(title), children)
36}
37
38fn kv(title: &str, query: &str) -> ViewNode {
39    kit::kv(kit::base().title(title).query(query), Vec::new())
40}
41
42fn table(title: &str, query: &str, columns: &[(&str, &str)]) -> ViewNode {
43    kit::sortable_table(
44        kit::base().title(title).query(query),
45        columns
46            .iter()
47            .map(|(key, t)| kit::table_column(key, t))
48            .collect::<Vec<TableColumn>>(),
49        Vec::new(),
50    )
51}
52
53fn spec(
54    id: &str,
55    title: &str,
56    route: &str,
57    description: &str,
58    queries: &[&str],
59    root: ViewNode,
60) -> ViewSpec {
61    let mut view = kit::view(id, title, route, root).description(description);
62    for q in queries {
63        view = view.query(kit::binding(q, "query"));
64    }
65    view.build()
66}

138/// The `manifest` op result — built with `pavo_driver_kit`'s
139/// `Manifest` fluent builder (verbs/capabilities default to
140/// `["manifest","serve"]`/`["pavo"]`, matching the literals this
141/// replaces).
142pub fn manifest() -> DriverManifest {
143    let mut navbar = kit::contribution("runs", "CI", "/bupha/runs");
144    navbar.priority = 20;
145    let mut dashboard = kit::contribution("status", "Bupha status", "/bupha/status");
146    dashboard.priority = 30;
147    let mut statusbar = kit::contribution("status", "Bupha queue", "/bupha/status");
148    statusbar.priority = 40;
149
150    let mut manifest = kit::Manifest::new("bupha", "Bupha")
151        .version(env!("CARGO_PKG_VERSION"))
152        .icon("🐦")
153        .navbar(navbar)
154        .dashboard(dashboard)
155        .statusbar(statusbar);
156    for view in specs() {
157        manifest = manifest.view(view);
158    }
159    manifest.build()
160}

164/// Every query name the driver answers — the `query` op rejects
165/// anything outside this set with `ok:false`.
166pub fn known_query(name: &str) -> bool {
167    matches!(name, "summary" | "runs" | "queue" | "repos" | "units")
168}
169
170/// `{"op":"query","query":name,"params":{…}}` → the data payload.
171/// Failures degrade to `{"error": …}`/`{"note": …}` objects, never
172/// `ok:false`.
173pub fn query_data(name: &str, params: &Map<String, Value>) -> Value {
174    match name {
175        "summary" => summary(params),
176        "runs" => runs_rows(params),
177        "queue" => queue_rows(params),
178        "repos" => repos_rows(params),
179        "units" => Value::Array(unit_rows()),
180        other => json!({"error": format!("unknown query '{other}'")}),
181    }
182}
183
184/// Default row cap for `runs`/`repos` — history.jsonl is unbounded;
185/// `--param limit=` overrides.
186const DEFAULT_LIMIT: u64 = 100;
187
188/// `summary` query — the status view's kv object: resolved plane
189/// plus live counts. Sub-read failures degrade to their message
190/// rather than failing the panel.
191fn summary(ctx: &Map<String, Value>) -> Value {
192    let plane = data::resolve(ctx);
193    let history = data::history_path(&plane.base);
194    let queue = match data::queue_entries(&plane.base, &plane.queue_kind) {
195        Ok(entries) => format!(
196            "{} key(s), {} pending",
197            entries.len(),
198            data::pending_total(&entries)
199        ),
200        Err(e) => e,
201    };
202    let mut out = Map::new();
203    out.insert("base".into(), json!(plane.base));
204    out.insert("queueKind".into(), json!(plane.queue_kind));
205    out.insert(
206        "history".into(),
207        json!(if history.exists() {
208            "present"
209        } else {
210            "absent"
211        }),
212    );
213    out.insert("queue".into(), json!(queue));
214    for (unit, status) in data::unit_statuses() {
215        out.insert(unit.trim_end_matches(".service").to_string(), json!(status));
216    }
217    Value::Object(out)
218}

220/// One `history.jsonl` record → table row. Pure — unit-tested with
221/// synthetic JSON.
222fn run_row(r: &Value) -> Value {
223    json!({
224        "repo": r["repo"],
225        "sha": r["sha"],
226        "status": r["status"],
227        "wall_secs": r["wall_secs"],
228    })
229}
230
231/// `runs` query — the tail of `history.jsonl`, newest last.
232fn runs_rows(ctx: &Map<String, Value>) -> Value {
233    let plane = data::resolve(ctx);
234    let repo = data::param_str(ctx, "repo");
235    let limit = data::param_u64(ctx, "limit").unwrap_or(DEFAULT_LIMIT) as usize;
236    Value::Array(
237        data::run_records(&plane.base, repo, limit)
238            .iter()
239            .map(run_row)
240            .collect(),
241    )
242}
243
244/// Short sha for the pending-shas preview column.
245fn short_sha(sha: &str) -> &str {
246    sha.get(..12).unwrap_or(sha)
247}
248
249/// One queue entry → table row. `pending_shas` previews up to five
250/// pending shas (short form); the count column stays exact.
251fn queue_row(key: &str, q: &bupha_queue::PersistedQueue) -> Value {
252    let shas: Vec<&str> = q.pending.iter().map(|s| short_sha(s)).collect();
253    let preview = if shas.len() > 5 {
254        format!("{} …", shas[..5].join(" "))
255    } else {
256        shas.join(" ")
257    };
258    json!({
259        "key": key,
260        "pending": q.pending.len(),
261        "running": q.running.as_deref().map(short_sha).unwrap_or("—"),
262        "pending_shas": preview,
263    })
264}
265
266/// `queue` query — persisted pending/running per key. Non-json
267/// queue kinds degrade to a note (retired redb/redis fleet state is
268/// unreadable here); a missing store reads as an empty queue.
269fn queue_rows(ctx: &Map<String, Value>) -> Value {
270    let plane = data::resolve(ctx);
271    match data::queue_entries(&plane.base, &plane.queue_kind) {
272        Ok(entries) => Value::Array(entries.iter().map(|(k, q)| queue_row(k, q)).collect()),
273        Err(e) => json!({"note": e}),
274    }
275}

277/// Latest history record per repo/job key — history is append-only,
278/// so the last write wins. Returns rows in repo-key order.
279fn latest_by_repo(records: &[Value]) -> Vec<Value> {
280    let mut latest: BTreeMap<String, &Value> = BTreeMap::new();
281    for r in records {
282        if let Some(repo) = r["repo"].as_str() {
283            latest.insert(repo.to_string(), r);
284        }
285    }
286    latest.values().map(|r| run_row(r)).collect()
287}
288
289/// `repos` query — per-key latest status (the "statuses on each
290/// sha" surface), optionally repo-filtered.
291fn repos_rows(ctx: &Map<String, Value>) -> Value {
292    let plane = data::resolve(ctx);
293    let repo = data::param_str(ctx, "repo");
294    // Latest-per-key is bounded by the number of watched keys, not
295    // history length — `limit` doesn't apply here.
296    let records = data::run_records(&plane.base, repo, usize::MAX);
297    Value::Array(latest_by_repo(&records))
298}

300/// `units` query — systemd state rows for the watcher units. The
301/// status line is `enabled=<state> active=<state>`; unparseable
302/// probes leave the fields at "?" with the raw line in `detail`.
303fn unit_rows() -> Vec<Value> {
304    data::unit_statuses()
305        .into_iter()
306        .map(|(unit, s)| {
307            let mut enabled = "?";
308            let mut active = "?";
309            let mut parsed = false;
310            for tok in s.split_whitespace() {
311                if let Some(v) = tok.strip_prefix("enabled=") {
312                    enabled = v;
313                    parsed = true;
314                } else if let Some(v) = tok.strip_prefix("active=") {
315                    active = v;
316                    parsed = true;
317                }
318            }
319            json!({
320                "unit": unit,
321                "enabled": enabled,
322                "active": active,
323                "detail": if parsed { "" } else { s.as_str() },
324            })
325        })
326        .collect()
327}

331/// `{"op":"view","view":id,"params":{…}}` → [`pavo_protocol::wire::ViewResult`].
332pub fn view_result(
333    view_id: &str,
334    params: &Map<String, Value>,
335) -> Result<pavo_protocol::wire::ViewResult, String> {
336    let spec = specs()
337        .into_iter()
338        .find(|v| v.id == view_id)
339        .ok_or_else(|| format!("unknown view: {view_id}"))?;
340
341    let mut data_map = Map::new();
342    for b in &spec.queries {
343        data_map.insert(b.name.clone(), query_data(&b.name, params));
344    }
345
346    Ok(pavo_protocol::wire::ViewResult {
347        view: spec,
348        data: data_map,
349    })
350}
