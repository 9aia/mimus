1//! sturn — the engine as a library. Everything that used to live
2//! in the binary lives here: Cli/Cmd, the WAL/bus/ws plumbing,
3//! claims/asks, snap/push, daemon/sub/mount surfaces.
4//! `run()` is the compat CLI's whole body; drivers (sturn bin,
5//! tamias-driver-sturn `run`) split it into [`parse`]/[`parse_from`]
6//! + [`dispatch`] — argv in, exit code out, ops callable as
7//!   `dispatch(&cli)`.
8//!
9//! sturn — evented coordination over the git object store.
10//!
11//! The WAL is the state; the tree is a projection; the bus is a
12//! cache. Composition root: wires ports (wal/bus/ws/vcs/config/
13//! service) to impls (git plumbing, redb, toml, systemd).

421/// One parsed command → exit code. The dispatch every driver
422/// shares; argv parsing is a caller concern.
423pub fn dispatch(cli: &Cli) -> i32 {
424    match execute(cli) {
425        Ok(()) => {
426            post_op(cli);
427            0
428        }
429        Err(e) => {
430            eprintln!("sturn: {e}");
431            1
432        }
433    }
434}

436/// Whether `cmd` moved refs — the `[hooks] post_op` list fires
437/// only after these land, so read verbs never spawn the hook.
438/// "Moved" includes appends (WAL events are commits under
439/// refs/sturn/wal/*), ref creates/moves (ws new, snap), pushes
440/// (remote-tracking refs move), and surfaces that write through
441/// a ws (hydrate/exec --writeback, mount --rw).
442fn mutates(cmd: &Cmd) -> bool {
443    match cmd {
444        Cmd::Claim { .. }
445        | Cmd::Release { .. }
446        | Cmd::Join { .. }
447        | Cmd::Leave { .. }
448        | Cmd::Write { .. }
449        | Cmd::Rename { .. }
450        | Cmd::Delete { .. }
451        | Cmd::Intent { .. }
452        | Cmd::Ask { .. }
453        | Cmd::Answer { .. }
454        | Cmd::Cancel { .. }
455        | Cmd::Review { .. }
456        | Cmd::Compact { .. }
457        | Cmd::Snap { .. }
458        | Cmd::Push { .. } => true,
459        Cmd::Ws { cmd } => matches!(
460            cmd,
461            WsCmd::New { .. }
462                | WsCmd::Hydrate {
463                    writeback: true,
464                    ..
465                }
466        ),
467        Cmd::Exec { writeback, .. } => *writeback,
468        Cmd::Mount { rw, .. } => *rw,
469        _ => false,
470    }
471}

473/// `[hooks] post_op` — spawn each configured command detached
474/// (`sh -c`, null stdio, cwd = repo dir) after a successful
475/// ref-moving verb. Best-effort by contract: a spawn failure
476/// warns on stderr, hook exit status is never observed, and the
477/// op's exit code already printed above stands. Off by default.
478fn post_op(cli: &Cli) {
479    if !mutates(&cli.cmd) {
480        return;
481    }
482    let cfg: Option<Config> = sturn_config_toml::TomlSource {
483        path: cli.config.clone(),
484    }
485    .load()
486    .ok();
487    let cmds = cfg
488        .as_ref()
489        .map(|c| c.hooks.clone())
490        .unwrap_or_default()
491        .post_ops();
492    if cmds.is_empty() {
493        return;
494    }
495    let repo = cli
496        .repo
497        .clone()
498        .or_else(|| cfg.as_ref().map(|c| c.repo.dir.clone()))
499        .unwrap_or_else(|| PathBuf::from("."));
500    for cmd in &cmds {
501        // Dropped Child never reaps — the hook outlives the op
502        // and the process exits anyway, so no zombie lingers.
503        if let Err(e) = std::process::Command::new("sh")
504            .arg("-c")
505            .arg(cmd)
506            .current_dir(&repo)
507            .stdin(std::process::Stdio::null())
508            .stdout(std::process::Stdio::null())
509            .stderr(std::process::Stdio::null())
510            .spawn()
511        {
512            eprintln!("sturn: post_op hook `{cmd}`: {e}");
513        }
514    }
515}

532fn execute(cli: &Cli) -> Result<(), String> {
533    // Config is optional — everything works flag-driven too.
534    let cfg: Option<Config> = sturn_config_toml::TomlSource {
535        path: cli.config.clone(),
536    }
537    .load()
538    .ok();
539    let repo = cli
540        .repo
541        .clone()
542        .or_else(|| cfg.as_ref().map(|c| c.repo.dir.clone()))
543        .unwrap_or_else(|| PathBuf::from("."));
544    let repo_id = cfg.as_ref().map(|c| c.repo.id()).unwrap_or_else(|| {
545        repo.canonicalize()
546            .ok()
547            .and_then(|p| p.file_name().map(|s| s.to_string_lossy().into_owned()))
548            .unwrap_or_else(|| "repo".into())
549    });
550    let branch = cfg
551        .as_ref()
552        .map(|c| c.repo.branch.clone())
553        .unwrap_or_else(|| "main".into());
554    let actor = cfg.as_ref().map(|c| c.actor.name()).unwrap_or_else(|| {
555        std::env::var("STURN_ACTOR")
556            .or_else(|_| std::env::var("USER"))
557            .unwrap_or_else(|_| "anon".into())
558    });
559    let wal = sturn_wal_git::GitWal {
560        ref_prefix: cfg
561            .as_ref()
562            .map(|c| c.wal.ref_prefix.clone())
563            .unwrap_or_else(|| "refs/sturn/wal".into()),
564        sign_key: cfg
565            .as_ref()
566            .and_then(|c| c.signing.key.clone())
567            .map(|k| k.replace('~', &std::env::var("HOME").unwrap_or_else(|_| ".".into())))
568            .map(PathBuf::from),
569    };
570    let ws_impl = sturn_ws_git::GitWs;
571    let bus_path = cfg
572        .as_ref()
573        .map(|c| c.bus.store.clone())
574        .unwrap_or_else(|| {
575            std::env::var("HOME")
576                .map(PathBuf::from)
577                .unwrap_or_else(|_| ".".into())
578                .join(".local/state/sturn/bus.redb")
579        });
580    let poll_ms = cfg.as_ref().map(|c| c.bus.poll_ms).unwrap_or(100);
581
582    match &cli.cmd {
583        Cmd::Init => {
584            if !Path::new(".git").exists() && !repo.join(".git").exists() {
585                return Err("not a git repo — run inside one".into());
586            }
587            if Path::new(".jj").exists() || repo.join(".jj").exists() {
588                println!("colocated jj detected — refs/sturn/* stays git-side");
589            }
590            if !Path::new("sturn.toml").exists() {
591                std::fs::write(
592                    "sturn.toml",
593                    "# sturn — coordination over git refs\n[repo]\ndir = \".\"\n",
594                )
595                .map_err(|e| e.to_string())?;
596                println!("wrote sturn.toml");
597            }
598            // Converge AGENTS.md to the managed agent-awareness
599            // block — markers scoped, prose untouched, idempotent.
600            match agentmd::sync(Path::new("."))? {
601                agentmd::Change::Created => {
602                    println!("wrote AGENTS.md (sturn agent-awareness block)")
603                }
604                agentmd::Change::Updated => {
605                    println!("updated AGENTS.md agent-awareness block")
606                }
607                agentmd::Change::Unchanged => {}
608            }
609            println!("sturn ready — WAL: refs/sturn/wal/*, ws: refs/sturn/ws/*");
610            Ok(())
611        }
612        Cmd::Ws { cmd } => match cmd {
613            WsCmd::New { name, base } => {
614                let base = base.clone().unwrap_or_else(|| "HEAD".into());
615                ws_impl
616                    .create(&repo, name, &base)
617                    .map_err(|e| e.to_string())?;
618                println!("ws {name} → {base}");
619                Ok(())
620            }
621            WsCmd::Ls => {
622                // Union: ws namespaces (pointer-only) + WAL refs.
623                let mut names = ws_impl.workspaces(&repo).unwrap_or_default();
624                names.extend(wal.workspaces(&repo).unwrap_or_default());
625                names.sort();
626                names.dedup();
627                names.iter().for_each(|n| println!("{n}"));
628                Ok(())
629            }
630            WsCmd::Read { name, path } => {
631                let (tip, pending) = ws_view(&repo, &wal, &ws_impl, name)?;
632                let ov = ws_overlay(&repo, &ws_impl, &tip, &pending);
633                match ov.get(path) {
634                    // Pending op wins — the ws's live view, same
635                    // fold `hydrate` materializes.
636                    Some(Some(sha)) => {
637                        let bytes = ws_impl.blob(&repo, sha).map_err(|e| e.to_string())?;
638                        print!("{}", String::from_utf8_lossy(&bytes));
639                    }
640                    Some(None) => return Err(format!("{path} deleted in ws overlay")),
641                    None => {
642                        let bytes = ws_impl.read(&repo, &tip, path).map_err(|e| e.to_string())?;
643                        print!("{}", String::from_utf8_lossy(&bytes));
644                    }
645                }
646                Ok(())
647            }
648            WsCmd::Paths { name, pattern } => {
649                let (tip, pending) = ws_view(&repo, &wal, &ws_impl, name)?;
650                let ov = ws_overlay(&repo, &ws_impl, &tip, &pending);
651                let paths = ws_paths(&repo, &ws_impl, &tip, &ov)?;
652                let pat = pattern.as_deref().unwrap_or("**");
653                for p in paths.iter().filter(|p| claims::glob_match(pat, p)) {
654                    println!("{p}");
655                }
656                Ok(())
657            }
658            WsCmd::Grep {
659                name,
660                pattern,
661                path,
662            } => {
663                let (tip, pending) = ws_view(&repo, &wal, &ws_impl, name)?;
664                let ov = ws_overlay(&repo, &ws_impl, &tip, &pending);
665                let paths = ws_paths(&repo, &ws_impl, &tip, &ov)?;
666                for p in &paths {
667                    if let Some(f) = path {
668                        if !claims::glob_match(f, p) {
669                            continue;
670                        }
671                    }
672                    let bytes = match ov.get(p) {
673                        Some(Some(sha)) => ws_impl.blob(&repo, sha).map_err(|e| e.to_string())?,
674                        // ws_paths already dropped pending deletes.
675                        Some(None) => continue,
676                        None => ws_impl.read(&repo, &tip, p).map_err(|e| e.to_string())?,
677                    };
678                    for (n, line) in String::from_utf8_lossy(&bytes).lines().enumerate() {
679                        if line.contains(pattern.as_str()) {
680                            println!("{p}:{}: {line}", n + 1);
681                        }
682                    }
683                }
684                Ok(())
685            }
686            WsCmd::Hydrate {
687                name,
688                dir,
689                paths,
690                writeback,
691            } => {
692                hydrate_ws(&repo, &wal, &ws_impl, name, dir, paths).map_err(|e| e.to_string())?;
693                if *writeback {
694                    let n = writeback_ops(&repo, &wal, &ws_impl, name, dir, &actor)?;
695                    println!("writeback: {n} op(s)");
696                }
697                println!("hydrated {name} → {}", dir.display());
698                Ok(())
699            }
700        },
701        Cmd::Claim {
702            uri,
703            ws,
704            ttl,
705            force,
706        } => {
707            let all = all_events(&wal, &repo)?;
708            let now = now();
709            let live = claims::active(&all, now);
710            // Overlap within the same ws is fine — one workspace,
711            // one writer.
712            let hits: Vec<_> = claims::conflicts(&live, uri)
713                .into_iter()
714                .filter(|c| c.ws != *ws)
715                .collect();
716            if !hits.is_empty() && !force {
717                for c in &hits {
718                    eprintln!("conflict: {} claimed {}", c.actor, claim_line(c));
719                }
720                return Err("claim denied — --force to override".into());
721            }
722            append(
723                &wal,
724                &repo,
725                ws,
726                &actor,
727                Op::Claim {
728                    glob: uri.clone(),
729                    ttl_secs: *ttl,
730                },
731            )?;
732            let fence = claims::active(&all_events(&wal, &repo)?, now)
733                .into_iter()
734                .find(|c| c.ws == *ws && c.uri == claims::normalize(uri))
735                .map(|c| c.fence)
736                .unwrap_or(0);
737            println!("claimed {uri} (ws {ws}, ttl {ttl}s, fence {fence})");
738            Ok(())
739        }
740        Cmd::Release { uri, ws } => {
741            append(&wal, &repo, ws, &actor, Op::Release { glob: uri.clone() })?;
742            println!("released {uri}");
743            Ok(())
744        }
745        Cmd::Join { ws } => {
746            append(
747                &wal,
748                &repo,
749                ws,
750                &actor,
751                Op::Presence {
752                    status: "joined".into(),
753                },
754            )?;
755            println!("{actor} joined {ws}");
756            Ok(())
757        }
758        Cmd::Leave { ws } => {
759            append(
760                &wal,
761                &repo,
762                ws,
763                &actor,
764                Op::Presence {
765                    status: "left".into(),
766                },
767            )?;
768            println!("{actor} left {ws} — claims swept");
769            Ok(())
770        }
771        Cmd::Verify { ws } => {
772            let signers = cfg
773                .as_ref()
774                .and_then(|c| c.signing.allowed_signers.clone())
775                .map(PathBuf::from)
776                .unwrap_or_else(|| {
777                    std::env::var("HOME")
778                        .map(PathBuf::from)
779                        .unwrap_or_else(|_| ".".into())
780                        .join(".config/sturn/allowed_signers")
781                });
782            if !signers.exists() {
783                return Err(format!(
784                    "no allowed_signers at {} — can't verify",
785                    signers.display()
786                ));
787            }
788            let names = match ws {
789                Some(w) => vec![w.clone()],
790                None => wal.workspaces(&repo).map_err(|e| e.to_string())?,
791            };
792            let (mut ok, mut nosig, mut bad) = (0u32, 0u32, 0u32);
793            for n in names {
794                for ev in wal.events(&repo, &n).map_err(|e| e.to_string())? {
795                    let Some(sig) = &ev.sig else {
796                        nosig += 1;
797                        continue;
798                    };
799                    let dir = std::env::temp_dir();
800                    let data = dir.join(format!("sturn-vfy-{}", std::process::id()));
801                    let sigf = dir.join(format!("sturn-vfy-{}.sig", std::process::id()));
802                    std::fs::write(&data, ev.signing_bytes()).map_err(|e| e.to_string())?;
803                    std::fs::write(&sigf, sig).map_err(|e| e.to_string())?;
804                    let st = std::process::Command::new("ssh-keygen")
805                        .args(["-Y", "verify", "-f"])
806                        .arg(&signers)
807                        .args(["-I", &ev.actor, "-n", "sturn-event", "-s"])
808                        .arg(&sigf)
809                        .stdin(std::process::Stdio::from(
810                            std::fs::File::open(&data).map_err(|e| e.to_string())?,
811                        ))
812                        .stdout(std::process::Stdio::null())
813                        .status();
814                    let _ = std::fs::remove_file(&data);
815                    let _ = std::fs::remove_file(&sigf);
816                    match st.map(|s| s.success()) {
817                        Ok(true) => ok += 1,
818                        _ => {
819                            bad += 1;
820                            eprintln!("bad sig: {} ({})", ev.id, ev.actor);
821                        }
822                    }
823                }
824            }
825            println!("verify: {ok} ok, {nosig} unsigned, {bad} bad");
826            if bad > 0 {
827                Err("bad signatures found".into())
828            } else {
829                Ok(())
830            }
831        }
832        Cmd::Claims => {
833            let all = all_events(&wal, &repo)?;
834            for c in claims::active(&all, now()) {
835                println!("{}", claim_line(&c));
836            }
837            Ok(())
838        }
839        Cmd::Check { uri, fence, watch } => {
840            let hits = check_hits(&wal, &repo, uri, *fence)?;
841            // Watch streams set changes — initial state is the
842            // first batch of `+` lines, same shape as `diff --live`.
843            for c in &hits {
844                println!("{}{}", if *watch { "+ " } else { "" }, claim_line(c));
845            }
846            if hits.is_empty() {
847                println!("no conflicts: {uri}");
848            }
849            if *watch {
850                return check_watch(&wal, &repo, uri, *fence, hits, poll_ms);
851            }
852            if hits.is_empty() {
853                Ok(())
854            } else {
855                Err(format!("{} conflicting claim(s)", hits.len()))
856            }
857        }
858        Cmd::Ask {
859            subject,
860            ws,
861            target,
862            options,
863            ttl,
864        } => {
865            let id = append(
866                &wal,
867                &repo,
868                ws,
869                &actor,
870                Op::Ask {
871                    subject: subject.clone(),
872                    target: target.clone(),
873                    options: options.clone(),
874                    ttl_secs: *ttl,
875                },
876            )?;
877            let exp = if *ttl == 0 {
878                "never".to_string()
879            } else {
880                format!("{}s", ttl)
881            };
882            println!("ask {id} open (ws {ws} → {target}, ttl {exp})");
883            Ok(())
884        }
885        Cmd::Asks { all, json } => {
886            let entries = sturn_ask::all(&all_events(&wal, &repo)?, now())
887                .into_iter()
888                .filter(|(_, s)| *all || matches!(s, sturn_ask::Status::Open))
889                .collect::<Vec<_>>();
890            if *json {
891                let arr: Vec<serde_json::Value> =
892                    entries.iter().map(|(o, s)| ask_json(o, s)).collect();
893                println!(
894                    "{}",
895                    serde_json::to_string(&arr).map_err(|e| e.to_string())?
896                );
897            } else {
898                for (o, s) in &entries {
899                    let exp = o
900                        .expires
901                        .map(|e| format!("expires {e}"))
902                        .unwrap_or_else(|| "no expiry".into());
903                    println!(
904                        "{} \"{}\" (ws {}, {} → {}, {exp}, {})",
905                        o.id,
906                        o.subject,
907                        o.ws,
908                        o.actor,
909                        o.target,
910                        status_label(s)
911                    );
912                }
913            }
914            Ok(())
915        }
916        Cmd::Answer { ask, verdict, ws } => {
917            match sturn_ask::status(&all_events(&wal, &repo)?, now(), ask) {
918                None => return Err(format!("no such ask: {ask}")),
919                Some(sturn_ask::Status::Open) => {}
920                Some(s) => return Err(format!("ask {ask} is {} — can't answer", status_label(&s))),
921            }
922            append(
923                &wal,
924                &repo,
925                ws,
926                &actor,
927                Op::Answer {
928                    ask: ask.clone(),
929                    verdict: verdict.clone(),
930                },
931            )?;
932            println!("{verdict} → ask {ask}");
933            Ok(())
934        }
935        Cmd::Cancel { ask, ws } => {
936            match sturn_ask::status(&all_events(&wal, &repo)?, now(), ask) {
937                None => return Err(format!("no such ask: {ask}")),
938                Some(sturn_ask::Status::Open) => {}
939                Some(s) => return Err(format!("ask {ask} is {} — can't cancel", status_label(&s))),
940            }
941            append(&wal, &repo, ws, &actor, Op::Cancel { ask: ask.clone() })?;
942            println!("cancelled ask {ask}");
943            Ok(())
944        }
945        Cmd::Wait { ask, timeout, poll } => match wait_ask(&wal, &repo, ask, *timeout, *poll)? {
946            None => {
947                println!("{}", serde_json::json!({"ask": ask, "status": "timeout"}));
948                std::process::exit(4);
949            }
950            Some(sturn_ask::Status::Answered { verdict, by }) => {
951                println!(
952                    "{}",
953                    serde_json::json!({"ask": ask, "status": "answered", "verdict": verdict, "by": by})
954                );
955                Ok(())
956            }
957            Some(s) => {
958                println!(
959                    "{}",
960                    serde_json::json!({"ask": ask, "status": status_label(&s)})
961                );
962                std::process::exit(wait_exit_code(&s));
963            }
964        },
965        Cmd::Write {
966            path,
967            ws,
968            file,
969            content,
970            intent,
971        } => {
972            let bytes = match (file, content) {
973                (Some(f), _) => std::fs::read(f).map_err(|e| e.to_string())?,
974                (None, Some(c)) => c.as_bytes().to_vec(),
975                (None, None) => {
976                    let mut b = Vec::new();
977                    std::io::stdin()
978                        .read_to_end(&mut b)
979                        .map_err(|e| e.to_string())?;
980                    b
981                }
982            };
983            let blob = sturn_git::store_blob(&repo, &bytes).map_err(|e| e.to_string())?;
984            let id = append(
985                &wal,
986                &repo,
987                ws,
988                &actor,
989                Op::Write {
990                    path: path.clone(),
991                    blob: blob.clone(),
992                    intent: intent.clone(),
993                },
994            )?;
995            println!("write {path} (blob {blob}, event {id})");
996            Ok(())
997        }
998        Cmd::Rename { from, to, ws } => {
999            append(
1000                &wal,
1001                &repo,
1002                ws,
1003                &actor,
1004                Op::Rename {
1005                    from: from.clone(),
1006                    to: to.clone(),
1007                },
1008            )?;
1009            println!("rename {from} → {to}");
1010            Ok(())
1011        }
1012        Cmd::Delete { path, ws } => {
1013            append(&wal, &repo, ws, &actor, Op::Delete { path: path.clone() })?;
1014            println!("delete {path}");
1015            Ok(())
1016        }
1017        Cmd::Intent { text, ws } => {
1018            append(&wal, &repo, ws, &actor, Op::Intent { text: text.clone() })?;
1019            Ok(())
1020        }
1021        Cmd::Compact { ws } => {
1022            let names = match ws {
1023                Some(w) => vec![w.clone()],
1024                None => wal.workspaces(&repo).map_err(|e| e.to_string())?,
1025            };
1026            for n in names {
1027                let dropped = wal.compact(&repo, &n).map_err(|e| e.to_string())?;
1028                println!("{n}: dropped {dropped} pre-snapshot events");
1029            }
1030            Ok(())
1031        }
1032        Cmd::Log {
1033            ws,
1034            topic: topic_name,
1035            limit,
1036            wait_ms,
1037        } => {
1038            let filtered = topic_name.is_some() || limit.is_some() || wait_ms.is_some();
1039            if !filtered {
1040                let names = match ws {
1041                    Some(w) => vec![w.clone()],
1042                    None => wal.workspaces(&repo).map_err(|e| e.to_string())?,
1043                };
1044                for n in names {
1045                    for ev in wal.events(&repo, &n).map_err(|e| e.to_string())? {
1046                        println!("{}", serde_json::to_string(&ev).map_err(|e| e.to_string())?);
1047                    }
1048                }
1049                return Ok(());
1050            }
1051            // Filtered form: collect the scope, optionally long-poll
1052            // for a new event, then topic-filter and tail-limit —
1053            // the daemon-free `subs` pull the MCP surface had.
1054            let names = match ws {
1055                Some(w) => vec![w.clone()],
1056                None => wal.workspaces(&repo).map_err(|e| e.to_string())?,
1057            };
1058            let scope_events = || -> Result<Vec<Event>, String> {
1059                let mut all = Vec::new();
1060                for n in &names {
1061                    all.extend(wal.events(&repo, n).map_err(|e| e.to_string())?);
1062                }
1063                Ok(all)
1064            };
1065            if let Some(ms) = wait_ms {
1066                if *ms > 0 {
1067                    let base: std::collections::HashSet<String> =
1068                        scope_events()?.into_iter().map(|e| e.id).collect();
1069                    let deadline = std::time::Instant::now() + Duration::from_millis(*ms);
1070                    while std::time::Instant::now() < deadline {
1071                        let fresh = scope_events()?.iter().any(|e| !base.contains(&e.id));
1072                        if fresh {
1073                            break;
1074                        }
1075                        std::thread::sleep(Duration::from_millis(50));
1076                    }
1077                }
1078            }
1079            let t = topic_name.as_deref();
1080            if let Some(t) = t {
1081                if !topic::ALL.contains(&t) {
1082                    return Err(format!(
1083                        "unknown topic `{t}` — ops|claims|review|presence|asks"
1084                    ));
1085                }
1086            }
1087            let mut events = scope_events()?;
1088            if let Some(t) = t {
1089                events.retain(|e| topic_keep(t, e));
1090            }
1091            if let Some(n) = limit {
1092                let start = events.len().saturating_sub(*n);
1093                events = events.split_off(start);
1094            }
1095            for ev in &events {
1096                println!("{}", serde_json::to_string(ev).map_err(|e| e.to_string())?);
1097            }
1098            Ok(())
1099        }
1100        Cmd::Diff { ws, json, live } => {
1101            // The initial batch is one tick with an empty cursor
1102            // map — it also seeds cursors for `--live`.
1103            let mut heads: HashMap<String, Option<String>> = HashMap::new();
1104            diff_tick(
1105                &repo,
1106                &wal,
1107                &ws_impl,
1108                ws.as_deref(),
1109                &mut heads,
1110                &mut |ev| println!("{}", diff_line(&repo, &ws_impl, ev, *json)),
1111            );
1112            if *live {
1113                return diff_live(&repo, &wal, &ws_impl, ws.as_deref(), heads, *json, poll_ms);
1114            }
1115            Ok(())
1116        }
1117        Cmd::Snap {
1118            ws,
1119            message,
1120            branch: b,
1121        } => {
1122            let require = cfg.as_ref().map(|c| c.review.require).unwrap_or(false);
1123            let br = b.as_deref().unwrap_or(&branch);
1124            if ws.len() == 1 {
1125                let commit = do_snap(&repo, &wal, &ws_impl, &ws[0], message, br, &actor, require)
1126                    .map_err(|e| e.to_string())?;
1127                println!("snap {commit} → {br}");
1128            } else {
1129                let commit = merge_snap(&repo, &wal, &ws_impl, ws, message, br, &actor, require)?;
1130                println!("snap {commit} → {br} ({} ws merged)", ws.len());
1131            }
1132            Ok(())
1133        }
1134        Cmd::Bridge { to, secret_env } => {
1135            let secret = std::env::var(secret_env).map_err(|_| format!("${secret_env} not set"))?;
1136            run_bridge(
1137                &repo, &wal, &ws_impl, to, &secret, &repo_id, &branch, poll_ms,
1138            )
1139        }
1140        Cmd::Exec {
1141            ws,
1142            keep,
1143            writeback,
1144            cmd,
1145        } => {
1146            let dir = std::env::temp_dir().join(format!("sturn-exec-{}", std::process::id()));
1147            hydrate_ws(&repo, &wal, &ws_impl, ws, &dir, &[])?;
1148            let status = std::process::Command::new(&cmd[0])
1149                .args(&cmd[1..])
1150                .current_dir(&dir)
1151                .status()
1152                .map_err(|e| e.to_string())?;
1153            if *writeback {
1154                let n = writeback_ops(&repo, &wal, &ws_impl, ws, &dir, &actor)?;
1155                eprintln!("writeback: {n} op(s)");
1156            }
1157            if !*keep {
1158                let _ = std::fs::remove_dir_all(&dir);
1159            } else {
1160                eprintln!("kept {}", dir.display());
1161            }
1162            if !status.success() {
1163                std::process::exit(status.code().unwrap_or(1));
1164            }
1165            Ok(())
1166        }
1167        Cmd::Review { event, verdict, ws } => {
1168            let v = match verdict.as_str() {
1169                "approve" => Verdict::Approve,
1170                "reject" => Verdict::Reject,
1171                _ => return Err("verdict: approve|reject".into()),
1172            };
1173            append(
1174                &wal,
1175                &repo,
1176                ws,
1177                &actor,
1178                Op::Review {
1179                    event: event.clone(),
1180                    verdict: v,
1181                },
1182            )?;
1183            println!("{verdict} {event}");
1184            Ok(())
1185        }
1186        Cmd::Push { remote } => {
1187            if sturn_git::push(&repo, remote) {
1188                println!("pushed heads + refs/sturn/* → {remote}");
1189                Ok(())
1190            } else {
1191                Err("push failed".into())
1192            }
1193        }
1194        // The poll loop is infinite by design — the driver is
1195        // excluded under coverage; `daemon_tick` (one iteration),
1196        // `handle_sub` and `fanout` carry the logic and are
1197        // unit-tested directly.
1198        #[cfg(not(coverage))]
1199        Cmd::Daemon => {
1200            daemon(&repo, &cfg, &wal, &bus_path, poll_ms, &repo_id).map_err(|e| e.to_string())
1201        }
1202        #[cfg(coverage)]
1203        Cmd::Daemon => Ok(()),
1204        Cmd::Sub { topic } => {
1205            let sock_path = bus_path.with_file_name("bus.sock");
1206            subscribe(&sock_path, &repo_id, topic)
1207        }
1208        Cmd::Mount { ws, dir, rw } => {
1209            let tip = ws_impl.tip(&repo, ws).map_err(|e| e.to_string())?;
1210            if *rw {
1211                sturn_fuse::SturnFs::new_rw(&repo, &tip, &ws_impl, &wal, ws, &actor)
1212                    .and_then(|fs| fs.mount(dir))
1213                    .map_err(|e| e.to_string())
1214            } else {
1215                sturn_fuse::SturnFs::new(&repo, &tip, &ws_impl)
1216                    .and_then(|fs| fs.mount(dir))
1217                    .map_err(|e| e.to_string())
1218            }
1219        }
1220        Cmd::Deploy { cmd } => match cmd {
1221            DeployCmd::Service => {
1222                let svc = sturn_systemd::SystemdService {
1223                    unit: "sturn-daemon.service".into(),
1224                    bin: std::env::current_exe().map_err(|e| e.to_string())?,
1225                    env: std::env::var("HOME")
1226                        .map(PathBuf::from)
1227                        .unwrap_or_else(|_| ".".into())
1228                        .join(".config/sturn/env"),
1229                    config: cli.config.clone().map(PathBuf::from),
1230                };
1231                let p = svc.write().map_err(|e| e.to_string())?;
1232                svc.enable().map_err(|e| e.to_string())?;
1233                println!("enabled {}", p.display());
1234                Ok(())
1235            }
1236            DeployCmd::Env => {
1237                let dir = std::env::var("HOME")
1238                    .map(PathBuf::from)
1239                    .unwrap_or_else(|_| ".".into())
1240                    .join(".config/sturn");
1241                let p = sturn_systemd::write_env_skeleton(&dir).map_err(|e| e.to_string())?;
1242                println!("env skeleton: {}", p.display());
1243                Ok(())
1244            }
1245        },
1246        Cmd::Destroy { cmd } => match cmd {
1247            DestroySub::Service => {
1248                let p = sturn_systemd::disable_service("sturn-daemon.service")
1249                    .map_err(|e| e.to_string())?;
1250                println!("removed {}", p.display());
1251                Ok(())
1252            }
1253        },
1254        Cmd::Service { cmd } => {
1255            // Toggle only — `deploy service` writes the unit, and the
1256            // file stays put through enable/disable cycles.
1257            let unit = "sturn-daemon.service";
1258            match cmd {
1259                ServiceCmd::Enable => {
1260                    sturn_systemd::enable_service(unit).map_err(|e| e.to_string())?;
1261                    println!("sturn-daemon enabled — running now and at login");
1262                }
1263                ServiceCmd::Disable => {
1264                    sturn_systemd::stop_service(unit).map_err(|e| e.to_string())?;
1265                    println!("sturn-daemon disabled — stopped, will not restart at login");
1266                }
1267                ServiceCmd::Status => {
1268                    let s = sturn_systemd::unit_status(unit).map_err(|e| e.to_string())?;
1269                    println!("{s}");
1270                }
1271                ServiceCmd::Restart => {
1272                    sturn_systemd::restart_service(unit).map_err(|e| e.to_string())?;
1273                    println!("sturn-daemon restarted");
1274                }
1275                ServiceCmd::Logs { follow } => {
1276                    let code =
1277                        sturn_systemd::service_logs(unit, *follow).map_err(|e| e.to_string())?;
1278                    if code != 0 {
1279                        std::process::exit(code);
1280                    }
1281                }
1282            }
1283            Ok(())
1284        }
1285    }
1286}
1287
1288fn now() -> u64 {
1289    std::time::SystemTime::now()
1290        .duration_since(std::time::UNIX_EPOCH)
1291        .map(|d| d.as_secs())

1300/// Fold a ws's pending ops (since last Snapshot) into a real commit
1301/// on `branch`, move the ws ref, append the Snapshot marker.
1302/// Shared by `sturn snap` and the generated agent surface's snap op.
1303#[allow(clippy::too_many_arguments)]
1304fn do_snap(
1305    repo: &Path,
1306    wal: &dyn Wal,
1307    wsi: &dyn Ws,
1308    ws: &str,
1309    message: &str,
1310    branch: &str,
1311    actor: &str,
1312    require_review: bool,
1313) -> std::io::Result<String> {
1314    let base = wsi.tip(repo, ws)?;
1315    let events = wal.events(repo, ws)?;
1316    let since = sturn_core::since_last_snapshot(&events);
1317    if require_review {
1318        // Approvals are swarm-wide — any ws's WAL may hold them.
1319        let mut approved = std::collections::HashSet::new();
1320        for w in wal.workspaces(repo)? {
1321            for e in wal.events(repo, &w)? {
1322                if let Op::Review {
1323                    event,
1324                    verdict: Verdict::Approve,
1325                } = &e.op
1326                {
1327                    approved.insert(event.clone());
1328                }
1329            }
1330        }
1331        let unreviewed: Vec<&Event> = events[since..]
1332            .iter()
1333            .filter(|e| {
1334                matches!(
1335                    e.op,
1336                    Op::Write { .. } | Op::Rename { .. } | Op::Delete { .. }
1337                )
1338            })
1339            .filter(|e| !approved.contains(&e.id))
1340            .collect();
1341        if !unreviewed.is_empty() {
1342            let ids: Vec<&str> = unreviewed.iter().map(|e| e.id.as_str()).collect();
1343            return Err(std::io::Error::other(format!(
1344                "review gate: {} unreviewed op(s) — {}",
1345                unreviewed.len(),
1346                ids.join(", ")
1347            )));
1348        }
1349    }
1350    let commit = sturn_git::snap_commit(repo, &base, &events[since..], message, branch)?;
1351    wsi.create(repo, ws, &commit)?;
1352    wal.append(
1353        repo,
1354        ws,
1355        &Event::new(
1356            ws,
1357            actor,
1358            Op::Snapshot {
1359                message: message.into(),
1360            },
1361        ),
1362    )?;
1363    Ok(commit)
1364}

1366/// POST a GitHub-shaped push webhook (HMAC-SHA256 signed) — bupha's
1367/// GitHubTrigger accepts it as an ordinary push. Pure std::net.
1368fn post_push(
1369    url: &str,
1370    secret: &str,
1371    repo_id: &str,
1372    branch: &str,
1373    sha: &str,
1374    sender: &str,
1375) -> Result<(), String> {
1376    use hmac::{Hmac, Mac};
1377    use sha2::Sha256;
1378    let body = format!(
1379        "{{\"ref\":\"refs/heads/{branch}\",\"after\":\"{sha}\",\
1380         \"repository\":{{\"full_name\":\"{repo_id}\"}},\"sender\":{{\"login\":\"{sender}\"}}}}"
1381    );
1382    let mut mac = Hmac::<Sha256>::new_from_slice(secret.as_bytes()).map_err(|e| e.to_string())?;
1383    mac.update(body.as_bytes());
1384    let sig = hex::encode(mac.finalize().into_bytes());
1385    // minimal HTTP/1.1 POST over TcpStream
1386    let url = url.trim_end_matches('/');
1387    let host = url.strip_prefix("http://").unwrap_or(url);
1388    let (hostport, path) = host.split_once('/').unwrap_or((host, ""));
1389    let mut s = std::net::TcpStream::connect(hostport).map_err(|e| e.to_string())?;
1390    let req = format!(
1391        "POST /{path} HTTP/1.1\r\nHost: {hostport}\r\nContent-Type: application/json\r\n\
1392         X-GitHub-Event: push\r\nX-Hub-Signature-256: sha256={sig}\r\n\
1393         X-GitHub-Delivery: sturn-{sha}\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
1394        body.len()
1395    );
1396    s.write_all(req.as_bytes()).map_err(|e| e.to_string())?;
1397    let mut resp = String::new();
1398    s.read_to_string(&mut resp).map_err(|e| e.to_string())?;
1399    if resp.starts_with("HTTP/1.1 2") || resp.starts_with("HTTP/1.0 2") {
1400        Ok(())
1401    } else {
1402        Err(format!(
1403            "bridge POST {url}: {}",
1404            resp.lines().next().unwrap_or("?")
1405        ))
1406    }
1407}

1409/// One bridge poll iteration — publish webhook posts for snapshot
1410/// events since each ws's last-seen head. Extracted so tests drive
1411/// it directly; the `loop` around it is coverage-excluded.
1412#[allow(clippy::too_many_arguments)]
1413fn bridge_tick(
1414    repo: &Path,
1415    wal: &dyn Wal,
1416    wsi: &dyn Ws,
1417    heads: &mut HashMap<String, Option<String>>,
1418    to: &str,
1419    secret: &str,
1420    repo_id: &str,
1421    branch: &str,
1422) {
1423    for ws in wal.workspaces(repo).unwrap_or_default() {
1424        let head = wal.head(repo, &ws).unwrap_or(None);
1425        let prev = heads.get(&ws).cloned().unwrap_or(None);
1426        if head == prev {
1427            continue;
1428        }
1429        for ev in wal
1430            .events_since(repo, &ws, prev.as_deref())
1431            .unwrap_or_default()
1432        {
1433            if let Op::Snapshot { .. } = ev.op {
1434                if let Ok(sha) = wsi.tip(repo, &ws) {
1435                    let _ = post_push(to, secret, repo_id, branch, &sha, &ev.actor);
1436                }
1437            }
1438        }
1439        heads.insert(ws, head);
1440    }
1441}

1443/// WAL→topic fan-out loop. Infinite by design — excluded under
1444/// coverage; everything it does per iteration lives in
1445/// `daemon_tick`, `handle_sub` and `fanout`, which are unit-tested.
1446#[cfg(not(coverage))]
1447fn daemon(
1448    repo: &Path,
1449    cfg: &Option<Config>,
1450    wal: &dyn Wal,
1451    bus_path: &Path,
1452    poll_ms: u64,
1453    repo_id: &str,
1454) -> std::io::Result<()> {
1455    let bus = sturn_bus_redb::RedbBus::open(bus_path.to_path_buf())?;
1456    // Push side: unix socket next to the store. Subscribers
1457    // send one line "repo|topic", then get JSONL forever.
1458    let sock_path = bus_path.with_file_name("bus.sock");
1459    if let Some(d) = sock_path.parent() {
1460        let _ = std::fs::create_dir_all(d);
1461    }
1462    let _ = std::fs::remove_file(&sock_path);
1463    let listener = UnixListener::bind(&sock_path)?;
1464    let subs: Subs = Arc::new(Mutex::new(HashMap::new()));
1465    {
1466        let subs = subs.clone();
1467        std::thread::spawn(move || {
1468            for conn in listener.incoming().flatten() {
1469                let subs = subs.clone();
1470                std::thread::spawn(move || handle_sub(conn, subs));
1471            }
1472        });
1473    }
1474    // Multi-repo: this repo + any [bus] repos. Topics key
1475    // per repo id; heads key per (dir, ws).
1476    let mut dirs: Vec<PathBuf> = vec![repo.to_path_buf()];
1477    dirs.extend(
1478        cfg.as_ref()
1479            .map(|c| c.bus.repos.clone())
1480            .unwrap_or_default(),
1481    );
1482    let mut heads: HashMap<(PathBuf, String), Option<String>> = HashMap::new();
1483    println!(
1484        "sturn daemon — fan-out {} repo(s) (poll {poll_ms}ms, sub {})",
1485        dirs.len(),
1486        sock_path.display()
1487    );
1488    loop {
1489        daemon_tick(wal, &dirs, repo, repo_id, &mut heads, &bus, &subs);
1490        std::thread::sleep(Duration::from_millis(poll_ms));
1491    }
1492}

1550/// One WAL→topic fan-out iteration across `dirs`: for each ws whose
1551/// head moved, publish the delta to `ops` plus the op-kind topic and
1552/// push to live socket subscribers.
1553fn daemon_tick(
1554    wal: &dyn Wal,
1555    dirs: &[PathBuf],
1556    repo: &Path,
1557    repo_id: &str,
1558    heads: &mut HashMap<(PathBuf, String), Option<String>>,
1559    bus: &sturn_bus_redb::RedbBus,
1560    subs: &Subs,
1561) {
1562    for dir in dirs {
1563        let rid = if *dir == repo {
1564            repo_id.to_string()
1565        } else {
1566            dir.file_name()
1567                .map(|s| s.to_string_lossy().into_owned())
1568                .unwrap_or_else(|| "repo".into())
1569        };
1570        for ws in wal.workspaces(dir).unwrap_or_default() {
1571            let head = wal.head(dir, &ws).unwrap_or(None);
1572            let key = (dir.clone(), ws.clone());
1573            let prev = heads.get(&key).cloned().unwrap_or(None);
1574            if head != prev {
1575                // Only the delta — refs moved, walk the range.
1576                for ev in wal
1577                    .events_since(dir, &ws, prev.as_deref())
1578                    .unwrap_or_default()
1579                {
1580                    let payload = Arc::new(serde_json::to_vec(&ev).unwrap_or_default());
1581                    let _ = bus.publish(&rid, topic::OPS, &payload);
1582                    fanout(subs, &rid, topic::OPS, &payload);
1583                    let t = match &ev.op {
1584                        Op::Claim { .. } | Op::Release { .. } => Some(topic::CLAIMS),
1585                        Op::Review { .. } => Some(topic::REVIEW),
1586                        Op::Presence { .. } => Some(topic::PRESENCE),
1587                        Op::Ask { .. } | Op::Answer { .. } | Op::Cancel { .. } => Some(topic::ASKS),
1588                        _ => None,
1589                    };
1590                    if let Some(t) = t {
1591                        let _ = bus.publish(&rid, t, &payload);
1592                        fanout(subs, &rid, t, &payload);
1593                    }
1594                }
1595                heads.insert(key, head);
1596            }
1597        }
1598    }
1599}

1601/// One `diff` pass over `scope` (`None` = ws namespaces ∪ WAL
1602/// refs): for each workspace whose WAL head moved since `heads`,
1603/// emit the new events. An empty cursor means the ws log replays
1604/// whole — sliced to post-snapshot so both a fresh `diff` and a ws
1605/// born mid-stream show only in-flight ops. Errors degrade to
1606/// silence, matching `daemon_tick` — the WAL stays the authority.
1607fn diff_tick(
1608    repo: &Path,
1609    wal: &dyn Wal,
1610    wsi: &dyn Ws,
1611    scope: Option<&str>,
1612    heads: &mut HashMap<String, Option<String>>,
1613    emit: &mut dyn FnMut(&Event),
1614) {
1615    let mut names: Vec<String> = match scope {
1616        Some(w) => vec![w.to_string()],
1617        None => {
1618            let mut n = wsi.workspaces(repo).unwrap_or_default();
1619            n.extend(wal.workspaces(repo).unwrap_or_default());
1620            n
1621        }
1622    };
1623    names.sort();
1624    names.dedup();
1625    for ws in names {
1626        let head = wal.head(repo, &ws).unwrap_or(None);
1627        let prev = heads.get(&ws).cloned().unwrap_or(None);
1628        if head == prev {
1629            continue;
1630        }
1631        let delta = wal
1632            .events_since(repo, &ws, prev.as_deref())
1633            .unwrap_or_default();
1634        for ev in &delta[sturn_core::since_last_snapshot(&delta)..] {
1635            emit(ev);
1636        }
1637        heads.insert(ws, head);
1638    }
1639}

1694/// Compact one-line rendering for every op kind — path(s) first;
1695/// Write carries its content hint (byte size via the blob, else the
1696/// blob id itself) and its intent when one was logged.
1697fn op_summary(repo: &Path, wsi: &dyn Ws, op: &Op) -> String {
1698    match op {
1699        Op::Write { path, blob, intent } => {
1700            let hint = wsi
1701                .blob(repo, blob)
1702                .map(|b| format!("({} B)", b.len()))
1703                .unwrap_or_else(|_| format!("(blob {blob})"));
1704            let why = intent
1705                .as_deref()
1706                .map(|i| format!(" \"{i}\""))
1707                .unwrap_or_default();
1708            format!("write {path} {hint}{why}")
1709        }
1710        Op::Rename { from, to } => format!("rename {from} → {to}"),
1711        Op::Delete { path } => format!("delete {path}"),
1712        Op::Claim { glob, ttl_secs } => format!("claim {glob} ttl={ttl_secs}s"),
1713        Op::Release { glob } => format!("release {glob}"),
1714        Op::Intent { text } => format!("intent \"{text}\""),
1715        Op::Snapshot { message } => format!("snapshot \"{message}\""),
1716        Op::Review { event, verdict } => format!(
1717            "review {} {}",
1718            match verdict {
1719                Verdict::Approve => "approve",
1720                Verdict::Reject => "reject",
1721            },
1722            short_id(event)
1723        ),
1724        Op::Presence { status } => format!("presence {status}"),
1725        Op::Ask {
1726            subject,
1727            target,
1728            ttl_secs,
1729            ..
1730        } => format!("ask \"{subject}\" → {target} ttl={ttl_secs}s"),
1731        Op::Answer { ask, verdict } => {
1732            format!("answer {verdict} → ask {}", short_id(ask))
1733        }
1734        Op::Cancel { ask } => format!("cancel ask {}", short_id(ask)),
1735    }
1736}

1820/// Hydrate a ws into `dir`: tip commit via `git archive` (+ optional
1821/// path filter), then the WAL overlay since the last snapshot —
1822/// hydrated dirs see uncommitted ops, matching `ws read`.
1823fn hydrate_ws(
1824    repo: &Path,
1825    wal: &dyn Wal,
1826    wsi: &dyn Ws,
1827    name: &str,
1828    dir: &Path,
1829    paths: &[String],
1830) -> Result<(), String> {
1831    let tip = wsi.tip(repo, name).map_err(|e| e.to_string())?;
1832    wsi.materialize(repo, &tip, dir, paths)
1833        .map_err(|e| e.to_string())?;
1834    let events = wal.events(repo, name).map_err(|e| e.to_string())?;
1835    let pending = &events[sturn_core::since_last_snapshot(&events)..];
1836    for (p, v) in sturn_core::overlay(pending, |p| wsi.entry(repo, &tip, p).ok().flatten()) {
1837        let f = dir.join(&p);
1838        match v {
1839            Some(sha) => {
1840                if let Some(d) = f.parent() {
1841                    std::fs::create_dir_all(d).map_err(|e| e.to_string())?;
1842                }
1843                std::fs::write(&f, wsi.blob(repo, &sha).map_err(|e| e.to_string())?)
1844                    .map_err(|e| e.to_string())?;
1845            }
1846            None => {
1847                let _ = std::fs::remove_file(&f);
1848            }
1849        }
1850    }
1851    Ok(())
1852}

1907/// Poll `asks` until `ask` reaches a terminal state or `timeout`
1908/// seconds pass (0 = wait until resolved). `Ok(Some)` is the
1909/// terminal state, `Ok(None)` the deadline hit, `Err` means the
1910/// ask id doesn't exist.
1911fn wait_ask(
1912    wal: &dyn Wal,
1913    repo: &Path,
1914    ask: &str,
1915    timeout: u64,
1916    poll: u64,
1917) -> Result<Option<sturn_ask::Status>, String> {
1918    let deadline = (timeout > 0).then(|| now() + timeout);
1919    loop {
1920        let s = sturn_ask::status(&all_events(wal, repo)?, now(), ask)
1921            .ok_or_else(|| format!("no such ask: {ask}"))?;
1922        if !matches!(s, sturn_ask::Status::Open) {
1923            return Ok(Some(s));
1924        }
1925        if deadline.is_some_and(|d| now() >= d) {
1926            return Ok(None);
1927        }
1928        std::thread::sleep(Duration::from_millis(poll));
1929    }
1930}

1951/// Live claims conflicting with `uri`. `fence` narrows to
1952/// usurpers — only claims whose fencing token beats it — which
1953/// turns `check` into the staleness probe for a held claim.
1954fn check_hits(
1955    wal: &dyn Wal,
1956    repo: &Path,
1957    uri: &str,
1958    fence: Option<u64>,
1959) -> Result<Vec<claims::Active>, String> {
1960    let live = claims::active(&all_events(wal, repo)?, now());
1961    Ok(claims::conflicts(&live, uri)
1962        .into_iter()
1963        .filter(|c| fence.is_none_or(|f| c.fence > f))
1964        .cloned()
1965        .collect())
1966}

1968/// Diff two conflict snapshots into stream lines: `+` for claims
1969/// that appeared, `-` for ones that left, `no conflicts` when the
1970/// set drains. The rendered line is the identity — a re-claim at
1971/// a new fence reads as `-`/`+`, which is the truthful story.
1972fn watch_tick(prev: &[claims::Active], hits: &[claims::Active], uri: &str) -> Vec<String> {
1973    let had: Vec<String> = prev.iter().map(claim_line).collect();
1974    let have: Vec<String> = hits.iter().map(claim_line).collect();
1975    let mut out = Vec::new();
1976    for l in &have {
1977        if !had.contains(l) {
1978            out.push(format!("+ {l}"));
1979        }
1980    }
1981    for l in &had {
1982        if !have.contains(l) {
1983            out.push(format!("- {l}"));
1984        }
1985    }
1986    if have.is_empty() && !had.is_empty() {
1987        out.push(format!("no conflicts: {uri}"));
1988    }
1989    out
1990}

2029/// Live subscribers, keyed `"<repo>|<topic>"` — the daemon's push
2030/// side. Dead senders get pruned on the next failed delivery.
2031type Subs = Arc<Mutex<HashMap<String, Vec<mpsc::Sender<Arc<Vec<u8>>>>>>>;

2033/// First line on the socket names the topic; then JSONL payloads
2034/// stream until the client hangs up.
2035fn handle_sub(mut conn: UnixStream, subs: Subs) {
2036    let mut key = String::new();
2037    // try_clone only fails on a dead fd — defensive.
2038    let mut rdr = BufReader::new(match conn.try_clone() {
2039        Ok(c) => c,
2040        Err(_) => return,
2041    });
2042    if rdr.read_line(&mut key).is_err() || key.trim().is_empty() {
2043        return;
2044    }
2045    let (tx, rx) = mpsc::channel::<Arc<Vec<u8>>>();
2046    subs.lock()
2047        .unwrap()
2048        .entry(key.trim().to_string())
2049        .or_default()
2050        .push(tx);
2051    for msg in rx {
2052        if conn
2053            .write_all(&msg)
2054            .and_then(|_| conn.write_all(b"\n"))
2055            .is_err()
2056        {
2057            break;
2058        }
2059    }
2060}

2071/// Walk `dir`, diff against the ws view (tip + pending overlay),
2072/// append Write/Delete ops for the difference. Returns ops written.
2073fn writeback_ops(
2074    repo: &Path,
2075    wal: &dyn Wal,
2076    wsi: &dyn Ws,
2077    ws: &str,
2078    dir: &Path,
2079    actor: &str,
2080) -> Result<usize, String> {
2081    let tip = wsi.tip(repo, ws).map_err(|e| e.to_string())?;
2082    let events = wal.events(repo, ws).map_err(|e| e.to_string())?;
2083    let pending = &events[sturn_core::since_last_snapshot(&events)..];
2084    let ov = sturn_core::overlay(pending, |p| wsi.entry(repo, &tip, p).ok().flatten());
2085    // expected path → content bytes (overlay wins, else tip)
2086    let mut expected: HashMap<String, Vec<u8>> = HashMap::new();
2087    for p in wsi.list(repo, &tip).map_err(|e| e.to_string())? {
2088        if let Ok(b) = wsi.read(repo, &tip, &p) {
2089            expected.insert(p, b);
2090        }
2091    }
2092    for (p, v) in &ov {
2093        match v {
2094            Some(sha) => {
2095                if let Ok(b) = wsi.blob(repo, sha) {
2096                    expected.insert(p.clone(), b);
2097                }
2098            }
2099            None => {
2100                expected.remove(p);
2101            }
2102        }
2103    }
2104    // walk the hydrated dir
2105    let mut seen = std::collections::HashSet::new();
2106    let mut stack = vec![dir.to_path_buf()];
2107    let mut n = 0;
2108    while let Some(d) = stack.pop() {
2109        for ent in std::fs::read_dir(&d).map_err(|e| e.to_string())? {
2110            let ent = ent.map_err(|e| e.to_string())?;
2111            let ft = ent.file_type().map_err(|e| e.to_string())?;
2112            if ft.is_dir() {
2113                stack.push(ent.path());
2114                continue;
2115            }
2116            if !ft.is_file() {
2117                continue;
2118            }
2119            let rel = ent
2120                .path()
2121                .strip_prefix(dir)
2122                .map_err(|e| e.to_string())?
2123                .to_string_lossy()
2124                .into_owned();
2125            seen.insert(rel.clone());
2126            let bytes = std::fs::read(ent.path()).map_err(|e| e.to_string())?;
2127            let differs = expected.get(&rel).map(|e| *e != bytes).unwrap_or(true);
2128            if differs {
2129                let blob = wsi.store(repo, &bytes).map_err(|e| e.to_string())?;
2130                wal.append(
2131                    repo,
2132                    ws,
2133                    &Event::new(
2134                        ws,
2135                        actor,
2136                        Op::Write {
2137                            path: rel,
2138                            blob,
2139                            intent: Some("writeback".into()),
2140                        },
2141                    ),
2142                )
2143                .map_err(|e| e.to_string())?;
2144                n += 1;
2145            }
2146        }
2147    }
2148    for p in expected.keys() {
2149        if !seen.contains(p) {
2150            wal.append(
2151                repo,
2152                ws,
2153                &Event::new(ws, actor, Op::Delete { path: p.clone() }),
2154            )
2155            .map_err(|e| e.to_string())?;
2156            n += 1;
2157        }
2158    }
2159    Ok(n)
2160}

2162/// Merge several ws's pending ops into ONE commit on `branch`.
2163/// `sturn_core::merge` folds them rename-aware — a write onto a
2164/// path another ws renamed away follows the rename; ambiguous
2165/// shapes warn on stderr instead of picking a silent winner.
2166/// Later ops still win on true same-path collisions — claims
2167/// should have prevented that upstream.
2168#[allow(clippy::too_many_arguments)]
2169fn merge_snap(
2170    repo: &Path,
2171    wal: &dyn Wal,
2172    wsi: &dyn Ws,
2173    wss: &[String],
2174    message: &str,
2175    branch: &str,
2176    actor: &str,
2177    require_review: bool,
2178) -> Result<String, String> {
2179    let mut pending: Vec<Event> = Vec::new();
2180    for ws in wss {
2181        let events = wal.events(repo, ws).map_err(|e| e.to_string())?;
2182        let since = sturn_core::since_last_snapshot(&events);
2183        pending.extend(events[since..].iter().cloned());
2184    }
2185    if require_review {
2186        let mut approved = std::collections::HashSet::new();
2187        for w in wal.workspaces(repo).map_err(|e| e.to_string())? {
2188            for e in wal.events(repo, &w).map_err(|e| e.to_string())? {
2189                if let Op::Review {
2190                    event,
2191                    verdict: Verdict::Approve,
2192                } = &e.op
2193                {
2194                    approved.insert(event.clone());
2195                }
2196            }
2197        }
2198        let unreviewed: Vec<&Event> = pending
2199            .iter()
2200            .filter(|e| {
2201                matches!(
2202                    e.op,
2203                    Op::Write { .. } | Op::Rename { .. } | Op::Delete { .. }
2204                )
2205            })
2206            .filter(|e| !approved.contains(&e.id))
2207            .collect();
2208        if !unreviewed.is_empty() {
2209            return Err(format!(
2210                "review gate: {} unreviewed op(s)",
2211                unreviewed.len()
2212            ));
2213        }
2214    }
2215    // Base = current branch head, not any ws tip — merged ops apply
2216    // to where the branch is now.
2217    let base =
2218        sturn_git::resolve(repo, &format!("refs/heads/{branch}")).unwrap_or_else(|| "HEAD".into());
2219    let merged = sturn_core::merge(&pending);
2220    for w in &merged.warnings {
2221        eprintln!("{w}");
2222    }
2223    let commit = sturn_git::snap_commit(repo, &base, &merged.events, message, branch)
2224        .map_err(|e| e.to_string())?;
2225    for ws in wss {
2226        wsi.create(repo, ws, &commit).map_err(|e| e.to_string())?;
2227        let _ = wal.append(
2228            repo,
2229            ws,
2230            &Event::new(
2231                ws,
2232                actor,
2233                Op::Snapshot {
2234                    message: message.into(),
2235                },
2236            ),
2237        );
2238    }
2239    Ok(commit)
2240}
