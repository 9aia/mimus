1//! data — one read of a repo's sturn state per wire op.
2//!
3//! The driver is a *reader*: every `view`/`query` resolves the
4//! target repo, folds every `refs/sturn/wal/*` log through the same
5//! pure functions the `sturn` CLI uses (`sturn_claim::active`,
6//! `sturn_ask::all`), and drops the result. No mutations — `sturn
7//! claim`/`write`/`snap` stay the write surface; this crate only
8//! needs what `sturn claims`/`sturn log`/`sturn asks` already show.
9//!
10//! Repo resolution order (first hit wins) — mirrors `sturn --repo`
11//! plus driver conveniences:
12//!
13//! 1. Request params — `dir`, `root`, or `repo` (all name the repo
14//!    dir), e.g. `pavo view sturn:claims --param dir=/path/to/repo`.
15//! 2. `$STURN_REPO` — driver-level convenience; the `sturn` CLI
16//!    itself has no repo env var.
17//! 3. `sturn.toml`'s `repo.dir` — config search identical to the
18//!    CLI: `--config`-equivalent param `config` → `$STURN_CONFIG` →
19//!    `./sturn.toml` → `~/.config/sturn/sturn.toml`. A relative
20//!    `repo.dir` resolves against the process cwd, same as the CLI.
21//! 4. `.` — the driver's cwd (pavo inherits the user's cwd, so
22//!    `pavo view sturn:claims` inside a wired repo reads it).
23//!
24//! When nothing resolves to a git repo, [`Snapshot::error`] carries
25//! the reason and every query degrades to an `{"error": …}` payload
26//! — the wire stays `ok:true` so hosts render the failure inline.
27

39/// One workspace row — union of `refs/sturn/ws/*` namespaces and
40/// `refs/sturn/wal/*` logs (same union `sturn ws ls` prints).
41#[derive(Debug, Clone)]
42pub struct WsRow {
43    /// Workspace name.
44    pub name: String,
45    /// `namespace` = `refs/sturn/ws/<name>` exists,
46    /// `wal` = `refs/sturn/wal/<name>` exists; both when both do.
47    pub kinds: Vec<&'static str>,
48    /// WAL events in the log (0 for namespace-only ws).
49    pub events: usize,
50    /// Events since the last `Snapshot` marker — ops still in
51    /// flight, not yet folded into a commit.
52    pub pending: usize,
53    /// WAL tip short sha, when the log exists.
54    pub tip: Option<String>,
55}
56
57/// Everything the views need, loaded once per op.
58pub struct Snapshot {
59    /// Resolved repo dir (may not be a git repo).
60    pub repo: PathBuf,
61    /// Repo id — `repo.id` from config, else dir basename.
62    pub repo_id: String,
63    /// Where `sturn.toml` came from (`<path>`), None = flag/env only.
64    pub config: Option<String>,
65    /// WAL ref prefix in effect (default `refs/sturn/wal`).
66    pub wal_prefix: String,
67    /// Union of ws namespaces + WAL logs.
68    pub workspaces: Vec<WsRow>,
69    /// Every event across every WAL, fold order per ws.
70    pub events: Vec<Event>,
71    /// Live claims at `now` — `sturn claims` output, structured.
72    pub claims: Vec<claims::Active>,
73    /// Every ask and its terminal state — `sturn asks --all`.
74    pub asks: Vec<(sturn_ask::Open, AskStatus)>,
75    /// Last `Presence` status per `(ws, actor)` — `joined` actors
76    /// are the live swarm; `left`/absent are gone.
77    pub presence: BTreeMap<(String, String), String>,
78    /// Unix seconds the snapshot was taken at.
79    pub now: u64,
80    /// Per-source load problems — surfaced, never fatal.
81    pub warnings: Vec<String>,
82    /// Why there is no usable repo state at all.
83    pub error: Option<String>,
84}
85
86impl Default for Snapshot {
87    fn default() -> Self {
88        Snapshot {
89            repo: PathBuf::from("."),
90            repo_id: String::new(),
91            config: None,
92            wal_prefix: "refs/sturn/wal".into(),
93            workspaces: Vec::new(),
94            events: Vec::new(),
95            claims: Vec::new(),
96            asks: Vec::new(),
97            presence: BTreeMap::new(),
98            now: 0,
99            warnings: Vec::new(),
100            error: None,
101        }
102    }
103}

105impl Snapshot {
106    /// The empty/degraded snapshot — `error` says why.
107    fn failed(error: String) -> Self {
108        Snapshot {
109            error: Some(error),
110            ..Snapshot::default()
111        }
112    }
113
114    /// The `{"error": …}` payload every query emits when the
115    /// snapshot couldn't load — the degradation marker pavo
116    /// renderers already understand.
117    pub fn error_payload(&self) -> serde_json::Value {
118        serde_json::json!({"error": self.error.clone().unwrap_or_else(|| "no sturn repo".into())})
119    }
120
121    /// Is there usable repo state?
122    pub fn ok(&self) -> bool {
123        self.error.is_none()
124    }
125}

127fn now() -> u64 {
128    std::time::SystemTime::now()
129        .duration_since(std::time::UNIX_EPOCH)
130        .map(|d| d.as_secs())
131        .unwrap_or(0)
132}

134/// `params` → `(repo_dir, config_path)` overrides. `dir`/`root`/
135/// `repo` all name the repo; `dir` wins when several are given
136/// (mirrors the CLI globals' precedence).
137fn overrides(
138    params: &serde_json::Map<String, serde_json::Value>,
139) -> (Option<PathBuf>, Option<String>) {
140    let repo = ["dir", "root", "repo"]
141        .iter()
142        .find_map(|k| params.get(*k).and_then(|v| v.as_str()))
143        .map(PathBuf::from);
144    let config = params
145        .get("config")
146        .and_then(|v| v.as_str())
147        .map(str::to_string);
148    (repo, config)
149}

151/// Which config file the source chain would read — same candidate
152/// order as `TomlSource`, reported for the status view.
153fn config_path_hint(explicit: Option<&str>) -> Option<String> {
154    let candidates: Vec<PathBuf> = match explicit
155        .map(str::to_string)
156        .or_else(|| std::env::var("STURN_CONFIG").ok().filter(|s| !s.is_empty()))
157    {
158        Some(p) => vec![PathBuf::from(p)],
159        None => vec![
160            PathBuf::from("sturn.toml"),
161            std::env::var("HOME")
162                .map(PathBuf::from)
163                .unwrap_or_else(|_| ".".into())
164                .join(".config/sturn/sturn.toml"),
165        ],
166    };
167    candidates
168        .into_iter()
169        .find(|p| p.exists())
170        .map(|p| p.display().to_string())
171}

173/// Load config via the same source chain the CLI uses. Failure is
174/// a warning, not fatal — sturn works flag-driven with no toml.
175fn load_config(path: Option<String>) -> (Option<Config>, Vec<String>) {
176    let mut warnings = Vec::new();
177    match (sturn_config_toml::TomlSource { path }).load() {
178        Ok(cfg) => (Some(cfg), warnings),
179        Err(e) => {
180            // "no sturn.toml found" is the normal case outside a
181            // wired repo — only surface real parse/read errors.
182            if !e.contains("no sturn.toml found") {
183                warnings.push(format!("config: {e}"));
184            }
185            (None, warnings)
186        }
187    }
188}

190/// Where a `cli.<verb>` re-exec should run — the repo dir plus the
191/// config file the same chain `load` uses would read, so engine ops
192/// land exactly where the views see state. Both come back absolute:
193/// the child's cwd is the repo dir, so a relative candidate like
194/// `sturn.toml` would re-resolve against the wrong root.
195pub fn resolve(params: &serde_json::Map<String, serde_json::Value>) -> (PathBuf, Option<String>) {
196    let (param_repo, param_config) = overrides(params);
197    let (cfg, _) = load_config(param_config.clone());
198    let repo = param_repo
199        .or_else(|| {
200            std::env::var("STURN_REPO")
201                .ok()
202                .filter(|s| !s.is_empty())
203                .map(PathBuf::from)
204        })
205        .or_else(|| cfg.as_ref().map(|c| c.repo.dir.clone()))
206        .unwrap_or_else(|| PathBuf::from("."));
207    let repo = std::fs::canonicalize(&repo).unwrap_or(repo);
208    let config = config_path_hint(param_config.as_deref())
209        .map(|p| {
210            std::fs::canonicalize(&p)
211                .map(|a| a.to_string_lossy().into_owned())
212                .unwrap_or(p)
213        })
214        .or(param_config);
215    (repo, config)
216}

218/// Resolve + load. `params` are the wire op's caller params —
219/// `dir`/`root`/`repo`/`config` steer resolution, the rest pass
220/// through to the query resolvers untouched.
221pub fn load(params: &serde_json::Map<String, serde_json::Value>) -> Snapshot {
222    let (param_repo, param_config) = overrides(params);
223    let config = config_path_hint(param_config.as_deref());
224    let (cfg, warnings) = load_config(param_config);
225
226    // Repo resolution — params → $STURN_REPO → cfg.repo.dir → cwd.
227    let repo = param_repo
228        .or_else(|| {
229            std::env::var("STURN_REPO")
230                .ok()
231                .filter(|s| !s.is_empty())
232                .map(PathBuf::from)
233        })
234        .or_else(|| cfg.as_ref().map(|c| c.repo.dir.clone()))
235        .unwrap_or_else(|| PathBuf::from("."));
236
237    // Not a git repo → honest degradation, no WAL reads attempted.
238    if sturn_odb::git_dir(&repo).is_err() {
239        return Snapshot::failed(format!(
240            "{}: not a git repo — sturn state lives in refs/sturn/* under a repo",
241            repo.display()
242        ));
243    }
244
245    let repo_id = cfg
246        .as_ref()
247        .map(|c| c.repo.id())
248        .or_else(|| {
249            repo.canonicalize()
250                .ok()
251                .and_then(|p| p.file_name().map(|s| s.to_string_lossy().into_owned()))
252        })
253        .unwrap_or_else(|| "repo".into());
254    let wal_prefix = cfg
255        .as_ref()
256        .map(|c| c.wal.ref_prefix.clone())
257        .unwrap_or_else(|| "refs/sturn/wal".into());
258
259    let wal = GitWal {
260        ref_prefix: wal_prefix.clone(),
261        // Read-only driver — never signs.
262        sign_key: None,
263    };
264    let ws_impl = sturn_ws_git::GitWs;
265    let now = now();
266    let mut snap = Snapshot {
267        repo,
268        repo_id,
269        config,
270        wal_prefix,
271        now,
272        warnings,
273        ..Snapshot::default()
274    };
275
276    // Workspace union — namespaces (`refs/sturn/ws/*`) plus WAL
277    // logs (`refs/sturn/wal/*`), same as `sturn ws ls`.
278    let mut names: Vec<String> = ws_impl
279        .workspaces(&snap.repo)
280        .map_err(|e| e.to_string())
281        .unwrap_or_else(|e| {
282            snap.warnings.push(format!("ws namespaces: {e}"));
283            Vec::new()
284        });
285    names.extend(wal.workspaces(&snap.repo).unwrap_or_else(|e| {
286        snap.warnings.push(format!("wal workspaces: {e}"));
287        Vec::new()
288    }));
289    names.sort();
290    names.dedup();
291
292    let mut events_by_ws: BTreeMap<String, Vec<Event>> = BTreeMap::new();
293    for name in &names {
294        let mut row = WsRow {
295            name: name.clone(),
296            kinds: Vec::new(),
297            events: 0,
298            pending: 0,
299            tip: None,
300        };
301        if ws_impl.tip(&snap.repo, name).is_ok() {
302            row.kinds.push("namespace");
303        }
304        // `wal.events` answers `Ok(vec![])` for a missing log — the
305        // head check is what distinguishes "wal-only/absent".
306        let head = wal.head(&snap.repo, name).unwrap_or(None);
307        if head.is_some() {
308            row.kinds.push("wal");
309            row.tip = head.map(|h| if h.len() > 8 { h[..8].to_string() } else { h });
310            match wal.events(&snap.repo, name) {
311                Ok(evs) => {
312                    row.events = evs.len();
313                    row.pending = evs
314                        .len()
315                        .saturating_sub(sturn_core::since_last_snapshot(&evs));
316                    events_by_ws.insert(name.clone(), evs);
317                }
318                Err(e) => snap.warnings.push(format!("wal {name}: {e}")),
319            }
320        }
321        snap.workspaces.push(row);
322    }
323
324    // Fold order: per-ws WAL order, concatenated — the same order
325    // `all_events` in the CLI feeds `claims::active`.
326    for evs in events_by_ws.values() {
327        snap.events.extend(evs.iter().cloned());
328    }
329
330    snap.claims = claims::active(&snap.events, now);
331    snap.asks = sturn_ask::all(&snap.events, now);
332    for ev in &snap.events {
333        if let Op::Presence { status } = &ev.op {
334            snap.presence
335                .insert((ev.ws.clone(), ev.actor.clone()), status.clone());
336        }
337    }
338    snap
339}

341/// The five bus topic suffixes under `repo.<id>.` — derived names,
342/// not stored state (the bus is a cache; topics are vocabulary).
343pub fn topic_rows(repo_id: &str) -> Vec<serde_json::Value> {
344    let descs = [
345        (topic::OPS, "every op append — the firehose"),
346        (topic::CLAIMS, "claim/release ops only — collision warnings"),
347        (topic::REVIEW, "approve/reject verdicts on events"),
348        (topic::PRESENCE, "actors joining/leaving workspaces"),
349        (topic::ASKS, "pending-response ops — asks, answers, cancels"),
350    ];
351    descs
352        .iter()
353        .map(|(suffix, desc)| {
354            serde_json::json!({
355                "topic": format!("repo.{repo_id}.{suffix}"),
356                "suffix": suffix,
357                "description": desc,
358            })
359        })
360        .collect()
361}
