1//! Engine ops — `cli.<verb>` wire queries dispatch into the `sturn`
2//! engine lib through this binary's own `run` verb: each call
3//! re-execs `tamias-driver-sturn run <argv>` with the request's repo
4//! as cwd, so `sturn::dispatch` writes to the child's captured
5//! stdout and never touches the serve loop's ndjson stream — and
6//! every mutation keeps the engine's own write-path semantics (WAL
7//! append, actor from config/`$STURN_ACTOR`, claim fencing).
8//!
9//! `ops` answers the ops manifest as JSON — *derived from the
10//! engine's own clap tree* ([`sturn::command`]), so a verb can't
11//! exist on the CLI and be missing here: leaf subcommands flatten
12//! into dotted names (`ws new` → `cli.ws.new`), declared args
13//! become typed params (`arg`/`opt`/`flag`), and doc strings ride
14//! along unchanged.
15//!
16//! Unbounded verbs (`daemon`, `sub`, `bridge`, `mount`) are
17//! deliberately *not* ops — they never return, so they live in
18//! `surfaces` instead: surface drivers (`tamias-driver-cli`/`-mcp`,
19//! the pavo-driver-kit `surface` layer) stream them via `run`.
20//! Streaming *flags* (`diff --live`, `check --watch`,
21//! `service logs -f`) are likewise excluded — the bounded form of
22//! those verbs stays a wire op.
23
24use clap::{Arg, ArgAction, Command};
25use pavo_protocol::wire::{QueryResult, Response};
26use serde_json::{json, Map, Value};
27use std::path::Path;
28
29/// Unbounded surfaces — verbs that never return, so they can
30/// never be request/response ops.
31const SURFACES: &[&str] = &["daemon", "sub", "bridge", "mount"];
32
33/// Params excluded from ops — flags that turn a bounded verb into
34/// a stream. The bounded form rides the wire; the stream rides
35/// `run` passthrough.
36const SKIP_PARAMS: &[&str] = &["live", "watch", "follow"];
37
38/// Read-only verbs — everything else appends WAL ops, writes
39/// files, pushes refs, or touches system state. Mutators belong
40/// behind a deliberate trigger, not ambient view polling.
41const READ_ONLY: &[&str] = &[
42    "log", "diff", "claims", "check", "asks", "wait", "verify", "ws.ls", "ws.read",
43];
44
45/// `(dotted name, clap spec)` for every leaf verb — groups (`ws`,
46/// `deploy`, `destroy`, `service`) aren't invocable themselves.
47fn verbs() -> Vec<(String, Command)> {
48    fn walk(prefix: &str, cmd: &Command, out: &mut Vec<(String, Command)>) {
49        for sub in cmd.get_subcommands() {
50            let name = sub.get_name();
51            if name == "help" {
52                continue;
53            }
54            let dotted = match prefix {
55                "" => name.to_string(),
56                _ => format!("{prefix}.{name}"),
57            };
58            if sub.has_subcommands() {
59                walk(&dotted, sub, out);
60            } else {
61                out.push((dotted, sub.clone()));
62            }
63        }
64    }
65    let mut out = Vec::new();
66    walk("", &sturn::command(), &mut out);
67    out
68}
69
70fn op_cmd(name: &str) -> Option<Command> {
71    verbs()
72        .into_iter()
73        .find(|(n, _)| n == name && !SURFACES.contains(&n.as_str()))
74        .map(|(_, c)| c)
75}
76
77/// Args that can't ride the wire: streaming flags (above) plus
78/// clap's own help/version actions.
79fn skipped(a: &Arg) -> bool {
80    SKIP_PARAMS.contains(&a.get_id().as_str())
81        || matches!(
82            a.get_action(),
83            ArgAction::Help | ArgAction::HelpShort | ArgAction::HelpLong | ArgAction::Version
84        )
85}
86
87/// A clap arg → the manifest's typed param kind — `arg` =
88/// positional, `opt` = `--name <v>`, `flag` = `--name` — the
89/// contract `pavo-driver-kit`'s surface layer renders args from.
90fn kind(a: &Arg) -> &'static str {
91    if a.is_positional() {
92        "arg"
93    } else if matches!(
94        a.get_action(),
95        ArgAction::SetTrue | ArgAction::SetFalse | ArgAction::Count
96    ) {
97        "flag"
98    } else {
99        "opt"
100    }
101}
102
103fn param_json(a: &Arg) -> Value {
104    // Wire name = the flag name (`--option`), not the field id
105    // (`options`) — generated surfaces spell params `--<name>`.
106    let id = a.get_id().as_str();
107    let name = match a.is_positional() {
108        true => id,
109        false => a.get_long().unwrap_or(id),
110    };
111    json!({
112        "name": name,
113        "kind": kind(a),
114        "doc": a.get_help().map(|d| d.to_string()).unwrap_or_default(),
115    })
116}

118/// The `ops` query/`ops` argv verb payload — self-describing so
119/// hosts and agents discover the engine surface without argv help.
120/// `dir`/`root`/`repo`/`config`/`actor` are the driver's globals —
121/// accepted by every op, listed once on the manifest itself.
122pub fn manifest() -> Value {
123    json!({
124        "engine": "sturn",
125        "invoke": "query `cli.<verb>` with the listed params, or argv `tamias-driver-sturn run <sturn argv>`",
126        "globals": [
127            {"name": "dir|root|repo", "kind": "opt", "doc": "Target repo dir — same resolution the view queries use (param → $STURN_REPO → sturn.toml repo.dir → cwd)."},
128            {"name": "config", "kind": "opt", "doc": "Config path — the driver's own chain resolves it and hands the child an absolute --config."},
129            {"name": "actor", "kind": "env", "doc": "Claim/verb actor — sets $STURN_ACTOR in the child (config actor.name still wins, same as the CLI)."},
130        ],
131        "ops": verbs().iter()
132            .filter(|(n, _)| !SURFACES.contains(&n.as_str()))
133            .map(|(n, c)| json!({
134                "query": format!("cli.{n}"),
135                "doc": c.get_about().map(|d| d.to_string()).unwrap_or_default(),
136                "params": c.get_arguments()
137                    .filter(|a| !skipped(a))
138                    .map(param_json)
139                    .collect::<Vec<_>>(),
140                "mutates": !READ_ONLY.contains(&n.as_str()),
141            }))
142            .collect::<Vec<_>>(),
143        "surfaces": SURFACES,
144    })
145}

147/// Is `query` an engine op (`cli.<verb>`)? Returns the verb name.
148pub fn op_of(query: &str) -> Option<String> {
149    query
150        .strip_prefix("cli.")
151        .filter(|n| op_cmd(n).is_some())
152        .map(str::to_string)
153}

155/// A wire scalar → its argv token (numbers/bools stringify).
156fn scalar(v: &Value) -> Option<String> {
157    match v {
158        Value::String(s) => Some(s.clone()),
159        Value::Number(n) => Some(n.to_string()),
160        Value::Bool(b) => Some(b.to_string()),
161        _ => None,
162    }
163}
164
165/// All values of a param — arrays flatten, scalars wrap (repeatable
166/// opts like `--option`, multi-positionals like `snap <ws>…`).
167fn values(v: &Value) -> Vec<String> {
168    match v {
169        Value::Array(xs) => xs.iter().filter_map(scalar).collect(),
170        other => scalar(other).into_iter().collect(),
171    }
172}

174/// `cli.<verb>` params → sturn argv (no argv[0] — `run` prepends
175/// it). Globals are resolved by [`run_verb`] (`--repo`/`--config`
176/// prepend here); the dotted verb name splits into the group path
177/// (`ws.new` → `ws new`). Declared params map by kind — positionals
178/// land last, in schema order.
179fn argv(
180    dir: &Path,
181    config: Option<&str>,
182    name: &str,
183    cmd: &Command,
184    params: &Map<String, Value>,
185) -> Vec<String> {
186    let mut out: Vec<String> = vec!["--repo".into(), dir.display().to_string()];
187    if let Some(c) = config {
188        out.extend(["--config".into(), c.into()]);
189    }
190    out.extend(name.split('.').map(String::from));
191    for a in cmd.get_arguments() {
192        if skipped(a) || a.is_positional() {
193            continue;
194        }
195        let id = a.get_id().as_str();
196        let long = a.get_long().unwrap_or(id);
197        let raw = params
198            .get(long)
199            .or_else(|| params.get(id))
200            .or_else(|| params.get(id.replace('_', "-").as_str()));
201        let Some(raw) = raw else { continue };
202        match kind(a) {
203            "flag" => {
204                if raw.as_bool().unwrap_or(false) {
205                    out.push(format!("--{long}"));
206                }
207            }
208            _ => {
209                for s in values(raw) {
210                    out.push(format!("--{long}"));
211                    out.push(s);
212                }
213            }
214        }
215    }
216    let mut positionals: Vec<String> = Vec::new();
217    for a in cmd.get_positionals() {
218        if skipped(a) {
219            continue;
220        }
221        let id = a.get_id().as_str();
222        if let Some(raw) = params
223            .get(id)
224            .or_else(|| params.get(id.replace('_', "-").as_str()))
225        {
226            positionals.extend(values(raw));
227        }
228    }
229    out.extend(positionals);
230    out
231}

233/// The bare-name entry the embedded surfaces use (`OpSpec.name` is
234/// `cli.`-stripped); the wire path goes `op_of` → `run_verb`.
235#[cfg(any(feature = "cli", feature = "mcp"))]
236pub fn run_op(op: &str, params: &Map<String, Value>) -> Response {
237    match op_of(&format!("cli.{}", op.strip_prefix("cli.").unwrap_or(op))) {
238        Some(name) => run_verb(&name, params),
239        None => Response::err(format!("unknown op: {op}")),
240    }
241}
242
243/// Run one engine op by name: re-exec `run` in a child whose cwd is
244/// the request's repo (same resolution the views use), and hand
245/// back `{op, argv, code, stdout, stderr}` as the query payload.
246/// `ok:false` is reserved for spawn failures — a nonzero `code` is
247/// the engine's own exit status (e.g. `check` contested), honest
248/// data. `params.actor` sets `STURN_ACTOR` for the child — claims
249/// stay actor-scoped, no bypass around the engine's own rules.
250pub fn run_verb(name: &str, params: &Map<String, Value>) -> Response {
251    let Some(cmd) = op_cmd(name) else {
252        return Response::err(format!("unknown op: {name}"));
253    };
254    let (dir, config) = crate::data::resolve(params);
255    let args = argv(&dir, config.as_deref(), name, &cmd, params);
256    let out = std::env::current_exe().and_then(|exe| {
257        let mut c = std::process::Command::new(exe);
258        c.arg("run").args(&args).current_dir(&dir);
259        if let Some(actor) = params.get("actor").and_then(Value::as_str) {
260            c.env("STURN_ACTOR", actor);
261        }
262        c.output()
263    });
264    let data = match out {
265        Ok(o) => json!({
266            "op": name,
267            "argv": args,
268            "code": o.status.code().unwrap_or(-1),
269            "stdout": String::from_utf8_lossy(&o.stdout),
270            "stderr": String::from_utf8_lossy(&o.stderr),
271        }),
272        Err(e) => return Response::err(format!("run {name}: {e}")),
273    };
274    Response::ok_typed(&QueryResult {
275        query: format!("cli.{name}"),
276        data,
277    })
278}

288    #[test]
289    fn op_of_accepts_cli_prefix_only() {
290        assert_eq!(op_of("cli.claims").unwrap(), "claims");
291        assert_eq!(op_of("cli.ws.new").unwrap(), "ws.new");
292        assert_eq!(op_of("cli.service.logs").unwrap(), "service.logs");
293        assert!(op_of("cli.daemon").is_none()); // surface, not an op
294        assert!(op_of("cli.mcp").is_none()); // the verb is gone — ops→MCP tools replaced it
295        assert!(op_of("claims").is_none()); // view query, not an op
296        assert!(op_of("cli.bogus").is_none());
297        assert!(op_of("cli.ws").is_none()); // group, not a leaf
298    }
299
300    #[test]
301    fn manifest_lists_every_leaf_as_cli_query() {
302        let m = manifest();
303        let ops = m["ops"].as_array().unwrap();
304        for (n, _) in verbs()
305            .iter()
306            .filter(|(n, _)| !SURFACES.contains(&n.as_str()))
307        {
308            assert!(
309                ops.iter().any(|o| o["query"] == json!(format!("cli.{n}"))),
310                "{n} missing from ops manifest"
311            );
312        }
313        assert_eq!(m["surfaces"], json!(SURFACES));
314        for o in ops {
315            for p in o["params"].as_array().unwrap() {
316                assert!(["arg", "opt", "flag"].contains(&p["kind"].as_str().unwrap()));
317                // streaming flags never ride the wire
318                assert!(!SKIP_PARAMS.contains(&p["name"].as_str().unwrap()));
319            }
320        }
321    }

323    #[test]
324    fn argv_maps_params_to_sturn_flags() {
325        let dir = PathBuf::from("/repo");
326        let cmd = op_cmd("claim").unwrap();
327        let mut p = Map::new();
328        p.insert("uri".into(), json!("fs:src/**"));
329        p.insert("ws".into(), json!("t"));
330        p.insert("ttl".into(), json!(600));
331        p.insert("force".into(), json!(true));
332        assert_eq!(
333            argv(&dir, None, "claim", &cmd, &p),
334            vec![
335                "--repo",
336                "/repo",
337                "claim",
338                "--ws",
339                "t",
340                "--ttl",
341                "600",
342                "--force",
343                "fs:src/**",
344            ]
345        );
346    }

348    #[test]
349    fn argv_splits_dotted_names_and_repeatables() {
350        let dir = PathBuf::from("/repo");
351        let cmd = op_cmd("ask").unwrap();
352        let mut p = Map::new();
353        p.insert("subject".into(), json!("ship it?"));
354        p.insert("ws".into(), json!("t"));

356        assert_eq!(

358            vec![
359                "--repo",
360                "/repo",
361                "--config",
362                "/cfg/sturn.toml",
363                "ask",
364                "--ws",
365                "t",
366                "--option",
367                "yes",
368                "--option",
369                "no",
370                "ship it?",
371            ]
372        );
373        let cmd = op_cmd("ws.new").unwrap();
374        let mut p = Map::new();
375        p.insert("name".into(), json!("w1"));
376        p.insert("base".into(), json!("HEAD~1"));
377        assert_eq!(
378            argv(&dir, None, "ws.new", &cmd, &p),
379            vec!["--repo", "/repo", "ws", "new", "--base", "HEAD~1", "w1"]
