1//! tamias-driver-sturn — the pavo view driver for a repo's sturn
2//! coordination state.
3//!
4//! Verbs (the tamias driver contract):
5//!
6//! - `manifest` — print the [`DriverManifest`] JSON (discovery
7//!   probe; never touches repo state).
8//! - `capabilities` — `{"capabilities": ["pavo"], "verbs": […]}`.
9//! - `ops` — the engine ops manifest: every `sturn` CLI verb as a
10//!   `cli.<verb>` wire query, derived from the engine's own clap
11//!   tree ([`sturn::command`]).
12//! - `run <argv>` — engine passthrough: `tamias-driver-sturn run
13//!   claims` ≡ `sturn claims`. Also the child side of `cli.*` wire
14//!   queries (`ops::run_verb` re-execs this same verb so engine
15//!   output never touches the serve stream).
16//! - `serve` — ndjson JSON-RPC on stdio: `{"op":"manifest"}`,
17//!   `{"op":"views"}`, `{"op":"view","view":…,"params":{…}}`,
18//!   `{"op":"query","query":…,"params":{…}}` →
19//!   `{"ok":true,"result":…}` / `{"ok":false,"error":…}`.
20//!   `query:"ops"` answers the ops manifest; `query:"cli.<verb>"`
21//!   runs the engine op.
22//!
23//! Optional surfaces: features `cli`/`mcp` embed
24//! pavo-driver-kit's surface layer — `cli <op> [args]` is a
25//! generated CLI over the ops manifest (`daemon`/`sub`/`bridge`/
26//! `mount` ride as streaming subcommands), `mcp` serves the
27//! ops as MCP tools on stdio. Both are equivalent to the standalone
28//! `tamias-driver-cli`/`tamias-driver-mcp` surface drivers, minus
29//! the PATH/discovery hop.
30//!
31//! Which repo: each `view`/`query` resolves its own target —
32//! `params.dir`/`params.root`/`params.repo` → `$STURN_REPO` →
33//! `sturn.toml`'s `repo.dir` (same config search as the `sturn`
34//! CLI) → cwd. See `data.rs` for the exact order. Engine ops are
35//! the exception by design: `cli.<verb>` re-execs `run` in the
36//! request's repo and lets `sturn::dispatch` own the outcome —
37//! writes keep the engine's own semantics (WAL append, actor from
38//! config/`$STURN_ACTOR`, claim fencing). `manifest`/`views` are
39//! repo-free so pavo can enumerate drivers anywhere.
40//!
41//! The view side is **read-only**: it folds `refs/sturn/wal/*`
42//! exactly like `sturn claims`/`sturn log` do — the WAL is the
43//! state, the tree is a projection, and views only ever read.

54fn capabilities_json() -> serde_json::Value {
55    let m = views::manifest();
56    serde_json::json!({
57        "capabilities": m.capabilities,
58        "verbs": m.verbs,
59        "views": m.views.len(),
60    })
61}
62
63/// Dispatch one parsed request to its response.
64fn dispatch(req: &Request) -> Response {
65    match req {
66        Request::Manifest => Response::ok_typed(&views::manifest()),
67        Request::Views => {
68            let summaries: Vec<ViewSummary> =
69                views::specs().iter().map(ViewSummary::from).collect();
70            Response::ok_typed(&summaries)
71        }
72        Request::View { view, params } => match views::view_result(view, params) {
73            Ok(result) => Response::ok_typed(&result),
74            Err(e) => Response::err(e),
75        },
76        Request::Query { query, params } => {
77            // `ops` and `cli.<verb>` are engine ops; everything
78            // else is a view query — unknown names fail the op,
79            // degraded snapshots travel as `{"error": …}` data
80            // (the marker renderers understand).
81            if query == "ops" {
82                Response::ok_typed(&wire::QueryResult {
83                    query: query.clone(),
84                    data: ops::manifest(),
85                })
86            } else if let Some(v) = ops::op_of(query) {
87                ops::run_verb(&v, params)
88            } else if views::known_query(query) {
89                let snap = data::load(params);
90                let payload = views::query_data(query, params, &snap);
91                Response::ok_typed(&wire::QueryResult {
92                    query: query.clone(),
93                    data: payload,
94                })
95            } else {
96                Response::err(format!("unknown query: {query}"))
97            }
98        }
99        // `render` — this driver ships no renderer plugins; hosts
100        // only call it for kinds the manifest claims, so answering
101        // an explicit error keeps the wire honest.
102        Request::Render { kind, .. } => Response::err(format!(
103            "tamias-driver-sturn claims no renderers (kind '{kind}')"
104        )),
105        // The protocol grew `submit`/`jobs`/`notifications` ops after
106        // this driver was written; mutations go through `cli.<verb>`,
107        // never `submit`, so they answer `ok:false` like any
108        // unsupported op.
109        Request::Submit { .. } | Request::Jobs | Request::Notifications => {
110            Response::err("tamias-driver-sturn implements no submit/jobs/notifications ops")
111        }
112    }
113}

115fn serve() -> i32 {
116    // The kit owns the ndjson loop — stdin EOF and a dead client
117    // both end it quietly, and either way the driver exits 0.
118    let _ = kit::serve_stdio(dispatch);
119    0
120}
121
122fn main() {
123    let args: Vec<String> = std::env::args().skip(1).collect();
124    let code = match args.first().map(String::as_str) {
125        Some("manifest") => {
126            println!("{}", kit::manifest_json(&views::manifest()));
127            0
128        }
129        Some("capabilities") => {
130            println!(
131                "{}",
132                serde_json::to_string_pretty(&capabilities_json()).unwrap_or_default()
133            );
134            0
135        }
136        Some("ops") => {
137            println!("{}", ops::manifest());
138            0
139        }
140        // Engine passthrough — `tamias-driver-sturn run claims` is
141        // `sturn claims`. Also the child side of `cli.*` wire
142        // queries (ops::run_verb re-execs this same verb).
143        Some("run") => {
144            let argv = std::iter::once("sturn".to_string()).chain(args.iter().skip(1).cloned());
145            std::process::exit(sturn::run_from(argv));
146        }
147        // Embedded surface verbs (feature-gated, optional): the
148        // pavo-driver-kit surface layer renders this driver's own
149        // ops manifest — `cli` = generated clap CLI (op →
150        // subcommand), `mcp` = ops → MCP tools over stdio. The
151        // standalone tamias-driver-cli/-mcp binaries do the same
152        // over process dispatch; these verbs need nothing on PATH.
153        #[cfg(feature = "cli")]
154        Some("cli") => {
155            let m = kit::surface::OpsManifest::from_value(&ops::manifest()).unwrap_or_default();
156            let argv: Vec<String> = std::iter::once("sturn".to_string())
157                .chain(args.iter().skip(1).cloned())
158                .collect();
159            kit::surface::cli::run("sturn", &m, &argv, &mut surface::Engine)
160        }
161        #[cfg(feature = "mcp")]
162        Some("mcp") => {
163            let m = kit::surface::OpsManifest::from_value(&ops::manifest()).unwrap_or_default();
164            match kit::surface::mcp::serve_stdio(&m, &mut surface::Engine) {
165                Ok(()) => 0,
166                Err(e) => {
167                    eprintln!("tamias-driver-sturn mcp: {e}");
168                    1
169                }
170            }
171        }
172        Some("serve") | None => serve(),
173        _ => {
174            eprintln!(
175                "usage: tamias-driver-sturn manifest|capabilities|ops|run <argv>|serve\n       (feature `cli` adds `cli <op> [args]`; feature `mcp` adds `mcp`)"
176            );
177            1
178        }
179    };
180    std::process::exit(code);
181}
