2//! tamias-tui — the interactive package-manager surface behind
3//! `tamias ui` (ratatui/crossterm, PL-0004). Phases 1–3: the Installed
4//! pane lists store entries with a `/` filter, `Enter` opens a detail
5//! view, and `v`/`u`/`p`/`r`/`c`/`d`/`i` run the same library calls
6//! `tamias-serve`/the CLI use on worker threads (confirm + consent
7//! modals, streamed phase log); the Registry pane searches a registry
8//! as-you-type and `i` installs a result. Services/Manifest/Author
9//! stay placeholders until their phases land. Terminal
10//! discipline mirrors `myco-tui`: alternate screen, raw mode, a
11//! drop guard, a panic hook, mouse capture, and a `notify` watcher
12//! on the store root that reloads the entry list when installs land
13//! from another `tamias` invocation.
14//!
15//! The crate stays engine-free: it reads the store through
16//! `tamias-store-local` directly and takes its roots from [`Ctx`] —
17//! `tamias ui` builds one from the CLI env; the standalone
18//! `tamias-tui` bin reads `TAMIAS_HOME`/`TAMIAS_CACHE`/`TAMIAS_REGISTRY`.
19//! Later phases call the shared ops (`tamias-install`,
20//! `tamias-registry-open`, `tamias-serve`) on worker threads — the
21//! Ctx fields are already the serve `Env` shape so nothing has to
22//! be re-plumbed.

40/// Where the TUI reads/writes — the same roots `tamias-serve`'s
41/// `Env` carries, so CLI, serve, and TUI resolve identically.
42/// `config_file: None` honors `TAMIAS_CONFIG` then `./tamias.*`
43/// discovery (per-op, like serve).
44#[derive(Clone, Debug)]
45pub struct Ctx {
46    /// The local store root (`<TAMIAS_HOME>/store` by default).
47    pub store_root: PathBuf,
48    /// The fetch-cache root.
49    pub cache_root: PathBuf,
50    /// The default local-registry root.
51    pub registry_root: PathBuf,
52    /// Explicit config file for registry-aware ops — `None` honors
53    /// `TAMIAS_CONFIG` then `./tamias.{yml,yaml,toml}` discovery.
54    pub config_file: Option<PathBuf>,
55}
56
57impl Ctx {
58    /// All four roots explicit.
59    #[must_use]
60    pub fn new(
61        store_root: impl Into<PathBuf>,
62        cache_root: impl Into<PathBuf>,
63        registry_root: impl Into<PathBuf>,
64        config_file: Option<PathBuf>,
65    ) -> Self {
66        Self {
67            store_root: store_root.into(),
68            cache_root: cache_root.into(),
69            registry_root: registry_root.into(),
70            config_file,
71        }
72    }
73
74    /// Roots from the `TAMIAS_*` env vars with the same
75    /// `~/.local/share`/`~/.cache` fallbacks the CLI uses — the
76    /// standalone bin's whole config story (no project config layer;
77    /// `tamias ui` is the full-fidelity entry).
78    #[must_use]
79    pub fn from_env() -> Self {
80        let data = || {
81            std::env::var_os("XDG_DATA_HOME")
82                .map(PathBuf::from)
83                .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share")))
84                .unwrap_or_else(|| PathBuf::from(".tamias"))
85        };
86        let cache = || {
87            std::env::var_os("XDG_CACHE_HOME")
88                .map(PathBuf::from)
89                .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".cache")))
90                .unwrap_or_else(|| PathBuf::from(".tamias"))
91        };
92        let store_root = std::env::var_os("TAMIAS_HOME")
93            .map(|h| PathBuf::from(h).join("store"))
94            .or_else(|| Some(data().join("tamias/store")))
95            .unwrap_or_else(|| PathBuf::from(".tamias/store"));
96        let cache_root = std::env::var_os("TAMIAS_CACHE")
97            .map(PathBuf::from)
98            .or_else(|| Some(cache().join("tamias/fetch")))
99            .unwrap_or_else(|| PathBuf::from(".tamias/fetch"));
100        let registry_root = std::env::var_os("TAMIAS_REGISTRY")
101            .map(PathBuf::from)
102            .or_else(|| Some(data().join("tamias/registry")))
103            .unwrap_or_else(|| PathBuf::from(".tamias/registry"));
104        Self::new(store_root, cache_root, registry_root, None)
105    }
106}

108/// Restore the terminal on any exit path.
109struct Guard;
110
111impl Drop for Guard {
112    fn drop(&mut self) {
113        let _ = disable_raw_mode();
114        let _ = execute!(io::stdout(), DisableMouseCapture, LeaveAlternateScreen);
115    }
116}
117
118/// Panic hook — restore the terminal *before* the default hook
119/// prints, otherwise the panic message is erased by the alt-screen
120/// exit and the shell is left in raw mode.
121fn install_panic_hook() {
122    let prev = std::panic::take_hook();
123    std::panic::set_hook(Box::new(move |info| {
124        let _ = disable_raw_mode();
125        let _ = execute!(io::stdout(), DisableMouseCapture, LeaveAlternateScreen);
126        prev(info);
127    }));
128}

130/// Launch the TUI against `ctx`'s roots.
131pub fn run(ctx: &Ctx) -> Result<(), String> {
132    let mut model = app::Model::open(ctx)?;
133    enable_raw_mode().map_err(|e| format!("raw mode: {e}"))?;
134    // Guard first — raw mode is on from here, so every fallible
135    // step below (alt screen, terminal init) must restore on `?`.
136    let _guard = Guard;
137    install_panic_hook();
138    let mut stdout = io::stdout();
139    execute!(stdout, EnterAlternateScreen, EnableMouseCapture)
140        .map_err(|e| format!("alt screen: {e}"))?;
141    let mut terminal =
142        Terminal::new(CrosstermBackend::new(stdout)).map_err(|e| format!("terminal: {e}"))?;
143
144    loop {
145        terminal
146            .draw(|f| render::draw(f, &mut model))
147            .map_err(|e| format!("draw: {e}"))?;
148        if event::poll(Duration::from_millis(250)).map_err(|e| e.to_string())? {
149            match event::read().map_err(|e| e.to_string())? {
150                Event::Key(k) if matches!(k.kind, KeyEventKind::Press | KeyEventKind::Repeat) => {
151                    if model.key(k) {
152                        break;
153                    }
154                }
155                Event::Mouse(m) => model.mouse(m),
156                Event::Resize(..) => {}
157                _ => {}
158            }
159        }
160        // auto-refresh — pick up installs/removes that landed while
161        // we were idle (watcher dirty flag + slow fallback scan)
162        model.tick();
163    }
164    Ok(())
165}

172    /// The hook body itself — a caught panic still runs it, which
173    /// exercises the terminal-restore path (a no-op without a tty).
174    #[test]
175    fn panic_hook_restores_the_terminal() {
176        install_panic_hook();
177        let r = std::panic::catch_unwind(|| panic!("boom"));
178        assert!(r.is_err());
179    }

181    #[test]
182    fn ctx_from_env_resolves_roots() {

184        // the shape for the standalone bin
185        let ctx = Ctx::from_env();
186        assert!(!ctx.store_root.as_os_str().is_empty());
187        assert!(ctx.config_file.is_none());
