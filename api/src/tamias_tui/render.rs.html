1//! Drawing — tab bar on top, the active pane filling the frame, an
2//! optional op-log strip, status bar at the bottom, and modal/prompt
3//! overlays. Chrome bars yield to the pane on short terminals, same
4//! budget rule as myco-tui: tabs need height ≥3, the status bar ≥5,
5//! the log strip ≥12 (and only while the log has lines).

19/// Chrome budget — `(tabs, status)`: tabs cost a row at height ≥3,
20/// the status bar only at height ≥5.
21fn chrome(height: u16) -> (bool, bool) {
22    (height >= 3, height >= 5)
23}
24
25/// One full frame. Also records the tab/pane geometry on `m` so the
26/// next mouse event can hit-test against where rows landed.
27pub fn draw(f: &mut Frame, m: &mut Model) {
28    let area = f.area();
29    let (show_tabs, show_status) = chrome(area.height);
30    let show_log = area.height >= 12 && !m.log.is_empty();
31    let mut constraints = Vec::with_capacity(4);
32    if show_tabs {
33        constraints.push(Constraint::Length(1));
34    }
35    constraints.push(Constraint::Min(0));
36    if show_log {
37        constraints.push(Constraint::Length(4));
38    }
39    if show_status {
40        constraints.push(Constraint::Length(1));
41    }
42    let chunks = Layout::vertical(constraints).split(area);
43
44    let mut next = 0;
45    if show_tabs {
46        draw_tabs(f, m, chunks[next]);
47        m.tabs_area = Some(chunks[next]);
48        next += 1;
49    } else {
50        m.tabs_area = None;
51    }
52    draw_pane(f, m, chunks[next]);
53    next += 1;
54    if show_log {
55        draw_log(f, m, chunks[next]);
56        next += 1;
57    }
58    if show_status {
59        draw_status(f, m, chunks[next]);
60    }
61    match &m.mode {
62        Mode::Help => draw_help(f, area),
63        Mode::Filter | Mode::Search | Mode::Command | Mode::Prompt(_) => draw_input(f, m, area),
64        Mode::Normal => {}
65    }
66    if let Some(modal) = &m.modal {
67        draw_modal(f, modal, area);
68    }
69}

71/// Tab captions — full names when they fit, initials when they
72/// don't (the `1`-`5`/`h`/`l` binds still work).
73fn tab_labels(width: usize) -> Vec<String> {
74    let full: Vec<String> = Pane::ALL
75        .iter()
76        .map(|p| format!(" {} ", p.label()))
77        .collect();
78    let need: usize = full.iter().map(|s| s.chars().count()).sum::<usize>() + full.len() + 1;
79    if need <= width {
80        full
81    } else {
82        Pane::ALL
83            .iter()
84            .map(|p| p.label()[..1].to_string())
85            .collect()
86    }
87}

89fn draw_tabs(f: &mut Frame, m: &Model, area: Rect) {
90    let mut spans = vec![Span::raw(" ")];
91    for (p, label) in Pane::ALL.iter().zip(tab_labels(area.width as usize)) {
92        let style = if *p == m.pane {
93            Style::default().fg(Color::White).bg(ACCENT)
94        } else {
95            Style::default().fg(MUTED)
96        };
97        spans.push(Span::styled(label, style));
98        spans.push(Span::raw(" "));
99    }
100    f.render_widget(Paragraph::new(Line::from(spans)), area);
101}

103/// Click hit-test for the tab row: which tab does column `x` land
104/// on in a `width`-wide bar? Mirrors the span layout `draw_tabs`
105/// emits — leading space + one separator per tab.
106pub fn tab_hit(width: u16, x: u16) -> Option<usize> {
107    let mut col = 1u16; // leading space
108    for (i, label) in tab_labels(width as usize).iter().enumerate() {
109        let w = label.chars().count() as u16;
110        if (col..col + w).contains(&x) {
111            return Some(i);
112        }
113        col += w + 1; // the separator after each tab
114    }
115    None
116}

118fn human_size(bytes: u64) -> String {
119    const UNITS: [&str; 5] = ["B", "KiB", "MiB", "GiB", "TiB"];
120    let mut v = bytes as f64;
121    let mut u = 0;
122    while v >= 1024.0 && u + 1 < UNITS.len() {
123        v /= 1024.0;
124        u += 1;
125    }
126    if u == 0 {
127        format!("{bytes} B")
128    } else {
129        format!("{v:.1} {}", UNITS[u])
130    }
131}

133fn short_sha(s: &str) -> String {
134    let bare = s.strip_prefix("sha256:").unwrap_or(s);
135    bare.chars().take(12).collect()
136}

138/// Trust badge for a store record — `record.trust` carries the
139/// `tamias-sign` wire spelling (`official`/`signed`/`pinned`/
140/// `unpinned`/`bad`); `None` on records that predate trust tiers,
141/// which render no badge.
142fn trust_badge(rec: &tamias_core::StoreRecord) -> String {
143    rec.trust
144        .as_deref()
145        .map_or_else(String::new, |t| format!(" [{t}]"))
146}

148fn draw_pane(f: &mut Frame, m: &mut Model, area: Rect) {
149    match m.pane {
150        Pane::Installed => {
151            if m.detail.is_some() {
152                draw_detail(f, m, area);
153            } else {
154                draw_installed(f, m, area);
155            }
156        }
157        Pane::Registry => draw_registry(f, m, area),
158        Pane::Services => draw_services(f, m, area),
159        Pane::Manifest => draw_manifest(f, m, area),
160        Pane::Author => draw_author(f, m, area),
161    }
162}

164/// The Services pane — every installed provider service with its
165/// unit state. Rows come from [`Job::ServicesRefresh`]; the table
166/// marks stale while a refresh is queued/running via the busy bar.
167fn draw_services(f: &mut Frame, m: &mut Model, area: Rect) {
168    if let Some(err) = &m.svc_err {
169        if m.svc_rows.is_empty() {
170            f.render_widget(
171                Paragraph::new(format!("services: {err}")).style(Style::default().fg(Color::Red)),
172                area,
173            );
174            m.board_area = Rect::default();
175            m.viewport_top = 0;
176            return;
177        }
178    }
179    if m.svc_rows.is_empty() {
180        let hint = if m.busy.is_some() {
181            "loading services…"
182        } else {
183            "no provider services — install a package with a `provider:` block"
184        };
185        f.render_widget(Paragraph::new(hint).style(Style::default().fg(MUTED)), area);
186        m.board_area = Rect::default();
187        m.viewport_top = 0;
188        return;
189    }
190    let header = Row::new(vec!["PACKAGE", "SERVICE", "UNIT", "STATE", "EXEC"])
191        .style(Style::default().fg(ACCENT).add_modifier(Modifier::BOLD));
192    let rows: Vec<Row> = m
193        .svc_rows
194        .iter()
195        .map(|r| {
196            Row::new(vec![
197                r.package.clone(),
198                r.service.clone(),
199                r.unit.clone(),
200                r.state.clone().unwrap_or_else(|| "-".into()),
201                r.exec.clone(),
202            ])
203        })
204        .collect();
205    let widths = [
206        Constraint::Min(14),
207        Constraint::Length(12),
208        Constraint::Min(14),
209        Constraint::Length(10),
210        Constraint::Min(16),
211    ];
212    let title = if m.svc_stale || m.busy.is_some() {
213        " services — refreshing… "
214    } else {
215        " services "
216    };
217    let table = Table::new(rows, widths)
218        .header(header)
219        .block(Block::default().borders(Borders::ALL).title(title))
220        .row_highlight_style(Style::default().bg(Color::Indexed(238)));
221    let mut state = ratatui::widgets::TableState::default().with_selected(Some(m.svc_sel));
222    f.render_stateful_widget(table, area, &mut state);
223    note_table_area(m, area, state.offset());
224}

226/// The Manifest pane — the project manifest's `deps:` as editable
227/// rows; every other field renders read-only above the table. Writes
228/// go through `Manifest::save` + `tamias.lock` sync on the worker.
229fn draw_manifest(f: &mut Frame, m: &mut Model, area: Rect) {
230    // Read-only header — path plus the manifest's scalar fields.
231    let mut info: Vec<Line> = Vec::new();
232    match (&m.project, &m.man_err) {
233        (Some(manifest), _) => {
234            info.push(Line::from(vec![
235                Span::styled("manifest: ", Style::default().fg(ACCENT)),
236                Span::raw(m.manifest_path.display().to_string()),
237            ]));
238            let mut scalars = Vec::new();
239            if !manifest.name.is_empty() {
240                scalars.push(format!("name {}", manifest.name));
241            }
242            if let Some(v) = &manifest.version {
243                scalars.push(format!("v{v}"));
244            }
245            if let Some(d) = &manifest.description {
246                scalars.push(d.clone());
247            }
248            if !scalars.is_empty() {
249                info.push(Line::from(Span::styled(
250                    scalars.join(" · "),
251                    Style::default().fg(MUTED),
252                )));
253            }
254            if !manifest.include.is_empty() {
255                info.push(Line::from(vec![
256                    Span::styled("include:  ", Style::default().fg(ACCENT)),
257                    Span::raw(manifest.include.join(", ")),
258                ]));
259            }
260            if manifest.provider.is_some() {
261                info.push(Line::from(vec![
262                    Span::styled("provider: ", Style::default().fg(ACCENT)),
263                    Span::raw("declared (read-only here)"),
264                ]));
265            }
266        }
267        (None, Some(err)) => {
268            info.push(Line::from(Span::styled(
269                format!("{err} — `:manifest` retries, `a` creates tamias.yaml on save"),
270                Style::default().fg(WARN),
271            )));
272        }
273        (None, None) => {
274            info.push(Line::from(Span::styled(
275                "no project manifest — `a` writes tamias.yaml + tamias.lock",
276                Style::default().fg(MUTED),
277            )));
278        }
279    }
280    // Deps table gets the rest of the frame; the info strip yields on
281    // short terminals.
282    let info_h = if area.height > 14 {
283        5.min(area.height / 3)
284    } else {
285        0
286    };
287    let (info_area, table_area) = if info_h > 0 {
288        let chunks = Layout::vertical([Constraint::Length(info_h), Constraint::Min(0)]).split(area);
289        (chunks[0], chunks[1])
290    } else {
291        (Rect::default(), area)
292    };
293    if info_h > 0 {
294        f.render_widget(Paragraph::new(info), info_area);
295    }
296    let deps: Vec<String> = m
297        .project
298        .as_ref()
299        .map(|p| p.deps.clone())
300        .unwrap_or_default();
301    if deps.is_empty() {
302        f.render_widget(
303            Paragraph::new("no deps — `a` add ref · rows edit via `e`/`x`")
304                .style(Style::default().fg(MUTED)),
305            table_area,
306        );
307        m.board_area = Rect::default();
308        m.viewport_top = 0;
309        return;
310    }
311    let header =
312        Row::new(vec!["#", "REF"]).style(Style::default().fg(ACCENT).add_modifier(Modifier::BOLD));
313    let rows: Vec<Row> = deps
314        .iter()
315        .enumerate()
316        .map(|(i, d)| Row::new(vec![(i + 1).to_string(), d.clone()]))
317        .collect();
318    let table = Table::new(rows, [Constraint::Length(4), Constraint::Min(30)])
319        .header(header)
320        .block(
321            Block::default()
322                .borders(Borders::ALL)
323                .title(format!(" deps — {} ", m.manifest_path.display())),
324        )
325        .row_highlight_style(Style::default().bg(Color::Indexed(238)));
326    let mut state = ratatui::widgets::TableState::default().with_selected(Some(m.man_sel));
327    f.render_stateful_widget(table, table_area, &mut state);
328    note_table_area(m, table_area, state.offset());
329}

331/// The Author pane — configured registries plus the pack/publish
332/// entry points; results land in the op log like every other job.
333fn draw_author(f: &mut Frame, m: &mut Model, area: Rect) {
334    let mut lines: Vec<Line> = vec![Line::from(vec![
335        Span::styled(
336            "authoring",
337            Style::default().fg(ACCENT).add_modifier(Modifier::BOLD),
338        ),
339        Span::raw(" — bundle pack/publish"),
340    ])];
341    lines.push(Line::from(""));
342    if m.author_regs.is_empty() {
343        lines.push(Line::from(Span::styled(
344            "no registries configured — publish routes to the default local root",
345            Style::default().fg(MUTED),
346        )));
347    } else {
348        lines.push(Line::from(Span::styled(
349            "registries:",
350            Style::default().fg(ACCENT),
351        )));
352        for (name, kind) in &m.author_regs {

355            } else {

359        }
360    }
361    lines.push(Line::from(""));
362    lines.push(Line::from(Span::styled(
363        "p  pack <dir> [out.tgz]      P  publish <dir> [registry]",
364        Style::default().fg(MUTED),
365    )));
366    lines.push(Line::from(Span::styled(
367        "r  :registries report        R  reload registry list",
368        Style::default().fg(MUTED),
369    )));
370    lines.push(Line::from(Span::styled(
371        "publish asks for confirmation; progress streams to the log",
372        Style::default().fg(MUTED),
373    )));
374    f.render_widget(
375        Paragraph::new(lines).block(Block::default().borders(Borders::ALL).title(" author ")),
376        area,
377    );
378    m.board_area = Rect::default();
379    m.viewport_top = 0;
380}

382/// Record where a bordered table's data rows landed — the border
383/// eats one row/column on each side; `state.offset()` is the scroll.
384fn note_table_area(m: &mut Model, area: Rect, offset: usize) {
385    m.board_area = Rect {
386        x: area.x.saturating_add(1),
387        y: area.y.saturating_add(1),
388        width: area.width.saturating_sub(2),
389        height: area.height.saturating_sub(2),
390    };
391    m.viewport_top = offset;
392}

394fn draw_installed(f: &mut Frame, m: &mut Model, area: Rect) {
395    let vis = m.visible();
396    if let Some(err) = &m.load_err {
397        f.render_widget(
398            Paragraph::new(format!("store: {err}")).style(Style::default().fg(Color::Red)),
399            area,
400        );
401        m.board_area = Rect::default();
402        m.viewport_top = 0;
403        return;
404    }
405    if m.entries.is_empty() {
406        f.render_widget(
407            Paragraph::new("no packages installed — `i` install-by-ref, `:install <ref>`, or `tamias install <ref>`")
408                .style(Style::default().fg(MUTED)),
409            area,
410        );
411        m.board_area = Rect::default();
412        m.viewport_top = 0;
413        return;
414    }
415    if vis.is_empty() {
416        f.render_widget(
417            Paragraph::new(format!("no entries match /{}", m.filter))
418                .style(Style::default().fg(MUTED)),
419            area,
420        );
421        m.board_area = Rect::default();
422        m.viewport_top = 0;
423        return;
424    }
425
426    let header = Row::new(vec!["NAME", "TYPE", "SOURCE", "RESOLVED", "DIGEST", "SIZE"])
427        .style(Style::default().fg(ACCENT).add_modifier(Modifier::BOLD));
428    let rows: Vec<Row> = vis
429        .iter()
430        .map(|&i| {
431            let r = &m.entries[i];
432            let rec = &r.entry.record;
433            Row::new(vec![
434                format!("{}{}", r.entry.label(), trust_badge(rec)),
435                rec.kind.clone().unwrap_or_else(|| "-".into()),
436                rec.source_ref.clone(),
437                short_sha(&rec.resolved),
438                short_sha(&rec.digest),
439                r.size.map_or("-".into(), human_size),
440            ])
441        })
442        .collect();
443    let widths = [
444        Constraint::Min(16),
445        Constraint::Length(12),
446        Constraint::Min(14),
447        Constraint::Length(14),
448        Constraint::Length(14),
449        Constraint::Length(9),
450    ];
451    let table = Table::new(rows, widths)
452        .header(header)
453        .block(Block::default().borders(Borders::ALL).title(" installed "))
454        .row_highlight_style(Style::default().bg(Color::Indexed(238)));
455    let mut state = ratatui::widgets::TableState::default().with_selected(Some(m.sel));
456    f.render_stateful_widget(table, area, &mut state);
457    note_table_area(m, area, state.offset());
458}

530                lines.push(Line::from(vec![
531                    Span::styled("deps:      ", Style::default().fg(ACCENT)),
532                    Span::raw(manifest.deps.join(", ")),
533                ]));
534            }

538                    .iter()

543                    Span::styled("provider:  ", Style::default().fg(ACCENT)),
544                    Span::raw(format!(
545                        "{} service(s){}",
546                        provider.services.len(),
547                        if services.is_empty() {
548                            String::new()
549                        } else {
550                            format!(" — {services}")
551                        }
552                    )),
553                ]));
554            }
555        }
556        None => lines.push(Line::from(vec![
557            Span::styled("manifest:  ", Style::default().fg(WARN)),
558            Span::raw(detail.manifest_err.clone().unwrap_or_else(|| "none".into())),
559        ])),
560    }
561    lines.push(Line::from(""));
562    lines.push(Line::from(Span::styled(
563        "esc/← back — v verify · u update · p pin · r remove · c cache clean · d doctor",
564        Style::default().fg(MUTED),
565    )));
566    let title = format!(" {} ", detail.entry.label());
567    f.render_widget(
568        Paragraph::new(lines).block(Block::default().borders(Borders::ALL).title(title)),
569        area,
570    );
571    m.board_area = Rect::default();
572    m.viewport_top = 0;
573}

575/// The Registry pane — search results table plus the target/query
576/// context in the block title.
577fn draw_registry(f: &mut Frame, m: &mut Model, area: Rect) {
578    if let Some(err) = &m.reg_err {
579        f.render_widget(
580            Paragraph::new(format!("registry: {err}")).style(Style::default().fg(Color::Red)),
581            area,
582        );
583        m.board_area = Rect::default();
584        m.viewport_top = 0;
585        return;
586    }
587    if m.reg_query.trim().is_empty() {
588        f.render_widget(
589            Paragraph::new(
590                "registry search — `/` to type a query (name/scope substring), \
591                 `:registry <name>` picks the target, `i` installs a result",
592            )
593            .style(Style::default().fg(MUTED)),
594            area,
595        );
596        m.board_area = Rect::default();
597        m.viewport_top = 0;
598        return;
599    }
600    if m.reg_results.is_empty() {
601        let hint = if m.busy.is_some() {
602            "searching…"
603        } else {
604            "no results"
605        };
606        f.render_widget(
607            Paragraph::new(format!("{hint} — /{query}", query = m.reg_query))
608                .style(Style::default().fg(MUTED)),
609            area,
610        );
611        m.board_area = Rect::default();
612        m.viewport_top = 0;
613        return;
614    }
615    let header = Row::new(vec!["NAME", "SCOPE", "TYPE", "VERSION", "DIGEST"])
616        .style(Style::default().fg(ACCENT).add_modifier(Modifier::BOLD));
617    let rows: Vec<Row> = m
618        .reg_results
619        .iter()
620        .map(|e| {
621            Row::new(vec![
622                e.label(),
623                e.scope.clone().unwrap_or_else(|| "-".into()),
624                e.kind.clone().unwrap_or_else(|| "-".into()),
625                e.version.clone().unwrap_or_else(|| "-".into()),
626                short_sha(&e.digest),
627            ])
628        })
629        .collect();
630    let widths = [
631        Constraint::Min(16),
632        Constraint::Length(12),
633        Constraint::Length(12),
634        Constraint::Length(10),
635        Constraint::Length(14),
636    ];
637    let title = format!(" registry — {} · /{} ", m.reg_target_label, m.reg_query);
638    let table = Table::new(rows, widths)
639        .header(header)
640        .block(Block::default().borders(Borders::ALL).title(title))
641        .row_highlight_style(Style::default().bg(Color::Indexed(238)));
642    let mut state = ratatui::widgets::TableState::default().with_selected(Some(m.reg_sel));
643    f.render_stateful_widget(table, area, &mut state);
644    note_table_area(m, area, state.offset());
645}

658                Style::default().fg(WARN)

666        Paragraph::new(text).block(Block::default().borders(Borders::TOP).title(" log ")),
667        area,
668    );

671fn draw_status(f: &mut Frame, m: &Model, area: Rect) {
672    let left = if let Some(busy) = &m.busy {
673        let frame = SPINNER[(m.tick_n as usize) % SPINNER.len()];

731                "esc back · v/u/p/r/c/d act on the row"

733                "j/k · Enter detail · v verify · u update · p pin · r remove · i install · c clean · d doctor · / filter · : cmds · q quit"
734            }

739                "streaming — x kill · q quit"

741                "j/k · s/S start/stop · r restart · e/E enable/disable · d deploy · x uninstall · t/T logs · n run · R refresh · : cmds · q quit"
742            }
743        }

746        }
747        Pane::Author => "p pack · P publish · r registries · R reload · : cmds · q quit",
748    };
749    let spans = vec![
750        Span::styled(left, style),
751        Span::raw("  "),
752        Span::styled(hint, Style::default().fg(MUTED)),
753    ];
754    f.render_widget(Paragraph::new(Line::from(spans)), area);
755}

757/// The `/`/`:`/prompt input line — drawn over the bottom of the pane
758/// while a text mode is active.
759fn draw_input(f: &mut Frame, m: &Model, area: Rect) {
760    let (label, text) = match &m.mode {
761        Mode::Filter => ("/", m.input.as_str()),
762        Mode::Search => ("/", m.input.as_str()),
763        Mode::Command => (":", m.cmdline.as_str()),
764        Mode::Prompt(kind) => {
765            let l = match kind {
766                PromptKind::Pin { .. } => "pin > ",
767                PromptKind::InstallRef => "install > ",
768                PromptKind::RegistryTarget => "registry > ",
769                PromptKind::AddDep => "dep+ > ",
770                PromptKind::SetDep { .. } => "dep→ > ",
771                PromptKind::DeployConfig { .. } => "deploy cfg > ",
772                PromptKind::PackSpec => "pack dir [out] > ",
773                PromptKind::PublishSpec => "publish dir [registry] > ",
774            };
775            (l, m.prompt_buf.as_str())
776        }
777        _ => return,
778    };
779    let y = area.bottom().saturating_sub(2).max(area.y);
780    let rect = Rect {
781        x: area.x,
782        y,
783        width: area.width,
784        height: 1,
785    };
786    let spans = vec![
787        Span::styled(label.to_string(), Style::default().fg(ACCENT)),
788        Span::raw(text.to_string()),
789        Span::styled("█", Style::default().fg(ACCENT)),
790    ];
791    f.render_widget(Paragraph::new(Line::from(spans)), rect);
792}

794/// A centered confirm/consent dialog — `y`/`n`, default `n`.
795fn draw_modal(f: &mut Frame, modal: &Modal, area: Rect) {
796    let (title, body): (String, Vec<String>) = match modal {
797        Modal::Confirm { title, body, .. } => (title.clone(), body.clone()),
798        Modal::Consent { reason, .. } => (
799            "bundled drivers — consent? [y/n]".to_string(),
800            vec![reason.clone()],
801        ),
802    };
803    let mut lines: Vec<Line> = body
804        .iter()
805        .map(|b| Line::from(Span::styled(b.clone(), Style::default().fg(WARN))))
806        .collect();
807    lines.push(Line::from(""));
808    lines.push(Line::from(Span::styled(
809        "y = yes · n/esc = no (default n)",
810        Style::default().fg(MUTED),
811    )));
812    let w = (area.width * 3 / 5).clamp(30, 72);
813    let h = (lines.len() as u16 + 2).min(area.height);
814    let x = area.x + area.width.saturating_sub(w) / 2;
815    let y = area.y + area.height.saturating_sub(h) / 2;
816    let rect = Rect::new(x, y, w, h);
817    f.render_widget(Clear, rect);
818    f.render_widget(
819        Paragraph::new(lines)
820            .block(Block::default().borders(Borders::ALL).title(title))
821            .style(Style::default().fg(Color::White))
822            .wrap(Wrap { trim: false }),
823        rect,
824    );
825}
