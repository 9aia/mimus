1//! `include:` base resolution — the install half of payload
2//! composition (PL-0007 / P-004).
3//!
4//! Each `Manifest::include` entry resolves **in manifest order** to a
5//! base payload dir:
6//!
7//! - an **installed store name** — full label (`@scope/name@version`),
8//!   scoped name (`@scope/name`), or bare name (`name`, ambiguous
9//!   across scopes → error) — via [`find_store_entry`] over
10//!   `Store::entries`. A missing name is a hard error naming it with a
11//!   hint to use an installable ref; `@`-prefixed entries never fall
12//!   through to path resolution.
13//! - an **installable ref** — anything the ref grammar claims
14//!   (`git:`/`cf:`/`http(s)`/paths) that isn't a store name — recurses
15//!   through [`crate::install_inner`], so the base gets the full
16//!   pipeline (fetch → *its own* includes → drivers → store commit)
17//!   and the outer package composes over the committed payload.
18//!
19//! Recursion is bounded two ways ([`IncludeState`]): a depth cap
20//! ([`MAX_INCLUDE_DEPTH`]) and a cycle set of in-progress package
21//! labels — a package including itself or an ancestor fails loudly.
22//!
23//! Composition itself is *not* transitive on the store-name path: a
24//! store entry's payload is whatever was committed at *its* install
25//! (already composed if it had `include:` then). We compose stored
26//! payloads as-is — never re-resolve their includes.
27

40/// Include-recursion cap — a chain of ref-includes deeper than this
41/// fails instead of recursing forever (cycles are caught earlier by
42/// [`IncludeState::in_progress`]; the cap bounds pathological DAGs).
43pub(crate) const MAX_INCLUDE_DEPTH: usize = 8;
44
45/// Cycle/depth state threaded through recursive installs. Created
46/// fresh per top-level [`crate::install_with_progress`] call.
47#[derive(Default)]
48pub(crate) struct IncludeState {
49    /// Labels (`@scope/name@version`, `name@<sha12>`) of packages whose
50    /// `include:` lists are currently resolving — the ancestor chain.
51    /// [`install_inner`] inserts its own label before resolving and
52    /// removes it after; [`resolve_base`] rejects a base whose store
53    /// label is already in the set.
54    pub in_progress: HashSet<String>,
55    /// Current include-recursion depth — `0` at the top-level install,
56    /// incremented around each ref-include's recursive install.
57    pub depth: usize,
58    /// Labels of packages whose `template.extends` parent resolution
59    /// is in progress — the ancestor chain for extends-cycle
60    /// detection ([`crate::extends`]). Separate from `in_progress`:
61    /// include composition finishes before extends materializes, so
62    /// the two windows never overlap.
63    pub extends_in_progress: HashSet<String>,
64    /// Current extends-recursion depth — incremented around each
65    /// ref-parent's recursive install.
66    pub extends_depth: usize,
67}

69/// The store-entry label for a not-yet-committed package — same rule
70/// as [`StoreEntry::label`] (`@scope/name@version`, sha short form
71/// when unversioned) so in-progress keys compare equal to store hits.
72pub(crate) fn pending_label(
73    name: &str,
74    scope: Option<&str>,
75    version: Option<&str>,
76    resolved: &str,
77) -> String {
78    let record = StoreRecord {
79        name: name.to_string(),
80        scope: scope.map(str::to_string),
81        version: version.map(str::to_string),
82        kind: None,
83        source_ref: String::new(),
84        resolved: resolved.to_string(),
85        digest: String::new(),
86        signature: None,
87        signer: None,
88        trust: None,
89        installed_at: 0,
90    };
91    StoreEntry {
92        dir: PathBuf::new(),
93        record,
94    }
95    .label()
96}

98/// Resolve every `include:` entry of `includes` to a base payload dir,
99/// **in manifest order** (earlier entries outrank later ones —
100/// [`tamias_core::overlay`] applies them last-to-first).
101///
102/// `report` sees a [`Progress::Including`] event per resolved base so
103/// surfaces can show composition happening; ref-includes additionally
104/// stream the recursive install's own events.
105///
106/// # Errors
107/// Fails on an ambiguous/missing store name, an unparseable ref, a
108/// cycle, depth-cap breach, or any recursive-install failure.
109#[allow(clippy::too_many_arguments)]
110pub(crate) fn resolve_bases(
111    env: &Env,
112    includes: &[String],
113    state: &mut IncludeState,
114    allow_drivers: bool,
115    surface: Surface,
116    report: &mut dyn FnMut(Progress),
117) -> Result<Vec<PathBuf>, Error> {
118    let store = Store::new(&env.store_root);
119    let mut bases = Vec::with_capacity(includes.len());
120    for text in includes {
121        // Re-read per entry: a ref-include earlier in the list installs
122        // into the store, and a later store-name entry may name it.
123        let entries = store.entries()?;
124        bases.push(resolve_base(
125            env,
126            &entries,
127            text,
128            state,
129            allow_drivers,
130            surface,
131            report,
132        )?);
133    }
134    Ok(bases)
135}

137/// One include entry → one base payload dir. See the module docs for
138/// the store-name-then-ref order.
139#[allow(clippy::too_many_arguments)]
140fn resolve_base(
141    env: &Env,
142    entries: &[StoreEntry],
143    text: &str,
144    state: &mut IncludeState,
145    allow_drivers: bool,
146    surface: Surface,
147    report: &mut dyn FnMut(Progress),
148) -> Result<PathBuf, Error> {
149    // 1. Store name — label / `@scope/name` / bare `name`.
150    if let Some(entry) = find_store_entry(entries, text)? {
151        let label = entry.label();
152        if state.in_progress.contains(&label) {
153            return Err(Error::Failed(format!(
154                "include cycle: '{text}' resolves to {label}, which is already being installed"
155            )));
156        }
157        report(Progress::Including { base: label });
158        return Ok(entry.dir);
159    }
160    // `@scope/name` is a *name*, never a path — a missing one errors
161    // here instead of falling through to fs-ref resolution.
162    if text.starts_with('@') {
163        return Err(missing_base(text, "include"));
164    }
165    // 2. Installable ref — recurse through the full pipeline.
166    let spec = crate::refs()
167        .parse(text)
168        .map_err(|_| missing_base(text, "include"))?;
169    // A remote include base is a transitive pull under this install's
170    // consent flags — warn when the ref isn't pinned so the operator
171    // knows provenance rides on "whatever the remote serves now".
172    // `cf:` refs need no warning: they're pinned by name@version plus
173    // the index-recorded payload digest.
174    if let Some(reason) = unpinned_remote_ref(&spec) {
175        eprintln!("warning: include '{text}' is an unpinned remote ref — {reason}");
176    }
177    if spec.is::<FsSpec>() {
178        // `FsSpec.path` is `expand_path(input)` — recompute it rather
179        // than downcasting the opaque spec.
180        if !expand_path(Path::new(text)).exists() {
181            return Err(missing_base(text, "include"));
182        }
183    }
184    if state.depth >= MAX_INCLUDE_DEPTH {
185        return Err(Error::Failed(format!(
186            "include '{text}' exceeds the {MAX_INCLUDE_DEPTH}-deep nesting cap"
187        )));
188    }
189    report(Progress::Including {
190        base: text.to_string(),
191    });
192    state.depth += 1;
193    let result = install_inner(env, text, allow_drivers, surface, report, state);
194    state.depth -= 1;
195    let entry = result?;
196    Ok(entry.dir)
197}

199/// Why a parsed include-base ref is an *unpinned remote pull*, or
200/// `None` when it's pinned/local. Remote refs fetched through `include:`
201/// are transitive pulls the top-level ref's consent flags cover — the
202/// warning is advisory (remote includes are legitimate), not a failure.
203///
204/// - `http(s)://…` without a `#sha256:` fragment → warn (unpinned).
205/// - `git:…` without a `#<rev>` that is a full 40-hex sha → warn
206///   (moving refs like `#main`/`#v1` resolve at fetch time). Local-path
207///   clones (`/`, `.`, `~`, `file://`) aren't remote pulls → no warn.
208/// - `cf:` is registry-pinned (name@version + index payload digest) →
209///   no warn; fs paths and store names never reach here as remote.
210pub(crate) fn unpinned_remote_ref(spec: &Spec) -> Option<&'static str> {
211    if let Some(http) = spec.downcast_ref::<HttpSpec>() {
212        return (http.digest.is_none()).then_some("http ref has no #sha256: digest pin");
213    }
214    if let Some(git) = spec.downcast_ref::<GitSpec>() {
215        let local = git.url.starts_with("file://")
216            || git.url.starts_with('/')
217            || git.url.starts_with('.')
218            || git.url.starts_with('~');
219        if !local {
220            let pinned = git
221                .rev
222                .as_deref()
223                .is_some_and(|rev| rev.len() == 40 && rev.bytes().all(|b| b.is_ascii_hexdigit()));
224            if !pinned {
225                return Some("git ref is not pinned to a full commit sha");
226            }
227        }
228    }
229    None
230}

232/// How an `include:` entry resolves for [`crate::plan_install`] — the
233/// store-name-then-ref classification of [`resolve_base`] minus the
234/// recursive install and any fetch.
235pub(crate) enum IncludePlan {
236    /// An installed store entry — carries its [`StoreEntry::label`].
237    Store(String),
238    /// Not a store name, but the ref grammar claims it — a real
239    /// install would recursively install it as a base.
240    WouldInstall,
241    /// Neither — carries why (an `@`-name that's not installed, an
242    /// ambiguous store name, an unparseable ref, or a path that
243    /// doesn't exist). A real install fails naming this entry.
244    Missing(String),
245}
246
247/// Classify one `include:` entry without installing anything — see
248/// [`IncludePlan`]. Mirrors [`resolve_base`]'s order: store name via
249/// [`find_store_entry`] first, `@`-prefixed names never fall through
250/// to ref resolution, fs paths must exist on disk.
251pub(crate) fn plan_include(entries: &[StoreEntry], text: &str) -> IncludePlan {
252    plan_base(entries, text, "include")
253}
254
255/// [`plan_include`] generalized — `noun` names the manifest field in
256/// error text (`include`, `template.extends`).
257pub(crate) fn plan_base(entries: &[StoreEntry], text: &str, noun: &str) -> IncludePlan {
258    match find_store_entry(entries, text) {
259        Ok(Some(entry)) => return IncludePlan::Store(entry.label()),
260        Ok(None) => {}
261        // Ambiguous bare name — the real install errors; surface the
262        // reason so the plan's problems list stays accurate.
263        Err(e) => return IncludePlan::Missing(e.to_string()),
264    }
265    if text.starts_with('@') {
266        return IncludePlan::Missing(missing_base(text, noun).to_string());
267    }
268    match crate::refs().parse(text) {
269        Ok(spec) => {
270            // `FsSpec.path` is `expand_path(input)` — same existence
271            // check `resolve_base` makes before recursing.
272            if spec.is::<FsSpec>() && !expand_path(Path::new(text)).exists() {
273                IncludePlan::Missing(missing_base(text, noun).to_string())
274            } else {
275                IncludePlan::WouldInstall
276            }
277        }
278        Err(_) => IncludePlan::Missing(missing_base(text, noun).to_string()),
279    }
280}

282/// The hard error for a base entry that is neither an installed
283/// store name nor a resolvable ref.
284fn missing_base(text: &str, noun: &str) -> Error {
285    Error::Failed(format!(
286        "{noun} '{text}' is not installed — store names must already be in the store; \
287         use an installable ref (path, git:, cf:, http:) or `tamias install {text}` first"
288    ))
289}

291/// Pin the composed payload's manifest to the *local* package's
292/// manifest file, byte-for-byte. [`tamias_core::overlay`] doc-merges
293/// `.yaml`/`.json` files — right for payload docs, wrong for the
294/// manifest itself (a base's `deps:`/`include:` arrays would concat
295/// into the composed manifest). The top package's manifest is
296/// authoritative: `include:` stays for provenance, nothing inherits.
297/// Stray manifest files a base contributed under a different
298/// [`MANIFEST_NAMES`](tamias_core::MANIFEST_NAMES) spelling are
299/// removed so `Manifest::load` sees exactly one.
300///
301/// # Errors
302/// Propagates copy/remove failures.
303pub(crate) fn pin_manifest(local: &Path, dest: &Path) -> Result<(), Error> {
304    let Some(src) = tamias_core::Manifest::path_in(local) else {
305        return Ok(());
306    };
307    let file_name = src
308        .file_name()
309        .map(|n| n.to_string_lossy().into_owned())
310        .unwrap_or_default();
311    for candidate in tamias_core::MANIFEST_NAMES {
312        let target = dest.join(candidate);
313        if candidate == file_name {
314            std::fs::copy(&src, &target).map_err(tamias_core::Error::io(format!(
315                "pin manifest {}",
316                target.display()
317            )))?;
318        } else if target.exists() {
319            std::fs::remove_file(&target).map_err(tamias_core::Error::io(format!(
320                "remove stray manifest {}",
321                target.display()
322            )))?;
323        }
324    }
325    Ok(())
326}

328#[cfg(test)]
329#[cfg_attr(coverage, coverage(off))]
330mod tests {
331    #![allow(clippy::unwrap_used)]
332
333    use super::*;
334
335    /// Parse `input` through the same grammar `resolve_base` uses.
336    fn parsed(input: &str) -> Spec {
337        crate::refs().parse(input).unwrap()
338    }
339
340    #[test]
341    fn unpinned_remote_ref_flags() {
342        // http(s) without #sha256: → warn; pinned → clean.
343        assert!(unpinned_remote_ref(&parsed("https://x/y.tgz")).is_some());
344        assert!(unpinned_remote_ref(&parsed("http://x/y.tgz")).is_some());
345        let sha = "a".repeat(64);
346        assert!(unpinned_remote_ref(&parsed(&format!("https://x/y.tgz#sha256:{sha}"))).is_none());
347        assert!(unpinned_remote_ref(&parsed(&format!("http://x/y.tgz#sha256={sha}"))).is_none());
348
349        // git: remote without a full-sha rev → warn.
350        assert!(unpinned_remote_ref(&parsed("git:https://h/o/r")).is_some());
351        assert!(unpinned_remote_ref(&parsed("git:org/repo#main")).is_some());
352        assert!(unpinned_remote_ref(&parsed("git:org/repo#v1.0")).is_some());
353        assert!(unpinned_remote_ref(&parsed("git:org/repo#abc1234")).is_some());
354        // full 40-hex sha → pinned.
355        let commit = "0123456789abcdef0123456789abcdef01234567";
356        assert!(unpinned_remote_ref(&parsed(&format!("git:org/repo#{commit}"))).is_none());
357        // local clones aren't remote pulls.
358        assert!(unpinned_remote_ref(&parsed("git:file:///tmp/r")).is_none());
359        assert!(unpinned_remote_ref(&parsed("git:/abs/path")).is_none());
360        assert!(unpinned_remote_ref(&parsed("git:./rel/path")).is_none());
361        assert!(unpinned_remote_ref(&parsed("git:~/path")).is_none());
362
363        // cf: is registry-pinned; fs paths and plain names are local.
364        assert!(unpinned_remote_ref(&parsed("cf:9aia/bucket/hello@0.1.0")).is_none());
365        let dir = tempfile::tempdir().unwrap();
366        let path = dir.path().join("pkg");
367        std::fs::create_dir_all(&path).unwrap();
368        assert!(unpinned_remote_ref(&parsed(path.to_str().unwrap())).is_none());
369    }
