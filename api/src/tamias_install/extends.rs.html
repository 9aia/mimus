1//! `template.extends:` parent materialization — the install half of
2//! template-pack composition (scarab's `type: template` driver).
3//!
4//! A pack manifest may declare `template.extends: <ref>` inside its
5//! driver-owned `template:` table. tamias never interprets the table
6//! beyond this one key: at install, after `include:` composition and
7//! the signature/consent gates, the parent payload is resolved
8//! (store name → that entry's committed payload; anything else the
9//! ref grammar claims → a recursive [`crate::install_inner`], so the
10//! parent gets the full pipeline — fetch, *its own* extends,
11//! drivers, store commit) and copied into the composed tree at
12//! [`PARENT_DIR`].
13//!
14//! Because the parent's committed payload already carries *its*
15//! `.scarab-parent/` (materialized by its own install), nesting is
16//! automatic: a grandparent lands at
17//! `.scarab-parent/.scarab-parent/`. The store commit therefore
18//! carries the whole ancestor chain, and every later driver verb —
19//! `install` (this dispatch), `deploy`, `uninstall` — sees it under
20//! `--dir <payload>` with no extra work.
21//!
22//! Recursion is bounded like `include:` ([`IncludeState`]): a depth
23//! cap ([`MAX_EXTENDS_DEPTH`]) and an in-progress label set
24//! ([`IncludeState::extends_in_progress`]) that catches
25//! `A extends B, B extends A` cycles through refs. Store-name
26//! parents can't cycle — their payloads are already committed and
27//! finite.
28//!
29//! The signature/consent/trust digest deliberately covers the tree
30//! *before* `.scarab-parent/` is added: publish doesn't flatten
31//! `extends`, so a signed parent-extending pack still verifies
32//! against its published bytes, and per-package driver consent binds
33//! to the package's own payload. `Store::install` digests the final
34//! tree (parents included), so `verify` stays honest.

46/// Extends-recursion cap — an `extends` chain deeper than this fails
47/// instead of recursing forever (ref cycles are caught earlier by
48/// [`IncludeState::extends_in_progress`]; the cap bounds pathological
49/// DAGs). Same bound as [`include::MAX_INCLUDE_DEPTH`].
50pub(crate) const MAX_EXTENDS_DEPTH: usize = 8;
51
52/// Payload subdir a resolved parent materializes into — the driver
53/// (`tamias-driver-template`) discovers the chain by this name.
54pub(crate) const PARENT_DIR: &str = ".scarab-parent";
55
56/// `template.extends` declared by the manifest inside `dir` —
57/// `Some(ref)` when present, `None` when absent or JSON-null.
58///
59/// The `template:` table is driver-owned: tamias re-reads the raw
60/// manifest file (the typed [`tamias_core::Manifest`] drops unknown
61/// keys) and validates only this one field — it must be a non-empty
62/// string; anything else fails the install.
63///
64/// # Errors
65/// Fails on manifest read/parse errors or a malformed `extends`.
66pub(crate) fn extends_ref(dir: &Path) -> Result<Option<String>, Error> {
67    let Some(path) = tamias_core::Manifest::path_in(dir) else {
68        return Ok(None);
69    };
70    let text =
71        std::fs::read_to_string(&path).map_err(Error::io(format!("read {}", path.display())))?;
72    // Same syntax dispatch as `Manifest::parse_at`: .yaml/.yml → YAML,
73    // everything else → TOML. Both normalize to JSON for the lookup.
74    let value: serde_json::Value = if matches!(
75        path.extension().and_then(|e| e.to_str()),
76        Some("yaml" | "yml")
77    ) {
78        let yaml: serde_yaml_ng::Value = serde_yaml_ng::from_str(&text)
79            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?;
80        serde_json::to_value(yaml)
81            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?
82    } else {
83        let table: toml::Table = toml::from_str(&text)
84            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?;
85        serde_json::to_value(table)
86            .map_err(|e| Error::Failed(format!("parse {}: {e}", path.display())))?
87    };
88    match value.get("template").and_then(|t| t.get("extends")) {
89        None | Some(serde_json::Value::Null) => Ok(None),
90        Some(serde_json::Value::String(s)) if !s.trim().is_empty() => Ok(Some(s.clone())),
91        Some(_) => Err(Error::Failed(format!(
92            "{}: template.extends must be a non-empty string ref",
93            path.display()
94        ))),
95    }
96}

98/// Resolve `parent_ref` to a payload dir and copy it into
99/// `child_root/.scarab-parent/` — replacing any dir already there
100/// (a declared `extends` is authoritative over vendored bytes).
101///
102/// `child_label` is the installing package's store label
103/// ([`include::pending_label`]) — the cycle-set key. `report` sees a
104/// [`Progress::Including`] event for the parent, and a ref-parent's
105/// recursive install streams its own events like an `include:` base.
106///
107/// # Errors
108/// Fails on an ambiguous/missing store name, an unparseable ref, a
109/// cycle, a depth-cap breach, or any recursive-install/copy failure.
110#[allow(clippy::too_many_arguments)]
111pub(crate) fn materialize(
112    env: &Env,
113    child_root: &Path,
114    child_label: &str,
115    parent_ref: &str,
116    state: &mut IncludeState,
117    allow_drivers: bool,
118    surface: Surface,
119    report: &mut dyn FnMut(Progress),
120) -> Result<(), Error> {
121    let parent = resolve_parent(
122        env,
123        child_label,
124        parent_ref,
125        state,
126        allow_drivers,
127        surface,
128        report,
129    )?;
130    let dest = child_root.join(PARENT_DIR);
131    if dest.exists() {
132        std::fs::remove_dir_all(&dest).map_err(Error::io(format!("clear {}", dest.display())))?;
133    }
134    // create_dir_all first: an empty parent payload must still leave a
135    // `.scarab-parent/` the driver can descend into (its manifest is
136    // what load_chain reads — a payload always has one).
137    std::fs::create_dir_all(&dest).map_err(Error::io(format!("create {}", dest.display())))?;
138    copy_tree(&parent, &dest)
139}

141/// One `template.extends` ref → one parent payload dir. Mirrors
142/// [`include::resolve_base`]'s store-name-then-ref order; see the
143/// module docs for the recursion bounds.
144#[allow(clippy::too_many_arguments)]
145fn resolve_parent(
146    env: &Env,
147    child_label: &str,
148    parent_ref: &str,
149    state: &mut IncludeState,
150    allow_drivers: bool,
151    surface: Surface,
152    report: &mut dyn FnMut(Progress),
153) -> Result<PathBuf, Error> {
154    // 1. Store name — label / `@scope/name` / bare `name`. The stored
155    //    payload already carries its own `.scarab-parent/` chain.
156    let entries = Store::new(&env.store_root).entries()?;
157    if let Some(entry) = find_store_entry(&entries, parent_ref)? {
158        report(Progress::Including {
159            base: entry.label(),
160        });
161        return Ok(entry.dir);
162    }
163    // `@scope/name` is a *name*, never a path.
164    if parent_ref.starts_with('@') {
165        return Err(missing_parent(parent_ref));
166    }
167    // 2. Installable ref — recurse through the full pipeline.
168    let spec = crate::refs()
169        .parse(parent_ref)
170        .map_err(|_| missing_parent(parent_ref))?;
171    if let Some(reason) = include::unpinned_remote_ref(&spec) {
172        eprintln!("warning: template.extends '{parent_ref}' is an unpinned remote ref — {reason}");
173    }
174    if spec.is::<FsSpec>() && !expand_path(Path::new(parent_ref)).exists() {
175        return Err(missing_parent(parent_ref));
176    }
177    if state.extends_in_progress.contains(child_label) {
178        return Err(Error::Failed(format!(
179            "template.extends cycle: {child_label} is already being installed as a parent"
180        )));
181    }
182    if state.extends_depth >= MAX_EXTENDS_DEPTH {
183        return Err(Error::Failed(format!(
184            "template.extends '{parent_ref}' exceeds the {MAX_EXTENDS_DEPTH}-deep nesting cap"
185        )));
186    }
187    report(Progress::Including {
188        base: parent_ref.to_string(),
189    });
190    state.extends_in_progress.insert(child_label.to_string());
191    state.extends_depth += 1;
192    let result = install_inner(env, parent_ref, allow_drivers, surface, report, state);
193    state.extends_depth -= 1;
194    state.extends_in_progress.remove(child_label);
195    Ok(result?.dir)
196}

198/// The hard error for an extends ref that is neither an installed
199/// store name nor a resolvable ref.
200fn missing_parent(parent_ref: &str) -> Error {
201    Error::Failed(format!(
202        "template.extends '{parent_ref}' is not installed — store names must already be in the \
203         store; use an installable ref (path, git:, cf:, http:) or `tamias install {parent_ref}` first"
204    ))
205}

207#[cfg(test)]
208#[cfg_attr(coverage, coverage(off))]
209mod tests {
210    #![allow(clippy::unwrap_used)]
211
212    use super::*;
213
214    /// Write a manifest file with the given `template:` table body.
215    fn manifest(dir: &Path, body: &str) {
216        std::fs::write(
217            dir.join("tamias.yaml"),
218            format!("name: x\ntype: template\ntemplate:\n{body}"),
219        )
220        .unwrap();
221    }
222
223    #[test]
224    fn extends_ref_reads_yaml_and_toml() {
225        let dir = tempfile::tempdir().unwrap();
226        manifest(dir.path(), "  extends: git:org/base\n");
227        assert_eq!(
228            extends_ref(dir.path()).unwrap().as_deref(),
229            Some("git:org/base")
230        );
231        // absent / null → None
232        manifest(dir.path(), "  files: []\n");
233        assert_eq!(extends_ref(dir.path()).unwrap(), None);
234        // non-string / empty → error
235        manifest(dir.path(), "  extends: 42\n");
236        assert!(extends_ref(dir.path()).is_err());
237        manifest(dir.path(), "  extends: \"  \"\n");
238        assert!(extends_ref(dir.path()).is_err());
239        // toml spelling
240        std::fs::write(
241            dir.path().join("tamias.toml"),
242            "name = \"x\"\ntype = \"template\"\n[template]\nextends = \"git:org/t\"\n",
243        )
244        .unwrap();
245        std::fs::remove_file(dir.path().join("tamias.yaml")).unwrap();
246        assert_eq!(
247            extends_ref(dir.path()).unwrap().as_deref(),
248            Some("git:org/t")
249        );
250    }
