11/// One indexed package.
12#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
13pub struct RegistryEntry {
14    /// Bare name.
15    pub name: String,
16    /// Scope, when namespaced.
17    #[serde(default, skip_serializing_if = "Option::is_none")]
18    pub scope: Option<String>,
19    /// Declared version.
20    #[serde(default, skip_serializing_if = "Option::is_none")]
21    pub version: Option<String>,
22    /// Content type.
23    #[serde(default, skip_serializing_if = "Option::is_none")]
24    pub kind: Option<String>,
25    /// `sha256:<hex>` of the packed payload.
26    pub digest: String,
27    /// Armored `ssh-keygen -Y` signature over the provenance anchors
28    /// (see `tamias-sign`) — written by `publish --sign`, checked by
29    /// `cf:` installs and `tamias verify`.
30    #[serde(default, skip_serializing_if = "Option::is_none")]
31    pub signature: Option<String>,
32    /// Publisher display identity carried by the index (P-005) —
33    /// display metadata only; nothing populates it yet (no publisher
34    /// identity exists), but the schema round-trips it through the
35    /// local and `cf` registries.
36    #[serde(default, skip_serializing_if = "Option::is_none")]
37    pub publisher: Option<String>,
38    /// Path to the `.tgz`, relative to the registry root.
39    pub file: String,
40}

42impl RegistryEntry {
43    /// `@scope/name@version` (or name-only fallbacks).
44    #[must_use]
45    pub fn label(&self) -> String {
46        let scoped = match &self.scope {
47            Some(scope) => format!("@{scope}/{}", self.name),
48            None => self.name.clone(),
49        };
50        match &self.version {
51            Some(v) => format!("{scoped}@{v}"),
52            None => scoped,
53        }
54    }
55}

57/// The result of [`Registry::publish`] — the new index entry plus the
58/// older versions the retention policy pruned from the index (P-024).
59#[derive(Clone, Debug)]
60pub struct PublishOutcome {
61    /// The published index entry.
62    pub entry: RegistryEntry,
63    /// `label()`s of index entries the retention policy removed —
64    /// empty when retention is unset/disabled. Payload blobs are never
65    /// pruned (index-entry prune only, v1): installed copies still
66    /// verify, and the `.tgz` stays fetchable by direct URL — but the
67    /// version no longer resolves through the index (`search`,
68    /// `cf:<name>@<version>` installs).
69    pub pruned: Vec<String>,
70}
71
72impl std::ops::Deref for PublishOutcome {
73    type Target = RegistryEntry;
74
75    fn deref(&self) -> &Self::Target {
76        &self.entry
77    }
78}

80/// Apply a retention policy to `entries` after `published` joined the
81/// index — the pure core both registry adapters share (P-024).
82///
83/// Only the published package's `(scope, name)` group is touched.
84/// Within it, versions that parse as semver order descending —
85/// prereleases count in the same ordering (`1.0.0-rc.1` sorts below
86/// `1.0.0`), publish order is irrelevant — and only the `keep` most
87/// recent stay indexed, plus `published` itself (never pruned, even
88/// when it sorts outside the window — e.g. re-publishing an old
89/// version). Entries with no version or a non-semver version can't be
90/// ordered, so they're always kept and don't count toward `keep`.
91/// `keep = 0` keeps only the published version. Returns the labels of
92/// the pruned entries, sorted for stable reporting.
93#[must_use]
94pub fn apply_retention(
95    entries: &mut Vec<RegistryEntry>,
96    keep: usize,
97    published: &RegistryEntry,
98) -> Vec<String> {
99    let same_pkg = |e: &RegistryEntry| e.name == published.name && e.scope == published.scope;
100    // Orderable = same package + semver-parseable version. The
101    // published entry's position is its dedup key (scope, name,
102    // version) — unique within the index.
103    let mut orderable: Vec<(usize, semver::Version, &str)> = entries
104        .iter()
105        .enumerate()
106        .filter(|(_, e)| same_pkg(e))
107        .filter_map(|(i, e)| {
108            let v = e.version.as_deref()?;
109            semver::Version::parse(v).ok().map(|sv| (i, sv, v))
110        })
111        .collect();
112    if orderable.is_empty() {
113        return Vec::new();
114    }
115    // Newest first; the version string breaks semver ties (build
116    // metadata doesn't participate in `Version::cmp`).
117    orderable.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| b.2.cmp(a.2)));
118    let orderable_idx: std::collections::HashSet<usize> =
119        orderable.iter().map(|(i, _, _)| *i).collect();
120    let mut retained: std::collections::HashSet<usize> =
121        orderable.iter().take(keep).map(|(i, _, _)| *i).collect();
122    // The published version is never pruned — force-retain it even
123    // when it sorts outside the top-`keep` window.
124    if let Some(pos) = entries
125        .iter()
126        .position(|e| same_pkg(e) && e.version == published.version && e.file == published.file)
127    {
128        retained.insert(pos);
129    }
130    let mut pruned: Vec<String> = entries
131        .iter()
132        .enumerate()
133        .filter(|(i, _)| orderable_idx.contains(i) && !retained.contains(i))
134        .map(|(_, e)| e.label())
135        .collect();
136    pruned.sort();
137    if pruned.is_empty() {
138        return pruned;
139    }
140    let drop: std::collections::HashSet<usize> =
141        orderable_idx.difference(&retained).copied().collect();
142    let mut i = 0usize;
143    entries.retain(|_| {
144        let keep_it = !drop.contains(&i);
145        i += 1;
146        keep_it
147    });
148    pruned
149}

151/// The registry port.
152pub trait Registry {
153    /// Every indexed entry.
154    ///
155    /// # Errors
156    /// Adapter-specific.
157    fn entries(&self) -> Result<Vec<RegistryEntry>, Error>;
158
159    /// Pack + index a payload. `signature` is the armored
160    /// `ssh-keygen -Y` signature over the payload's provenance anchors
161    /// (`tamias-sign`), stored on the index entry when `Some`.
162    ///
163    /// # Errors
164    /// Adapter-specific.
165    fn publish(
166        &self,
167        payload_root: &Path,
168        manifest: &Manifest,
169        signature: Option<&str>,
170    ) -> Result<PublishOutcome, Error>;
171
172    /// Entries whose name/scope contains `query` (case-insensitive).
173    ///
174    /// # Errors
175    /// Adapter-specific (delegates to [`Registry::entries`]).
176    fn search(&self, query: &str) -> Result<Vec<RegistryEntry>, Error> {
177        let q = query.to_lowercase();
178        Ok(self
179            .entries()?
180            .into_iter()
181            .filter(|e| {
182                e.name.to_lowercase().contains(&q)
183                    || e.scope.as_deref().unwrap_or("").to_lowercase().contains(&q)
184            })
185            .collect())
186    }
187}

189/// A registry ref — a local dir, or an http(s) base URL.
190#[derive(Clone, Debug, PartialEq, Eq)]
191pub enum RegistryRef {
192    /// A directory path.
193    Local(PathBuf),
194    /// An http(s) base URL.
195    Http(String),
196}
197
198/// Parse a `--registry` value into a [`RegistryRef`].
199#[must_use]
200pub fn parse_registry_ref(value: &str) -> RegistryRef {
201    if value.starts_with("http://") || value.starts_with("https://") {
202        RegistryRef::Http(value.trim_end_matches('/').to_string())
203    } else {
204        RegistryRef::Local(PathBuf::from(value))
205    }
206}
