1//! Check port — verify a claim against reality.
2//! Impls: `mimus-check-fs` (paths), `mimus-check-cmd`
3//! (commands), `mimus-check-symbol` (kuda index), `check-llm`
4//! (behavior prose — semantic tier, advisory). New checker =
5//! new adapter crate.
6
7use mimus_core::{Claim, ClaimKind, Violation};
8
9/// A verifier for some subset of `ClaimKind`s.
10pub trait Checker: Send + Sync {
11    /// The claim kinds this checker can prove or refute.
12    fn handles(&self) -> &[ClaimKind];
13    /// `Ok(Some(v))` = claim refuted with evidence.
14    /// `Ok(None)` = verified — OR unverifiable. Evidence-or-silence:
15    /// a checker that can't prove the claim false emits nothing.
16    fn check(&self, claim: &Claim) -> Result<Option<Violation>, String>;
17}
18
19/// Route claims to the checkers that handle their kind.
20pub fn run<'a>(
21    checkers: &'a [Box<dyn Checker>],
22    claims: &'a [Claim],
23) -> impl Iterator<Item = Result<Violation, String>> + 'a {
24    claims.iter().filter_map(move |c| {
25        checkers
26            .iter()
27            .find(|ch| ch.handles().contains(&c.kind))
28            .map(|ch| ch.check(c))
29            .and_then(|r| r.transpose())
30    })
31}
