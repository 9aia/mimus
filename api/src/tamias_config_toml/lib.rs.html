3//! TOML config impl (`tamias-config-toml`) — the **alternative** to the
4//! standard YAML. Claims `tamias.toml`.
5
6use tamias_config::{Config, ConfigFormat};
7use tamias_core::Error;
8
9/// The TOML config format.
10#[derive(Clone, Copy, Debug, Default)]
11pub struct TomlFormat;
12
13impl ConfigFormat for TomlFormat {
14    fn extensions(&self) -> &'static [&'static str] {
15        &["toml"]
16    }
17
18    fn parse(&self, text: &str) -> Result<Config, Error> {
19        toml::from_str(text).map_err(|e| Error::Failed(format!("parse toml config: {e}")))
20    }
21
22    fn render(&self, config: &Config) -> Result<String, Error> {
23        toml::to_string(config).map_err(|e| Error::Failed(format!("render toml config: {e}")))
24    }
25}
26
27#[cfg(test)]
28#[cfg_attr(coverage, coverage(off))]
29mod tests {
30    use super::*;
31
32    #[test]
33    fn parses_toml_and_rejects_yaml() {
34        let f = TomlFormat;
35        assert_eq!(f.extensions(), &["toml"]);
36        let cfg = f
37            .parse(
38                "[store]\nkind = \"local\"\n[publish]\ndefault = \"cloud\"\n[search]\ndefault = \"mirror\"\n[registries.cloud]\nkind = \"http\"\nurl = \"https://x\"\ntoken_env = \"T\"\n[registries.cloud.limits]\nmax_upload = \"256MiB\"\n[registries.cloud.retention]\nkeep = 3\n[limits]\nmax_download = 1073741824\n[limits.packages.\"kuda-*\"]\nmax_upload = \"2GiB\"\n[retention]\nkeep = 10\n[retention.packages.\"big-*\"]\nkeep = 50\n",
39            )
40            .unwrap();
41        assert_eq!(cfg.store.kind.as_deref(), Some("local"));
42        assert_eq!(cfg.publish.default.as_deref(), Some("cloud"));
43        assert_eq!(cfg.search.default.as_deref(), Some("mirror"));
44        assert_eq!(cfg.registries["cloud"].url.as_deref(), Some("https://x"));
45        assert_eq!(cfg.registries["cloud"].limits.max_upload, Some(256 << 20));
46        assert_eq!(cfg.limits.max_download, Some(1 << 30));
47        assert_eq!(cfg.limits.packages["kuda-*"].max_upload, Some(2 << 30));
48        assert_eq!(cfg.retention.keep, Some(10));
49        assert_eq!(cfg.retention.packages["big-*"].keep, Some(50));
50        assert_eq!(cfg.registries["cloud"].retention.keep, Some(3));
51        assert!(f.parse("name: x").is_err());
52        // render round-trips
53        let text = f.render(&cfg).unwrap();
54        assert_eq!(f.parse(&text).unwrap(), cfg);
55        assert!(f.render(&Config::default()).is_ok());
56    }
57}
