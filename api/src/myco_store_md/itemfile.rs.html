1//! Item files — one md file per issue/proposal/adr/plan:
2//!
3//! ```text
4//! issues/issue-NNNN-slug.md    proposals/proposal-NNNN-slug.md
5//! adrs/NNNN-slug.md            plans/pl-NNNN-slug.md
6//! ```
7//!
8//! `NNNN` = id number padded to 4 digits. TOML frontmatter in
9//! `+++` fences (id, title, status, created, owner, priority,
10//! labels, blocks, links, note, superseded_by, supersedes, epic,
11//! milestone, due, estimate, cycle — extendable: unknown keys are
12//! ignored on read for forward-compat); body = free-form markdown
13//! (verbatim; kind templates are `myco-template`'s job).
14

22/// The `project/` subdirectory a file-backed kind lives in —
23/// the spec's `dir` (`issues`, `spikes`); `doc`-storage kinds
24/// have no files (`None`).
25pub fn item_dir(spec: &KindSpec) -> Option<String> {
26    (spec.storage == Storage::Dir).then(|| spec.dir_name())
27}

29/// Filename prefix before the number — `issue`, `proposal`, `pl`,
30/// the spec's `file_prefix` for user kinds, or empty for ADRs
31/// (`adrs/0004-slug.md`).
32fn file_prefix(spec: &KindSpec) -> Option<String> {
33    (spec.storage == Storage::Dir).then(|| spec.file_prefix())
34}

36/// Canonical filename for an item — `issue-0003-memory-leak.md`,
37/// `0004-hybrid-store.md`. Slug = `slug::slugify(title)`,
38/// `"untitled"` when the title slugifies to nothing.
39pub fn item_filename(spec: &KindSpec, id: &ItemId, title: &str) -> Option<String> {
40    let prefix = file_prefix(spec)?;
41    let slug = {
42        let s = slug::slugify(title);
43        if s.is_empty() {
44            "untitled".to_string()
45        } else {
46            s
47        }
48    };
49    Some(if prefix.is_empty() {
50        format!("{:04}-{slug}.md", id.number())
51    } else {
52        format!("{prefix}-{:04}-{slug}.md", id.number())
53    })
54}

56/// `issues/issue-0003-x.md` → `I-003`; `None` when the filename
57/// doesn't fit the kind's pattern. The number must fit `u32` —
58/// the same bound [`ItemId`]'s own parse applies, so anything
59/// [`item_filename`] can emit parses back (id numbers are `u32`,
60/// i.e. up to 10 digits).
61pub fn parse_item_filename(spec: &KindSpec, filename: &str) -> Option<ItemId> {
62    let prefix = file_prefix(spec)?;
63    let stem = filename.strip_suffix(".md")?;
64    let rest = if prefix.is_empty() {
65        stem
66    } else {
67        stem.strip_prefix(prefix.as_str())?.strip_prefix('-')?
68    };
69    let (digits, slug) = rest.split_once('-')?;
70    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) || slug.is_empty() {
71        return None;
72    }
73    let number: u32 = digits.parse().ok()?;
74    if number == 0 {
75        return None;
76    }
77    Some(ItemId::new(spec.id_kind(), number))
78}

80/// Deepest line the `frontmatter_id` probe will read — real
81/// frontmatter is ~10 keys; past this the `+++` block is
82/// certainly unclosed (or the file is pathological).
83const FRONTMATTER_PROBE_MAX: usize = 128;
84
85/// `id = "…"` pulled out of a *closed* `+++` frontmatter block —
86/// the cheap probe for when the strict parse isn't needed (or
87/// already failed). `None` for files with no `+++` fence, no
88/// closing fence within [`FRONTMATTER_PROBE_MAX`] lines, or no
89/// parseable `id` line — including frontmatter-less files that
90/// merely *borrow* an id number via their filename. The close
91/// requirement is what keeps a body's `id = "…"` (a code block,
92/// a quoted example) from posing as frontmatter.
93pub fn frontmatter_id(src: &str) -> Option<ItemId> {
94    let mut lines = src.lines();
95    if lines.next()?.trim() != "+++" {
96        return None;
97    }
98    let mut found = None;
99    for l in lines.take(FRONTMATTER_PROBE_MAX) {
100        let t = l.trim();
101        // the close ends the block — an id only counts when the
102        // block actually closed (else a body's `id = "…"` would
103        // pose as frontmatter)
104        if t == "+++" {
105            return found;
106        }
107        if found.is_none() {
108            if let Some(rest) = t.strip_prefix("id") {
109                if let Some(v) = rest.trim_start().strip_prefix('=') {
110                    if let Ok(id) = v.trim().trim_matches('"').parse::<ItemId>() {
111                        found = Some(id);
112                    }
113                }
114            }
115        }
116    }
117    None // window exhausted — never closed
118}

120/// Deepest line [`repair_frontmatter`] will read — same bound as
121/// [`FRONTMATTER_PROBE_MAX`]; real frontmatter is ~10 keys.
122const REPAIR_MAX: usize = FRONTMATTER_PROBE_MAX;
123
124/// Strip one layer of `'`/`"` quoting a hand-written YAML scalar
125/// carries.
126fn unquote(s: &str) -> &str {
127    let b = s.as_bytes();
128    if b.len() >= 2
129        && ((b[0] == b'"' && b[b.len() - 1] == b'"') || (b[0] == b'\'' && b[b.len() - 1] == b'\''))
130    {
131        &s[1..s.len() - 1]
132    } else {
133        s
134    }
135}

137/// Escape a string for a TOML basic string.
138fn toml_escape(s: &str) -> String {
139    s.replace('\\', "\\\\").replace('"', "\\\"")
140}

142/// `key` is a TOML bare key — letter/`_` then alnum/`_`/`-`.
143fn is_bare_key(key: &str) -> bool {
144    let mut chars = key.chars();
145    match chars.next() {
146        Some(c) if c.is_ascii_alphabetic() || c == '_' => {}
147        _ => return false,
148    }
149    chars.all(|c| c.is_ascii_alphanumeric() || c == '_' || c == '-')
150}

152/// Rewrite a leading `---` YAML frontmatter block as the canonical
153/// `+++` TOML fence — the repair half of [`parse_item_file`]'s
154/// strictness. Agents hand-authoring item files reach for the
155/// ubiquitous `---` + `key: value` convention; this converts the
156/// simple subset — top-level `key: scalar`, `key: [a, b]` inline
157/// lists, `#` comments, blank lines — and returns the source with
158/// a `+++` fence, still *unvalidated*: callers re-parse. `None`
159/// when the file doesn't open with a closed `---` block, the block
160/// exceeds [`REPAIR_MAX`] lines, or a frontmatter line isn't
161/// `key: value` at column 0 (nested maps, `- ` list items — shapes
162/// this doesn't claim to understand).
163pub fn repair_frontmatter(src: &str) -> Option<String> {
164    let mut lines = src.split_inclusive('\n');
165    let first = lines.next()?;
166    if first.trim_end() != "---" {
167        return None;
168    }
169    let fm_start = first.len();
170    let mut off = fm_start;
171    let mut close = None;
172    for line in lines.by_ref().take(REPAIR_MAX) {
173        if line.trim_end() == "---" {
174            close = Some((off, off + line.len()));
175            break;
176        }
177        off += line.len();
178    }
179    let (fm_end, body_start) = close?;
180    let mut toml = String::new();
181    for raw in src[fm_start..fm_end].lines() {
182        // indented lines are nested YAML — out of scope
183        if raw.starts_with(|c: char| c.is_whitespace()) {
184            if !raw.trim().is_empty() {
185                return None;
186            }
187            continue;
188        }
189        let l = raw.trim();
190        if l.is_empty() || l.starts_with('#') {
191            continue;
192        }
193        let (key, val) = l.split_once(':')?;
194        if !is_bare_key(key) {
195            return None;
196        }
197        let val = val.trim();
198        if val.starts_with('[') && val.ends_with(']') {
199            let inner = &val[1..val.len() - 1];
200            let items: Vec<String> = inner
201                .split(',')
202                .map(str::trim)
203                .filter(|s| !s.is_empty())
204                .map(|s| format!("\"{}\"", toml_escape(unquote(s))))
205                .collect();
206            toml.push_str(&format!("{} = [{}]\n", key, items.join(", ")));
207        } else {
208            let v = unquote(val);
209            // a bare u32 (`estimate: 3`) — everything else is a
210            // TOML string, which every stringly-typed frontmatter
211            // field (and `de_opt_created`'s date forms) accepts
212            if !v.is_empty() && v.bytes().all(|b| b.is_ascii_digit()) {
213                toml.push_str(&format!("{key} = {v}\n"));
214            } else {
215                toml.push_str(&format!("{key} = \"{}\"\n", toml_escape(v)));
216            }
217        }
218    }
219    let mut out = String::from("+++\n");
220    out.push_str(&toml);
221    out.push_str("+++\n");
222    out.push_str(&src[body_start..]);
223    Some(out)
224}

226/// Frontmatter DTO — stringly-typed so bad values point at the
227/// frontmatter block, not toml's internals.
228#[derive(Debug, Deserialize)]
229struct FrontmatterIn {
230    id: String,
231    title: String,
232    status: Status,
233    #[serde(default, deserialize_with = "de_opt_created")]
234    created: Option<String>,
235    owner: Option<String>,
236    #[serde(default, deserialize_with = "de_opt_priority")]
237    priority: Option<Priority>,
238    #[serde(default)]
239    labels: Vec<String>,
240    #[serde(default)]
241    blocks: Vec<String>,
242    #[serde(default)]
243    links: Vec<String>,
244    note: Option<String>,
245    superseded_by: Option<String>,
246    supersedes: Option<String>,
247    epic: Option<String>,
248    milestone: Option<String>,
249    #[serde(default, deserialize_with = "de_opt_created")]
250    due: Option<String>,
251    estimate: Option<u32>,
252    cycle: Option<String>,
253}
254
255/// `priority` reads like the line grammar — `low|med|high|crit`,
256/// the spelling emit writes — plus the serde full names
257/// (`medium`, `critical`) older files and hand-edits carry, via
258/// [`Priority`]'s `FromStr` (lenient read, strict canonical
259/// write). Deriving `Priority` would accept full names only.
260fn de_opt_priority<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<Priority>, D::Error> {
261    Option::<String>::deserialize(d)?
262        .map(|s| s.parse().map_err(serde::de::Error::custom))
263        .transpose()
264}
265
266/// `created` accepts both the quoted string emit writes
267/// (`created = "2026-02-01"`) and a bare TOML date
268/// (`created = 2026-02-01`) — hand-edited files use the native
269/// form. Either way the value lands as text for [`date::parse`]
270/// to validate.
271fn de_opt_created<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Option<String>, D::Error> {
272    match Option::<toml::Value>::deserialize(d)? {
273        None => Ok(None),
274        Some(toml::Value::String(s)) => Ok(Some(s)),
275        Some(toml::Value::Datetime(dt)) => Ok(Some(dt.to_string())),
276        Some(v) => Err(serde::de::Error::custom(format!(
277            "expected a `YYYY-MM-DD` date, got {v}"
278        ))),
279    }
280}

321/// Parse an item file's source. `spec` declares the kind the
322/// directory the file was found in belongs to — an id whose
323/// prefix isn't the spec's (`P-002` inside `issues/`) is an
324/// error.
325pub fn parse_item_file(src: &str, spec: &KindSpec, path: &str) -> Result<ItemFile, ParseError> {
326    let src = mdx::normalize(src);
327    let at = |off: usize, len: usize, m: String| ParseError::at(path, src.clone(), off, len, m);
328    let whole = |m: &str| ParseError::whole_file(path, src.clone(), m);
329    let mut lines = src.split_inclusive('\n');
330    let first = lines.next().map(str::trim_end);
331    if first != Some("+++") {
332        // `---` is the YAML fence hand-authored files reach for —
333        // name the repair so the reader isn't stuck on the spec
334        let msg = if first == Some("---") {
335            "expected `+++` TOML frontmatter fence on line 1 — `---` YAML \
336             frontmatter isn't the file spec; `myco check --fix` rewrites it"
337        } else {
338            "expected `+++` TOML frontmatter fence on line 1"
339        };
340        return Err(whole(msg));
341    }
342    // find the closing `+++` line (byte offsets into src)
343    let fm_start = "+++\n".len();
344    let mut fm_end = None;
345    let mut off = fm_start;
346    for line in lines {
347        if line.trim_end() == "+++" {
348            fm_end = Some((off, off + line.len()));
349            break;
350        }
351        off += line.len();
352    }
353    let Some((fm_end, fence_end)) = fm_end else {
354        return Err(whole("unclosed `+++` frontmatter fence"));
355    };
356    let fm_src = &src[fm_start..fm_end];
357    let fm: FrontmatterIn = toml::from_str(fm_src).map_err(|e| {
358        let (o, l) = e
359            .span()
360            .map(|s| (fm_start + s.start, s.end.saturating_sub(s.start)))
361            .unwrap_or((fm_start, fm_end - fm_start));
362        at(o, l, e.to_string())
363    })?;
364    // DTO → Item; value errors point at the whole frontmatter block.
365    let fm_err = |m: String| at(fm_start, fm_end - fm_start, m);
366    let id: ItemId = fm
367        .id
368        .parse()
369        .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?;
370    if id.kind() != spec.id_kind() {
371        return Err(fm_err(format!(
372            "id {id} doesn't belong in this file (expected a {} id)",
373            spec.name
374        )));
375    }
376    // filename ↔ frontmatter: a canonical filename's number must
377    // agree with the id the file claims — `issue-0003-x.md`
378    // holding `I-009` reserves 3 in the id scan while claiming 9,
379    // which would let the allocator reissue a live id. A name
380    // that doesn't fit the kind's pattern carries no number to
381    // check against (that's a filename finding, not this error).
382    let basename = std::path::Path::new(path)
383        .file_name()
384        .and_then(|n| n.to_str())
385        .unwrap_or(path);
386    if let Some(fname_id) = parse_item_filename(spec, basename) {
387        if fname_id != id {
388            return Err(fm_err(format!(
389                "filename says {fname_id} but frontmatter id is {id}"
390            )));
391        }
392    }
393    let owner = fm
394        .owner
395        .map(|o| Actor::new(&o).map_err(|e| fm_err(e.to_string())))
396        .transpose()?;
397    let created = fm
398        .created
399        .map(|c| date::parse(&c).map_err(|e| fm_err(format!("bad created date — {e}"))))
400        .transpose()?;
401    let mut labels = BTreeSet::new();
402    for l in fm.labels {
403        if !myco_core::is_valid_label(&l) {
404            return Err(fm_err(format!("invalid label {l:?}")));
405        }
406        labels.insert(l);
407    }
408    let mut blocks = BTreeSet::new();
409    for b in fm.blocks {
410        blocks.insert(
411            b.parse::<ItemId>()
412                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
413        );
414    }
415    let mut links = BTreeSet::new();
416    for l in fm.links {
417        links.insert(
418            l.parse::<ItemId>()
419                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?,
420        );
421    }
422    let mut item = Item::new(id, fm.title);
423    item.status = fm.status;
424    item.priority = fm.priority.unwrap_or_default();
425    item.owner = owner;
426    item.labels = labels;
427    item.blocks = blocks;
428    item.links = links;
429    item.created = created;
430    item.note = fm.note;
431    item.superseded_by = fm
432        .superseded_by
433        .map(|s| {
434            s.parse::<ItemId>()
435                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
436        })
437        .transpose()?;
438    item.supersedes = fm
439        .supersedes
440        .map(|s| {
441            s.parse::<ItemId>()
442                .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))
443        })
444        .transpose()?;
445    for (key, name) in [
446        ("epic", &fm.epic),
447        ("milestone", &fm.milestone),
448        ("cycle", &fm.cycle),
449    ] {
450        if let Some(name) = name {
451            if !myco_core::is_valid_group_name(name) {
452                return Err(fm_err(format!("invalid {key} name {name:?}")));
453            }
454        }
455    }
456    item.epic = fm.epic;
457    item.milestone = fm.milestone;
458    item.due = fm
459        .due
460        .map(|d| date::parse(&d).map_err(|e| fm_err(format!("bad due date — {e}"))))
461        .transpose()?;
462    if let Some(e) = fm.estimate {
463        if e > myco_core::MAX_ESTIMATE {
464            return Err(fm_err(format!(
465                "estimate {e} exceeds max {}",
466                myco_core::MAX_ESTIMATE
467            )));
468        }
469        item.estimate = Some(e);
470    }
471    item.cycle = fm.cycle;
472    if item.blocks.contains(&id)
473        || item.links.contains(&id)
474        || item.supersedes == Some(id)
475        || item.superseded_by == Some(id)
476    {
477        return Err(fm_err(format!(
478            "{id} can't block, link, or supersede itself"
479        )));
480    }
481    // body: everything after the closing fence, canonicalized —
482    // leading blank lines go, the first content line's own
483    // indentation survives (an indented code block stays a code
484    // block)
485    let file = ItemFile {
486        meta: item,
487        body: myco_core::canonical_body(&src[fence_end..]).to_string(),
488    };
489    // parse-safe: the body must be markdown the parser accepts
490    if !file.body.is_empty() {
491        mdx::parse(&file.body).map_err(|m| at(fence_end, src.len() - fence_end, m))?;
492    }
493    Ok(file)
494}

496/// Canonical item file — `+++\n<toml>+++\n\n<body>\n`. The model
497/// is validated first ([`line::check_item`] — the same validity
498/// source as the line grammar) so emit can't write a file parse
499/// would reject; children are an error, never silently dropped.
500pub fn emit_item_file(file: &ItemFile) -> Result<String, EmitError> {
501    line::check_item(&file.meta, "item file")?;
502    if !file.meta.children.is_empty() {
503        return Err(EmitError::new(
504            "item file",
505            format!(
506                "{} carries children — sub-items only exist in TODO.md/BACKLOG.md",
507                file.meta.id
508            ),
509        ));
510    }
511    let fm = FrontmatterOut {
512        id: file.meta.id.to_string(),
513        title: &file.meta.title,
514        status: file.meta.status,
515        created: file.meta.created.map(date::render),
516        owner: file.meta.owner.as_ref().map(ToString::to_string),
517        priority: file.meta.priority.token(),
518        labels: file.meta.labels.iter().map(String::as_str).collect(),
519        blocks: file.meta.blocks.iter().map(ItemId::to_string).collect(),
520        links: file.meta.links.iter().map(ItemId::to_string).collect(),
521        note: file.meta.note.as_deref(),
522        superseded_by: file.meta.superseded_by.map(|i| i.to_string()),
523        supersedes: file.meta.supersedes.map(|i| i.to_string()),
524        epic: file.meta.epic.as_deref(),
525        milestone: file.meta.milestone.as_deref(),
526        due: file.meta.due.map(date::render),
527        estimate: file.meta.estimate,
528        cycle: file.meta.cycle.as_deref(),
529    };
530    let toml = toml::to_string(&fm).map_err(|e| EmitError::new("item file", e.to_string()))?;
531    if file.body != myco_core::canonical_body(&file.body) {
532        // a body parse can't reproduce byte-for-byte would emit
533        // then read back as a different model — non-canonical
534        // bodies are the caller's bug, not content to reshape
535        return Err(EmitError::new(
536            "item file",
537            "body isn't canonical — leading blank lines or trailing whitespace",
538        ));
539    }
540    let mut out = String::from("+++\n");
541    out.push_str(&toml);
542    out.push_str("+++\n");
543    if !file.body.is_empty() {
544        out.push('\n');
545        out.push_str(&file.body);
546        out.push('\n');
547    }
548    Ok(out)
549}
