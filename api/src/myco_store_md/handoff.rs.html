1//! Handoff files — `handoffs/<ID>.md`, one append-only file per
2//! item, uniform across all kinds (tasks can't hold this inline —
3//! done tasks leave `TODO.md` — and uniformity beats special-casing):
4//!
5//! ```text
6//! +++
7//! item = "T-001"
8//! +++
9//!
10//! # Handoffs — T-001
11//!
12//! ## 2026-10-29 · devin · session:cli:abc123
13//! Done: store landed. Next: wire verbs. Open: archive semantics.
14//! ```
15//!
16//! A `##` heading is `<YYYY-MM-DD> · <actor>[ · session:<id>]`;
17//! everything up to the next `##` is that entry's freeform note,
18//! verbatim. Files are append-only — `myco handoff` never rewrites
19//! an entry; `myco check --fix` moves orphaned files (their item
20//! was removed) to `handoffs/archive/`, never rewrites them either.
21

57/// Parse `handoffs/<ID>.md` — returns the `item` id (frontmatter
58/// and the `# Handoffs —` title must agree) and the entries in
59/// file order. Strict: a `##` heading that doesn't fit the entry
60/// grammar is an error with a span.
61pub fn parse_handoff_file(
62    src: &str,
63    path: &str,
64) -> Result<(ItemId, Vec<HandoffEntry>), ParseError> {
65    let src = mdx::normalize(src);
66    let at = |off: usize, len: usize, m: String| ParseError::at(path, src.clone(), off, len, m);
67    let whole = |m: String| ParseError::whole_file(path, src.clone(), m);
68    let mut lines = src.split_inclusive('\n');
69    if lines.next().map(str::trim_end) != Some("+++") {
70        return Err(whole(
71            "expected `+++` TOML frontmatter fence on line 1".into(),
72        ));
73    }
74    // find the closing `+++` line (byte offsets into src)
75    let fm_start = "+++\n".len();
76    let mut fm_end = None;
77    let mut off = fm_start;
78    for line in lines {
79        if line.trim_end() == "+++" {
80            fm_end = Some((off, off + line.len()));
81            break;
82        }
83        off += line.len();
84    }
85    let Some((fm_end, fence_end)) = fm_end else {
86        return Err(whole("unclosed `+++` frontmatter fence".into()));
87    };
88    let fm: FrontmatterIn = toml::from_str(&src[fm_start..fm_end]).map_err(|e| {
89        let (o, l) = e
90            .span()
91            .map(|s| (fm_start + s.start, s.end.saturating_sub(s.start)))
92            .unwrap_or((fm_start, fm_end - fm_start));
93        at(o, l, e.to_string())
94    })?;
95    let fm_err = |m: String| at(fm_start, fm_end - fm_start, m);
96    let item: ItemId = fm
97        .item
98        .parse()
99        .map_err(|e: myco_core::ParseItemIdError| fm_err(e.to_string()))?;
100
101    // body: mdast gives block structure + spans; offsets are into
102    // `body` (i.e. relative to `fence_end` in `src`). Parsed nodes
103    // always carry positions — a missing one degrades to the
104    // whole-file span.
105    let body = &src[fence_end..];
106    let err_at = |node: &Node, m: String| {
107        let (o, l) = node.position().map_or((0, body.len()), |pos| {
108            (
109                pos.start.offset,
110                pos.end.offset.saturating_sub(pos.start.offset),
111            )
112        });
113        at(fence_end + o, l, m)
114    };
115    let ast = mdx::parse(body).map_err(|m| at(fence_end, src.len() - fence_end, m))?;
116    let mut blocks = ast.children().into_iter().flatten();
117    let title_msg = "expected `# Handoffs — <ID>` title".to_string();
118    let Some(node @ Node::Heading(h1)) = blocks.next() else {
119        return Err(whole(title_msg.clone()));
120    };
121    if h1.depth != 1 {
122        return Err(err_at(node, title_msg));
123    }
124    let text = mdx::single_text(node).unwrap_or_default();
125    let title_id = parse_title(text)
126        .and_then(|s| s.parse::<ItemId>().ok())
127        .ok_or_else(|| err_at(node, title_msg.clone()))?;
128    if title_id != item {
129        return Err(err_at(
130            node,
131            format!("title id {title_id} doesn't match item = \"{item}\""),
132        ));
133    }
134
135    // entries: each h2 starts one; the note is the raw markdown
136    // between its heading and the next h2 (verbatim)
137    let mut entries: Vec<HandoffEntry> = Vec::new();
138    let mut bounds: Vec<(usize, usize)> = Vec::new(); // (h2 start, h2 end) in body
139    for block in blocks {
140        match block {
141            Node::Heading(h) if h.depth == 2 => {
142                let (start, end) = block
143                    .position()
144                    .map(|p| (p.start.offset, p.end.offset))
145                    .unwrap_or((0, 0));
146                let text = mdx::single_text(block)
147                    .ok_or_else(|| err_at(block, "entry heading must be plain text".to_string()))?;
148                let mut e = parse_entry_heading(text).map_err(|m| err_at(block, m))?;
149                e.note = String::new();
150                bounds.push((start, end));
151                entries.push(e);
152            }
153            other => {
154                if entries.is_empty() {
155                    return Err(err_at(
156                        other,
157                        "content before the first `##` entry heading".to_string(),
158                    ));
159                }
160                // inside the last entry's note — the raw range covers it
161            }
162        }
163    }
164    for (i, e) in entries.iter_mut().enumerate() {
165        let start = bounds[i].1;
166        let end = bounds.get(i + 1).map_or(body.len(), |b| b.0);
167        // canonical form — leading blank lines go, but a note that
168        // opens with an indented code block keeps its indent
169        e.note = myco_core::canonical_body(&body[start..end]).to_string();
170    }
171    Ok((item, entries))
172}

174/// `# Handoffs — T-001` / `Handoffs - T-1` → the id text (`None`
175/// when the title isn't in that shape).
176fn parse_title(h1: &str) -> Option<&str> {
177    let rest = h1.strip_prefix("Handoffs")?.trim_start();
178    let rest = rest.strip_prefix('—').or_else(|| rest.strip_prefix('-'))?;
179    Some(rest.trim())
180}

182/// `<YYYY-MM-DD> · <actor>[ · session:<id>]` — the `##` heading
183/// grammar. Field order is fixed; anything else is malformed.
184fn parse_entry_heading(text: &str) -> Result<HandoffEntry, String> {
185    let bad =
186        || format!("entry heading must be `<date> · <actor>[ · session:<id>]` — got {text:?}");
187    let parts: Vec<&str> = text.split(" · ").collect();
188    if parts.len() < 2 || parts.len() > 3 {
189        return Err(bad());
190    }
191    let date = date::parse(parts[0])
192        .map_err(|_| format!("bad entry date {:?} — expected YYYY-MM-DD", parts[0]))?;
193    let actor = Actor::new(parts[1]).map_err(|e| e.to_string())?;
194    let session = match parts.get(2) {
195        None => None,
196        Some(p) => {
197            let s = p
198                .strip_prefix("session:")
199                .filter(|s| is_valid_session(s))
200                .ok_or_else(|| format!("bad session field {p:?} — expected session:<id>"))?;
201            Some(s.to_string())
202        }
203    };
204    Ok(HandoffEntry {
205        date,
206        actor,
207        session,
208        note: String::new(),
209    })
210}

212/// One `## <date> · <actor>[ · session:<id>]` block + note —
213/// `append_handoff` writes this verbatim. The note may not contain
214/// a `##` heading (it would read back as the next entry).
215pub fn emit_entry(entry: &HandoffEntry) -> Result<String, EmitError> {
216    let err = |m: &str| EmitError::new("handoff entry", m);
217    if let Some(s) = &entry.session {
218        if !is_valid_session(s) {
219            return Err(err("session id must match `[A-Za-z0-9._:-]+`"));
220        }
221    }
222    // markdown-rs is effectively infallible for CommonMark — an
223    // unparseable note can't hide a `##` heading, so emit proceeds
224    let ast = mdx::parse(&entry.note).unwrap_or(Node::Root(Root {
225        children: Vec::new(),
226        position: None,
227    }));
228    if ast
229        .children()
230        .into_iter()
231        .flatten()
232        .any(|n| matches!(n, Node::Heading(h) if h.depth == 2))
233    {
234        return Err(err("note can't contain a `##` heading"));
235    }
236    let mut head = format!("{} · {}", date::render(entry.date), entry.actor);
237    if let Some(s) = &entry.session {
238        head.push_str(&format!(" · session:{s}"));
239    }
240    // `_`/`*` are legal actor chars — escape inline markup or the
241    // heading wouldn't read back as a single text run (heading
242    // context: a trailing ` #`-run would be an ATX closer)
243    let head = mdx::heading(&head);
244    if entry.note != myco_core::canonical_body(&entry.note) {
245        // a note parse can't reproduce byte-for-byte would come
246        // back as a different entry — non-canonical notes are the
247        // caller's bug, not content to reshape
248        return Err(err(
249            "note isn't canonical — leading blank lines or trailing whitespace",
250        ));
251    }
252    if entry.note.is_empty() {
253        return Ok(format!("## {head}\n"));
254    }
255    Ok(format!("## {head}\n{}\n", entry.note))
256}

258/// Canonical `handoffs/<ID>.md` — header + entries in file order.
259pub fn emit_handoff_file(id: &ItemId, entries: &[HandoffEntry]) -> Result<String, EmitError> {
260    let fm = FrontmatterOut {
261        item: id.to_string(),
262    };
263    let toml = toml::to_string(&fm).map_err(|e| EmitError::new("handoff file", e.to_string()))?;
264    let mut out = String::from("+++\n");
265    out.push_str(&toml);
266    out.push_str("+++\n\n# Handoffs — ");
267    out.push_str(&id.to_string());
268    out.push('\n');
269    for e in entries {
270        out.push('\n');
271        out.push_str(&emit_entry(e)?);
272    }
273    Ok(out)
274}

358    #[test]
359    fn emit_entry_variants() {
360        assert_eq!(
361            emit_entry(&entry()).unwrap(),
362            "## 2026-10-29 · devin · session:cli:abc123\nDone: x. Next: y.\n"
363        );
364        let mut bare = entry();
365        bare.session = None;
366        bare.note = String::new();
367        assert_eq!(emit_entry(&bare).unwrap(), "## 2026-10-29 · devin\n");
368        // bad session + h2 in the note are emit errors — the
369        // charset is whatever survives the `##` heading byte-for-
370        // byte: no `·` (field sep), no whitespace, no `#` (ATX
371        // closing run), no `\r` (normalize → `\n` splits the line)
372        for bad in ["a·b", "two\nlines", "a b", "x #", "a\rb"] {
373            let mut e = entry();
374            e.session = Some(bad.into());
375            assert!(emit_entry(&e).is_err(), "session {bad:?}");
376        }
377        let mut e = entry();
378        e.note = "text\n\n## sneaky\n".into();
379        assert!(emit_entry(&e).is_err());
380        // `#`/list content in the note is fine
381        let mut e = entry();
382        e.note = "# ok heading\n\n- item\n\n### deeper ok".into();
383        assert!(emit_entry(&e).is_ok());
384    }

413    #[test]
414    fn parse_errors() {
415        for (src, why) in [
416            ("no fence\n", "no +++"),
417            ("+++\nitem = \"T-001\"\n", "unclosed"),
418            ("+++\n+++\n\n# Handoffs — T-001\n", "missing item key"),
419            ("+++\nitem = \"X-001\"\n+++\n\n# Handoffs — T-001\n", "bad id"),
420            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-002\n", "title mismatch"),
421            ("+++\nitem = \"T-001\"\n+++\n\n## 2026-01-01 · devin\nnote\n", "no title"),
422            ("+++\nitem = \"T-001\"\n+++\n", "empty body"),
423            ("+++\nitem = \"T-001\"\n+++\n\njust a paragraph\n", "non-heading block"),
424            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\nstray\n", "content before entry"),
425            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## not a date · devin\nn\n", "bad date"),
426            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-01-01\ntext\n", "one field"),
427            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-01-01 · devin · bogus:x\nn\n", "bad session"),
428            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-01-01 · a b c\nn\n", "bad actor"),
429            ("+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-01-01 · devin · session:x · extra\nn\n", "too many fields"),
430        ] {
431            assert!(
432                parse_handoff_file(src, "handoffs/T-001.md").is_err(),
433                "{why}: {src:?}"
434            );
435        }
436        // deeper headings inside a note are fine
437        let src = "+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-01-01 · devin\n### sub\nnote\n";
438        let (_, entries) = parse_handoff_file(src, "handoffs/T-001.md").unwrap();
439        assert_eq!(entries[0].note, "### sub\nnote");
440        // a `##` inside a fenced code block is not an entry
441        let src = "+++\nitem = \"T-001\"\n+++\n\n# Handoffs — T-001\n\n## 2026-01-01 · devin\n```\n## not an entry\n```\n";
442        let (_, entries) = parse_handoff_file(src, "handoffs/T-001.md").unwrap();
443        assert_eq!(entries.len(), 1);
444        assert_eq!(entries[0].note, "```\n## not an entry\n```");
445    }
