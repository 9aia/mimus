4use markdown::mdast::{Node, Paragraph, Text};

7/// `ParseOptions`: CommonMark + GFM task-list items only — autolinks,
8/// tables etc. stay off so item lines stay plain.
9pub(crate) fn parse_options() -> ParseOptions {
10    ParseOptions {
11        constructs: Constructs {
12            gfm_task_list_item: true,
13            ..Constructs::default()
14        },
15        ..ParseOptions::default()
16    }
17}

19/// `to_mdast` with myco's options; the error is flattened to text.
20pub(crate) fn parse(src: &str) -> Result<Node, String> {
21    to_mdast(src, &parse_options()).map_err(|m| m.to_string())
22}
23
24/// Emit options — canonical style: `-` bullets, `*` emphasis, ATX
25/// headings, fenced code with backticks.
26pub(crate) fn emit_options() -> mdast_util_to_markdown::Options {
27    mdast_util_to_markdown::Options {
28        bullet: '-',
29        ..Default::default()
30    }
31}

33/// Serialize one inline string — escapes markdown specials (`*`,
34/// `[`, leading `#`, …) so hand-emitted lines re-parse to the same
35/// text. Uses `mdast_util_to_markdown` but verifies the result by
36/// re-parsing: 0.0.2's escaper corrupts sequences like `· #_x`
37/// (byte/char index drift — `_` is eaten and the next char is
38/// entity-encoded). On divergence it falls back to [`escape`].
39pub(crate) fn inline(text: &str) -> String {
40    let node = Node::Paragraph(Paragraph {
41        children: vec![Node::Text(Text {
42            value: text.to_string(),
43            position: None,
44        })],
45        position: None,
46    });
47    let via_lib = mdast_util_to_markdown::to_markdown_with_options(&node, &emit_options())
48        .unwrap_or_default();
49    let via_lib = via_lib.trim_end_matches('\n');
50    let verified = parse(via_lib)
51        .ok()
52        .as_ref()
53        .and_then(Node::children)
54        .and_then(|c| c.first())
55        .and_then(single_text)
56        == Some(text);
57    if verified {
58        via_lib.to_string()
59    } else {
60        escape(text)
61    }
62}

64/// Serialize inline text for an ATX heading (`# TODO — <repo>`,
65/// `## <section>`, `## <date> · <actor>`). [`inline`]'s output is
66/// verified in *paragraph* context, which can't see the
67/// heading-only hazard: a trailing ` #`+ run is the ATX closing
68/// sequence — `## x ###` reads back as `x`, not `x ###`. Verify
69/// against a real heading; on divergence each `#` of the closing
70/// run is `\`-escaped (an escaped hash is content, never a
71/// closer).
72pub(crate) fn heading(text: &str) -> String {
73    let via_lib = inline(text);
74    if reads_back_in_heading(&via_lib, text) {
75        return via_lib;
76    }
77    let escaped = escape_trailing_closing(&via_lib);
78    if reads_back_in_heading(&escaped, text) {
79        return escaped;
80    }
81    escape_all(text)
82}
83
84/// `## {out}` re-parses to a heading whose single text child is
85/// `text` — [`inline`]'s verification run in the context callers
86/// actually emit into.
87fn reads_back_in_heading(out: &str, text: &str) -> bool {
88    parse(&format!("## {out}\n"))
89        .ok()
90        .as_ref()
91        .and_then(Node::children)
92        .and_then(|c| c.first())
93        .and_then(single_text)
94        == Some(text)
95}

97/// `\`-escape each `#` of a trailing hash run that's whitespace-
98/// separated (or is the whole string — the `## ` prefix supplies
99/// the separating space either way).
100fn escape_trailing_closing(s: &str) -> String {
101    let body = s.trim_end_matches('#');
102    let run = s.len() - body.len();
103    if run == 0 || (!body.is_empty() && !body.chars().last().is_some_and(|c| c == ' ' || c == '\t'))
104    {
105        // `x#`, `x\##` — no whitespace before the run, so it's
106        // content, not a closing sequence
107        return s.to_string();
108    }
109    let mut out = String::with_capacity(s.len() + run);
110    out.push_str(body);
111    for _ in 0..run {
112        out.push_str("\\#");
113    }
114    out
115}

117/// Last-resort escaper — `\` before every ASCII punctuation
118/// (a backslash escape re-parses as the literal char anywhere,
119/// including inside a heading).
120fn escape_all(text: &str) -> String {
121    let mut out = String::with_capacity(text.len() + 8);
122    for c in text.chars() {
123        if c.is_ascii_punctuation() {
124            out.push('\\');
125        }
126        out.push(c);
127    }
128    out
129}

131/// Minimal CommonMark escaper — `\` before every ASCII punctuation
132/// that can open an inline construct anywhere (always-safe: `\x`
133/// for ASCII punctuation re-parses to literal `x`), plus the
134/// block-marker chars that only matter at text start (`#` heading,
135/// `+`/`-`/`.` list, `>` quote, `=` setext). `·` is *not* escaped —
136/// it's banned from titles/values by the grammar.
137pub(crate) fn escape(text: &str) -> String {
138    const ANYWHERE: &[char] = &['\\', '`', '*', '_', '[', ']', '<', '&', '!'];
139    const LINE_START: &[char] = &['#', '+', '-', '.', '>', '='];
140    let mut out = String::with_capacity(text.len() + 8);
141    for (i, c) in text.chars().enumerate() {
142        if ANYWHERE.contains(&c) || (i == 0 && LINE_START.contains(&c)) {
143            out.push('\\');
144        }
145        out.push(c);
146    }
147    out
148}

150/// The text of a node that must be exactly one `Text` child —
151/// heading names and item lines are plain by spec.
152pub(crate) fn single_text(node: &Node) -> Option<&str> {
153    let children = node.children()?;
154    if children.len() != 1 {
155        return None;
156    }
157    match &children[0] {
158        Node::Text(t) => Some(t.value.as_str()),
159        _ => None,
160    }
161}

163/// Normalize CRLF → LF up front so spans match the stored source.
164pub(crate) fn normalize(src: &str) -> String {
165    if src.contains('\r') {
166        src.replace("\r\n", "\n").replace('\r', "\n")
167    } else {
168        src.to_string()
169    }
170}

172/// Byte ranges in `src` covered by code — fenced/indented blocks
173/// and `inline` spans — so prose scans (`myco check`'s
174/// `dangling-ref`) can whitelist them. Sorted, non-overlapping.
175/// Empty when `src` doesn't parse (callers scan unwhitelisted —
176/// a stale ref inside a broken fence is still worth flagging).
177pub fn code_spans(src: &str) -> Vec<(usize, usize)> {
178    // explicit-stack walk — a deeply nested doc can't overflow
179    // the call stack
180    let mut out = Vec::new();
181    let Ok(ast) = parse(src) else {
182        return out;
183    };
184    let mut stack = vec![&ast];
185    while let Some(node) = stack.pop() {
186        if matches!(node, Node::Code(_) | Node::InlineCode(_)) {
187            if let Some(p) = node.position() {
188                out.push((p.start.offset, p.end.offset));
189            }
190            continue; // code nodes can't nest scannable prose
191        }
192        if let Some(children) = node.children() {
193            // reversed so the pop order is document order —
194            // the span list stays sorted
195            stack.extend(children.iter().rev());
196        }
197    }
198    out
199}

207    #[test]
208    fn inline_escapes_specials() {
209        assert_eq!(inline("plain"), "plain");
210        assert_eq!(inline("a *b* c"), "a \\*b\\* c");
211        assert_eq!(inline("#hash"), "\\#hash");
212        assert_eq!(
213            inline("T-1 t · #label · status:open"),
214            "T-1 t · #label · status:open"
215        );
216        assert_eq!(inline(""), "");
217    }
218
219    #[test]
220    fn inline_falls_back_on_lib_corruption() {
221        // mdast_util_to_markdown 0.0.2 mangles `· #_x` — verify our
222        // verifier catches it and the fallback round-trips.
223        let out = inline("T-001 · #_a-a");
224        assert_eq!(out, "T-001 · #\\_a-a");
225        let ast = parse(&format!("- {out}\n")).unwrap();
226        let list = &ast.children().unwrap()[0];
227        assert!(matches!(list, Node::List(_)));
228        let li = &list.children().unwrap()[0];
229        assert!(matches!(li, Node::ListItem(_)));
230        let para = &li.children().unwrap()[0];
231        assert_eq!(single_text(para), Some("T-001 · #_a-a"));
232    }

234    #[test]
235    fn heading_round_trips_atx_closing_shapes() {
236        // `## x ###` is the audit case: a trailing ` #`+ run reads
237        // back as the ATX closing sequence, silently truncating
238        // the name — the paragraph-context check in `inline`
239        // can't see it
240        for s in [
241            "plain", "#hash", "a *b* c", "###", "x #", "x ###", "##", "x \t##",
242        ] {
243            let out = heading(s);
244            assert!(reads_back_in_heading(&out, s), "{s:?} emitted as {out:?}");
245        }
246        // the fix is surgical — escape the run, keep the rest
247        assert_eq!(heading("x ###"), "x \\#\\#\\#");
248        assert_eq!(heading("x #"), "x \\#");
249        // mid-line and non-whitespace-separated hashes stay put
250        assert_eq!(heading("a #b"), "a #b");
251        assert_eq!(heading("x#"), "x#");
252    }

326    #[test]
327    fn code_spans_cover_fences_and_inline() {
328        let src = "see `T-001` here\n\n```\nT-002 in a fence\n```\n\nplain T-003\n";
329        let spans = code_spans(src);
330        // inline code + the fenced block, nothing else
331        assert_eq!(spans.len(), 2, "{spans:?}");
332        let inline = &src[spans[0].0..spans[0].1];
333        assert_eq!(inline, "`T-001`", "{inline}");
334        let fence = &src[spans[1].0..spans[1].1];
335        assert!(fence.contains("T-002 in a fence"), "{fence}");
336        // a hand-built node without position is skipped silently —
337        // and unparseable input yields no spans rather than an error
338        assert!(code_spans("plain text only\n").is_empty());
