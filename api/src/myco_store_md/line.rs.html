1//! The item-line grammar shared by `TODO.md` (`- [ ] T-012 …`) and
2//! `BACKLOG.md` (`- T-034 …`):
3//!
4//! ```text
5//! <id> <title> [· status:<s>] [· pri:<p>] [· owner:<a>]
6//!     [· created:<date>] [· #<label>]… [· blocks:<id>[,<id>]…]
7//!     [· links:<id>[,<id>]…] [· epic:<name>] [· ms:<name>]
8//!     [· due:<date>] [· est:<n>] [· cycle:<name>] [· note:<text>]
9//! ```
10//!
11//! Metadata tokens come after the title, `·`-separated, in the
12//! canonical order above. Titles may not contain `·`. Defaults:
13//! `status` = the doc's default (`open` for TODO, `deferred` for
14//! BACKLOG), `pri` = `med`. Unknown/malformed tokens are errors —
15//! `myco check` should catch typos, not swallow them.
16//!
17//! Items nest: a `-`/`[ ]` list indented under an item is its
18//! `children` (canonical emit = 2 spaces per level; arbitrary
19//! depth up to [`MAX_ITEM_DEPTH`], one level recommended). Child
20//! lines carry the same grammar — full items, not prose.
21

30/// Deepest item nesting the docs accept — the spec says arbitrary
31/// depth, but parse and emit walk the tree recursively, so past
32/// this bound it's an error, not a stack overflow. 64 is far past
33/// any real `TODO.md`/`BACKLOG.md` nesting.
34pub(crate) const MAX_ITEM_DEPTH: usize = 64;
35
36/// Per-document defaults for line parse/emit.
37pub(crate) struct LineSpec {
38    /// Kind ids must have (tasks only — `T-NNN`).
39    pub kind: ItemKind,
40    /// Status when no `status:` token is present.
41    pub default_status: Status,
42    /// Which file we're in — for emit errors.
43    pub file: &'static str,
44}
45
46fn bad(msg: impl Into<String>) -> String {
47    msg.into()
48}

50/// A list item's shape: its `Paragraph` head line plus at most one
51/// nested unordered `List` (the item's children). Anything else —
52/// a second paragraph, an ordered list, a second nested list —
53/// breaks the grammar.
54pub(crate) fn item_parts(
55    li: &markdown::mdast::ListItem,
56) -> Result<(&Node, Option<&markdown::mdast::List>), String> {
57    let mut children = li.children.iter();
58    let Some(head @ Node::Paragraph(_)) = children.next() else {
59        return Err(bad("item must be a single line"));
60    };
61    let mut nested = None;
62    for c in children {
63        match c {
64            Node::List(l) if !l.ordered && nested.is_none() => nested = Some(l),
65            _ => {
66                return Err(bad(
67                    "unexpected block under item — only a nested `-` list is allowed",
68                ))
69            }
70        }
71    }
72    Ok((head, nested))
73}

75/// Parse the text inside a list item (after `- [ ] ` / `- `).
76pub(crate) fn parse_text(text: &str, spec: &LineSpec) -> Result<Item, String> {
77    if text.contains('\n') {
78        return Err(bad("item must be a single line"));
79    }
80    let mut parts = text.split('·');
81    let head = parts.next().unwrap_or_default().trim();
82    let mut head_parts = head.splitn(2, char::is_whitespace);
83    let id_str = head_parts
84        .next()
85        .filter(|s| !s.is_empty())
86        .ok_or_else(|| bad("expected item id (T-NNN)"))?;
87    let id: ItemId = id_str.parse().map_err(|e| format!("{e}"))?;
88    if id.kind() != spec.kind {
89        return Err(bad(format!(
90            "wrong item kind {id} — {file} holds {} items only",
91            spec.kind.prefix(),
92            file = spec.file
93        )));
94    }
95    let title = head_parts.next().unwrap_or("").trim().to_string();
96
97    let mut status = None;
98    let mut priority = None;
99    let mut owner = None;
100    let mut created = None;
101    let mut blocks_seen = false;
102    let mut links_seen = false;
103    let mut epic = None;
104    let mut milestone = None;
105    let mut due = None;
106    let mut estimate = None;
107    let mut cycle = None;
108    let mut note = None;
109    let mut item = Item::new(id, title);
110
111    for raw in parts {
112        let tok = raw.trim();
113        if tok.is_empty() {
114            return Err(bad("empty metadata token — double `·` or trailing `·`"));
115        }
116        if let Some(label) = tok.strip_prefix('#') {
117            if !is_valid_label(label) {
118                return Err(bad(format!("invalid label #{label:?}")));
119            }
120            item.labels.insert(label.to_string());
121            continue;
122        }
123        let (key, value) = tok.split_once(':').ok_or_else(|| {
124            bad(format!(
125                "malformed metadata token {tok:?} — expected key:value or #label"
126            ))
127        })?;
128        let value = value.trim();
129        if value.is_empty() {
130            return Err(bad(format!("empty value in token {tok:?}")));
131        }
132        match key {
133            "status" => {
134                if status.is_some() {
135                    return Err(bad("duplicate status: token"));
136                }
137                status = Some(value.parse::<Status>().map_err(|e| format!("{e}"))?);
138            }
139            "pri" => {
140                if priority.is_some() {
141                    return Err(bad("duplicate pri: token"));
142                }
143                priority = Some(value.parse::<Priority>().map_err(|e| format!("{e}"))?);
144            }
145            "owner" => {
146                if owner.is_some() {
147                    return Err(bad("duplicate owner: token"));
148                }
149                owner = Some(Actor::new(value).map_err(|e| format!("{e}"))?);
150            }
151            "created" => {
152                if created.is_some() {
153                    return Err(bad("duplicate created: token"));
154                }
155                created = Some(date::parse(value).map_err(|e| format!("bad created: date — {e}"))?);
156            }
157            "blocks" => {
158                if blocks_seen {
159                    return Err(bad("duplicate blocks: token"));
160                }
161                blocks_seen = true;
162                for bid in value.split(',') {
163                    let bid = bid.trim();
164                    if bid.is_empty() {
165                        return Err(bad("empty id in blocks: list"));
166                    }
167                    item.blocks
168                        .insert(bid.parse::<ItemId>().map_err(|e| format!("{e}"))?);
169                }
170            }
171            "links" => {
172                if links_seen {
173                    return Err(bad("duplicate links: token"));
174                }
175                links_seen = true;
176                for lid in value.split(',') {
177                    let lid = lid.trim();
178                    if lid.is_empty() {
179                        return Err(bad("empty id in links: list"));
180                    }
181                    item.links
182                        .insert(lid.parse::<ItemId>().map_err(|e| format!("{e}"))?);
183                }
184            }
185            "epic" => {
186                if epic.is_some() {
187                    return Err(bad("duplicate epic: token"));
188                }
189                if !is_valid_group_name(value) {
190                    return Err(bad(format!("invalid epic name {value:?}")));
191                }
192                epic = Some(value.to_string());
193            }
194            "ms" => {
195                if milestone.is_some() {
196                    return Err(bad("duplicate ms: token"));
197                }
198                if !is_valid_group_name(value) {
199                    return Err(bad(format!("invalid milestone name {value:?}")));
200                }
201                milestone = Some(value.to_string());
202            }
203            "due" => {
204                if due.is_some() {
205                    return Err(bad("duplicate due: token"));
206                }
207                due = Some(date::parse(value).map_err(|e| format!("bad due: date — {e}"))?);
208            }
209            "est" => {
210                if estimate.is_some() {
211                    return Err(bad("duplicate est: token"));
212                }
213                let n: u32 = value.parse().map_err(|_| {
214                    bad(format!("bad est: value {value:?} — expected a point count"))
215                })?;
216                if n > MAX_ESTIMATE {
217                    return Err(bad(format!("est:{n} exceeds max {MAX_ESTIMATE}")));
218                }
219                estimate = Some(n);
220            }
221            "cycle" => {
222                if cycle.is_some() {
223                    return Err(bad("duplicate cycle: token"));
224                }
225                if !is_valid_group_name(value) {
226                    return Err(bad(format!("invalid cycle name {value:?}")));
227                }
228                cycle = Some(value.to_string());
229            }
230            "note" => {
231                if note.is_some() {
232                    return Err(bad("duplicate note: token"));
233                }
234                note = Some(value.to_string());
235            }
236            other => {
237                return Err(bad(format!(
238                    "unknown metadata key {other:?} — expected status|pri|owner|created|blocks|links|epic|ms|due|est|cycle|note or #label"
239                )));
240            }
241        }
242    }
243
244    if item.blocks.contains(&id) {
245        return Err(bad(format!("{id} can't block itself")));
246    }
247    if item.links.contains(&id) {
248        return Err(bad(format!("{id} can't link itself")));
249    }
250    item.status = status.unwrap_or(spec.default_status);
251    item.priority = priority.unwrap_or_default();
252    item.owner = owner;
253    item.created = created;
254    item.epic = epic;
255    item.milestone = milestone;
256    item.due = due;
257    item.estimate = estimate;
258    item.cycle = cycle;
259    item.note = note;
260    Ok(item)
261}

263/// Validate what the line grammar can't express. Shared by TODO and
264/// BACKLOG emit — the caller adds status rules on top.
265pub(crate) fn check_item(item: &Item, file: &'static str) -> Result<(), EmitError> {
266    let err = |m: &str| EmitError::new(file, format!("{m} ({})", item.id));
267    if item.title != item.title.trim() {
268        return Err(err("untrimmed title"));
269    }
270    if item.title.contains('·') {
271        return Err(err("title contains `·`"));
272    }
273    if item.title.contains('\n') || item.title.contains('\r') {
274        return Err(err("title contains a newline"));
275    }
276    for label in &item.labels {
277        if !is_valid_label(label) {
278            return Err(err(&format!("invalid label #{label:?}")));
279        }
280    }
281    for (what, name) in [
282        ("epic", item.epic.as_deref()),
283        ("milestone", item.milestone.as_deref()),
284        ("cycle", item.cycle.as_deref()),
285    ] {
286        if let Some(name) = name {
287            if !is_valid_group_name(name) {
288                return Err(err(&format!("invalid {what} name {name:?}")));
289            }
290        }
291    }
292    if let Some(e) = item.estimate {
293        if e > MAX_ESTIMATE {
294            return Err(err(&format!("est:{e} exceeds max {MAX_ESTIMATE}")));
295        }
296    }
297    if let Some(note) = &item.note {
298        if note.is_empty() || note != note.trim() {
299            return Err(err("untrimmed/empty note"));
300        }
301        if note.contains('·') || note.contains('\n') || note.contains('\r') {
302            return Err(err("note contains `·` or a newline"));
303        }
304    }
305    if item.blocks.contains(&item.id)
306        || item.links.contains(&item.id)
307        || item.supersedes == Some(item.id)
308        || item.superseded_by == Some(item.id)
309    {
310        return Err(err(
311            "self-reference — can't block, link, or supersede itself",
312        ));
313    }
314    Ok(())
315}

317/// The line text (`<id> <title> · <tokens>`) — checkbox/marker is
318/// the caller's job.
319pub(crate) fn text(item: &Item, spec: &LineSpec) -> Result<String, EmitError> {
320    check_item(item, spec.file)?;
321    let mut s = item.id.to_string();
322    if !item.title.is_empty() {
323        s.push(' ');
324        s.push_str(&item.title);
325    }
326    let mut tokens: Vec<String> = Vec::new();
327    if item.status != spec.default_status {
328        tokens.push(format!("status:{}", item.status.token()));
329    }
330    if item.priority != Priority::Medium {
331        tokens.push(format!("pri:{}", item.priority.token()));
332    }
333    if let Some(owner) = &item.owner {
334        tokens.push(format!("owner:{owner}"));
335    }
336    if let Some(d) = item.created {
337        tokens.push(format!("created:{}", date::render(d)));
338    }
339    for label in &item.labels {
340        tokens.push(format!("#{label}"));
341    }
342    if !item.blocks.is_empty() {
343        let ids: Vec<String> = item.blocks.iter().map(ItemId::to_string).collect();
344        tokens.push(format!("blocks:{}", ids.join(",")));
345    }
346    if !item.links.is_empty() {
347        let ids: Vec<String> = item.links.iter().map(ItemId::to_string).collect();
348        tokens.push(format!("links:{}", ids.join(",")));
349    }
350    if let Some(epic) = &item.epic {
351        tokens.push(format!("epic:{epic}"));
352    }
353    if let Some(ms) = &item.milestone {
354        tokens.push(format!("ms:{ms}"));
355    }
356    if let Some(d) = item.due {
357        tokens.push(format!("due:{}", date::render(d)));
358    }
359    if let Some(e) = item.estimate {
360        tokens.push(format!("est:{e}"));
361    }
362    if let Some(c) = &item.cycle {
363        tokens.push(format!("cycle:{c}"));
364    }
365    if let Some(note) = &item.note {
366        tokens.push(format!("note:{note}"));
367    }
368    for t in tokens {
369        s.push_str(" · ");
370        s.push_str(&t);
371    }
372    Ok(s)
373}

578        // self-references — an item can't block or supersede
579        // itself, however the model got built (I-021)
580        i.note = None;
581        i.blocks.insert(ItemId::new(ItemKind::Task, 1));
582        assert!(text(&i, &TODO).is_err());
583        i.blocks.clear();
584        i.supersedes = Some(ItemId::new(ItemKind::Task, 1));
585        assert!(text(&i, &TODO).is_err());
586        i.supersedes = None;
587        i.superseded_by = Some(ItemId::new(ItemKind::Task, 1));
588        assert!(text(&i, &TODO).is_err());
589    }
