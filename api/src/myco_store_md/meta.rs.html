1//! `meta.toml` — the one piece of state that isn't derivable:
2//! repo name + persistent per-kind `[next]` counters (done tasks
3//! leave `TODO.md`; ids are never reused).
4
5use myco_core::{ItemKind, ProjectMeta};
6
7use crate::error::{EmitError, ParseError};
8
9const FILE: &str = "meta.toml";
10
11/// Parse `meta.toml`. TOML errors keep their span.
12pub fn parse_meta(src: &str) -> Result<ProjectMeta, ParseError> {
13    toml::from_str::<ProjectMeta>(src).map_err(|e| {
14        let (offset, len) = e
15            .span()
16            .map(|s| (s.start, s.end.saturating_sub(s.start)))
17            .unwrap_or((0, 1));
18        ParseError::at(FILE, src.to_string(), offset, len, e.to_string())
19    })
20}
21
22/// Canonical `meta.toml`.
23///
24/// ```toml
25/// name = "myco"
26/// [next]
27/// t = 13
28/// i = 4
29/// p = 3
30/// adr = 5
31/// pl = 2
32/// ```
33pub fn emit_meta(meta: &ProjectMeta) -> Result<String, EmitError> {
34    if meta.name.trim().is_empty() {
35        return Err(EmitError::new(FILE, "empty repo name"));
36    }
37    // the name fills `# TODO — <name>` — it must be a single
38    // trimmed line; an interior newline would smuggle extra
39    // content into TODO.md as a paragraph
40    if meta.name.trim() != meta.name || meta.name.contains('\n') || meta.name.contains('\r') {
41        return Err(EmitError::new(
42            FILE,
43            "untrimmed repo name or newline — it fills `# TODO — <name>`",
44        ));
45    }
46    // a 0 counter would write `t = 0` — a file parse_meta then
47    // refuses (ids are 1-based; T-000 can't be named)
48    for kind in ItemKind::ALL {
49        if meta.next.get(kind) == 0 {
50            return Err(EmitError::new(
51                FILE,
52                format!("{} counter is 0 — id counters start at 1", kind.prefix()),
53            ));
54        }
55    }
56    toml::to_string(meta).map_err(|e| EmitError::new(FILE, e.to_string()))
57}

59#[cfg(test)]
60#[allow(clippy::unwrap_used)]
61mod tests {
62    use super::*;
63    use myco_core::ItemKind;
64
65    #[test]
66    fn roundtrip() {
67        let src = "name = \"myco\"\n\n[next]\nt = 13\ni = 4\np = 3\nadr = 5\npl = 2\n";
68        let meta = parse_meta(src).unwrap();
69        assert_eq!(meta.name, "myco");
70        assert_eq!(meta.next.get(ItemKind::Task), 13);
71        assert_eq!(meta.next.get(ItemKind::Adr), 5);
72        assert_eq!(emit_meta(&meta).unwrap(), src);
73    }
74
75    #[test]
76    fn partial_next_defaults_to_one() {
77        let meta = parse_meta("name = \"x\"\n\n[next]\nt = 9\n").unwrap();
78        assert_eq!(meta.next.get(ItemKind::Task), 9);
79        assert_eq!(meta.next.get(ItemKind::Issue), 1);
80        // no [next] at all
81        let meta = parse_meta("name = \"x\"\n").unwrap();
82        assert_eq!(meta.next.get(ItemKind::Task), 1);
83    }
84
85    #[test]
86    fn parse_errors() {
87        assert!(parse_meta("").is_err());
88        assert!(parse_meta("name = 3\n").is_err());
89        assert!(parse_meta("name = \"x\"\n[next]\nt = \"z\"\n").is_err());
90        assert!(parse_meta("name = \"x\"\nnext = [").is_err());
91        let e = parse_meta("name = \"x\"\n\nnext = [\n").unwrap_err();
92        assert!(e.line() >= 1);
93    }
94
95    #[test]
96    fn emit_rejects_empty_name() {
97        assert!(emit_meta(&ProjectMeta::new(" ")).is_err());
98        assert!(emit_meta(&ProjectMeta::new("myco")).is_ok());
99    }
100
101    #[test]
102    fn typo_keys_error_instead_of_resetting() {
103        // `[netx]` — a typo'd table must not silently drop the
104        // counters (they'd restart at 1 and reissue done ids)
105        for src in [
106            "name = \"x\"\n\n[netx]\nt = 9\n",
107            "name = \"x\"\nnamme = \"y\"\n",
108        ] {
109            assert!(parse_meta(src).is_err(), "{src:?}");
110        }
111        // unknown *counter* keys ride `extra` (P-008: they may be
112        // user-kind counters — meta parses without the table); a
113        // typo'd builtin lands there too, which check flags as a
114        // counter that shadows nothing declared
115        let m = parse_meta("name = \"x\"\n\n[next]\ntx = 9\n").unwrap();
116        assert_eq!(m.next.extra.get("tx"), Some(&9));
117    }
118
119    #[test]
120    fn zero_counter_rejected_at_both_ends() {
121        assert!(parse_meta("name = \"x\"\n\n[next]\nt = 0\n").is_err());
122        let mut meta = ProjectMeta::new("x");
123        meta.next.t = 0;
124        assert!(emit_meta(&meta).is_err());
125    }
