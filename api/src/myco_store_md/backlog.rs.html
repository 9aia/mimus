1//! `BACKLOG.md` — `# Backlog` plus optional `##` sections of
2//! `- T-NNN …` items. No checkboxes: backlog items are `Deferred`
3//! (a `status:` token must say `deferred` if present). Items nest
4//! like TODO.md — a `-` list indented under an item is its
5//! `children` (2 spaces per level, canonical).
6
7use markdown::mdast::Node;
8use myco_core::{BacklogDoc, Item, ItemKind, Section, Status};

10use crate::error::{EmitError, ParseError};
11use crate::line::{self, LineSpec};
12use crate::mdx;

14const FILE: &str = "BACKLOG.md";
15const SPEC: LineSpec = LineSpec {
16    kind: ItemKind::Task,
17    default_status: Status::Deferred,
18    file: FILE,
19};

21fn err(src: &str, node: &Node, msg: impl Into<String>) -> ParseError {
22    match node.position() {
23        Some(pos) => ParseError::at_position(FILE, src.to_string(), pos, msg),
24        None => ParseError::whole_file(FILE, src.to_string(), msg),
25    }
26}

28/// Parse `BACKLOG.md` source into a [`BacklogDoc`].
29pub fn parse_backlog(src: &str) -> Result<BacklogDoc, ParseError> {
30    let src = mdx::normalize(src);
31    let ast = mdx::parse(&src).map_err(|m| ParseError::whole_file(FILE, src.clone(), m))?;
32    // to_mdast always yields Root; children() keeps it non-fatal
33    let mut blocks = ast.children().into_iter().flatten();
34    let Some(node @ Node::Heading(h1)) = blocks.next() else {
35        return Err(ParseError::whole_file(
36            FILE,
37            src,
38            "expected `# Backlog` title",
39        ));
40    };
41    if h1.depth != 1 || mdx::single_text(node).map(str::trim) != Some("Backlog") {
42        return Err(err(&src, node, "expected `# Backlog` title"));
43    }
44    let mut doc = BacklogDoc::default();
45    for block in blocks {
46        match block {
47            Node::Heading(h) if h.depth == 2 => {
48                let name = mdx::single_text(block)
49                    .ok_or_else(|| err(&src, block, "section heading must be plain text"))?;
50                doc.sections.push(Section::named(name.trim()));
51            }
52            Node::List(list) if !list.ordered => {
53                if doc.sections.is_empty() {
54                    doc.sections.push(Section::unsectioned());
55                }
56                if let Some(section) = doc.sections.last_mut() {
57                    parse_list(&src, list, section)?;
58                }
59            }
60            other => {
61                return Err(err(
62                    &src,
63                    other,
64                    "unexpected block — BACKLOG.md holds `##` sections and `-` items only",
65                ));
66            }
67        }
68    }
69    Ok(doc)
70}

72fn parse_list(
73    src: &str,
74    list: &markdown::mdast::List,
75    section: &mut Section,
76) -> Result<(), ParseError> {
77    for item in &list.children {
78        // list children are ListItem by construction; skip anything else
79        if let Node::ListItem(li) = item {
80            section.items.push(parse_li(src, item, li, 0)?);
81        }
82    }
83    Ok(())
84}

86/// One `- T-NNN …` list item → an [`Item`]; a nested `-` list
87/// under it parses recursively into `children` (bounded by
88/// [`line::MAX_ITEM_DEPTH`]).
89fn parse_li(
90    src: &str,
91    node: &Node,
92    li: &markdown::mdast::ListItem,
93    depth: usize,
94) -> Result<Item, ParseError> {
95    if depth > line::MAX_ITEM_DEPTH {
96        return Err(err(src, node, "items nested too deep"));
97    }
98    if li.checked.is_some() {
99        return Err(err(
100            src,
101            node,
102            "backlog items have no checkbox — they're deferred",
103        ));
104    }
105    let (head, nested) = line::item_parts(li).map_err(|m| err(src, node, m))?;
106    let text =
107        mdx::single_text(head).ok_or_else(|| err(src, node, "item line must be plain text"))?;
108    let mut parsed = line::parse_text(text, &SPEC).map_err(|m| err(src, node, m))?;
109    if parsed.status != Status::Deferred {
110        return Err(err(
111            src,
112            node,
113            "backlog items are deferred — no other status is allowed",
114        ));
115    }
116    if let Some(list) = nested {
117        for child in &list.children {
118            if let Node::ListItem(cli) = child {
119                parsed.children.push(parse_li(src, child, cli, depth + 1)?);
120            }
121        }
122    }
123    Ok(parsed)
124}

126/// Canonical `BACKLOG.md` — same layout as [`crate::emit_todo`]
127/// minus the `[ ]` checkboxes. Fails on non-deferred items and the
128/// same line-grammar limits as TODO.md.
129pub fn emit_backlog(doc: &BacklogDoc) -> Result<String, EmitError> {
130    let err = |m: &str| EmitError::new(FILE, m);
131    let mut out = String::from("# Backlog\n");
132    for (i, section) in doc.sections.iter().enumerate() {
133        for item in &section.items {
134            check_subtree(item, 0)?;
135        }
136        out.push('\n');
137        match &section.name {
138            Some(name) => {
139                if name.trim() != *name || name.is_empty() {
140                    return Err(err("empty or untrimmed section name"));
141                }
142                if name.contains('\n') || name.contains('\r') {
143                    return Err(err("section name contains a newline"));
144                }
145                out.push_str("## ");
146                out.push_str(&mdx::heading(name));
147                out.push('\n');
148                if !section.items.is_empty() {
149                    out.push('\n');
150                }
151            }
152            None => {
153                if i != 0 {
154                    return Err(err("unsectioned items must come before the first `##`"));
155                }
156                if section.items.is_empty() {
157                    return Err(err("empty unsectioned section is unrepresentable"));
158                }
159            }
160        }
161        for item in &section.items {
162            emit_item(&mut out, item, 0)?;
163        }
164    }
165    Ok(out)
166}

168/// [`line::check_item`] + the deferred-only rule, applied to a
169/// whole subtree. Depth matches the parse-side cap so emit can't
170/// write a file parse would reject (or overflow the stack on).
171fn check_subtree(item: &Item, depth: usize) -> Result<(), EmitError> {
172    if depth > line::MAX_ITEM_DEPTH {
173        return Err(EmitError::new(
174            FILE,
175            format!("item {} nested too deep", item.id),
176        ));
177    }
178    line::check_item(item, FILE)?;
179    if item.status != Status::Deferred {
180        return Err(EmitError::new(
181            FILE,
182            format!("non-deferred item {} in BACKLOG.md", item.id),
183        ));
184    }
185    for c in &item.children {
186        check_subtree(c, depth + 1)?;
187    }
188    Ok(())
189}
190
191/// One item line plus its nested children — 2-space indent per
192/// level under the parent's bullet.
193fn emit_item(out: &mut String, item: &Item, depth: usize) -> Result<(), EmitError> {
194    for _ in 0..depth {
195        out.push_str("  ");
196    }
197    out.push_str("- ");
198    out.push_str(&mdx::inline(&line::text(item, &SPEC)?));
199    out.push('\n');
200    for c in &item.children {
201        emit_item(out, c, depth + 1)?;
202    }
203    Ok(())
204}

212    #[test]
213    fn roundtrip() {
214        let src = "# Backlog\n\n## Someday\n\n- T-034 Old idea · pri:low · #maybe\n- T-040 Another\n\n## Rejected\n\n- T-050 Meh\n";
215        let doc = parse_backlog(src).unwrap();
216        assert_eq!(doc.sections.len(), 2);
217        assert_eq!(doc.items().count(), 3);
218        assert!(doc.items().all(|i| i.status == Status::Deferred));
219        assert_eq!(emit_backlog(&doc).unwrap(), src);
220    }

358    #[test]
359    fn nesting_depth_is_capped() {
360        // past MAX_ITEM_DEPTH parse errors instead of recursing
361        // into a stack overflow; emit enforces the same bound
362        // (I-021)
363        let mut src = String::from("# Backlog\n\n");
364        for d in 0..=line::MAX_ITEM_DEPTH + 2 {
365            src.push_str(&"  ".repeat(d));
366            src.push_str("- T-001 t\n");
367        }
368        assert!(parse_backlog(&src).is_err());
369        let mut item = Item::new(ItemId::new(ItemKind::Task, 1), "t").status(Status::Deferred);
370        for _ in 0..line::MAX_ITEM_DEPTH + 2 {
371            item = Item::new(ItemId::new(ItemKind::Task, 1), "t")
372                .status(Status::Deferred)
373                .child(item);
374        }
375        let mut doc = BacklogDoc::default();
376        doc.sections.push(Section::named("s"));
377        doc.sections[0].items.push(item);
378        assert!(emit_backlog(&doc).is_err());
379    }
