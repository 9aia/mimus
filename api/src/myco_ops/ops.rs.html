1//! Shared plumbing for the lifecycle verbs — the `project/`
2//! scaffold and the auto-commit-per-verb machinery.
3//!
4//! Every mutating verb ends in one atomic commit on the `project`
5//! branch (`--no-commit` or `$MYCO_NO_COMMIT` opts out, e.g. for
6//! tests and dry runs). Commits happen only when the worktree
7//! actually changed — a no-op verb leaves no noise in the audit
8//! trail.
9

15/// Dirs `myco init` / `check --fix` scaffold (with `.gitkeep`).
16const SCAFFOLD_DIRS: [&str; 5] = ["issues", "proposals", "adrs", "plans", "handoffs"];

18/// Name of the lock file — inside the repo's shared git dir
19/// (`<main>/.git/myco.lock`) so it never lands in a ledger commit
20/// (`git add -A` can't see it) and covers the `project/` worktree
21/// of every checkout of the repo.
22const LOCK_FILE: &str = "myco.lock";
23
24/// RAII guard for a worktree lock — an OS advisory lock (`flock`)
25/// on the repo's lock file. The lock releases when the guard
26/// drops (or the process dies — the OS cleans up).
27#[derive(Debug)]
28pub struct WorktreeLock {
29    _file: std::fs::File,
30}
31
32/// The file the lock lives on: `<git-common-dir>/myco.lock` — the
33/// common dir is shared by a repo's worktrees and exists before
34/// `myco init` creates the worktree, so one path covers init and
35/// every verb. A dir outside any repo falls back to a dotfile
36/// inside itself (nothing can autocommit there anyway, and the
37/// ledger spec ignores dotfiles).
38fn lock_path(dir: &Path) -> Result<PathBuf, String> {
39    match crate::vcs::common_dir(dir) {
40        Ok(git) => Ok(git.join(LOCK_FILE)),
41        Err(_) => {
42            if dir.is_dir() {
43                Ok(dir.join(".myco.lock"))
44            } else {
45                Err(format!("{} is not a directory", dir.display()))
46            }
47        }
48    }
49}
50
51fn lock_at(dir: &Path, exclusive: bool) -> Result<WorktreeLock, String> {
52    let path = lock_path(dir)?;
53    let file = std::fs::OpenOptions::new()
54        .create(true)
55        .truncate(false)
56        .write(true)
57        .open(&path)
58        .map_err(|e| format!("open {}: {e}", path.display()))?;
59    let taken = if exclusive {
60        file.try_lock()
61    } else {
62        file.try_lock_shared()
63    };
64    match taken {
65        Ok(()) => Ok(WorktreeLock { _file: file }),
66        Err(std::fs::TryLockError::WouldBlock) => {
67            Err("another myco process is writing — retry".into())
68        }
69        Err(std::fs::TryLockError::Error(e)) => Err(format!("lock {}: {e}", path.display())),
70    }
71}

73/// The exclusive mutation lock — every load→mutate→save→commit
74/// verb holds it for the verb's whole life (including the
75/// autocommit). Fail-fast on contention: concurrent writers
76/// corrupt the ledger (dup ids, last-writer-wins), so the second
77/// one gets a retry hint instead.
78///
79/// `dir` is any path inside the repo — the worktree or the repo
80/// root; the lock lands on the shared git dir either way.
81pub fn lock_worktree(dir: &Path) -> Result<WorktreeLock, String> {
82    lock_at(dir, true)
83}
84
85/// The shared read lock — concurrent readers don't block each
86/// other, but a reader contending with a writer fails fast
87/// rather than rendering a mid-verb ledger.
88pub fn lock_shared(dir: &Path) -> Result<WorktreeLock, String> {
89    lock_at(dir, false)
90}

108/// Create every missing scaffold piece in the worktree; existing
109/// files are never overwritten. `root` is the store's worktree
110/// dir (the `.gitkeep` sweep is raw fs — the port has no "list
111/// dir" verb). Returns the rel paths created.
112pub fn scaffold(store: &impl Store, root: &Path, name: &str) -> Result<Vec<String>, StoreError> {
113    let mut created = Vec::new();
114    let missing = |rel: &str| !root.join(rel).exists();
115    if missing("meta.toml") {
116        store.save_meta(&ProjectMeta::new(name))?;
117        created.push("meta.toml".to_string());
118    }
119    if missing("TODO.md") {
120        store.save_todo(&TodoDoc::new(name))?;
121        created.push("TODO.md".to_string());
122    }
123    if missing("BACKLOG.md") {
124        store.save_backlog(&BacklogDoc::default())?;
125        created.push("BACKLOG.md".to_string());
126    }
127    if missing("DONE.md") {
128        store.save_done(&myco_core::DoneDoc::default())?;
129        created.push("DONE.md".to_string());
130    }
131    if missing("index.md") {
132        store.save_index(&index_template(name))?;
133        created.push("index.md".to_string());
134    }
135    for dir in SCAFFOLD_DIRS {
136        let keep = root.join(dir).join(".gitkeep");
137        let empty_dir = match std::fs::read_dir(root.join(dir)) {
138            Ok(mut it) => it.next().is_none(),
139            Err(_) => true,
140        };
141        if empty_dir && !keep.exists() {
142            std::fs::create_dir_all(root.join(dir))?;
143            std::fs::write(&keep, "")?;
144            created.push(format!("{dir}/.gitkeep"));
145        }
146    }
147    Ok(created)
148}

156/// Fallback actor for the `Actor:` commit trailer and every
157/// actor default — the local CLI wrote it. The last rung of
158/// [`resolve_actor`]'s chain; verbs that name an actor
159/// (`--actor`/`--owner`) resolve through the chain instead.
160pub const DEFAULT_ACTOR: &str = "cli";
161
162/// Sanitize a `user.name` into actor-id shape — actor ids allow
163/// `[A-Za-z0-9._:-]`; anything else collapses to `-` so "Luis P."
164/// becomes `human:Luis-P.` rather than failing validation.
165fn sanitize_actor_name(name: &str) -> String {
166    let mut out = String::with_capacity(name.len());
167    let mut dash = false;
168    for c in name.trim().chars() {
169        if c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-') {
170            out.push(c);
171            dash = false;
172        } else if !dash {
173            out.push('-');
174            dash = true;
175        }
176    }
177    out.trim_matches('-').to_string()
178}

180/// The actor-resolution chain every verb shares — explicit flag
181/// first, then environment, then git identity, then the CLI
182/// fallback:
183///
184/// 1. `explicit` — the verb's `--actor`/`--owner` value.
185/// 2. `$MYCO_ACTOR` — the agent/hook-set identity (invalid values
186///    fall through rather than fail).
187/// 3. `human:<git user.name>` — the repo's configured identity,
188///    sanitized into actor-id shape. `dir` is any path inside the
189///    repo (the project worktree works); outside a repo or with
190///    no `user.name` this rung skips.
191/// 4. [`DEFAULT_ACTOR`] — `cli`.
192///
193/// There is no session-env rung: sepia sessions reach myco as
194/// `--session` values on the continuity verbs, not as an actor
195/// convention — nothing in the org sets one.
196pub fn resolve_actor(dir: Option<&Path>, explicit: Option<&Actor>) -> Actor {
197    if let Some(a) = explicit {
198        return a.clone();
199    }
200    if let Ok(v) = std::env::var("MYCO_ACTOR") {
201        if let Ok(a) = Actor::new(v.trim()) {
202            return a;
203        }
204    }
205    if let Some(d) = dir {
206        if let Ok(name) = crate::vcs::git_out(d, &["config", "user.name"]) {
207            let sanitized = sanitize_actor_name(name.trim());
208            if !sanitized.is_empty() {
209                if let Ok(a) = Actor::new(&format!("human:{sanitized}")) {
210                    return a;
211                }
212            }
213        }
214    }
215    Actor::new(DEFAULT_ACTOR).unwrap_or_else(|_| unreachable!("DEFAULT_ACTOR is a valid actor id"))
216}

218/// `msg` + the `Actor:` trailer — every myco commit names the
219/// verb's actor in the message body (T-011: actor-tagged
220/// trailers, not per-actor git authors — author/committer stay
221/// the repo's stable `user.name`/`user.email` identity, or the
222/// `myco@localhost` fallback). `None` resolves to
223/// [`DEFAULT_ACTOR`]. The subject line is untouched, so `%s`
224/// parses (e.g. `report`'s velocity walk) never see it.
225pub fn commit_msg(msg: &str, actor: Option<&Actor>) -> String {
226    let actor = actor.map(Actor::as_str).unwrap_or(DEFAULT_ACTOR);
227    format!("{msg}\n\nActor: {actor}")
228}

230/// Mutating verbs must not write a ledger they cannot commit. A
231/// detached `project/` worktree is the failure mode that costs the
232/// most: every write lands, no branch records it, and the audit
233/// trail is gone the moment the worktree moves. `check` reports and
234/// repairs it; every write path refuses it first.
235pub fn ensure_committable(dir: &Path) -> Result<(), String> {
236    if crate::vcs::is_detached(dir) == Some(true) {
237        return Err(format!(
238            "{} is on a detached HEAD — refusing to write a ledger whose commits would be \
239             unreachable; run `myco check --fix` to reattach it",
240            dir.display()
241        ));
242    }
243    Ok(())
244}

246/// Refuse a mutation while the ledger branch is behind its
247/// tracked remote — every verb commit would deepen the divergence
248/// `myco sync` exists to close.
249///
250/// `auto_fetch` (`sync.auto_fetch`, default on) runs a
251/// best-effort `git fetch origin <branch>` first: the cached
252/// `refs/remotes/origin/<branch>` can only *under*-report
253/// divergence, so refreshing it before the check is what makes
254/// the gate honest. A failed fetch (offline, no remote, bad
255/// creds) is never fatal — the check falls back to the stale
256/// ref exactly as before. `false` skips the fetch entirely.
257/// Untracked branches (never pushed, no remote) pass — pushing is
258/// a repo-level choice, not this verb's business.
259pub fn ensure_synced(dir: &Path, auto_fetch: bool) -> Result<(), String> {
260    let Some(branch) = crate::vcs::worktree_branch(dir) else {
261        return Ok(());
262    };
263    if auto_fetch {
264        let _ = crate::vcs::fetch_branch(dir, "origin", &branch);
265    }
266    if !crate::vcs::tracks_branch(dir, "origin", &branch) {
267        return Ok(());
268    }
269    let remote = format!("refs/remotes/origin/{branch}");
270    let Some(behind) = crate::vcs::commits_behind(dir, &remote) else {
271        return Ok(());
272    };
273    if behind > 0 {
274        return Err(format!(
275            "the ledger is {behind} commit(s) behind origin/{branch} — run `myco sync` \
276             first so mutations land on the published tip"
277        ));
278    }
279    Ok(())
280}

282/// One atomic commit of everything under `dir` — the
283/// auto-commit-per-verb path. `msg` gains the `Actor:` trailer
284/// ([`commit_msg`]; `actor` is the verb's actor, `None` →
285/// [`resolve_actor`] over `dir` — `$MYCO_ACTOR` → `human:<git
286/// user.name>` → `cli`). No commit when disabled or when the
287/// tree is clean. Returns whether a commit was made.
288pub fn autocommit(
289    dir: &Path,
290    msg: &str,
291    actor: Option<&Actor>,
292    enabled: bool,
293) -> Result<bool, String> {
294    if !enabled || !crate::vcs::has_changes(dir).map_err(|e| e.to_string())? {
295        return Ok(false);
296    }
297    let resolved;
298    let actor = match actor {
299        Some(a) => a,
300        None => {
301            resolved = resolve_actor(Some(dir), None);
302            &resolved
303        }
304    };
305    crate::vcs::commit_all(dir, &commit_msg(msg, Some(actor)))
306        .map_err(|e| format!("git commit failed in {}: {e}", dir.display()))?;
307    Ok(true)
308}

310/// Commit only `paths` inside `dir` — used for repo-root edits
311/// (`.gitignore`, `pm/` removal) so an unrelated dirty tree isn't
312/// swept into the verb's commit. Returns whether a commit was made.
313///
314/// Scoping is per *file*, not per hunk: callers must pass only
315/// paths myco itself wrote this invocation. If the user had
316/// pre-existing uncommitted edits inside one of those files, they
317/// land in the commit under myco's message — an accepted
318/// limitation (hunk-level staging via `git apply --cached` was
319/// judged disproportionate for the managed-file call sites).
320pub fn commit_paths(
321    dir: &Path,
322    msg: &str,
323    paths: &[&str],
324    actor: Option<&Actor>,
325    enabled: bool,
326) -> Result<bool, String> {
327    if !enabled {
328        return Ok(false);
329    }
330    let mut args = vec!["status", "--porcelain", "--"];
331    args.extend_from_slice(paths);
332    let dirty = crate::vcs::git_out(dir, &args).map_err(|e| e.to_string())?;
333    if dirty.is_empty() {
334        return Ok(false);
335    }
336    let mut add = vec!["add", "--"];
337    add.extend_from_slice(paths);
338    crate::vcs::git_out(dir, &add).map_err(|e| e.to_string())?;
339    let resolved;
340    let actor = match actor {
341        Some(a) => a,
342        None => {
343            resolved = resolve_actor(Some(dir), None);
344            &resolved
345        }
346    };
347    let msg = commit_msg(msg, Some(actor));
348    let mut commit = vec!["commit", "-m", msg.as_str(), "--"];
349    commit.extend_from_slice(paths);
350    crate::vcs::git_ident(dir, &commit)
351        .map_err(|e| e.to_string())
352        .map(|_| true)
353}

355/// What [`undo_last`] reverted — the `undo` verb's report payload.
356#[derive(Debug)]
357pub struct UndoOutcome {
358    /// The commit that was reset away.
359    pub hash: String,
360    /// Its subject line.
361    pub subject: String,
362    /// Its author date (`%as`).
363    pub date: String,
364    /// Caveats — e.g. multi-commit verbs leave sibling commits a
365    /// worktree-scoped reset can't reach.
366    pub warnings: Vec<String>,
367}

369/// Subjects of verbs that also commit *outside* this worktree —
370/// `init`/`check --fix`/`migrate` land legs on the main branch,
371/// `item move` commits in a second repo. Undoing their
372/// project-side tip can't revert the whole verb; the caller gets
373/// a warning instead of a silent half-undo.
374const MULTI_COMMIT_SUBJECTS: &[&str] = &[
375    "chore: init",
376    "chore: migrate",
377    "chore: move",
378    "chore: check --fix",
379];

381/// `myco undo` — move the worktree's branch tip back one commit
382/// (`git reset --hard HEAD~1`). Every mutating verb is exactly one
383/// commit on the ledger branch, so the reset *is* the inverse
384/// mutation — including file deletions (`todo done`'s removed
385/// line comes back whole). The commit stays reachable through the
386/// reflog; nothing is garbage-collected.
387///
388/// Guards, in order: the caller holds [`lock_worktree`]; the
389/// checkout must not be detached ([`ensure_committable`]); the
390/// worktree must be clean — `--no-commit` writes are uncommitted
391/// ledger state a hard reset would silently destroy; `HEAD~1`
392/// must exist (the orphan root has nothing behind it); and a tip
393/// already published on `origin` is refused unless `force` —
394/// rewriting pushed audit-trail history is the one real hazard.
395///
396/// `dir` is the `project/` worktree (or any dir inside it).
397pub fn undo_last(dir: &Path, force: bool) -> Result<UndoOutcome, String> {
398    ensure_committable(dir)?;
399    let entry = crate::vcs::log(dir, Some(1))
400        .map_err(|e| e.to_string())?
401        .into_iter()
402        .next()
403        .ok_or_else(|| "nothing to undo — the ledger branch has no commits".to_string())?;
404    let base = crate::vcs::git_out(dir, &["rev-parse", "--verify", "HEAD~1"])
405        .map_err(|_| format!("nothing to undo — {} is the root commit", entry.hash))?;
406    if crate::vcs::has_changes(dir).map_err(|e| e.to_string())? {
407        return Err(
408            "uncommitted changes in the worktree — `myco commit` them or clean the tree first; \
409             undo only rewinds committed state"
410                .into(),
411        );
412    }
413    if let Some(branch) = crate::vcs::worktree_branch(dir) {
414        let remote = format!("refs/remotes/origin/{branch}");
415        let published = crate::vcs::tracks_branch(dir, "origin", &branch)
416            && crate::vcs::commits_ahead(dir, &remote) == Some(0);
417        if published && !force {
418            return Err(format!(
419                "the tip commit is already published on origin/{branch} — undo would rewrite \
420                 pushed history; pass --force to override"
421            ));
422        }
423    }
424    crate::vcs::git_out(dir, &["reset", "--hard", "HEAD~1"])
425        .map_err(|e| format!("git reset failed in {}: {e}", dir.display()))?;
426    let mut warnings = multi_commit_warnings(&entry.subject);
427    // Record the undone commit for `redo`. The journal self-
428    // invalidates: if its top anchor isn't the HEAD we just left,
429    // the timeline moved and the stale stack is dropped first.
430    let mut j = crate::history::load(dir);
431    if !crate::history::top_valid(&j, &entry.hash) {
432        crate::history::clear(&mut j);
433    }
434    crate::history::push(
435        &mut j,
436        crate::history::HistoryEntry {
437            target: entry.hash.clone(),
438            base: base.trim().to_string(),
439            subject: entry.subject.clone(),
440            date: entry.date.clone(),
441        },
442    );
443    if let Err(e) = crate::history::save(dir, &j) {
444        warnings.push(format!(
445            "undo committed but the redo journal could not be saved ({e}) — redo may be unavailable"
446        ));
447    }
448    Ok(UndoOutcome {
449        hash: entry.hash,
450        subject: entry.subject,
451        date: entry.date,
452        warnings,
453    })
454}

456/// `myco redo` — restore the commit the last undo moved away
457/// from: pop the top of the undo journal
458/// (`<git-common-dir>/myco-history.json`, see
459/// [`crate::history`]) and `git reset --hard` to its target.
460/// Repeatable — each pop exposes the next-older undone commit,
461/// which is the redo stack single-step reflog tricks can't
462/// express.
463///
464/// Guards mirror undo's: committable checkout, clean worktree (a
465/// hard reset would destroy `--no-commit` writes), and the top
466/// entry's `base` must still equal HEAD — any commit, pull, or
467/// manual git op since the undo breaks the anchor and the stack
468/// self-clears. No published-tip guard: re-applying a commit
469/// that was already the tip rewrites nothing new.
470pub fn redo_last(dir: &Path) -> Result<UndoOutcome, String> {
471    ensure_committable(dir)?;
472    let head = crate::history::head_sha(dir)
473        .map_err(|_| "nothing to redo — the ledger branch has no commits".to_string())?;
474    let mut j = crate::history::load(dir);
475    let Some(top) = j.future.last().cloned() else {
476        return Err("nothing to redo — no undone commit is waiting".into());
477    };
478    if top.base != head {
479        crate::history::clear(&mut j);
480        let _ = crate::history::save(dir, &j);
481        return Err("nothing to redo — the ledger moved since the last undo".into());
482    }
483    if crate::vcs::has_changes(dir).map_err(|e| e.to_string())? {
484        return Err(
485            "uncommitted changes in the worktree — `myco commit` them or clean the tree first; \
486             redo only rewinds committed state"
487                .into(),
488        );
489    }
490    if crate::vcs::git_out(
491        dir,
492        &[
493            "rev-parse",
494            "--verify",
495            &format!("{}^{{commit}}", top.target),
496        ],
497    )
498    .is_err()
499    {
500        return Err(format!(
501            "nothing to redo — {} no longer exists",
502            &top.target[..12]
503        ));
504    }
505    crate::vcs::git_out(dir, &["reset", "--hard", &top.target])
506        .map_err(|e| format!("git reset failed in {}: {e}", dir.display()))?;
507    j.future.pop();
508    let mut warnings = multi_commit_warnings(&top.subject);
509    if let Err(e) = crate::history::save(dir, &j) {
510        warnings.push(format!(
511            "redo committed but the redo journal could not be saved ({e}) — further redo may be unavailable"
512        ));
513    }
514    Ok(UndoOutcome {
515        hash: top.target,
516        subject: top.subject,
517        date: top.date,
518        warnings,
519    })
520}

522/// How many commits `redo` could still restore — 0 when the
523/// journal is empty or its top anchor no longer matches HEAD.
524pub fn redo_depth(dir: &Path) -> usize {
525    match crate::history::head_sha(dir) {
526        Ok(head) => crate::history::depth(dir, &head),
527        Err(_) => 0,
528    }
529}

536/// Sibling-leg warnings for verbs that commit outside the
537/// worktree (`init`, `migrate`, `move`, `check --fix`).
538fn multi_commit_warnings(subject: &str) -> Vec<String> {
539    MULTI_COMMIT_SUBJECTS
540        .iter()
541        .filter(|p| subject.starts_with(**p))
542        .map(|p| format!("{p}… also commits outside this worktree — those sibling commits remain"))
543        .collect()
544}

554/// Default pager — `-F` quits instantly under one screen, `-R`
555/// keeps ANSI colors, `-S` chops instead of wrapping rows, `-X`
556/// leaves output in the scrollback.
557const DEFAULT_PAGER: &str = "less -FRSX";

559/// Pager program: `$MYCO_PAGER` → `$PAGER` → [`DEFAULT_PAGER`].
560/// An explicitly empty value disables paging.
561fn pager_program() -> Option<String> {
562    for var in ["MYCO_PAGER", "PAGER"] {
563        if let Ok(v) = std::env::var(var) {
564            return if v.trim().is_empty() { None } else { Some(v) };
565        }
566    }
567    Some(DEFAULT_PAGER.to_string())
568}

570/// `sh -c <pager>` with `text` on its stdin — the pager inherits
571/// our stdout/stderr, so scrolling and colors work as usual.
572/// `$MYCO_PAGER`/`$PAGER` are executed verbatim (git-parity
573/// behavior) — an env var is a trusted setting, never user data.
574fn spawn_pager(pager: &str, text: &str) -> std::io::Result<()> {
575    use std::io::Write;
576    let mut child = std::process::Command::new("sh")
577        .args(["-c", pager])
578        .stdin(std::process::Stdio::piped())
579        .spawn()?;
580    if let Some(mut stdin) = child.stdin.take() {
581        // quitting early (`q`) closes the pipe — with SIGPIPE at
582        // its default disposition we die like any unix tool, which
583        // is what the user expects; ignore the write error for the
584        // SIG_IGN case
585        let _ = stdin.write_all(text.as_bytes());
586    }
587    child.wait().map(|_| ())
588}

590/// Print `text`, paging it through the user's pager when stdout is
591/// a tty — `less -F` exits instantly for one-screen output, so
592/// callers can page unconditionally. Non-tty output, a disabled or
593/// unspawnable pager all fall back to `println!`.
594pub fn print_paged(text: &str) {
595    use std::io::IsTerminal;
596    if std::io::stdout().is_terminal() {
597        if let Some(pager) = pager_program() {
598            if spawn_pager(&pager, text).is_ok() {
599                return;
600            }
601        }
602    }
603    println!("{text}");
604}
