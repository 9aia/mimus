1//! Config-declared verbs — `[verbs]` from merged config layers
2//! (P-007). Resolution mirrors `view.rs`: config files are I/O so
3//! the merge lives here; the verbs themselves are executed by the
4//! front-end (cli). The builtin surface is injected — ops doesn't
5//! know the registry, callers hand it in like `ConfigLoader`.
6//!
7//! Config verbs are read-only by construction: an alias may only
8//! target a read-only builtin, and a view-emit can only evaluate a
9//! spec. Collisions mirror view-shadowing except the *builtin*
10//! wins — a config verb can never hide the stable surface.
11

19/// One builtin verb as the front-end's registry sees it — the
20/// facts `[verbs]` resolution needs for collision and
21/// alias-target checks.
22#[derive(Debug, Clone, Copy)]
23pub struct BuiltinVerb {
24    /// The canonical name (`todo.list`, `status`).
25    pub name: &'static str,
26    /// Can write the ledger/index — alias targets must be
27    /// read-only, so `mutates` disqualifies.
28    pub mutates: bool,
29    /// Has a CLI handler — wire-only builtins can't be aliased
30    /// either.
31    pub cli: bool,
32}
33
34/// A resolved config-declared verb — shadowed/invalid entries are
35/// dropped from the effective set and reported as warnings.
36#[derive(Debug, Clone)]
37pub struct NamedVerb {
38    /// The `[verbs.<name>]` name.
39    pub name: String,
40    /// The declared verb.
41    pub verb: ConfigVerb,
42}
43
44/// Verb names are top-level CLI words — `[A-Za-z0-9_-]+`, never
45/// flag-shaped.
46pub fn is_valid_verb_name(name: &str) -> bool {
47    !name.is_empty()
48        && !name.starts_with('-')
49        && name
50            .bytes()
51            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_'))
52}
53
54/// `[verbs]` for a repo — same layered read as [`crate::view`]:
55/// `<root>/myco.toml` when it exists (else the impl's default
56/// probe), merged by the injected `cfg`. "No config anywhere" is
57/// empty, not an error.
58fn config_verbs(
59    root: &Path,
60    strict: bool,
61    cfg: ConfigLoader,
62) -> Result<BTreeMap<String, ConfigVerb>, String> {
63    let repo_cfg = root.join("myco.toml");
64    match cfg.load(repo_cfg.exists().then_some(repo_cfg.as_path())) {
65        Ok(c) => Ok(c.verbs),
66        // the documented search-miss — absence isn't a failure
67        Err(e) if e.contains("no myco.toml found") => Ok(BTreeMap::new()),
68        Err(e) if !strict && e.contains("not found") => Ok(BTreeMap::new()),
69        Err(e) => Err(e),
70    }
71}

73/// The warning a config-vs-builtin name collision earns — the
74/// builtin wins (the stable surface is part of the audit trail).
75fn shadow_warning(name: &str) -> String {
76    format!("verb \"{name}\" shadows a built-in — the built-in wins")
77}
78
79/// Why a config verb was dropped from the effective set.
80fn skip_reason(name: &str, verb: &ConfigVerb, builtins: &[BuiltinVerb]) -> Option<String> {
81    if !is_valid_verb_name(name) {
82        return Some(format!(
83            "verb \"{name}\": invalid name — verbs match `[A-Za-z0-9_-]+`"
84        ));
85    }
86    if builtins.iter().any(|b| b.name == name) {
87        return Some(shadow_warning(name));
88    }
89    if let ConfigVerb::Alias(a) = verb {
90        match builtins.iter().find(|b| b.name == a.target) {
91            None => {
92                return Some(format!(
93                    "verb \"{name}\": unknown alias target \"{}\" — skipped",
94                    a.target
95                ))
96            }
97            Some(b) if b.mutates || !b.cli => {
98                return Some(format!(
99                    "verb \"{name}\": alias target \"{}\" isn't a read-only verb — skipped",
100                    a.target
101                ))
102            }
103            _ => {}
104        }
105    }
106    None
107}

109/// Config verbs effective for a repo — invalid names, builtin
110/// collisions, and bad alias targets warn + skip.
111fn merged_verbs(
112    root: &Path,
113    strict: bool,
114    cfg: ConfigLoader,
115    builtins: &[BuiltinVerb],
116) -> Result<(Vec<NamedVerb>, Vec<String>), String> {
117    let mut warnings = Vec::new();
118    let mut out = Vec::new();
119    for (name, verb) in config_verbs(root, strict, cfg)? {
120        match skip_reason(&name, &verb, builtins) {
121            Some(w) => warnings.push(w),
122            None => out.push(NamedVerb { name, verb }),
123        }
124    }
125    Ok((out, warnings))
126}

128/// [`merged_verbs`] with strict config handling — `myco serve`,
129/// `myco check`.
130pub fn named_verbs(
131    root: &Path,
132    cfg: ConfigLoader,
133    builtins: &[BuiltinVerb],
134) -> Result<(Vec<NamedVerb>, Vec<String>), String> {
135    merged_verbs(root, true, cfg, builtins)
136}

138/// [`merged_verbs`] tolerant of a *missing* config — the verb
139/// dispatch path: `$MYCO_CONFIG` naming a missing file degrades
140/// to no config verbs; an unparseable config still errors.
141pub fn named_verbs_lenient(
142    root: &Path,
143    cfg: ConfigLoader,
144    builtins: &[BuiltinVerb],
145) -> Result<(Vec<NamedVerb>, Vec<String>), String> {
146    merged_verbs(root, false, cfg, builtins)
147}

149/// One *raw* `[verbs]` entry by name — lenient, no validation.
150/// For the "is this builtin shadowed" check the effective set
151/// can't answer (shadowed entries are dropped from it).
152pub fn raw_verb(root: &Path, name: &str, cfg: ConfigLoader) -> Result<Option<ConfigVerb>, String> {
153    Ok(config_verbs(root, false, cfg)?.remove(name))
154}

156/// Resolve one config verb by name — builtin names never resolve
157/// (the builtin wins); unknown names error listing the defined
158/// config verbs. A view-emit's spec is validated before return.
159pub fn resolve(
160    root: &Path,
161    name: &str,
162    cfg: ConfigLoader,
163    builtins: &[BuiltinVerb],
164) -> Result<NamedVerb, String> {
165    resolve_in(root, name, true, cfg, builtins)
166}

168/// [`resolve`] tolerant of a *missing* config — the dispatch path.
169pub fn resolve_lenient(
170    root: &Path,
171    name: &str,
172    cfg: ConfigLoader,
173    builtins: &[BuiltinVerb],
174) -> Result<NamedVerb, String> {
175    resolve_in(root, name, false, cfg, builtins)
176}

178fn resolve_in(
179    root: &Path,
180    name: &str,
181    strict: bool,
182    cfg: ConfigLoader,
183    builtins: &[BuiltinVerb],
184) -> Result<NamedVerb, String> {
185    if let Some(reason) = config_verbs(root, strict, cfg)?
186        .get(name)
187        .and_then(|v| skip_reason(name, v, builtins))
188    {
189        return Err(reason);
190    }
191    let (verbs, _) = merged_verbs(root, strict, cfg, builtins)?;
192    let verb = verbs.into_iter().find(|v| v.name == name).ok_or_else(|| {
193        let names: Vec<String> = config_verbs(root, strict, cfg)
194            .map(|m| m.keys().cloned().collect())
195            .unwrap_or_default();
196        if names.is_empty() {
197            format!("unknown verb \"{name}\"")
198        } else {
199            format!(
200                "unknown verb \"{name}\" — config verbs: {}",
201                names.join(", ")
202            )
203        }
204    })?;
205    if let ConfigVerb::ViewEmit(v) = &verb.verb {
206        v.spec.validate()?;
207    }
208    Ok(verb)
209}

211/// Lint the repo-layer config only — deterministic, so `myco
212/// check` reports the same findings for every user and CI.
213/// `Ok` = per-verb validation errors (`(name, error)`); `Err` =
214/// the file is unreadable/unparseable. Missing file is
215/// `Ok(vec![])`. Unlike views (where shadowing is legal), a
216/// verb-name collision is dead config — it is a finding.
217pub fn validate_repo_verbs(
218    root: &Path,
219    loader: ConfigLoader,
220    builtins: &[BuiltinVerb],
221) -> Result<Vec<(String, String)>, String> {
222    let path = root.join("myco.toml");
223    if !path.exists() {
224        return Ok(vec![]);
225    }
226    let cfg = loader.load_file(&path)?;
227    Ok(cfg
228        .verbs
229        .iter()
230        .flat_map(|(name, verb)| {
231            let mut errs: Vec<String> = skip_reason(name, verb, builtins).into_iter().collect();
232            if let ConfigVerb::ViewEmit(v) = verb {
233                if let Err(e) = v.spec.validate() {
234                    errs.push(e);
235                }
236            }
237            errs.into_iter().map(move |e| (name.clone(), e))
238        })
239        .collect())
240}

282    #[test]
283    fn resolves_alias_and_view_emit() {
284        let dir = tempfile::tempdir().unwrap();
285        std::fs::write(
286            dir.path().join("myco.toml"),
287            "[verbs.standup]\ntarget = \"view\"\nargs = [\"board\"]\n\n\
288             [verbs.hot]\nemit = \"first\"\nfilter = { status = [\"open\"] }\n",
289        )
290        .unwrap();
291        no_env(|| {
292            let (verbs, warnings) = named_verbs(dir.path(), test_loader(), BUILTINS).unwrap();
293            assert!(warnings.is_empty());
294            let names: Vec<&str> = verbs.iter().map(|v| v.name.as_str()).collect();
295            assert_eq!(names, ["hot", "standup"]);
296            let v = resolve(dir.path(), "hot", test_loader(), BUILTINS).unwrap();
297            assert!(matches!(v.verb, ConfigVerb::ViewEmit(_)));
298            let v = resolve(dir.path(), "standup", test_loader(), BUILTINS).unwrap();
299            match &v.verb {
300                ConfigVerb::Alias(a) => {
301                    assert_eq!(a.target, "view");
302                    assert_eq!(a.args, ["board"]);
303                }
304                _ => panic!("standup should be an alias"),
305            }
306        });
307    }
