1//! tamias-driver-bupha — dual-surface driver for bupha's CI state.
2//!
3//! One binary, two consumers, disjoint argv:
4//!
5//! **pavo view driver** (verbs take no `--dir`/`--name`):
6//!
7//! - `manifest` — print the [`pavo_protocol::DriverManifest`] JSON
8//!   (discovery probe; never touches bupha state).
9//! - `capabilities` — merged JSON advertising BOTH surfaces: pavo's
10//!   `{"capabilities": ["pavo"], "verbs": […"serve"…]}` shape plus
11//!   tamias's `"types": ["bupha"]` and the provider verbs in `verbs`.
12//!   pavo's discovery greps `capabilities`/`verbs` for
13//!   `"pavo"`/`"serve"`; tamias's greps `types` for the content kind
14//!   and warns on verbs missing from `verbs` — one payload satisfies
15//!   both.
16//! - `serve` — ndjson JSON-RPC on stdio (see `bupha_driver::serve`).
17//!   `query:"ops"` answers the engine ops manifest; `query:"cli.<verb>"`
18//!   runs a bounded `bupha` CLI verb in a captured child.
19//!
20//! Engine surface (P-021 — the `bupha` CLI as a generated surface):
21//!
22//! - `ops` — the ops manifest: every bounded `bupha` verb as a
23//!   `cli.<verb>` wire query, derived from the engine's own clap
24//!   tree (`bupha::command`). Unbounded daemons (`watch`, `agent`,
25//!   `gha-runner`) are `surfaces`, never ops.
26//! - `run <argv>` — engine passthrough: `tamias-driver-bupha run
27//!   doctor --json` ≡ `bupha doctor --json`. Also the child side of
28//!   `cli.*` wire queries (`engine_ops::run_verb` re-execs this same
29//!   verb so engine output never touches the serve stream).
30//! - `cli <argv>` — the tamias `bins[]` shim verb (T-012): `tamias
31//!   bin exec` runs `<driver> cli --dir <payload> --name <pkg>
32//!   <bupha argv>` for a declared `shim: cli` bin. bupha's engine
33//!   IS the CLI, so this is a passthrough — strip the two standard
34//!   driver flags and hand the rest to `bupha::run_from` with
35//!   inherited stdio. That keeps full engine parity a generated
36//!   ops→CLI surface can't express: globals before the verb
37//!   (`--config`), `--help`, the unbounded `watch`/`agent`/
38//!   `gha-runner` surfaces, and stdin-bound verbs all behave
39//!   exactly like the retired `bupha` binary. There is no `cli`
40//!   feature: `cli` is the shim verb, not a generated surface —
41//!   the engine's own argv grammar is the contract.
42//!
43//! Optional surfaces: feature `mcp` embeds pavo-driver-kit's
44//! surface layer — `mcp` serves the ops as MCP tools on stdio,
45//! equivalent to the standalone `tamias-driver-mcp` surface
46//! driver minus the PATH/discovery hop.
47//!
48//! **tamias content-type driver** (`type: bupha` packages — tamias
49//! always passes `--dir <payload> --name <pkg> [--config <f>]`, so
50//! these verbs are unreachable from pavo's bare probes):
51//!
52//! - `install` — no-op (the operator config is `deploy`'s job; install
53//!   only sees the evictable fetch-cache payload).
54//! - `host` — engine-host mode: run the watcher pipeline
55//!   (`bupha-watch`) in-process against the deployed operator
56//!   config — what provider units exec (`tamias-driver-bupha host
57//!   --config ~/.config/bupha/bundles/<name>.toml`). Blocks until
58//!   the sources stop; `--dir`/`--name` accepted per the tamias
59//!   dispatch contract (`--name` supplies the default
60//!   `bundles/<name>.toml` path when `--config` is absent).
61//!   **Standalone/dev mode** (ADR-0007): the trigger listener this
62//!   hosts is the fallback ingress — production webhook ingress is
63//!   a peregrin campaign listener enqueuing `ci-job` via
64//!   `ci.enqueue`.
65//! - `worker` — the `ci-job` consumption loop (P-002 §4.1): polls
66//!   `kind="ci-job"` entities on a hypha/peregrin node
67//!   (`[worker] node` / `--node` / `BUPHA_WORKER_NODE`, credential
68//!   via `[worker] node_token_env` / `--node-token-env` /
69//!   `BUPHA_NODE_TOKEN`), `ci.claim`s pending jobs this box's
70//!   `[worker] labels` cover, runs them through the same
71//!   `bupha-watch` pipeline, `ci.complete`s the result. The
72//!   **second service** a fleet deployment runs — worker boxes
73//!   execute jobs the entity plane hands out.
74//! - `deploy` — merge the payload's `fleet.toml` with
75//!   `tamias.yaml provider.ingress[]` (`tunnel.public_url`,
76//!   `trigger.listen`/`kind`) and write
77//!   `~/.config/bupha/bundles/<name>.toml` — the file the provider
78//!   unit's `host` verb consumes.
79//! - `uninstall` — remove that file.
80//! - `hooks register|uninstall [--dry-run] [--api-base <url>]
81//!   [--token-env <NAME>]` — upsert/remove the GitHub webhooks
82//!   `provider.ingress[].hooks[]` declare, via `bupha_github`'s REST
83//!   path. Secrets resolve by NAME (env → cuora → keyring →
84//!   `~/.config/bupha/env`; cuora canonical per T-056); values are
85//!   never printed.
86//!   **Standalone mode** (ADR-0007): registers hooks at the bupha
87//!   listener — production ingress is peregrin campaigns, so use
88//!   this only for deployments with no peregrin.
89//! - `health` — JSON probe: webhook `ping` per ingress
90//!   `listen`+`path`, else `systemctl --user is-active` per unit.
91//! - `status` — JSON per-unit `is-active`/`is-enabled`.
92//! - `logs [--lines N]` — `journalctl --user` over the provider units.
93//! - `config-schema` — JSON Schema (draft 2020-12) of the operator
94//!   config `deploy` writes (tamias's reserved verb).
95//!
96//! See `bupha_driver::ops` for the implementation and
97//! `bupha_driver::data` for the pavo read plane.
98
99use bupha_driver as drv;
100#[cfg(feature = "mcp")]
101use pavo_driver_kit as kit;
102
103fn main() {
104    let mut args = std::env::args().skip(1);
105    match args.next().as_deref() {
106        Some("manifest") => {
107            println!(
108                "{}",
109                serde_json::to_string_pretty(&drv::manifest_value()).unwrap_or_default()
110            );
111        }
112        Some("capabilities") => {
113            println!(
114                "{}",
115                serde_json::to_string_pretty(&drv::capabilities_value()).unwrap_or_default()
116            );
117        }
118        Some("serve") => std::process::exit(drv::serve()),
119        // Engine ops surface — `ops` prints the manifest derived
120        // from `bupha::command()`; `run` is the engine passthrough
121        // and the child side of `cli.*` wire queries.
122        Some("ops") => {
123            println!("{}", drv::engine_ops::manifest());
124        }
125        Some("run") => {
126            let argv = std::iter::once("bupha".to_string()).chain(args);
127            std::process::exit(bupha::run_from(argv));
128        }
129        // The tamias `bins[]` shim verb (T-012): `tamias bin exec`
130        // runs `<driver> cli --dir <payload> --name <pkg> <bupha
131        // argv>` for a declared `shim: cli` bin. bupha's engine IS
132        // the CLI, so this is a passthrough — strip the two standard
133        // driver flags and hand the rest to `bupha::run_from` with
134        // inherited stdio. That keeps full engine parity a generated
135        // ops→CLI surface can't express: globals before the verb
136        // (`--config`), `--help`, the unbounded `watch`/`agent`/
137        // `gha-runner` surfaces, and stdin-bound verbs all behave
138        // exactly like the retired `bupha` binary. The tamias
139        // content-type verbs below stay disjoint — `cli` is only
140        // ever the first arg from `bin exec`.
141        Some("cli") => {
142            let argv = engine_argv(&args.collect::<Vec<_>>());
143            std::process::exit(bupha::run_from(argv));
144        }
145        // Embedded surface verb (feature-gated, optional): the
146        // pavo-driver-kit surface layer renders this driver's own
147        // ops manifest as MCP tools over stdio. The standalone
148        // tamias-driver-mcp binary does the same over process
149        // dispatch; this verb needs nothing on PATH.
150        #[cfg(feature = "mcp")]
151        Some("mcp") => {
152            let m = kit::surface::OpsManifest::from_value(&drv::engine_ops::manifest())
153                .unwrap_or_default();
154            match kit::surface::mcp::serve_stdio(&m, &mut drv::surface::Engine) {
155                Ok(()) => {}
156                Err(e) => {
157                    eprintln!("tamias-driver-bupha mcp: {e}");
158                    std::process::exit(1);
159                }
160            }
161        }
162        // Tamias verbs — always invoked with `--dir`/`--name` by the
163        // tamias dispatch contract; pavo never passes those flags, so
164        // subcommand-first dispatch keeps the surfaces disjoint.
165        Some(
166            verb @ ("install" | "deploy" | "uninstall" | "hooks" | "health" | "status" | "logs"
167            | "host" | "worker" | "config-schema"),
168        ) => {
169            std::process::exit(drv::ops::run(verb, &args.collect::<Vec<_>>()));
170        }
171        _ => {
172            eprintln!("usage: tamias-driver-bupha manifest|capabilities|ops|run <argv>|cli <bupha argv>|serve");
173            eprintln!("       (feature `mcp` adds `mcp`)");
174            eprintln!("       tamias-driver-bupha <install|deploy|uninstall|hooks|health|status|logs|host|worker|config-schema> --dir <d> --name <n> [--config <c>]");
175            std::process::exit(1);
176        }
177    }
178}

180/// `cli` verb argv (after the `cli` token) → engine argv (argv[0]
181/// included). `tamias bin exec`/`driver exec` prepend the standard
182/// driver flags — `--dir <payload>` `--name <pkg>` — which the
183/// engine doesn't know; consume the leading run of those two
184/// (`--flag v` and `--flag=v` forms) and pass everything else
185/// through untouched. `--config` is deliberately NOT stripped:
186/// it's a bupha global, so a user-passed one must reach the
187/// engine's own parse.
188fn engine_argv(args: &[String]) -> Vec<String> {
189    let mut out = vec!["bupha".to_string()];
190    let mut i = 0;
191    while i < args.len() {
192        let a = args[i].as_str();
193        if a == "--dir" || a == "--name" {
194            // flag + separate value — skip both (a trailing flag
195            // with no value just ends the scan).
196            i = (i + 2).min(args.len());
197        } else if a.starts_with("--dir=") || a.starts_with("--name=") {
198            i += 1;
199        } else {
200            break;
201        }
202    }
203    out.extend(args[i..].iter().cloned());
204    out
205}
