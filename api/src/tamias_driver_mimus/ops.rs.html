1//! Engine ops — `cli.<verb>` wire queries dispatch into the `mimus`
2//! engine lib through this binary's own `run` verb: each call
3//! re-execs `tamias-driver-mimus run <argv>` with the request's
4//! resolved dir as cwd, so `mimus::dispatch` writes to the child's
5//! captured stdout and never touches the serve loop's ndjson stream
6//! — and every mutation keeps the engine's own write-path
7//! semantics (config search, `.mimus/` cache, claim rewrites).
8//!
9//! `ops` answers the ops manifest as JSON — *derived from the
10//! engine's own clap tree* ([`mimus::command`]), so a verb can't
11//! exist on the CLI and be missing here: declared args become
12//! typed params (`arg`/`opt`/`flag`), doc strings ride along
13//! unchanged.
14//!
15//! mimus has no unbounded verbs today — `surfaces` is empty by
16//! design, kept so a future streaming verb (watch mode, a serve
17//! loop) has an obvious home: surfaces never return, so they ride
18//! `run` passthrough, never request/response ops. Streaming
19//! *flags* would join `SKIP_PARAMS` for the same reason.
20//!
21//! `mutates` is honest: `check`/`report` write `.mimus/triples.db`
22//! (the prose-triple cache), `fix` rewrites claim text, `init`
23//! scaffolds `mimus.toml` + `rules/` — everything else only reads.
24
25use clap::{Arg, ArgAction, Command};
26use pavo_protocol::wire::{QueryResult, Response};
27use serde_json::{json, Map, Value};
28
29/// Unbounded surfaces — verbs that never return, so they can
30/// never be request/response ops. None today.
31const SURFACES: &[&str] = &[];
32
33/// Params excluded from ops — flags that turn a bounded verb into
34/// a stream. None today.
35const SKIP_PARAMS: &[&str] = &[];
36
37/// Read-only verbs — `init` writes config, `check`/`report` write
38/// the `.mimus/` triples cache, `fix` rewrites claims. Mutators
39/// belong behind a deliberate trigger, not ambient view polling.
40const READ_ONLY: &[&str] = &["audit", "claims", "coverage", "doctor"];
41
42/// `(name, clap spec)` for every leaf verb. Hidden plumbing
43/// (`gen-cli-docs`) is a docs-pipeline verb, not an engine op —
44/// it stays reachable through `run` passthrough.
45fn verbs() -> Vec<(String, Command)> {
46    fn walk(prefix: &str, cmd: &Command, out: &mut Vec<(String, Command)>) {
47        for sub in cmd.get_subcommands() {
48            let name = sub.get_name();
49            if name == "help" || sub.is_hide_set() {
50                continue;
51            }
52            let dotted = match prefix {
53                "" => name.to_string(),
54                _ => format!("{prefix}.{name}"),
55            };
56            if sub.has_subcommands() {
57                walk(&dotted, sub, out);
58            } else {
59                out.push((dotted, sub.clone()));
60            }
61        }
62    }
63    let mut out = Vec::new();
64    walk("", &mimus::command(), &mut out);
65    out
66}
67
68fn op_cmd(name: &str) -> Option<Command> {
69    verbs()
70        .into_iter()
71        .find(|(n, _)| n == name && !SURFACES.contains(&n.as_str()))
72        .map(|(_, c)| c)
73}
74
75/// Args that can't ride the wire: streaming flags (above), the
76/// driver-resolved `--config` global, and clap's own help/version
77/// actions.
78fn skipped(a: &Arg) -> bool {
79    SKIP_PARAMS.contains(&a.get_id().as_str())
80        || a.get_id() == "config"
81        || matches!(
82            a.get_action(),
83            ArgAction::Help | ArgAction::HelpShort | ArgAction::HelpLong | ArgAction::Version
84        )
85}
86
87/// A clap arg → the manifest's typed param kind — `arg` =
88/// positional, `opt` = `--name <v>`, `flag` = `--name` — the
89/// contract `pavo-driver-kit`'s surface layer renders args from.
90fn kind(a: &Arg) -> &'static str {
91    if a.is_positional() {
92        "arg"
93    } else if matches!(
94        a.get_action(),
95        ArgAction::SetTrue | ArgAction::SetFalse | ArgAction::Count
96    ) {
97        "flag"
98    } else {
99        "opt"
100    }
101}
102
103fn param_json(a: &Arg) -> Value {
104    // Wire name = the flag name (`--no-cmd`), not the field id
105    // (`no_cmd`) — generated surfaces spell params `--<name>`.
106    let id = a.get_id().as_str();
107    let name = match a.is_positional() {
108        true => id,
109        false => a.get_long().unwrap_or(id),
110    };
111    json!({
112        "name": name,
113        "kind": kind(a),
114        "doc": a.get_help().map(|d| d.to_string()).unwrap_or_default(),
115    })
116}

118/// The `ops` query/`ops` argv verb payload — self-describing so
119/// hosts and agents discover the engine surface without argv help.
120/// `dir`/`root`/`repo`/`config` are the driver's globals —
121/// accepted by every op, listed once on the manifest itself.
122pub fn manifest() -> Value {
123    json!({
124        "engine": "mimus",
125        "invoke": "query `cli.<verb>` with the listed params, or argv `tamias-driver-mimus run <mimus argv>`",
126        "globals": [
127            {"name": "dir|root|repo", "kind": "opt", "doc": "Search dir the config search starts from — same resolution the view queries use (param → $MIMUS_CONFIG → <dir>/mimus.toml → ~/.config/mimus/); the repo root is the config's parent dir."},
128            {"name": "config", "kind": "opt", "doc": "mimus.toml path — the driver's own chain resolves it and hands the child an absolute --config."},
129            {"name": "format", "kind": "opt", "doc": "Engine output format — human | json (the CLI's --format global)."},
130        ],
131        "ops": verbs().iter()
132            .filter(|(n, _)| !SURFACES.contains(&n.as_str()))
133            .map(|(n, c)| json!({
134                "query": format!("cli.{n}"),
135                "doc": c.get_about().map(|d| d.to_string()).unwrap_or_default(),
136                "params": c.get_arguments()
137                    .filter(|a| !skipped(a))
138                    .map(param_json)
139                    .collect::<Vec<_>>(),
140                "mutates": !READ_ONLY.contains(&n.as_str()),
141            }))
142            .collect::<Vec<_>>(),
143        "surfaces": SURFACES,
144    })
145}

147/// Is `query` an engine op (`cli.<verb>`)? Returns the verb name.
148pub fn op_of(query: &str) -> Option<String> {
149    query
150        .strip_prefix("cli.")
151        .filter(|n| op_cmd(n).is_some())
152        .map(str::to_string)
153}

155/// A wire scalar → its argv token (numbers/bools stringify).
156fn scalar(v: &Value) -> Option<String> {
157    match v {
158        Value::String(s) => Some(s.clone()),
159        Value::Number(n) => Some(n.to_string()),
160        Value::Bool(b) => Some(b.to_string()),
161        _ => None,
162    }
163}
164
165/// All values of a param — arrays flatten, scalars wrap.
166fn values(v: &Value) -> Vec<String> {
167    match v {
168        Value::Array(xs) => xs.iter().filter_map(scalar).collect(),
169        other => scalar(other).into_iter().collect(),
170    }
171}

173/// `cli.<verb>` params → mimus argv (no argv[0] — `run` prepends
174/// it). Globals are resolved by [`run_verb`] (`--config` prepends
175/// here; the child cwd is the resolved dir, which is how the CLI
176/// finds its repo — mimus has no `--repo` flag). The dotted verb
177/// name splits into the group path; declared params map by kind —
178/// positionals land last, in schema order.
179fn argv(
180    config: Option<&str>,
181    name: &str,
182    cmd: &Command,
183    params: &Map<String, Value>,
184) -> Vec<String> {
185    let mut out: Vec<String> = Vec::new();
186    if let Some(c) = config {
187        out.extend(["--config".into(), c.into()]);
188    }
189    // --format is a clap *global*: it never appears in a
190    // subcommand's get_arguments(), so the param loop below can't
191    // see it — map it by hand. Same for --config (resolved, above).
192    if let Some(f) = params.get("format").and_then(scalar) {
193        out.extend(["--format".into(), f]);
194    }
195    out.extend(name.split('.').map(String::from));
196    for a in cmd.get_arguments() {
197        if skipped(a) || a.is_positional() {
198            continue;
199        }
200        let id = a.get_id().as_str();
201        let long = a.get_long().unwrap_or(id);
202        let raw = params
203            .get(long)
204            .or_else(|| params.get(id))
205            .or_else(|| params.get(id.replace('_', "-").as_str()));
206        let Some(raw) = raw else { continue };
207        match kind(a) {
208            "flag" => {
209                if raw.as_bool().unwrap_or(false) {
210                    out.push(format!("--{long}"));
211                }
212            }
213            _ => {
214                for s in values(raw) {
215                    out.push(format!("--{long}"));
216                    out.push(s);
217                }
218            }
219        }
220    }
221    let mut positionals: Vec<String> = Vec::new();
222    for a in cmd.get_positionals() {
223        if skipped(a) {
224            continue;
225        }
226        let id = a.get_id().as_str();
227        if let Some(raw) = params
228            .get(id)
229            .or_else(|| params.get(id.replace('_', "-").as_str()))
230        {
231            positionals.extend(values(raw));
232        }
233    }
234    out.extend(positionals);
235    out
236}

238/// The bare-name entry the embedded surfaces use (`OpSpec.name` is
239/// `cli.`-stripped); the wire path goes `op_of` → `run_verb`.
240#[cfg(any(feature = "cli", feature = "mcp"))]
241pub fn run_op(op: &str, params: &Map<String, Value>) -> Response {
242    match op_of(&format!("cli.{}", op.strip_prefix("cli.").unwrap_or(op))) {
243        Some(name) => run_verb(&name, params),
244        None => Response::err(format!("unknown op: {op}")),
245    }
246}
247
248/// Run one engine op by name: re-exec `run` in a child whose cwd is
249/// the request's resolved dir (same resolution the views use), and
250/// hand back `{op, argv, code, stdout, stderr}` as the query
251/// payload. `ok:false` is reserved for spawn failures — a nonzero
252/// `code` is the engine's own exit status (e.g. `check` findings),
253/// honest data.
254pub fn run_verb(name: &str, params: &Map<String, Value>) -> Response {
255    let Some(cmd) = op_cmd(name) else {
256        return Response::err(format!("unknown op: {name}"));
257    };
258    let (dir, config) = crate::data::resolve(params);
259    let args = argv(config.as_deref(), name, &cmd, params);
260    let out = std::env::current_exe().and_then(|exe| {
261        std::process::Command::new(exe)
262            .arg("run")
263            .args(&args)
264            .current_dir(&dir)
265            .output()
266    });
267    let data = match out {
268        Ok(o) => json!({
269            "op": name,
270            "argv": args,
271            "code": o.status.code().unwrap_or(-1),
272            "stdout": String::from_utf8_lossy(&o.stdout),
273            "stderr": String::from_utf8_lossy(&o.stderr),
274        }),
275        Err(e) => return Response::err(format!("run {name}: {e}")),
276    };
277    Response::ok_typed(&QueryResult {
278        query: format!("cli.{name}"),
279        data,
280    })
281}

283#[cfg(test)]
284mod tests {
285    #![allow(clippy::unwrap_used)]
286    use super::*;
287    use serde_json::json;

289    #[test]
290    fn op_of_accepts_cli_prefix_only() {
291        assert_eq!(op_of("cli.audit").unwrap(), "audit");
292        assert_eq!(op_of("cli.check").unwrap(), "check");
293        assert!(op_of("cli.gen-cli-docs").is_none()); // hidden plumbing
294        assert!(op_of("audit").is_none()); // view query, not an op
295        assert!(op_of("cli.bogus").is_none());
296    }
297
298    #[test]
299    fn manifest_lists_every_leaf_as_cli_query() {
300        let m = manifest();
301        let ops = m["ops"].as_array().unwrap();
302        for (n, _) in verbs()
303            .iter()
304            .filter(|(n, _)| !SURFACES.contains(&n.as_str()))
305        {
306            assert!(
307                ops.iter().any(|o| o["query"] == json!(format!("cli.{n}"))),
308                "{n} missing from ops manifest"
309            );
310        }
311        assert_eq!(m["surfaces"], json!(SURFACES));
312        for o in ops {
313            for p in o["params"].as_array().unwrap() {
314                assert!(["arg", "opt", "flag"].contains(&p["kind"].as_str().unwrap()));
315                // streaming flags never ride the wire
316                assert!(!SKIP_PARAMS.contains(&p["name"].as_str().unwrap()));
317                // --config is a driver-resolved global, never a per-op param
318                assert_ne!(p["name"], json!("config"));
319            }
320        }
321        // mutates honesty — the write set is exactly these
322        let mutators: Vec<&str> = ops
323            .iter()
324            .filter(|o| o["mutates"] == json!(true))
325            .filter_map(|o| o["query"].as_str())
326            .collect();
327        assert_eq!(
328            mutators,
329            vec!["cli.init", "cli.check", "cli.fix", "cli.report"]
330        );
331    }

333    #[test]
334    fn argv_maps_params_to_mimus_flags() {
335        let cmd = op_cmd("check").unwrap();
336        let mut p = Map::new();
337        p.insert("all".into(), json!(true));
338        p.insert("no-cmd".into(), json!(true));
339        p.insert("diff".into(), json!("origin/main...HEAD"));
340        assert_eq!(
341            argv(Some("/cfg/mimus.toml"), "check", &cmd, &p),
342            vec![
343                "--config",
344                "/cfg/mimus.toml",
345                "check",
346                "--all",
347                "--no-cmd",
348                "--diff",
349                "origin/main...HEAD",
350            ]
351        );
352    }
