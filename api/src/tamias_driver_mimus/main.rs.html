1//! tamias-driver-mimus — the pavo view driver for a repo's mimus
2//! critic state.
3//!
4//! Verbs (the tamias driver contract):
5//!
6//! - `manifest` — print the [`DriverManifest`] JSON (discovery
7//!   probe; never touches repo state).
8//! - `capabilities` — `{"capabilities": ["pavo"], "verbs": […]}`.
9//! - `ops` — the engine ops manifest: every `mimus` CLI verb as a
10//!   `cli.<verb>` wire query, derived from the engine's own clap
11//!   tree ([`mimus::command`]).
12//! - `run <argv>` — engine passthrough: `tamias-driver-mimus run
13//!   audit` ≡ `mimus audit`. Also the child side of `cli.*` wire
14//!   queries (`ops::run_verb` re-execs this same verb so engine
15//!   output never touches the serve stream).
16//! - `serve` — ndjson JSON-RPC on stdio: `{"op":"manifest"}`,
17//!   `{"op":"views"}`, `{"op":"view","view":…,"params":{…}}`,
18//!   `{"op":"query","query":…,"params":{…}}` →
19//!   `{"ok":true,"result":…}` / `{"ok":false,"error":…}`.
20//!   `query:"ops"` answers the ops manifest; `query:"cli.<verb>"`
21//!   runs the engine op.
22//!
23//! - `cli <mimus argv>` — the tamias `bins[]` shim verb (P-021):
24//!   `tamias bin exec` runs `tamias-driver-mimus cli --dir <pkg>
25//!   --name mimus <argv>` for a declared `shim: cli` bin; the two
26//!   driver flags are stripped and the rest goes to
27//!   `mimus::run_from` with inherited stdio — full engine parity
28//!   (`--config` globals before the verb, `--help`, stdin-bound
29//!   verbs, the unbounded surfaces should mimus grow any).
30//!
31//! Optional surface: feature `mcp` embeds pavo-driver-kit's
32//! surface layer — `mcp` serves the ops as MCP tools on stdio,
33//! equivalent to the standalone `tamias-driver-mcp` surface
34//! driver, minus the PATH/discovery hop.
35//!
36//! Which repo: each `view`/`query` resolves its own target —
37//! `params.dir`/`params.root`/`params.repo`/`params.config` →
38//! `$MIMUS_CONFIG` → `<dir>/mimus.toml` → `~/.config/mimus/mimus.toml`,
39//! exactly the `mimus` CLI's config search; the repo root is the
40//! config's parent dir. See `data.rs` for the exact order. Engine
41//! ops are the exception by design: `cli.<verb>` re-execs `run` in
42//! the resolved dir and lets `mimus::dispatch` own the outcome —
43//! writes keep the engine's own semantics (config search,
44//! `.mimus/` cache, claim rewrites). `manifest`/`views` are
45//! repo-free so pavo can enumerate drivers anywhere.
46//!
47//! The view side is **read-only**: it reads `mimus.toml`,
48//! `rules/*.toml`, and `.kuda` card provenance in-process, and
49//! shells out to `mimus report --format json` for the violations
50//! view — mimus owns the check pipeline; the driver only ever
51//! reads its output.

63fn capabilities_json() -> Value {
64    let m = views::manifest();
65    // The kit's shape plus the view count every sibling driver
66    // advertises.
67    let mut v = kit::capabilities_value(&m);
68    v["views"] = json!(m.views.len());
69    v
70}

72/// One parsed wire [`Request`] → response. The kit's serve loop owns
73/// the ndjson plumbing and turns parse failures into `ok:false`;
74/// ops this driver doesn't implement (`render`, `submit`, `jobs`,
75/// `notifications`) get the same `ok:false` convention every
76/// non-provider driver answers.
77fn dispatch(req: &Request) -> Response {
78    match req {
79        Request::Manifest => Response::ok_typed(&views::manifest()),
80        Request::Views => {
81            let summaries: Vec<ViewSummary> =
82                views::specs().iter().map(ViewSummary::from).collect();
83            Response::ok_typed(&summaries)
84        }
85        Request::View { view, params } => match views::view_result(view, params) {
86            Ok(result) => Response::ok_typed(&result),
87            Err(e) => Response::err(e),
88        },
89        Request::Query { query, params } => {
90            // `ops` and `cli.<verb>` are engine ops; everything
91            // else is a view query — unknown names fail the op,
92            // degraded snapshots travel as `{"error": …}` data
93            // (the marker renderers understand).
94            if query == "ops" {
95                Response::ok_typed(&pavo_protocol::wire::QueryResult {
96                    query: query.clone(),
97                    data: ops::manifest(),
98                })
99            } else if let Some(v) = ops::op_of(query) {
100                ops::run_verb(&v, params)
101            } else if views::known_query(query) {
102                let snap = data::load(params);
103                let payload = views::query_data(query, params, &snap);
104                Response::ok_typed(&pavo_protocol::wire::QueryResult {
105                    query: query.clone(),
106                    data: payload,
107                })
108            } else {
109                Response::err(format!("unknown query: {query}"))
110            }
111        }
112        // `render` — this driver ships no renderer plugins; hosts
113        // only call it for kinds the manifest claims, so answering
114        // an explicit error keeps the wire honest.
115        Request::Render { kind, .. } => Response::err(format!(
116            "tamias-driver-mimus claims no renderers (kind '{kind}')"
117        )),
118        // The protocol grew `submit`/`jobs`/`notifications` ops after
119        // this driver was written; mutations go through `cli.<verb>`,
120        // never `submit`, so they answer `ok:false` like any
121        // unsupported op.
122        Request::Submit { .. } | Request::Jobs | Request::Notifications => {
123            Response::err("tamias-driver-mimus implements no submit/jobs/notifications ops")
124        }
125    }
126}

128fn serve() -> i32 {
129    // The kit owns the ndjson loop — stdin EOF and a dead client
130    // both end it quietly, and either way the driver exits 0.
131    let _ = kit::serve_stdio(dispatch);
132    0
133}

135fn main() {
136    let args: Vec<String> = std::env::args().skip(1).collect();
137    let code = match args.first().map(String::as_str) {
138        Some("manifest") => {
139            println!("{}", kit::manifest_json(&views::manifest()));
140            0
141        }
142        Some("capabilities") => {
143            println!(
144                "{}",
145                serde_json::to_string_pretty(&capabilities_json()).unwrap_or_default()
146            );
147            0
148        }
149        Some("ops") => {
150            println!("{}", ops::manifest());
151            0
152        }
153        // Engine passthrough — `tamias-driver-mimus run audit` is
154        // `mimus audit`. Also the child side of `cli.*` wire
155        // queries (ops::run_verb re-execs this same verb).
156        Some("run") => {
157            let argv = std::iter::once("mimus".to_string()).chain(args.iter().skip(1).cloned());
158            std::process::exit(mimus::run_from(argv));
159        }
160        // The tamias `bins[]` shim verb (P-021): `tamias bin exec`
161        // runs `<driver> cli --dir <pkg> --name <pkg> <mimus argv>`
162        // for a declared `shim: cli` bin. mimus's engine IS the
163        // CLI, so this is a passthrough — strip the two standard
164        // driver flags and hand the rest to `mimus::run_from` with
165        // inherited stdio. That keeps full engine parity a
166        // generated ops→CLI surface can't express: globals before
167        // the verb (`--config`), `--help`, and stdin-bound verbs
168        // all behave exactly like the `mimus` binary.
169        Some("cli") => {
170            let argv = engine_argv(&args[1..]);
171            std::process::exit(mimus::run_from(argv));
172        }
173        // Optional embedded surface (feature-gated): the
174        // pavo-driver-kit surface layer serves this driver's own
175        // ops manifest as MCP tools over stdio — the same ops the
176        // `cli.*` wire queries expose.
177        #[cfg(feature = "mcp")]
178        Some("mcp") => {
179            let m = kit::surface::OpsManifest::from_value(&ops::manifest()).unwrap_or_default();
180            match kit::surface::mcp::serve_stdio(&m, &mut surface::Engine) {
181                Ok(()) => 0,
182                Err(e) => {
183                    eprintln!("tamias-driver-mimus mcp: {e}");
184                    1
185                }
186            }
187        }
188        Some("serve") | None => serve(),
189        _ => {
190            eprintln!(
191                "usage: tamias-driver-mimus manifest|capabilities|ops|run <argv>|cli <mimus argv>|serve\n       (feature `mcp` adds `mcp`)"
192            );
193            1
194        }
195    };
196    std::process::exit(code);
197}

199/// `cli` verb argv (after the `cli` token) → engine argv (argv[0]
200/// included). `tamias bin exec`/`driver exec` prepend the standard
201/// driver flags — `--dir <payload>` `--name <pkg>` — which the
202/// engine doesn't know; consume the leading run of those two
203/// (`--flag v` and `--flag=v` forms) and pass everything else
204/// through untouched. `--config` is deliberately NOT stripped:
205/// it's a mimus global, so a user-passed one must reach the
206/// engine's own parse.
207fn engine_argv(args: &[String]) -> Vec<String> {
208    let mut out = vec!["mimus".to_string()];
209    let mut i = 0;
210    while i < args.len() {
211        let a = args[i].as_str();
212        if a == "--dir" || a == "--name" {
213            // flag + separate value — skip both (a trailing flag
214            // with no value just ends the scan).
215            i = (i + 2).min(args.len());
216        } else if a.starts_with("--dir=") || a.starts_with("--name=") {
217            i += 1;
218        } else {
219            break;
220        }
221    }
222    out.extend(args[i..].iter().cloned());
223    out
224}
