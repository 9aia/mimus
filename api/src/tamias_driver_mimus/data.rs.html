1//! data — one read of a repo's mimus state per wire op.
2//!
3//! mimus is unusual among the 9aia tools: it keeps **no persistent
4//! state**. There is no `~/.local/share/mimus`, no verdict log, no
5//! history — `mimus.toml` plus the repo's own sources *are* the
6//! state, and every `mimus` verb recomputes on read. The only
7//! on-disk artifacts are caches: `.mimus/triples.db` (prose-triple
8//! extraction cache, ADR-0002 ph3) and the kuda index the symbol/
9//! graph checkers read (`[check].kuda_index`).
10//!
11//! So this driver's "snapshot" is resolution, not loading:
12//!
13//! - `mimus:status`, `mimus:rules`, `mimus:audit` read in-process —
14//!   config via `mimus-config`'s `FileSource` (the same source the
15//!   CLI uses), rules via `mimus-rules::load_rules`, card trust scores
16//!   via the same `.kuda/modules/*.yaml` provenance-hash walk
17//!   `mimus audit` performs.
18//! - `mimus:violations` is the expensive op: it shells out to
19//!   `mimus report --format json` (the full check pipeline —
20//!   claim extraction, fs/cmd/symbol checkers, rules, graph
21//!   buckets). Reimplementing that pipeline here would fork the
22//!   critic; the CLI is the authority.
23//!
24//! Root/config resolution mirrors the `mimus` CLI exactly
25//! (`mimus-cli/src/main.rs` `root()` + `mimus-config`'s
26//! `FileSource`):
27//!
28//! 1. Request params — `config` names a config file directly;
29//!    `dir`/`root`/`repo` name the repo dir the config search
30//!    starts from (the CLI's `./mimus.toml` becomes
31//!    `<dir>/mimus.toml`).
32//! 2. `$MIMUS_CONFIG` — same env var the CLI honors.
33//! 3. `<search-dir>/mimus.toml` — `./mimus.toml` for the CLI.
34//! 4. `~/.config/mimus/mimus.toml` — the global fallback.
35//!
36//! The repo root is the config file's parent directory (the CLI's
37//! `root()`), or `.`/the `dir` param when no config exists — the
38//! same place `mimus audit`/`mimus doctor` would look.
39
40use std::path::{Path, PathBuf};
41use std::time::Duration;
42
43use mimus_config::ConfigSource;
44use mimus_core::Config;
45use serde_json::{Map, Value};
46
47/// Everything the views need, resolved once per op.
48pub struct Snapshot {
49    /// Repo root — config file's parent, `dir` param, or `.`.
50    pub root: PathBuf,
51    /// The `mimus.toml` that loaded, display string; `None` when
52    /// the default search found nothing.
53    pub config_path: Option<String>,
54    /// Parsed config — `None` when no mimus.toml exists (mimus
55    /// verbs that need it degrade; `audit`/`doctor` don't).
56    pub config: Option<Config>,
57    /// `mimus` executable found on PATH (for the violations view).
58    pub mimus_bin: Option<PathBuf>,
59    /// Per-source load problems — surfaced, never fatal.
60    pub warnings: Vec<String>,
61}
62
63impl Default for Snapshot {
64    fn default() -> Self {
65        Snapshot {
66            root: PathBuf::from("."),
67            config_path: None,
68            config: None,
69            mimus_bin: None,
70            warnings: Vec::new(),
71        }
72    }
73}

75/// `params` → `(repo_dir, config_path)` overrides. `dir`/`root`/
76/// `repo` all name the search dir; `dir` wins (mirrors the CLI
77/// globals' precedence and sturn-driver's convention).
78fn overrides(params: &Map<String, Value>) -> (Option<PathBuf>, Option<String>) {
79    let dir = ["dir", "root", "repo"]
80        .iter()
81        .find_map(|k| params.get(*k).and_then(|v| v.as_str()))
82        .map(PathBuf::from);
83    let config = params
84        .get("config")
85        .and_then(|v| v.as_str())
86        .map(str::to_string);
87    (dir, config)
88}

90/// The config-file candidates in CLI order, given the explicit
91/// path/env override and the directory `./mimus.*` resolves
92/// against — `mimus_config::candidates_in` is the same scarab
93/// discovery the CLI's `FileSource` runs, rooted at `search_dir`.
94/// Returns the first that exists.
95fn find_config(explicit: Option<String>, search_dir: &Path) -> Option<PathBuf> {
96    mimus_config::candidates_in(search_dir, explicit)
97        .into_iter()
98        .find(|p| p.is_file())
99}

101/// Parse a config file through the same `FileSource` the CLI
102/// wires — explicit path only, so the search semantics stay here.
103fn parse_config(path: &Path) -> Result<Config, String> {
104    mimus_config::FileSource {
105        path: Some(path.display().to_string()),
106    }
107    .load()
108}

110/// `which mimus` — the violations view shells out; record whether
111/// the binary exists so status can say so honestly.
112fn find_mimus() -> Option<PathBuf> {
113    let path = std::env::var_os("PATH")?;
114    for dir in std::env::split_paths(&path) {
115        let cand = dir.join("mimus");
116        if cand.is_file() {
117            return Some(cand);
118        }
119    }
120    None
121}

123/// `(dir, config)` for an engine-op child — the resolution `load`
124/// uses, minus the snapshot: `dir` is the cwd the re-exec'd `run`
125/// inherits (config's parent, else the search dir), `config` is
126/// the `mimus.toml` the child gets as an absolute `--config`.
127/// `None` config leaves the child's own search in charge.
128pub fn resolve(params: &Map<String, Value>) -> (PathBuf, Option<String>) {
129    let (param_dir, param_config) = overrides(params);
130    let search_dir = param_dir.unwrap_or_else(|| PathBuf::from("."));
131    let found = find_config(param_config, &search_dir);
132    let dir = found
133        .as_ref()
134        .and_then(|p| p.parent())
135        .filter(|p| !p.as_os_str().is_empty())
136        .map(Path::to_path_buf)
137        .unwrap_or(search_dir);
138    let dir = std::fs::canonicalize(&dir).unwrap_or(dir);
139    let config = found.map(|p| {
140        std::fs::canonicalize(&p)
141            .map(|a| a.to_string_lossy().into_owned())
142            .unwrap_or_else(|_| p.display().to_string())
143    });
144    (dir, config)
145}

147/// Resolve + load. `params` are the wire op's caller params —
148/// `dir`/`root`/`repo`/`config` steer resolution, the rest pass
149/// through to the query resolvers untouched.
150pub fn load(params: &Map<String, Value>) -> Snapshot {
151    let (param_dir, param_config) = overrides(params);
152    let search_dir = param_dir.clone().unwrap_or_else(|| PathBuf::from("."));
153
154    let mut snap = Snapshot {
155        mimus_bin: find_mimus(),
156        ..Snapshot::default()
157    };
158
159    let found = find_config(param_config, &search_dir);
160    snap.config_path = found.as_ref().map(|p| p.display().to_string());
161
162    match &found {
163        Some(path) => {
164            // The CLI's root() — the config file's parent dir.
165            snap.root = path
166                .parent()
167                .filter(|p| !p.as_os_str().is_empty())
168                .map(Path::to_path_buf)
169                .unwrap_or_else(|| PathBuf::from("."));
170            match parse_config(path) {
171                Ok(cfg) => snap.config = Some(cfg),
172                Err(e) => snap
173                    .warnings
174                    .push(format!("config {}: {e}", path.display())),
175            }
176        }
177        None => {
178            // No mimus.toml anywhere — the CLI would fail `check`/
179            // `report` here, but `audit`/`doctor` still run against
180            // "."; keep that honest degradation.
181            snap.root = search_dir;
182        }
183    }
184    snap
185}

187/// Run `mimus report --format json` in the resolved root and parse
188/// the violations array. This is the one expensive op — everything
189/// else reads files in-process. A hard timeout keeps a configured
190/// semantic tier (`[check].acp_cmd`) from hanging the wire forever.
191pub fn violations_json(snap: &Snapshot) -> Result<Vec<Value>, String> {
192    let Some(bin) = &snap.mimus_bin else {
193        return Err(
194            "mimus CLI not found on PATH — install mimus to populate mimus:violations".into(),
195        );
196    };
197    if snap.config_path.is_none() {
198        return Err(
199            "no mimus.toml found (searched $MIMUS_CONFIG, <dir>/mimus.toml, ~/.config/mimus/)"
200                .into(),
201        );
202    }
203    let mut cmd = std::process::Command::new(bin);
204    cmd.arg("report")
205        .arg("--format")
206        .arg("json")
207        .current_dir(&snap.root)
208        .stdout(std::process::Stdio::piped())
209        .stderr(std::process::Stdio::piped());
210    let mut child = cmd
211        .spawn()
212        .map_err(|e| format!("spawn {}: {e}", bin.display()))?;
213    // Poll with a deadline — std has no wait-with-timeout.
214    let deadline = std::time::Instant::now() + Duration::from_secs(60);
215    loop {
216        match child.try_wait() {
217            Ok(Some(_)) => break,
218            Ok(None) if std::time::Instant::now() < deadline => {
219                std::thread::sleep(Duration::from_millis(50));
220            }
221            Ok(None) => {
222                let _ = child.kill();
223                let _ = child.wait();
224                return Err("mimus report timed out after 60s".into());
225            }
226            Err(e) => return Err(format!("wait mimus report: {e}")),
227        }
228    }
229    let out = child
230        .wait_with_output()
231        .map_err(|e| format!("read mimus report: {e}"))?;
232    let text = String::from_utf8_lossy(&out.stdout);
233    // `mimus report` exits 1 when error-severity violations exist —
234    // the JSON array is still the payload; only a non-JSON stdout
235    // (config error, crash) is a real failure.
236    match serde_json::from_str::<Vec<Value>>(&text) {
237        Ok(v) => Ok(v),
238        Err(e) => {
239            let err = String::from_utf8_lossy(&out.stderr);
240            let msg: String = err
241                .lines()
242                .next()
243                .unwrap_or("mimus report produced no JSON")
244                .to_owned();
245            Err(format!("mimus report: {msg} ({e})"))
246        }
247    }
248}
